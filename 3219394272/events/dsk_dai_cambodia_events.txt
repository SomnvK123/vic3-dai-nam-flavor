# Event 5: Khmer cầu viện Xiêm La (gửi sang SIA)
# Chỉ xảy ra khi thanh DAI giảm dưới 40
dsk_dai_cambodia_events.5 = {
	type = country_event
	placement = root
	title = dsk_dai_cambodia_events.5.t
	desc = dsk_dai_cambodia_events.5.d
	flavor = dsk_dai_cambodia_events.5.f
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3

	trigger = {
		has_state_in_state_region = STATE_CAMBODIA
		NOT = { has_global_variable = dsk_sia_asked_to_join }
		game_date >= 1841.1.1		scope:journal_entry = {
			"scripted_bar_progress(dsk_dai_cambodia_progress_bar)" < 75
		}
	}

	# Event này xảy ra ở DAI nhưng gửi event cho SIA
	immediate = {
		set_global_variable = dsk_sia_asked_to_join
		c:SIA ?= {
			trigger_event = { id = dsk_dai_cambodia_events.10 days = 10 }
		}
	}

	option = {
		name = dsk_dai_cambodia_events.5.a
		# DAI chỉ biết tin, không thể ngăn cản
	}
}

# Event 7: DAI thua -> SIA phục quốc CAM
dsk_dai_cambodia_events.7 = {
	type = country_event
	placement = root
	title = dsk_dai_cambodia_events.7.t
	desc = dsk_dai_cambodia_events.7.d
	flavor = dsk_dai_cambodia_events.7.f
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3

	option = {
		name = dsk_dai_cambodia_events.7.a
		if = {
			limit = {
				NOT = { exists = c:CAM }
			}
			create_country = {
				tag = CAM
				origin = c:DAI
				province = p:x36D7BA
			}
		}
		s:STATE_CAMBODIA = {
			set_owner_of_provinces = {
				country = c:CAM
				provinces = { x36D7BA xDF21D5 xEFEF20 x989BA9 x2E84A3 x810F9E xFA2C44 xD45279 x81EF79 xA3B8D9 xA9542D xB5755C x65369D x20D113 x7070A0 x9EEF57 }
			}
		}
		if = {
			limit = {
				exists = c:CAM
				c:CAM = { is_subject = no }
			}
			c:CAM = {
				create_diplomatic_pact = {
					country = ROOT
					type = puppet
				}
			}
		}
		add_prestige = 15
	}
}

# Event 10: Khmer cầu viện - SIA quyết định (SIA nhận)

	type = country_event
	placement = root
	title = dsk_dai_cambodia_events.10.t
	desc = dsk_dai_cambodia_events.10.d
	flavor = dsk_dai_cambodia_events.10.f
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3

	option = {
		# Đồng ý tham chiến - hỗ trợ khởi nghĩa
		name = dsk_dai_cambodia_events.10.a
		add_treasury = -100000
		set_global_variable = dsk_sia_joined_cambodia_war
		if = {
			limit = { exists = c:DAI }
			change_relations = { country = c:DAI value = -50 }
		}
		add_loyalists = { value = 0.1 interest_group = ig:ig_armed_forces }
	}

	option = {
		# Từ chối - không muốn gây chiến với DAI
		name = dsk_dai_cambodia_events.10.b
		default_option = yes
		add_radicals = { value = 0.05 interest_group = ig:ig_armed_forces }
	}
}
