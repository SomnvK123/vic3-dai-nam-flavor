####################################
# Event 1: Tiếng Thì Thầm Trong Cấm Thành
# ("Whispers in the Forbidden City")
#
# Kích hoạt: Từ immediate block của JE je_dvf_hong_bao_coup
#            (chạy một lần khi JE lần đầu kích hoạt)
# Scope: Đại Nam (country_event)
#
# Lịch sử: Ngay khi Tự Đức lên ngôi, triều đình
# nhận tin Hồng Bảo đang ngấm ngầm tập hợp lực lượng.
####################################
dvf_hong_bao.1 = {
	type = country_event
	placement = root
	title = dvf_hong_bao.1.t
	desc = dvf_hong_bao.1.d
	flavor = dvf_hong_bao.1.f
	icon = "gfx/interface/icons/event_icons/event_political.dds"
	duration = 3

	event_image = {
		video = "asia_court_politics"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	is_triggered_only = yes

	immediate = {
		# Lưu Hồng Bảo vào scope để dùng trong options
		random_scope_character = {
			limit = {
				has_template = DAI_hong_bao
				is_agitator = yes
			}
			save_scope_as = hong_bao_scope
		}
	}

	# Lựa chọn A: Bắt giam ngay (Lịch sử — Âm mưu bị dập tắt)
	# Hồng Bảo bị giết → JE fail condition sẽ kích hoạt tự động
	option = {
		name = dvf_hong_bao.1.a
		default_option = yes

		# Bắt và tử hình Hồng Bảo (lịch sử: tự vẫn trong ngục 1854)
		if = {
			limit = { exists = scope:hong_bao_scope }
			scope:hong_bao_scope = {
				kill_character = { hidden = no }
			}
		}

		add_prestige = 10
		ig:ig_landowners = {
			add_loyalists = { value = 0.05 }
		}

		# Đánh dấu đã xử lý — JE fail sẽ trigger event 5
		set_global_variable = dvf_hong_bao_coup_resolved
	}

	# Lựa chọn B: Theo dõi tình hình (JE bắt đầu giằng co ngầm)
	option = {
		name = dvf_hong_bao.1.b

		# Tiến trình khởi đầu ở 1 (Hồng Bảo đang tập hợp lực lượng)
		set_variable = { name = dvf_hong_bao_coup_progress value = 1 }

		add_prestige = -5
		ig:ig_armed_forces = {
			add_radicals = { value = 0.03 }
		}
	}
}

####################################
# Event 4: Đảo Chính Thành Công!
# ("The Coup Succeeds!")
#
# Kích hoạt: on_complete của JE (tiến trình đạt 5)
# Alt-History: Hồng Bảo đánh úp Hoàng thành,
#              Tự Đức bị phế truất.
####################################
dvf_hong_bao.4 = {
	type = country_event
	placement = root
	title = dvf_hong_bao.4.t
	desc = dvf_hong_bao.4.d
	flavor = dvf_hong_bao.4.f
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3

	event_image = {
		video = "asia_westerners_arriving"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	is_triggered_only = yes

	immediate = {
		# Lưu Tự Đức vào scope (để xử lý số phận vua cũ ở Event 10)
		random_scope_character = {
			limit = { has_template = DAI_tu_duc }
			save_scope_as = tu_duc_scope
		}
		# Hồng Bảo lên ngôi Hoàng đế ngay trong immediate
		random_scope_character = {
			limit = { has_template = DAI_hong_bao }
			save_scope_as = hong_bao_scope
		}
		# FIX #4: replace_character_roles phải gọi sau khi đã save scope,
		# không được đặt trong random_scope_character block
		if = {
			limit = { exists = scope:hong_bao_scope }
			scope:hong_bao_scope = {
				replace_character_roles = { add = { ruler } remove_all = yes }
			}
		}
	}

	option = {
		name = dvf_hong_bao.4.a
		default_option = yes

		# Tự Đức tạm thời nhận modifier "bị lưu đày"
		# Số phận thực sự sẽ do người chơi quyết định ở Event 10
		if = {
			limit = { exists = scope:tu_duc_scope }
			scope:tu_duc_scope = {
				add_modifier = { name = dvf_tu_duc_exiled months = -1 }
			}
		}

		# Đảo chính gây chấn động: Nho gia & Phe đạo phẫn nộ
		add_prestige = -50
		ig:ig_landowners = {
			add_radicals = { value = 0.40 }
		}
		ig:ig_devout = {
			add_radicals = { value = 0.20 }
		}

		# Chính quyền mới: Quân đội & Trí thức ủng hộ Hồng Bảo
		ig:ig_armed_forces = {
			add_loyalists = { value = 0.30 }
		}
		ig:ig_intelligentsia = {
			add_loyalists = { value = 0.20 }
		}

		# Buff canh tân ban đầu
		add_modifier = {
			name = dvf_military_awakening
			years = 10
		}

		# Modifier Triều đình Duy Tân (Mục 4) — kéo dài đến khi JE Canh Tân kết thúc
		# Buff Intelligentsia/Armed Forces/Industrialists, debuff Landowners/Devout
		add_modifier = {
			name = dvf_hong_bao_reform_court
			months = -1
		}

		# Kích hoạt chuỗi Ổn Định Hậu Đảo Chính (§3.3)
		trigger_event = { id = dvf_hong_bao.10 days = 30 }
	}
}

####################################
# Event 10: Xử Trí Tự Đức (Ổn Định Hậu Đảo Chính)
# ("What to do with Tu Duc?")
#
# Kích hoạt: 30 ngày sau Event 4 (Đảo chính thành công)
# §3.3: Event đầu tiên chuỗi Ổn Định Hậu Đảo Chính

dvf_hong_bao.10 = {
	type = country_event
	placement = root
	title = dvf_hong_bao.10.t
	desc = dvf_hong_bao.10.d
	flavor = dvf_hong_bao.10.f
	icon = "gfx/interface/icons/event_icons/event_political.dds"
	duration = 3

	event_image = {
		video = "asia_court_politics"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	is_triggered_only = yes

	# Lựa chọn A: Lưu đày về lăng tẩm (Nhân đạo — tốt nhất)
	option = {
		name = dvf_hong_bao.10.a
		default_option = yes

		# country_legitimacy_base_add = +10 (định nghĩa trong modifier)
		add_modifier = {
			name = dvf_post_coup_legitimacy_high
			months = -1
		}
		# FIX: Report yêu cầu "Radicals -5%" — add_radicals không nhận giá trị âm,
		# dùng add_loyalists để đạt hiệu ứng dịu hóa phe bảo thủ tương đương
		add_loyalists = { value = 0.05 }

		trigger_event = { id = dvf_hong_bao.11 days = 60 }
	}

	# Lựa chọn B: Giam cầm vĩnh viễn
	option = {
		name = dvf_hong_bao.10.b

		# country_legitimacy_base_add = +5
		add_modifier = {
			name = dvf_post_coup_legitimacy_mid
			months = -1
		}
		# FIX #3: add_radicals phải chỉ định nhóm — phe bảo thủ phẫn nộ
		ig:ig_landowners = {
			add_radicals = { value = 0.10 }
		}
		ig:ig_devout = {
			add_radicals = { value = 0.05 }
		}

		trigger_event = { id = dvf_hong_bao.11 days = 60 }
	}

	# Lựa chọn C: Xử tử (Tàn bạo — mất uy tín quốc tế)
	option = {
		name = dvf_hong_bao.10.c

		# country_legitimacy_base_add = -15
		add_modifier = {
			name = dvf_post_coup_legitimacy_low
			months = -1
		}
		# FIX #3: add_radicals phải chỉ định nhóm — toàn bộ phe bảo thủ nổi loạn
		ig:ig_landowners = {
			add_radicals = { value = 0.25 }
		}
		ig:ig_devout = {
			add_radicals = { value = 0.15 }
		}
		every_country = {
			limit = { country_rank >= rank_value:great_power }
			change_relations = { country = ROOT value = -20 }
		}

		trigger_event = { id = dvf_hong_bao.11 days = 60 }
	}
}

####################################
# Event 12: Phong Thưởng Công Thần Canh Tân
# ("Rewarding the Reformist Ministers")
#
# Kích hoạt: 60 ngày sau Event 11
# Tự động — Hồng Bảo phong chức cho nhà canh tân.
# Đây là event cuối chuỗi Ổn Định Hậu Đảo Chính.
# Sau event này, JE Canh Tân Đất Nước kích hoạt.
####################################
dvf_hong_bao.12 = {
	type = country_event
	placement = root
	title = dvf_hong_bao.12.t
	desc = dvf_hong_bao.12.d
	flavor = dvf_hong_bao.12.f
	icon = "gfx/interface/icons/event_icons/event_political.dds"
	duration = 3

	event_image = {
		video = "asia_court_politics"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	is_triggered_only = yes

	immediate = {
		# Spawn Nguyễn Trường Tộ làm Cố vấn Canh tân nếu chưa có
		if = {
			limit = {
				NOT = {
					any_scope_character = { has_template = DAI_nguyen_truong_to }
				}
			}
			create_character = {
				template = DAI_nguyen_truong_to
				on_created = {
					replace_character_roles = { add = { politician } remove_all = yes }
				}
			}
		}
		# Spawn Đặng Huy Trứ làm Cục trưởng Cơ khí nếu chưa có
		if = {
			limit = {
				NOT = {
					any_scope_character = { has_template = DAI_dang_huy_tru }
				}
			}
			create_character = {
				template = DAI_dang_huy_tru
				on_created = {
					# FIX #5: Lịch sử Đặng Huy Trứ là nhà ngoại thương/cơ khí,
					# không phải tướng quân — dùng politician phù hợp hơn
					replace_character_roles = { add = { politician } remove_all = yes }
				}
			}
		}
	}

	option = {
		name = dvf_hong_bao.12.a
		default_option = yes

		# FIX: Report yêu cầu approval_add = +5, không phải add_loyalists
		# Áp dụng lên country scope (interest_group_ig_*_approval_add là country modifier)
		add_modifier = {
			name = dvf_reform_court_intelligentsia_approval
			years = 5
		}
		add_modifier = {
			name = dvf_reform_court_armed_forces_approval
			years = 5
		}

		# Quyền lực tập trung — country_authority_add = +100 (trong modifier)
		add_modifier = {
			name = dvf_new_court_authority
			months = -1
		}

		# Mở khóa JE Canh Tân Đất Nước (je_dvf_hong_bao_restoration)
		set_global_variable = dvf_hong_bao_restoration_unlocked
	}
}
