# Event 1: Tiếng sấm từ Quảng Châu (Cú Sốc Nam Kinh)
# Lịch sử: Hiệp ước Nam Kinh ký 29/8/1842. Vua đang trị vì: Thiệu Trị (1841-1847).
# Thiệu Trị là người bảo thủ, KHÔNG tiếp tục các chính sách canh tân của Minh Mạng.
# Ông chọn tăng cường cấm đạo và bế quan hơn nữa.
dvf_opium_war_events.1 = {
	type = country_event
	placement = root
	title = dvf_opium_war_events.1.t
	desc = dvf_opium_war_events.1.d
	flavor = dvf_opium_war_events.1.f
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3

	event_image = {
		video = "asia_westerners_arriving"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		exists = c:DAI
		c:DAI ?= THIS
		NOT = { has_variable = dvf_opium_shock_happened }
		exists = c:CHI
		c:CHI = {
			has_modifier = opium_wars_lost
		}
	}

	immediate = {
		set_variable = dvf_opium_shock_happened
	}

	# Lựa chọn 1: Bế quan tỏa cảng (Đúng lịch sử - Thiệu Trị thụ động)
	# Thiệu Trị tăng cường cấm đạo, hạn chế thương mại phương Tây nhưng vẫn quan hệ Thanh bình thường
	option = {
		name = dvf_opium_war_events.1.a
		default_option = yes

		ai_chance = {
			base = 80
		}

		# Tăng sức mạnh Địa chủ (Nho giáo thủ cựu)
		ig:ig_landowners = {
			add_modifier = {
				name = dvf_isolationist_policy
				years = 15
			}
		}

		# Chính sách cấm đạo + hạn chế thương mại phương Tây
		# Nhưng KHÔNG cắt đứt ngoại giao Thanh (vẫn triều cống)
		add_modifier = {
			name = dvf_closed_door_policy
			years = 15
		}

		# Lịch sử: Ngay sau 1842, quân đội Đại Nam bắt đầu tụt hậu dần
		# Thiệu Trị/Tự Đức không tiếp tục canh tân vũ khí của Minh Mạng
		# Debuff nhẹ cấp 1 — vẫn còn thời gian sửa nếu hoàn thành JE Tự Cường
		add_modifier = {
			name = dvf_military_backwardness
			years = 15
		}

		add_loyalists = { value = 0.05 interest_group = ig:ig_landowners }
		add_loyalists = { value = 0.05 interest_group = ig:ig_devout }

		# Kích hoạt event Đà Nẵng sau ~5 năm (1842 + 5 = 1847 - khớp lịch sử!)
		set_variable = { name = dvf_tourane_timer years = 5 }
	}

	# Lựa chọn 2: Tiếp nối di sản Minh Mạng - Cử sứ bộ mua vũ khí (Alternate History)
	# Thực tế: Minh Mạng (mất 1841) đã mua tàu hơi nước Pháp 1838, chế tạo thành công tàu
	# hơi nước 4/1839 (Hoàng Văn Lịch, Võ Huy Trinh). Cử Trần Viết Xương đi Châu Âu 1839.
	# Đây là con đường giả định: vua tiếp tục canh tân thay vì thu mình như Thiệu Trị thực tế.
	option = {
		name = dvf_opium_war_events.1.b

		ai_chance = {
			base = 20
		}

		# Tốn tiền cử sứ bộ mua vũ khí và tàu máy hơi nước
		add_treasury = -50000

		# Buff quân sự nhưng bất ổn nội bộ (phe Nho giáo phản đối)
		ig:ig_armed_forces = {
			add_modifier = {
				name = dvf_military_awakening
				years = 15
			}
		}

		add_modifier = {
			name = dvf_military_reform_unrest
			years = 10
		}

		add_radicals = { value = 0.05 interest_group = ig:ig_landowners }
		add_loyalists = { value = 0.05 interest_group = ig:ig_armed_forces }

		# Kích hoạt event Đà Nẵng sau ~5 năm
		set_variable = { name = dvf_tourane_timer years = 5 }
	}
}

# Event 2: Sự cố Vịnh Tourane - Đà Nẵng (15/4/1847)
# Lịch sử: 2 tàu chiến Pháp là Gloire (Đại tá de Lapierre) và Victorieuse (Tr.tá Rigault de Genouilly)
# đến Đà Nẵng đòi thả giáo sĩ, Thiệu Trị cự tuyệt. Pháp bắn chìm 5 thuyền chiến bọc đồng của Đại Nam,
# khoảng 230 người chết và bị thương. Pháp gần như không tổn thất.
# Rigault de Genouilly chính là người sẽ chỉ huy tấn công Đà Nẵng 1858.
dvf_opium_war_events.2 = {
	type = country_event
	placement = root
	title = dvf_opium_war_events.2.t
	desc = dvf_opium_war_events.2.d
	flavor = dvf_opium_war_events.2.f
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3

	event_image = {
		video = "unspecific_naval_battle"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		exists = c:DAI
		c:DAI ?= THIS
		has_variable = dvf_opium_shock_happened
		NOT = { has_variable = dvf_tourane_incident_happened }
		NOT = { has_variable = dvf_tourane_timer }
		exists = c:FRA
	}

	immediate = {
		set_variable = dvf_tourane_incident_happened
		c:FRA = {
			save_scope_as = fra_scope
		}
	}

	# Lựa chọn 1: Đánh trả bảo vệ quốc thể! (Đúng lịch sử - Thiệu Trị chọn đối đầu)
	# 5 thuyền chiến bọc đồng bị đánh chìm, ~230 người thương vong
	option = {
		name = dvf_opium_war_events.2.a
		default_option = yes

		ai_chance = {
			base = 80
		}

		# 5 thuyền chiến bọc đồng bị bắn chìm - hải quân tổn thất nặng 5 năm
		add_modifier = {
			name = dvf_fleet_destroyed_danang
			years = 5
		}

		# Nhục nhã quốc thể - triều đình và hải quân mất tinh thần
		# Lịch sử: Thiệu Trị sốc nặng, tăng cường phòng thủ nhưng uy tín sụt giảm
		add_modifier = {
			name = dvf_danang_humiliation
			years = 5
		}

		# Pháp có cớ tấn công (Rigault de Genouilly sẽ dùng sự kiện này làm cớ 1858)
		# years = 20 để kéo dài đến 1867 - bao phủ toàn bộ giai đoạn xâm lược
		c:FRA = {
			add_modifier = {
				name = dvf_french_casus_belli_dai
				years = 20
			}
		}

		# Tăng cường phòng thủ Đà Nẵng (đúng lịch sử: Thiệu Trị làm vậy sau sự cố)
		add_loyalists = { value = 0.05 interest_group = ig:ig_armed_forces }
		add_prestige = -25
	}

	# Lựa chọn 2: Nhượng bộ tạm thời (Alternate History)
	# Thực tế lịch sử: Thiệu Trị KHÔNG nhượng bộ. Pháp vẫn có cớ tấn công bất kể.
	option = {
		name = dvf_opium_war_events.2.b

		ai_chance = {
			base = 20
		}

		# Nhượng bộ: chấp nhận tự do truyền đạo tạm thời
		add_modifier = {
			name = dvf_western_concession
			years = 10
		}

		# Phe Nho giáo nổi giận - phản đối cực mạnh vì nhục quốc thể
		add_radicals = { value = 0.15 interest_group = ig:ig_landowners }
		add_radicals = { value = 0.15 interest_group = ig:ig_devout }
		add_prestige = -15

		# Pháp VẪN ghi nhận như một điểm yếu, quan hệ cải thiện ít
		# (thực tế Pháp vẫn tiếp tục gây áp lực bất kể Việt Nam nhượng bộ)
		change_relations = {
			country = c:FRA
			value = 15
		}

		# Đánh dấu đã nhượng bộ → kích hoạt chuỗi Event 2C sau ~5 năm
		set_variable = dvf_concession_made
		set_variable = { name = dvf_french_pressure_timer years = 5 }
	}
}

	trigger = {
	}

	option = {
		name = dvf_opium_war_events.4.a
		default_option = yes

		# Xóa debuff cấp 1 nếu còn, thay bằng debuff cấp 2 TRẦM TRỌNG

			years = 15
		}
		add_prestige = -30
	}
