####################################
# Event 1: Đại Nam Duy Tân (Thành Công)
#
# Kích hoạt: on_complete của je_dvf_hong_bao_restoration
####################################
dvf_hong_bao_restoration.1 = {
	type = country_event
	placement = root
	title = dvf_hong_bao_restoration.1.t
	desc = dvf_hong_bao_restoration.1.d
	flavor = dvf_hong_bao_restoration.1.f
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	duration = 3

	event_image = {
		video = "asia_westerners_arriving"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	is_triggered_only = yes

	option = {
		name = dvf_hong_bao_restoration.1.a
		default_option = yes
		
		add_prestige = 100
		
		# Gỡ bỏ buff cũ
		if = {
			limit = { has_modifier = dvf_military_awakening }
			remove_modifier = dvf_military_awakening
		}
		
		# Add buff vĩnh viễn
		add_modifier = {
			name = dvf_great_modernization
			months = -1
		}
		
		# Nhóm Trí thức & Quân đội vươn lên mạnh mẽ
		ig:ig_intelligentsia = {
			add_loyalists = { value = 0.20 }
		}
		ig:ig_armed_forces = {
			add_loyalists = { value = 0.20 }
		}
	}
}

####################################
# Event 2: Giấc Mộng Duy Tân Tan Vỡ (Thất Bại)
#
# Kích hoạt: on_fail của je_dvf_hong_bao_restoration
####################################
dvf_hong_bao_restoration.2 = {
	type = country_event
	placement = root
	title = dvf_hong_bao_restoration.2.t
	desc = dvf_hong_bao_restoration.2.d
	flavor = dvf_hong_bao_restoration.2.f
	icon = "gfx/interface/icons/event_icons/event_political.dds"
	duration = 3

	event_image = {
		video = "asia_court_politics"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	is_triggered_only = yes

	option = {
		name = dvf_hong_bao_restoration.2.a
		default_option = yes
		
		# Phạt nặng
		add_modifier = {
			name = dvf_hong_bao_dynasty_collapse
			years = 10
		}
		
		# Xóa buff
		if = {
			limit = { has_modifier = dvf_military_awakening }
			remove_modifier = dvf_military_awakening
		}
		if = {
			limit = { has_modifier = dvf_great_modernization }
			remove_modifier = dvf_great_modernization
		}
		
		every_interest_group = { add_radicals = { value = 0.20 } }
	}
}
