# MỤC 15: ÁP LỰC THỜI GIAN TỪ PHÁP (1856-1858)

####################################
# Event 1: Chiến Hạm Pháp Nả Pháo Cảnh Cáo Đà Nẵng (1856)
# "French Warships Bombard Da Nang"
#
# Kích hoạt: on_yearly_pulse (năm ≥ 1856)
# Điều kiện: Đại Nam (DAI) đang chơi path Hồng Bảo
#           VÀ vẫn còn cấm đạo (law_state_religion)
#           VÀ chưa từng kích hoạt event này
# Lịch sử: 1856, chiến hạm Pháp Catinat nả pháo Đà Nẵng
#          trả thù bách hại giáo sĩ Schoeffler & Bonnard.
#          Đây là lần đầu Pháp dùng vũ lực trực tiếp.
####################################
dvf_france.1 = {
	type = country_event
	placement = root
	title = dvf_france.1.t
	desc = dvf_france.1.d
	flavor = dvf_france.1.f
	icon = "gfx/interface/icons/event_icons/event_naval.dds"
	duration = 3

	event_image = {
		video = "asia_westerners_arriving"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	is_triggered_only = yes

	# Kiểm tra trigger trong immediate để không show popup nếu không thỏa điều kiện
	trigger = {
		# Chỉ DAI mới nhận event này
		tag = DAI
		# Đang chơi path Hồng Bảo
		has_global_variable = dvf_hong_bao_coup_success
		# Năm game ≥ 1856
		game_date >= 1856.1.1
		# Vẫn còn cấm đạo Công giáo
		has_law = law_type:law_state_religion
		# Chưa từng fire event này
		NOT = { has_variable = dvf_france_bombardment_fired }
	}

	immediate = {
		# Đánh dấu đã fire để không lặp
		set_variable = dvf_france_bombardment_fired
		# Modifier áp lực 3 năm
		add_modifier = {
			name = dvf_france_looming_threat
			months = 36
		}
	}

	# Lựa chọn A: Khẩn cấp thông qua Tự do Tín ngưỡng (Bắt đầu đổi luật)
	option = {
		name = dvf_france.1.a
		ai_chance = { base = 80 }

		# Ban hành luật ngay lập tức (activate_law là effect chuẩn V3, dùng cả trong
		# history lẫn event - đã verify qua joi_flavor_gov.txt của mod JoI)
		activate_law = law_type:law_freedom_of_conscience
		# Tốn 10 Prestige (chi phí chính trị thay cho -50 Authority không hợp lệ trong V3)
		add_prestige = -10
		# Flag để tracking
		set_variable = dvf_france_warned_reformed
	}

	# Lựa chọn B: Tăng cường quân đội phòng thủ (Không đổi luật)
	option = {
		name = dvf_france.1.b
		ai_chance = { base = 20 }

		# Unit defense bonus 5 năm
		add_modifier = {
			name = dvf_france_defense_boost
			months = 60
		}
		set_variable = dvf_france_warned_fortified
	}
}

####################################
# Event 2: Viễn Chinh Đông Dương (1858) - SỰ KIỆN QUYẾT ĐỊNH
# "French Expedition to Indochina"
#
# Kích hoạt: on_yearly_pulse (năm ≥ 1858)
# Rẽ nhánh tùy theo hành động người chơi:
# - NĂNH A: Đã thông qua law_freedom_of_conscience → Pháp MẤT CỞS → Thắng lợi Ngoại giao
# - NHÁNH B: Vẫn cấm đạo → Pháp TẤN CÔNG → Diplomatic Play / War
####################################
dvf_france.2 = {
	type = country_event
	placement = root
	title = dvf_france.2.t
	desc = dvf_france.2.d
	flavor = dvf_france.2.f
	icon = "gfx/interface/icons/event_icons/event_naval.dds"
	duration = 3

	event_image = {
		video = "asia_westerners_arriving"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	is_triggered_only = yes

	trigger = {
		tag = DAI
		has_global_variable = dvf_hong_bao_coup_success
		game_date >= 1858.1.1
		NOT = { has_variable = dvf_france_invasion_resolved }
	}

	immediate = {
		set_variable = dvf_france_invasion_resolved
	}

	# NHÁNH A: ĐÃ THÔNG QUA TỰ DO TÍN NGƯỠNG
	option = {
		name = dvf_france.2.a
		# Option này chỉ hiện nếu đã có law_freedom_of_conscience
		trigger = {
			has_law = law_type:law_freedom_of_conscience
		}

		# COMPATCH LOGIC: Backup xóa claim/biến JoI nếu còn sót lại
		# (remove_claim là effect của STATE scope, set_secret_goal dùng tham số
		# country/secret_goal - đã verify qua dvf_compatch_on_actions.txt)
		if = {
			limit = { has_variable = indochina_campaign_target }
			remove_variable = indochina_campaign_target
			if = {
				limit = { exists = c:FRA }
				s:STATE_TONKIN = { remove_claim = c:FRA }
				s:STATE_ANNAM = { remove_claim = c:FRA }
				s:STATE_MEKONG = { remove_claim = c:FRA }
				s:STATE_CAMBODIA = { remove_claim = c:FRA }
				s:STATE_LAOS = { remove_claim = c:FRA }
				c:FRA = {
					set_secret_goal = { country = ROOT secret_goal = none }
				}
			}
		}

		# Phần thưởng Ngoại giao lớn
		add_prestige = 25
		# Relations +20 với Anh/Phổ (cường quốc đối thủ Pháp)
		if = { limit = { exists = c:GBR } change_relations = { country = c:GBR value = 20 } }
		if = { limit = { exists = c:PRU } change_relations = { country = c:PRU value = 20 } }
		# Modifier 10 năm: Thắng lợi Ngoại giao
		add_modifier = {
			name = dvf_diplomatic_victory_vs_france
			months = 120
		}
	}

	# NHÁNH B: VẪN CẤM ĐẠO / CHƯA ĐỔI LUẬT
	option = {
		name = dvf_france.2.b
		# Option này chỉ hiện nếu CHƯA có law_freedom_of_conscience
		trigger = {
			NOT = { has_law = law_type:law_freedom_of_conscience }
		}

		# COMPATCH CHECK: Nếu JoI đang active (có biến indochina_campaign_target)
		# → Để JoI tự xử lý Diplomatic Play (không duplicate)
		if = {
			limit = { has_variable = indochina_campaign_target }
			# Chỉ thêm debuff chiến đấu, JoI sẽ trigger DP
			add_modifier = {
				name = dvf_french_invasion_debuff
				months = -1  # Vĩnh viễn cho đến khi hết chiến tranh
			}
		}
		# NẾU JOI KHÔNG ACTIVE (Không có biến) → DVF tự fallback tạo áp lực
		else = {
			# Thêm claim cho Pháp lên Annam & Mekong để trigger Diplomatic Play tự nhiên
			# (add_claim là effect của STATE scope; create_incident dùng tham số
			# country/value - đã verify qua joi_flavor_fra.txt của mod JoI)
			if = {
				limit = { exists = c:FRA }
				s:STATE_ANNAM = { add_claim = c:FRA }
				s:STATE_MEKONG = { add_claim = c:FRA }
				c:FRA = {
					# Tăng tension
					create_incident = { country = ROOT value = 20 }
				}
			}
			# Debuff bị xâm lược
			add_modifier = {
				name = dvf_french_invasion_debuff
				months = -1
			}
		}
	}
}

####################################
# Event 3: Kết Thúc Xâm Lược (Cleanup)
# Triggered khi chiến tranh với Pháp kết thúc (thắng/thua)
# Hoặc qua on_action on_war_end
# Dùng để xóa modifier debuff
####################################
dvf_france.3 = {
	type = country_event
	placement = root
	title = dvf_france.3.t
	desc = dvf_france.3.d
	icon = "gfx/interface/icons/event_icons/event_diplomatic.dds"
	duration = 1

	is_triggered_only = yes
	hide_window = yes

	trigger = {
		tag = DAI
		has_modifier = dvf_french_invasion_debuff
		# Không còn ở chiến tranh với Pháp
		NOT = { is_at_war_with = c:FRA }
	}

	immediate = {
		remove_modifier = dvf_french_invasion_debuff
		# Nếu thắng Pháp (Pháp không còn claim hoặc quan hệ tốt)
		if = {
			limit = {
				OR = {
					relations = { country = c:FRA value >= 0 }
					NOT = { exists = c:FRA }
				}
			}
			add_prestige = 50
			add_modifier = {
				name = dvf_france_defense_boost
				months = 120
			}
		}
	}
}
