namespace = dvf_flavor

####################################
# Event 1: Bác Vật Tân Biên (Dịch sách khoa học)
# Kích hoạt ngẫu nhiên từ JE Canh Tân khi:
# institution_schools >= 2 & literacy >= 0.25
####################################
dvf_flavor.1 = {
	type = country_event
	placement = root
	title = dvf_flavor.1.t
	desc = dvf_flavor.1.d
	flavor = dvf_flavor.1.f
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	duration = 3

	event_image = {
		video = "asia_westerners_arriving"
	}

	is_triggered_only = yes
	
	trigger = {
		has_global_variable = dvf_hong_bao_restoration_unlocked
		institution_investment_level = { institution = institution_schools value >= 2 }
		literacy_rate >= 0.25
		NOT = { has_global_variable = dvf_bac_vat_tan_bien_fired }
	}

	option = {
		name = dvf_flavor.1.a
		default_option = yes
		set_global_variable = dvf_bac_vat_tan_bien_fired
		
		add_technology_progress = { progress = 2500 technology = empiricism }
		add_prestige = 10
	}
}
