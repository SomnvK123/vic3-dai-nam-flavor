####################################
# Decision: Expand Defense Industry
# Spends funds and imperial authority
# to accelerate construction of arms
# factories and shipyards.
####################################
dvf_expand_military_industry = {
	is_shown = {
		exists = c:DAI
		c:DAI ?= THIS
		has_variable = dvf_opium_shock_happened
		NOT = { has_modifier = dvf_military_industry_drive }
	}

	possible = {
		c:DAI ?= THIS
		money >= 25000
	}

	when_taken = {
		add_treasury = -25000
		add_modifier = {
			name = dvf_military_industry_drive
			years = 5
		}
		ig:ig_armed_forces = {
			add_loyalists = { value = 0.05 }
		}
	}

	ai_chance = {
		value = 50
	}
}
