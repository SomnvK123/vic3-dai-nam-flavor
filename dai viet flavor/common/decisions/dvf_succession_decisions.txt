dvf_secure_succession = {
	is_shown = {
		c:DAI ?= this
		ruler ?= { has_template = DAI_thieu_tri }
		NOT = { has_variable = dvf_tu_duc_spawned }
	}

	possible = {
		c:DAI ?= this
		ruler ?= { has_template = DAI_thieu_tri }
		can_generate_scripted_heirs = yes
		NOT = { has_variable = dvf_tu_duc_spawned }
	}

	when_taken = {
		set_variable = dvf_tu_duc_spawned

		# Dọn heir ngẫu nhiên nếu vanilla đã tự sinh trước khi quyết nghị
		# này được thực hiện (thay cho on_heir_born cũ).
		if = {
			limit = { heir ?= { NOT = { has_template = DAI_tu_duc } } }
			heir = { kill_character = { hidden = yes } }
		}

		# Tạo Tự Đức làm heir lịch sử
		create_character = {
			template = DAI_tu_duc
			on_created = {
				replace_character_roles = { add = { heir } remove_all = yes }
			}
		}

		# Tạo Hồng Bảo ngay là agitator — bắt đầu kích động từ đây
		create_character = {
			template = DAI_hong_bao
			on_created = {
				replace_character_roles = { add = { agitator } remove_all = yes }
			}
		}

		# Thông báo kế vị
		trigger_event = { id = dvf_succession.1 }
	}

	ai_chance = {
		value = 100
	}
}
