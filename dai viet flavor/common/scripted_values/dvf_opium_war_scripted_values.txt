# Scripted values for Military Investment Journal Entry
building_arms_industry_count = {
	value = 0
	every_scope_building = {
		limit = {
			is_building_type = building_arms_industry
		}
		add = 1
	}
}

building_shipyards_count = {
	value = 0
	every_scope_building = {
		limit = {
			is_building_type = building_shipyards
		}
		add = 1
	}
}

# Total military investment score (5 arms + 2 shipyards = 7 max)
dvf_military_investment_score = {
	value = 0
	add = building_arms_industry_count
	add = building_shipyards_count
}
