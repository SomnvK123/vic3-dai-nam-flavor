# FIX: on_heir_born đã bị XÓA khỏi đây. error.log xác nhận on_action
# này trùng với 1 mod thứ 3 trong playset (cmf_heir_blocker.txt) — vì
# on_actions không merge cộng dồn giữa các file (chỉ file load sau
# cùng được giữ, effect của file kia bị loại bỏ hoàn toàn), việc dùng
# on_heir_born có thể vô hiệu hóa hẳn logic của mod kia (hoặc ngược
# lại, mình bị vô hiệu hóa tùy thứ tự load) — rủi ro 2 chiều không thể
# kiểm soát vì không có quyền sửa mod đó. Chuyển việc "dọn heir ngẫu
# nhiên" xuống on_yearly_pulse_country bên dưới (đã dùng sẵn cho việc
# spawn Tự Đức/Hồng Bảo) — không đụng tới on_heir_born nữa.

# Mỗi năm: Kiểm tra và spawn Tự Đức + Hồng Bảo
# dưới thời Thiệu Trị (cách an toàn nhất để đảm bảo
# spawn đúng thời điểm bất kể Thiệu Trị chết lúc nào)
on_yearly_pulse_country = {
	effect = {
		# Phase 1: Tạo Tự Đức (heir) và Hồng Bảo (agitator) khi Thiệu Trị trị vì
		if = {
			limit = {
				c:DAI ?= this
				can_generate_scripted_heirs = yes
				ruler ?= {
					has_template = DAI_thieu_tri_nguyen_phuoc
				}
				NOT = { has_variable = dvf_tu_duc_spawned }
			}
			# Dọn heir ngẫu nhiên nếu vanilla đã tự sinh trước khi mình kịp
			# tạo nhân vật lịch sử (thay cho on_heir_born cũ)
			if = {
				limit = { heir ?= { NOT = { has_template = DAI_tu_duc } } }
				heir = { kill_character = { hidden = yes } }
			}

			set_variable = dvf_tu_duc_spawned

			# Tạo Tự Đức làm heir lịch sử
			create_character = {
				template = DAI_tu_duc
				on_created = {
					replace_character_roles = { add = { heir } remove_all = yes }
				}
			}

			# Tạo Hồng Bảo ngay là agitator — bắt đầu kích động từ đây
			create_character = {
				template = DAI_hong_bao
				on_created = {
					replace_character_roles = { add = { agitator } remove_all = yes }
				}
			}

			# Thông báo kế vị
			trigger_event = { id = dvf_succession.1 }
		}
	}
}
