# DAI NAM - Kế Vị & Hồng Bảo Xuất Hiện
#
# Lịch sử: Thiệu Trị băng hà năm 1847.
# Trương Đăng Quế phế Hồng Bảo, đưa
# Hồng Nhậm (Tự Đức) lên ngôi.
# Hồng Bảo bắt đầu hoạt động kích động
# ngay từ năm 1847.
####################################

# Khi Thiệu Trị là ruler, tiêu diệt heir ngẫu nhiên
# để đảm bảo chỉ có Tự Đức là heir duy nhất.
on_heir_born = {
	effect = {
		if = {
			limit = {
				owner ?= {
					c:DAI ?= this
					ruler ?= {
						has_template = DAI_thieu_tri_nguyen_phuoc
					}
					NOT = { has_variable = dvf_tu_duc_spawned }
				}
			}
			kill_character = { hidden = yes }
		}
	}
}

# Mỗi năm: Kiểm tra và spawn Tự Đức + Hồng Bảo
# dưới thời Thiệu Trị (cách an toàn nhất để đảm bảo
# spawn đúng thời điểm bất kể Thiệu Trị chết lúc nào)
on_yearly_pulse_country = {
	effect = {
		# Phase 1: Tạo Tự Đức (heir) và Hồng Bảo (agitator) khi Thiệu Trị trị vì
		if = {
			limit = {
				c:DAI ?= this
				can_generate_scripted_heirs = yes
				ruler ?= {
					has_template = DAI_thieu_tri_nguyen_phuoc
				}
				NOT = { has_variable = dvf_tu_duc_spawned }
			}
			set_variable = dvf_tu_duc_spawned

			# Tạo Tự Đức làm heir lịch sử
			create_character = {
				template = DAI_tu_duc
				on_created = {
					replace_character_roles = { add = { heir } remove_all = yes }
				}
			}

			# Tạo Hồng Bảo ngay là agitator — bắt đầu kích động từ đây
			create_character = {
				template = DAI_hong_bao
				on_created = {
					replace_character_roles = { add = { agitator } remove_all = yes }
				}
			}

			# Thông báo kế vị
			trigger_event = { id = dvf_succession.1 }
		}
	}
}
