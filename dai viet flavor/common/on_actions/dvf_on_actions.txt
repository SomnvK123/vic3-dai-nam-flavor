	# WAR END CLEANUP (Mục 15)

	# event đó tự xử lý remove_modifier + phần thưởng (tránh trùng logic).

	# FIX: common/on_actions KHÔNG hỗ trợ field "trigger" đứng ngang hàng

	# Cú pháp cũ gây lỗi parse "Unexpected token: on_war_end" /
	# "on_character_death" trong error.log, khiến CẢ FILE này không load,
	# và có dấu hiệu domino sang cả modifier/template khác của mod báo
	# "Invalid database object". Chuyển điều kiện vào if/limit trong effect.

	on_war_end = {
		effect = {
			if = {
				limit = {
					tag = DAI
					has_global_variable = dvf_hong_bao_coup_success
					has_modifier = dvf_french_invasion_debuff
					NOT = { is_at_war_with = c:FRA }
				}
				trigger_event = { id = dvf_france.3 days = 5 }
			}
		}
	}

	# ============================================================
	# CHARACTER DEATH HOOKS
	# Tang lễ Nguyễn Trường Tộ khi ông mất (lịch sử: 1871)

	on_character_death = {
		effect = {
			if = {
				limit = {
					character = DAI_nguyen_truong_to
					ROOT = { has_global_variable = dvf_hong_bao_restoration_unlocked }
					ROOT = { NOT = { has_variable = dvf_nguyen_truong_to_death_done } }
				}
				ROOT = { set_variable = dvf_nguyen_truong_to_death_done }
				ROOT = { trigger_event = { id = dvf_hong_bao_restoration_chars.5 days = 10 } }
			}
		}
	}
