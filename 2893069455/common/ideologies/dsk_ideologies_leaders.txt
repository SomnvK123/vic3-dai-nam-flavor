ideology_nordic_pan_nationalist = { #泛北欧
	icon = "gfx/interface/icons/ideology_icons/ideology_leader/unused/ideology_leader_nationalism.dds"
	
	character_ideology = yes
	
	lawgroup_citizenship = {

	country_trigger = {
		any_primary_culture = { 
			has_discrimination_trait = language_scandinavian 
		}
	}
	
	interest_group_leader_trigger = {
        OR = {
			is_interest_group_type = ig_intelligentsia
			is_interest_group_type = ig_trade_unions
			is_interest_group_type = ig_petty_bourgeoisie
			is_interest_group_type = ig_armed_forces
			is_interest_group_type = ig_rural_folk
		}
    }
	
	non_interest_group_leader_trigger = {
        interest_group = {
            OR = {
		    is_interest_group_type = ig_devout
			is_interest_group_type = ig_intelligentsia
			is_interest_group_type = ig_trade_unions
			is_interest_group_type = ig_petty_bourgeoisie
			is_interest_group_type = ig_armed_forces
			is_interest_group_type = ig_rural_folk
		    }
        }
    }
	
	interest_group_leader_weight = {
		value = 100
		if = { 
			limit = {
				owner ?= {
					NOR = {
						has_law_or_variant = law_type:law_cultural_exclusion
						has_law_or_variant = law_type:law_freedom_of_conscience
						has_law_or_variant = law_type:law_migration_controls
					}
				}
			}
			add = -50
		}

			limit = {
				owner ?= { NOT = { has_technology_researched = nationalism } }
			}
			add = -50
		}

		if = {
			limit = {
				scope:interest_group = {
					is_interest_group_type = ig_landowners
				}
			}
			add = -25
		}
		# more likely for armed forces and intelligentsia
		if = {
			limit = {
				scope:interest_group = {
					OR = {
						is_interest_group_type = ig_intelligentsia

					}
				}
			}
			add = 25
		}
		# More likely when SoL is high

					average_sol_for_primary_cultures >= 15
