	}
	lawgroup_free_speech = {
		law_protected_speech = strongly_disapprove
		law_right_of_assembly = disapprove
		law_censorship = approve
		law_outlawed_dissent = strongly_approve
		law_IJ_newspaper = approve		#报纸出版条例
		law_IJ_defamation = strongly_approve		#舆论管制
	}
	lawgroup_bureaucracy = {
		law_elected_bureaucrats = strongly_disapprove
		law_hereditary_bureaucrats = approve
		law_appointed_bureaucrats = strongly_approve
		law_IJ_cabinet = strongly_approve
	}
	interest_group_leader_trigger = {
		ideology_dsk_kmt_valid_trigger = yes
	}
	non_interest_group_leader_trigger = {
		NOT = {
			has_role_of_type = agitator
		}
		interest_group = {
			ideology_dsk_kmt_valid_trigger = yes
		}
	}
	country_trigger = {
		AND = {
			c:AUS ?= this
			year < 1870
			has_law_or_variant = law_type:law_monarchy
		}
	}
	interest_group_leader_weight = {
		value = 100
		if = {
			limit = {
				owner ?= {
					any_scope_character = {
						is_ruler_of_own_country = yes
						OR = {
							has_trait = ambitious
						}
					}
				}
			}
			add = 20
		}
		if = {
			limit = {
				owner ?= {
					has_law_or_variant = law_type:law_monarchy
				}
			}
			multiply = 2
		}
		if = {
			limit = {
				owner ?= {
					owns_entire_state_region = STATE_CENTRAL_HUNGARY
				}
			}
			add = 75
		}
	}
	non_interest_group_leader_weight = {
		value = 100
		if = {
			limit = {
				interest_group ?= {
					OR = {
						has_variable = communist_ig_var
						has_variable = chose_vanguardism
						has_variable = chose_anarchism
						has_variable = chose_communism
						has_variable = chose_conservatism
					}
				}
			}
			multiply = 0.1
		}
	}
}
