		}
		scope:target_country = {

						value = -70

					remove_diplomatic_pact = {
						country = root
						type = representacion_en_cortes
					}
				}

					change_relations = {
						country = ROOT
						value = -50
					}
				}
			}

		requirement_to_maintain = {
