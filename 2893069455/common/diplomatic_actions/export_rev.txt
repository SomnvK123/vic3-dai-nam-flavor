rev_target_country = {
	groups = {
		general
	}
	requires_approval = no
	selectable = {
		NOT = {
			is_country_type = decentralized
		}
		is_revolutionary = no
	}
	potential = {
		has_law_or_variant = law_type:law_council_republic
		not = {
			any_country = {
				has_modifier = modifier_rev_target_country
			}
		}
	}
	possible = {
		has_technology_researched = socialism
		has_technology_researched = political_agitation
		lca_is_in_diplomatic_play_or_war = no		#	和平才能点决议煽动
		scope:target_country = {
			NOT = {
				is_country_type = decentralized
			}
			is_revolutionary = no
			NOT = {
				has_law_or_variant = law_type:law_council_republic
			}			#禁止互开左籍
		}
	}
	accept_effect = {
		set_variable = start_to_export_rev
		scope:target_country = {
			add_modifier = {
				name = modifier_rev_target_country
				months = 24				#两年内大概
				is_decaying = yes
			}
			create_political_movement = {
				type = movement_communist
			}
			create_political_movement = {
				type = movement_socialist
			}
			save_scope_as = rev_target_country
		}
		add_modifier = {
			name = modifier_rev_investment
			multiplier = money_amount_multiplier_medium
			days = 7
		}
	}
	ai = {
		propose_score = {
			value = 0
		}
		will_propose = {
			always = no
		}
	}
}

help_build_country = {
	groups = {
		general
	}
	requires_approval = no
	potential = {
		country_rank >= rank_value:great_power
		gold_reserves >= 3000000
		not = {
			any_country = {
				has_modifier = modifier_help_build_country
			}
		}
		scope:target_country = {
			has_law_or_variant = law_type:law_council_republic
		}
	}
	accept_effect = {
		scope:target_country = {
			change_relations = {
				country = root
				value = 200
			}
			add_modifier = {
				name = modifier_help_build_country
				years = 2
			}
			add_modifier = {
				name = modifier_construction_time
				years = 10
			}
			add_treasury = money_amount_multiplier_very_large
			add_treasury = money_amount_multiplier_very_large
			add_treasury = money_amount_multiplier_very_large
			add_treasury = money_amount_multiplier_very_large
			add_treasury = money_amount_multiplier_very_large
			add_treasury = money_amount_multiplier_very_large
			add_treasury = money_amount_multiplier_very_large
			add_treasury = money_amount_multiplier_very_large
			add_treasury = money_amount_multiplier_very_large
			add_treasury = money_amount_multiplier_very_large
			add_treasury = money_amount_multiplier_very_large
		}
		save_scope_as = help_build_country
		add_modifier = {
			name = modifier_rev_investment
			multiplier = money_amount_multiplier_very_small
			days = 7
		}

	}

			always = no
		}
	}
}

stop_help_build_country = {
	groups = {
		general
	}
	requires_approval = no
	potential = {
		scope:target_country = {
			has_modifier = modifier_help_build_country
		}
	}
	accept_effect = {
		scope:target_country = {
			remove_modifier = modifier_help_build_country
		}
		remove_global_variable = helping_build
	}
	ai = {
		propose_score = {
			value = 0
		}
		will_propose = {
			always = no
		}
	}
}

		general
	}

			is_revolutionary = yes
			has_modifier = modifier_wartime_law
			NOT = {
				has_law_or_variant = law_type:law_council_republic
			}
		}
	}
	accept_effect = {
		scope:target_country = {
			activate_law = law_type:law_council_republic
		}
	}
	ai = {
		propose_score = {
			value = 0
		}
		will_propose = {
			always = no
		}
	}
}
