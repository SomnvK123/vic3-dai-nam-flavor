calculo_pirateria_mora_paises = {
	set_global_variable = {
		name = lucha_pirateria_global
		value = 0
	}
	set_global_variable = {
		name = lucha_pirateria_global_policia
		value = 0
	}
	every_country = {
		limit = {
			has_journal_entry = je_pirateria_mar_jolo_duennos_phi
		}
		# Variables para indicar
		set_variable = {
			name = lucha_pirateria_local
			value = {
				value = this.naval_power_projection
				if = {
					limit = {
						capital.region = sr:region_indonesia
					}
					divide = 100
				}
				else_if = {
					limit = {
						any_scope_state = {
							region = sr:region_indonesia
						}
					}
					divide = 500
				}
				else = {
					divide = 1000
				}
				if = {
					limit = {
						has_modifier = luchando_contra_la_pirateria_mod
					}
					multiply = 2
				}
			}
		}
		set_variable = {
			name = lucha_pirateria_local_policia
			value = {
				if = {
					limit = {
						has_law_or_variant = law_type:law_no_police
					}
					value = 0
				}
				else = {
					if = {
						limit = {
							has_law_or_variant = law_type:law_local_police
						}
						value = 0.01
					}
					else_if = {
						limit = {
							has_law_or_variant = law_type:law_dedicated_police
						}
						value = 0.02
					}
					else = {
						# Policia Militarizada
						value = 0.03
					}
					multiply = institution:institution_police.investment
					if = {
						limit = {
							OR = {
								AND = {
									exists = c:MGD
									this ?= c:MGD
								}
								AND = {
									exists = c:SUL
									this ?= c:SUL
								}
							}
						}
						multiply = 3
					}
				}
			}
		}
		change_global_variable = {
			name = lucha_pirateria_global
			add = {
				value = this.naval_power_projection
				if = {
					limit = {
						capital.region = sr:region_indonesia
					}
					divide = 100
				}
				else_if = {
					limit = {
						any_scope_state = {
							region = sr:region_indonesia
						}
					}
					divide = 500
				}
				else = {
					divide = 1000
				}
				if = {
					limit = {
						has_modifier = luchando_contra_la_pirateria_mod
					}
					multiply = 2
				}
			}
		}
		change_global_variable = {
			name = lucha_pirateria_global_policia
			add = {
				if = {
					limit = {
						has_law_or_variant = law_type:law_no_police
					}
					value = 0
				}
				else = {
					if = {
						limit = {
							has_law_or_variant = law_type:law_local_police
						}
						value = 0.01
					}
					else_if = {
						limit = {
							has_law_or_variant = law_type:law_dedicated_police
						}
						value = 0.02
					}
					else = {
						# Policia Militarizada
						value = 0.03
					}
					multiply = institution:institution_police.investment
					if = {
						limit = {
							OR = {
								AND = {
									exists = c:MGD
									this ?= c:MGD
								}
								AND = {
									exists = c:SUL
									this ?= c:SUL
								}
							}
						}
						multiply = 3
					}
				}
			}
		}
	}
}

limipar_variables_locales = {
	remove_variable = lucha_pirateria_local
	remove_variable = lucha_pirateria_local_policia
}

limpiar_variables_globales = {
	if = {
		limit = {
			has_global_variable = lucha_pirateria_global
		}
		remove_global_variable = lucha_pirateria_global
	}
	if = {
		limit = {
			has_global_variable = lucha_pirateria_global_policia
		}
		remove_global_variable = lucha_pirateria_global_policia
	}
}
