fra_apoyar_reclamos_espannoles_guinea = {
	custom_tooltip = fra_apoyar_reclamos_espannoles_guinea_tt
	root = {
		save_scope_as = scope_yolo
	}
	set_variable = fra_apoyo_reclamo_guinea_espanna
	if = {
		limit = {
			has_variable = fra_rechazo_apoyo_guinea_espanna
		}
		remove_variable = fra_rechazo_apoyo_guinea_espanna
		change_global_variable = {
			name = fra_votos_negativos_espana
			subtract = 1
		}
	}
	change_global_variable = {
		name = fra_votos_positivos_espana
		add = 1
	}
	set_variable = {
		name = fra_ha_tomado_decision
		months = 12
	}
	hidden_effect = {
		c:FRA ?= {
			post_notification = fra_voto_positivo
		}
	}
}

je_support_china_effect = {
	custom_tooltip = je_support_china_effect_tt  # 本地化键：支持中国提案的效果提示
	ROOT = {
			save_scope_as = scope_yolo
		}
	# 设置投票状态（支持）
	set_variable = je_vote_support
	
	# 如果之前投过反对票，则修正投票记录
	if = {
		limit = { has_variable = je_vote_oppose }
		remove_variable = je_vote_oppose
		change_global_variable = {
			name = je_anti_treaty_votes
			subtract = 1
		}
	}
	
	# 增加支持票
	change_global_variable = {
		name = je_pro_treaty_votes
		add = 1
	}
	
	# 设置决策冷却变量（防止重复投票）
	set_variable = {
		name = je_vote_cooldown
		months = 12
	}
	
	# 通知中国
	hidden_effect = {
		every_country = {
			limit = {
				OR = {
					country_rank >= rank_value:great_power
					c:CHI ?= this

				}
				
			}
			post_notification = je_country_supported_us 
		}
	}
}

je_clean_conference_vars = {
	# 清理全局变量
	remove_global_variable = je_pro_treaty_votes
	remove_global_variable = je_anti_treaty_votes
	remove_global_variable = je_conference_months_remaining
	remove_global_variable = je_participating_countries
	
	# 清理所有国家的投票相关变量
	every_country = {
		limit = { OR = { has_variable = je_vote_support has_variable = je_vote_oppose } }
		remove_variable = je_vote_support
		remove_variable = je_vote_oppose
		remove_variable = je_vote_cooldown
	}
}

government_leader_james_polk_effect = {
	if = { #俄勒冈边界
		limit = {
                        NOT = {
			any_country = {
                                NOT = { THIS = ROOT }
				NOT = { is_subject_of = ROOT }
                                NOT = { is_country_type = decentralized }
                                any_scope_state = {
                                   OR = {
		                   state_region = s:STATE_WASHINGTON
		                   state_region = s:STATE_OREGON
		                   state_region = s:STATE_IDAHO
                                   }
		                   }
		}
		}
		}
		change_variable = { name = government_leader_james_polk_counter add = 1 }
	}
	if = { #取得加利福尼亚
		limit = {
                        NOT = {
			any_country = {
                                NOT = { THIS = ROOT }
				NOT = { is_subject_of = ROOT }
                                NOT = { is_country_type = decentralized }
                                any_scope_state = {
                                   OR = {
		                   state_region = s:STATE_CALIFORNIA
		                   state_region = s:STATE_NEVADA
		                   state_region = s:STATE_ARIZONA
		                   state_region = s:STATE_NEW_MEXICO
		                   state_region = s:STATE_COLORADO
		                   state_region = s:STATE_UTAH
                                   }
		                   }
		}
		}
		}
		change_variable = { name = government_leader_james_polk_counter add = 1 }
	}
	if = { #降低关税
		limit = {
			has_law_or_variant = law_type:law_free_trade
			market_capital.market = {
				market_trade = {
				  value >= 8000
				}
			}
		}
		change_variable = { name = government_leader_james_polk_counter add = 1 }
	}
	if = { #合并德克萨斯
		limit = {
                                owns_entire_state_region = STATE_TEXAS
                                any_scope_state = {
		                   state_region = s:STATE_TEXAS
				   is_incorporated = yes
		                }
		                }
		change_variable = { name = government_leader_james_polk_counter add = 1 }
	}
	if = { #独立国库法
		limit = {
			gold_reserves > 0
                        NOT = {
                                any_scope_state = {
				owner = ROOT
				is_incorporated = yes
			        NOT = { has_building = building_government_administration }
		                }
		                }
		                }
		change_variable = { name = government_leader_james_polk_counter add = 1 }
	}
}
