lca_add_claim_id = {
	if = {
		limit = {
			NOR = {
				owns_entire_state_region = $STATE_ID$
				has_claim = s:$STATE_ID$
			}
		}
		s:$STATE_ID$ = {
			add_claim = ROOT
		}
	}
}

lca_leave_customs_union_from = {
	if = {
		limit = {
			is_junior_in_customs_union = yes
			$TARGET$ = {
				has_diplomatic_pact = {
					who = prev
					type = personal_union
				}
			}
		}
		$TARGET$ = {
			remove_diplomatic_pact = {
				country = prev
				type = personal_union
			}
		}
	}
}

lca_weak_gain_independence = {
	# 关税同盟、朝贡国和受保护国可以独立
	if = {
		limit = {
			OR = {
				is_junior_in_customs_union = yes
				is_subject_type = subject_type_protectorate
				is_subject_type = subject_type_tributary
			}
		}
		# independence
		every_country = {
			limit = {
				has_diplomatic_pact = {
					who = prev
					type = personal_union
				}
			}
			remove_diplomatic_pact = {
				country = prev
				type = personal_union
			}
		}
		make_independent = yes
	}
}

lca_remove_alliance_or_defensive_pact_with = {
	# 断绝盟约和防御协定
	if = {
		limit = {
			any_scope_treaty = {
				binds = $WHO$
				any_scope_article = {
					has_type = alliance
				}
			}
		}
		every_scope_treaty = {
			limit = {
				binds = $WHO$
				any_scope_article = {
					has_type = alliance
				}
			}
			withdraw = {
				country = root
			}
		}
	}
	if = {
		limit = {
			any_scope_treaty = {
				binds = $WHO$
				any_scope_article = {
					has_type = alliance
				}
			}
		}
		every_scope_treaty = {
			limit = {
				binds = $WHO$
				any_scope_article = {
					has_type = defensive_pact
				}
			}
			withdraw = {
				country = root
			}
		}
	}
}

lca_call_allies_for_current_war = {
	every_scope_ally = {
		limit = {
			lca_is_in_diplomatic_play_or_war = no
			any_scope_treaty = {
				binds = $ACTOR$
				any_scope_article = {
					has_type = alliance
				}
			}
			NOT = {
				has_truce_with = $TARGET$
			}
			NOT = {
				lca_has_alliance_or_defensive_pact_with = {
					WHO = $TARGET$
				}
			}
			lca_has_hostile_attitude = {
				WHO = $TARGET$
			}
			$TARGET$ = {
				relations:prev < relations_threshold:cordial
			}
		}
		violate_sovereignty_join = {
			violator = $ACTOR$
			target = this
			join_violator = yes
		}
	}
	$TARGET$ = {
		if = {
			limit = {
				is_non_customs_union_subject = yes
				exists = overlord
			}
			overlord = {
				violate_sovereignty_join = {
					violator = $ACTOR$
					target = this
					join_violator = no
				}
			}
		}
		every_scope_ally = {
			limit = {
				lca_is_in_diplomatic_play_or_war = no
			}
			violate_sovereignty_join = {
				violator = $ACTOR$
				target = this
				join_violator = no
			}
		}
	}
}

update_chi_progressbar_tooltips = {
	je:je_difficult_china = {
		#三月时期进度计算
		if = {
			limit = {
				"scripted_bar_progress(je_difficult_china_progress_bar)" < 0
			}
			set_bar_progress = {
				value = 0
				name = je_difficult_china_progress_bar
			}
		}
		if = {
			limit = {
				"scripted_bar_progress(je_difficult_china_progress_bar)" > 100
			}
			set_bar_progress = {
				value = 100
				name = je_difficult_china_progress_bar
			}
		}
		set_variable = {
			name = er_conservative_progress
			value = "scripted_bar_progress(je_difficult_china_progress_bar)"
		}
		set_variable = {
			name = er_liberal_progress
			value = 100
		}
		change_variable = {
			name = er_liberal_progress
			subtract = var:er_conservative_progress
		}
	}
}
