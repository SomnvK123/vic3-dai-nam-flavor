# Limpiar variables
limpiar_variables_regencia_fail = {
	remove_variable = contador
	if = {
		limit = {
			has_modifier = regencia_espartero_esp
		}
		remove_modifier = regencia_espartero_esp
	}
}

			has_modifier = regencia_espartero_esp

}

# Efectos en si
invalid_regencia = {
	limpiar_variables_regencia_invalid = yes
	if = {
		limit = {
			NOT = {
				heir = {
					has_template = spa_isabel_de_borbon_template
				}
			}
			NOT = {
				ruler ?= {
					has_template = spa_isabel_de_borbon_template
				}
			}
			has_law_or_variant = law_type:law_monarchy
		}
		create_character = {
			last_name = "de_Borbon"
			ruler = yes
		}
	}
}

fail_regencia_elegir_nuevo_regente = {
	limpiar_variables_regencia_fail = yes
	trigger_event = {
		id = esp.6
		popup = yes
	}
	# Poner un periodo de gracia de un año
	if = {
		limit = {
			has_variable = periodo_de_gracia_regencia
		}
		remove_variable = periodo_de_gracia_regencia
	}
	set_variable = {
		name = periodo_de_gracia_regencia
		months = 12
	}
	# Si esta Espartero, añadimos el modificador
	if = {
		limit = {
			any_scope_character = {
				is_ruler_of_own_country = no
				popularity >= 50
				OR = {
					interest_group.leader = this
					has_role_of_type = general
					has_role_of_type = admiral
				}
				interest_group ?= {
					ig_counts_as_marginal = no
				}
				has_variable = is_esp
			}
		}
		add_modifier = {
			name = regencia_espartero_esp
			days = normal_modifier_time
		}
	}
	else_if = {
		limit = {
			any_scope_character = {
				is_ruler_of_own_country = no
				has_variable = is_esp
			}
		}
		custom_tooltip = je_regencia_esp_indicacion9
	}
	custom_tooltip = je_regencia_esp_indicacion10
}

elegir_nuevo_regente = {
	if = {
		limit = {
			ruler ?= {
				has_template = spa_isabel_de_borbon_template
			}
		}
		set_heir = ruler
		heir = {
			if = {
				limit = {
					age < define:NCharacters|ADULT_AGE
					NOT = {
						has_trait = trait_child
					}
				}
				add_trait = trait_child
			}
		}
	}
	# Revisar si tenemos a Espartero 
	if = {
		limit = {
			any_scope_character = {
				trigger_personaje_valido_regente = yes
				has_variable = is_esp
			}
		}
		random_scope_character = {
			limit = {
				trigger_personaje_valido_regente = yes
				has_variable = is_esp
			}
			save_scope_as = nuevo_regente_scope
		}
	}
	else_if = {
		# Si no, guardamos a quien sea
		limit = {
			any_scope_character = {
				trigger_personaje_valido_regente = yes
			}
		}
		ordered_scope_character = {
			limit = {
				trigger_personaje_valido_regente = yes
			}
			order_by = popularity
			position = 0
			save_scope_as = nuevo_regente_scope
		}
	}
	else_if = {
		# Intentamos pillar a quien sea
		limit = {
			any_scope_character = {
				trigger_personaje_valido_regente_sin_popularidad = yes
			}
		}
		random_scope_character = {
			limit = {
				trigger_personaje_valido_regente_sin_popularidad = yes
			}
			save_scope_as = nuevo_regente_scope
		}
	}
	else = {
		# Si no hay nadie, creamos un general nuevo
		create_character = {
			is_general = yes
			on_created = {
				save_scope_as = nuevo_regente_scope
			}
			save_scope_as = nuevo_regente_scope
		}
	}
	# Poner el nuevo personaje como regente
	scope:nuevo_regente_scope = {
		set_character_as_ruler = yes
		set_variable = is_married
	}
	add_journal_entry = {
		type = je_regencia_espartero_esp
	}
}
