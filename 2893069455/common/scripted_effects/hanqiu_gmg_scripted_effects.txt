gmg_update_green_magical_girl = {
	# 每次刷新前，先清掉旧变量，避免角色死亡/消失后 GUI 继续读旧对象
	if = {
		limit = {
			has_variable = gmg_green_magical_girl_var
		}
		remove_variable = gmg_green_magical_girl_var
	}

	# 在当前国家的角色池里寻找模板为 AUS_mofashaonv 的角色
	random_scope_character = {
		limit = {
			has_variable = dsk_go_travel
		}

		save_temporary_scope_as = gmg_green_magical_girl_scope
	}

	# 如果找到了，就把这个角色保存到国家变量里
	# GUI 用 JournalEntry.GetCountry.MakeScope.Var('gmg_green_magical_girl_var').GetCharacter 读取
	if = {
		limit = {
			exists = scope:gmg_green_magical_girl_scope
		}

		set_variable = {
			name = gmg_green_magical_girl_var
			value = scope:gmg_green_magical_girl_scope
		}
	}
}
