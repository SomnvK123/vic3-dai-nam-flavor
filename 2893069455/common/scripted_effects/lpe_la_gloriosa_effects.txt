add_deslegitimacion_coronacion_pronta_modifer_edad = {
	if = {
		limit = {
			root.heir.age <= 5
		}
		add_modifier = {
			name = deslegitimacion_coronacion_pronta
			multiplier = var:agitacion_regencia
			years = 12
		}
	}
	else_if = {
		limit = {
			root.heir.age <= 6
		}
		add_modifier = {
			name = deslegitimacion_coronacion_pronta
			multiplier = var:agitacion_regencia
			years = 11

	}
	else_if = {
		limit = {
			root.heir.age <= 7
		}
		add_modifier = {
			name = deslegitimacion_coronacion_pronta
			multiplier = var:agitacion_regencia

	else_if = {
		limit = {
			root.heir.age <= 8
		}
		add_modifier = {
			name = deslegitimacion_coronacion_pronta
			multiplier = var:agitacion_regencia
			years = 9
		}
	}
	else_if = {
		limit = {
			root.heir.age <= 9
		}
		add_modifier = {
			name = deslegitimacion_coronacion_pronta
			multiplier = var:agitacion_regencia
			years = 8
		}
	}
	else_if = {
		limit = {
			root.heir.age <= 10
		}
		add_modifier = {
			name = deslegitimacion_coronacion_pronta
			multiplier = var:agitacion_regencia
			years = 7
		}
	}
	else_if = {
		limit = {
			root.heir.age <= 11
		}
		add_modifier = {
			name = deslegitimacion_coronacion_pronta
			multiplier = var:agitacion_regencia
			years = 6
		}
	}
	else_if = {
		limit = {
			root.heir.age <= 12
		}
		add_modifier = {
			name = deslegitimacion_coronacion_pronta
			multiplier = var:agitacion_regencia
			years = 5
		}
	}
	else_if = {
		limit = {
			root.heir.age <= 13
		}
		add_modifier = {
			name = deslegitimacion_coronacion_pronta
			multiplier = var:agitacion_regencia
			years = 4
		}
	}
	else_if = {
		limit = {
			root.heir.age <= 14
		}
		add_modifier = {
			name = deslegitimacion_coronacion_pronta
			multiplier = var:agitacion_regencia
			years = 3
		}
	}
	else_if = {
		limit = {
			root.heir.age <= 15
		}
		add_modifier = {
			name = deslegitimacion_coronacion_pronta
			multiplier = var:agitacion_regencia
			years = 2
		}
	}
	else = {
		add_modifier = {
			name = deslegitimacion_coronacion_pronta
			multiplier = var:agitacion_regencia
			years = 1
		}
	}
}

calcular_poder_carlista = {
	# Contador general
	# set_variable = { name = carlistas_progess_next value = 0 }
	# Legitimidad del gobierno
	set_variable = {
		name = carlistas_legitimidad
		value = root.government_legitimacy
	}
	change_variable = {
		name = carlistas_legitimidad
		subtract = 50
	}
	change_variable = {
		name = carlistas_legitimidad
		divide = -10
	}
	# Poder de los terratenientes
	set_variable = {
		name = carlistas_de_poder_terratenientes
		value = 0
	}
	set_variable = {
		name = carlistas_terratenientes_poder
		value = ig:ig_landowners.ig_clout
	}
	set_variable = {
		name = carlistas_terratenientes_felicidad
		value = ig:ig_landowners.ig_approval
	}
	change_variable = {
		name = carlistas_terratenientes_felicidad
		divide = -1
	}
	set_variable = {
		name = carlistas_terratenientes_felicidad_true
		value = var:carlistas_terratenientes_felicidad
	}
	change_variable = {
		name = carlistas_terratenientes_felicidad_true
		add = 5
	}
	change_variable = {
		name = carlistas_de_poder_terratenientes
		add = var:carlistas_terratenientes_poder
	}
	change_variable = {
		name = carlistas_de_poder_terratenientes
		multiply = var:carlistas_terratenientes_felicidad_true
	}
	# Poder de la iglesia
	set_variable = {
		name = carlistas_de_poder_iglesia
		value = 0
	}
	set_variable = {
		name = carlistas_iglesia_poder
		value = ig:ig_devout.ig_clout
	}
	set_variable = {
		name = carlistas_iglesia_felicidad
		value = ig:ig_devout.ig_approval
	}
	change_variable = {
		name = carlistas_iglesia_felicidad
		divide = -1
	}
	set_variable = {
		name = carlistas_iglesia_felicidad_true
		value = var:carlistas_iglesia_felicidad
	}
	change_variable = {
		name = carlistas_iglesia_felicidad_true
		add = 5
	}
	change_variable = {
		name = carlistas_de_poder_iglesia
		add = var:carlistas_iglesia_poder
	}
	change_variable = {
		name = carlistas_de_poder_iglesia
		multiply = var:carlistas_iglesia_felicidad_true
	}
	# Poder del ámbito rural
	set_variable = {
		name = carlistas_de_poder_granjeros
		value = 0
	}
	set_variable = {
		name = carlistas_granjeros_poder
		value = ig:ig_rural_folk.ig_clout
	}
	set_variable = {
		name = carlistas_granjeros_felicidad
		value = ig:ig_rural_folk.ig_approval
	}

		name = carlistas_granjeros_felicidad
		divide = -1
	}
	set_variable = {
		name = carlistas_granjeros_felicidad_true
		value = var:carlistas_granjeros_felicidad
	}
	change_variable = {
		name = carlistas_granjeros_felicidad_true
		add = 5
	}
	change_variable = {
		name = carlistas_de_poder_granjeros
		add = var:carlistas_granjeros_poder
	}
	change_variable = {
		name = carlistas_de_poder_granjeros
		multiply = var:carlistas_granjeros_felicidad_true
	}
	# Lideres compatibles ideologia carlista
	set_variable = {
		name = carlistas_lideres_tradicionalistas2
		value = 0
	}
	every_scope_character = {
		limit = {
			is_ruler_of_own_country = no
			is_heir_of_own_country = no
			OR = {
				has_role_of_type = politician
				has_role_of_type = agitator
			}
			OR = {
				has_ideology = ideology:ideology_traditionalist
				has_ideology = ideology:ideology_luddite
				has_ideology = ideology:ideology_theocrat
			}
		}
		root = {
			change_variable = {
				name = carlistas_lideres_tradicionalistas2
				add = prev.popularity
			}
		}
	}
	change_variable = {
		name = carlistas_lideres_tradicionalistas2
		divide = 20
	}
	# Líderes autoritarios
	set_variable = {
		name = carlistas_lideres_autoritarios2
		value = 0
	}
	if = {
		limit = {
			NOR = {
				has_law_or_variant = law_type:law_oligarchy
				has_law_or_variant = law_type:law_autocracy
				has_law_or_variant = law_type:law_single_party_state
			}
		}
		every_scope_character = {
			limit = {
				is_ruler_of_own_country = no
				is_heir_of_own_country = no
				OR = {
					has_role_of_type = politician
					has_role_of_type = agitator
				}
				has_ideology = ideology:ideology_authoritarian
			}
			root = {
				change_variable = {
					name = carlistas_lideres_autoritarios2
					add = prev.popularity
				}
			}
		}
		change_variable = {
			name = carlistas_lideres_autoritarios2
			divide = 20
		}
	}
	# Ley de policia
	if = {
		limit = {
			has_law_or_variant = law_type:law_no_police
		}
		set_variable = {
			name = carlistas_ley_policia
			value = 2
		}
	}
	else_if = {
		limit = {
			has_law_or_variant = law_type:law_local_police
		}
		set_variable = {
			name = carlistas_ley_policia
			value = 0.5
		}
		change_variable = {
			name = carlistas_ley_policia
			divide = institution:institution_police.investment
		}
	}
	else_if = {
		limit = {
			has_law_or_variant = law_type:law_dedicated_police
		}
		set_variable = {
			name = carlistas_ley_policia
			value = -0.5
		}
		change_variable = {
			name = carlistas_ley_policia
			multiply = institution:institution_police.investment
		}
	}
	else = {
		# Policia Militarizada
		set_variable = {
			name = carlistas_ley_policia
			value = -1
		}
		change_variable = {
			name = carlistas_ley_policia
			multiply = institution:institution_police.investment
		}
	}
	# change_variable = { name = carlistas_progess_next add = var:carlistas_legitimidad }
	# change_variable = { name = carlistas_progess_next add = var:carlistas_de_poder_terratenientes }
	# change_variable = { name = carlistas_progess_next add = var:carlistas_lideres_tradicionalistas2 }
	# change_variable = { name = carlistas_progess_next add = var:carlistas_lideres_autoritarios2 }
	# change_variable = { name = carlistas_progess_next add = var:carlistas_de_poder_granjeros }
	# change_variable = { name = carlistas_progess_next add = var:carlistas_ley_policia }
}

	}
	# Líderes tradicionalistas
	set_variable = {
		name = carlistas_lideres_tradicionalistas
		value = 0
	}
	every_scope_character = {
		limit = {
			is_ruler_of_own_country = no
			is_heir_of_own_country = no
			OR = {
				has_role_of_type = politician
				has_role_of_type = agitator
			}
			OR = {
				has_ideology = ideology:ideology_traditionalist
				has_ideology = ideology:ideology_theocrat
				has_ideology = ideology:ideology_luddite
			}
		}
		root = {
			change_variable = {
				name = carlistas_lideres_tradicionalistas
				add = prev.popularity
			}
		}
	}
	change_variable = {
		name = carlistas_lideres_tradicionalistas
		divide = 10
	}
	# Líderes autoritarios

		name = carlistas_lideres_autoritarios

			NOR = {

				has_law_or_variant = law_type:law_single_party_state
			}
		}

			limit = {
				is_ruler_of_own_country = no
				is_heir_of_own_country = no
				OR = {
					has_role_of_type = politician
					has_role_of_type = agitator
				}
				has_ideology = ideology:ideology_authoritarian
			}
			root = {
				change_variable = {
					name = carlistas_lideres_autoritarios
					add = prev.popularity
				}
			}
		}
		change_variable = {
			name = carlistas_lideres_autoritarios
			divide = 10
		}
	}
	# Popularidad del Gobernante
	set_variable = {
		name = carlistas_popularidad_lider
		value = root.ruler.popularity
	}
	change_variable = {
		name = carlistas_popularidad_lider
