update_er_progressbar_tooltips = {
	je:je_European_Restoration = {
		#三月时期进度计算
		if = {
			limit = {
				"scripted_bar_progress(je_European_Restoration_progress_bar)" < 0
			}
			set_bar_progress = {
				value = 0
				name = je_European_Restoration_progress_bar
			}
		}
		if = {
			limit = {
				"scripted_bar_progress(je_European_Restoration_progress_bar)" > 100
			}
			set_bar_progress = {
				value = 100
				name = je_European_Restoration_progress_bar
			}
		}
		set_variable = {
			name = er_conservative_progress
			value = "scripted_bar_progress(je_European_Restoration_progress_bar)"
		}
		set_variable = {
			name = er_liberal_progress
			value = 100
		}
		change_variable = {
			name = er_liberal_progress
			subtract = var:er_conservative_progress
		}
	}
}

aus_monthly_pulse_intelligentsia_progress = {
	#科罗夫拉特的月进度
	set_variable = {
		name = aus_intelligentsia_progress_from_characters
		value = 0
	}
	set_variable = {
		name = aus_intelligentsia_progress_from_igs
		value = 0
	}
	set_variable = {
		name = aus_intelligentsia_progress_from_building
		value = 0
	}
	set_variable = {
		name = aus_intelligentsia_progress_from_legitimacy
		value = root.government_legitimacy
	}
	#绝对主义支持者的人气
	every_scope_character = {
		limit = {
			OR = {
				has_ideology = ideology:ideology_neo_absolutism_leader
				has_ideology = ideology:ideology_royalist
				has_ideology = ideology:ideology_ideology_protectionist
				has_ideology = ideology:ideology_authoritarian
			}
		}
		ROOT = {
			change_variable = {
				name = aus_intelligentsia_progress_from_characters
				add = prev.popularity
			}
			change_variable = {
				name = aus_intelligentsia_progress_from_characters
				divide = 10
			}
		}
	}
	#绝对主义和相关政治倾向的支持者
	every_interest_group = {
		limit = {
			leader = {
				OR = {
					has_ideology = ideology:ideology_neo_absolutism_leader
					has_ideology = ideology:ideology_royalist
					has_ideology = ideology:ideology_ideology_protectionist
					has_ideology = ideology:ideology_authoritarian
				}
			}
		}
		ROOT = {
			set_variable = {
				name = ig_weight
				value = prev.ig_clout
			}
		}
		if = {
			limit = {
				is_in_government = yes
			}
			ROOT = {
				change_variable = {
					name = ig_weight
					multiply = 30
				}
			}
		}
		ROOT = {
			change_variable = {
				name = aus_intelligentsia_progress_from_igs
				add = var:ig_weight
			}
		}
	}
	change_variable = {
		name = aus_intelligentsia_progress_from_igs
		divide = 10
	}
	#政府合法性
	change_variable = {
		name = aus_intelligentsia_progress_from_legitimacy
		divide = 20
	}
	change_variable = {
		name = aus_intelligentsia_progress_from_legitimacy
		subtract = 0.25
	}
	#政府建筑
	every_scope_building = {
		limit = {
			OR = {
				is_building_group = bg_bureaucracy
				is_building_group = bg_government
			}
			occupancy >= 0.5
		}
		ROOT = {
			change_variable = {
				name = aus_intelligentsia_progress_from_building
				add = prev.level
			}
		}
	}
	change_variable = {
		name = aus_intelligentsia_progress_from_building
		divide = 10
	}
}

spa_monthly_centralists_progress = {
	# 初始化变量
	set_variable = {
		name = spa_centralists_progress_from_characters
		value = 0
	}
	set_variable = {
		name = spa_centralists_progress_from_igs
		value = 0
	}
	set_variable = {
		name = spa_centralists_progress_from_infra
		value = 0
	}
	set_variable = {
		name = spa_centralists_progress_from_legitimacy
		value = root.government_legitimacy
	}
	set_variable = {
		name = spa_centralists_french_relations
		value = 0
	}
	# 左翼支持者影响力（每点人气+0.8）
	every_scope_character = {
		limit = {
			OR = {
				#has_ideology = ideology:ideology_moderado
				#has_ideology = ideology:ideology_progresista
				#has_ideology = ideology:ideology_humanitarian_royalist 
				#has_ideology = ideology:ideology_liberal_leader
				has_ideology = ideology:ideology_republican_leader
				has_ideology = ideology:ideology_radical
				has_ideology = ideology:ideology_social_democrat
				has_ideology = ideology:ideology_communist
				has_ideology = ideology:ideology_vanguardist
				has_ideology = ideology:ideology_anarchist
			}
		}
		ROOT = {
			change_variable = {
				name = spa_centralists_progress_from_characters
				add = prev.popularity
			}
			change_variable = {
				name = spa_centralists_progress_from_characters
				divide = 10
			}
		}
	}
	# 左翼集团支持（政府内影响力×25）
	every_interest_group = {
		limit = {
			OR = {
				is_interest_group_type = ig_trade_unions
				is_interest_group_type = ig_rural_folk
				is_interest_group_type = ig_intelligentsia
			}
			is_powerful = yes
			is_in_government = yes
			is_insurrectionary = no
		}
		ROOT = {
			set_variable = {
				name = ig_weight
				value = prev.ig_clout
			}
			if = {
				limit = {
					is_in_government = yes
				}
				change_variable = {
					name = ig_weight
					multiply = 25
				}
			}
			change_variable = {
				name = spa_centralists_progress_from_igs
				add = var:ig_weight
			}
			change_variable = {
				name = spa_centralists_progress_from_igs
				divide = 10
			}
		}
	}
	# 兵工厂和行政
	every_scope_building = {
		limit = {
			OR = {
				is_building_group = bg_military
				is_building_group = bg_government
			}
			occupancy >= 0.5
		}
		ROOT = {
			change_variable = {
				name = spa_centralists_progress_from_infra
				add = prev.level
			}
			change_variable = {
				name = spa_centralists_progress_from_infra
				divide = 10
			}
		}
	}
	# 合法性转换（每点合法性+1.2）
	change_variable = {
		name = spa_centralists_progress_from_legitimacy
		divide = 20
	}
	change_variable = {
		name = spa_centralists_progress_from_legitimacy
		subtract = 0.25
	}
	if = {
		has_variable = dsk_spa.32_var
		ROOT = {
			change_variable = {
				name = spa_centralists_french_relations
				add = 2
			}
			change_variable = {
				name = spa_regionalists_british_support
				add = -1
			}
		}
	}
}
