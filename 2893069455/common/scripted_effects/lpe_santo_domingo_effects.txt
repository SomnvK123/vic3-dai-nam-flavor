calcular_deseo_anexion = {
	if = {
		limit = {
			any_political_lobby = {
				is_political_lobby_type = lobby_pro_country
				target = c:SPA
				appeasement >= 5
				any_lobby_member = {
					is_in_government = yes
					root.ruler ?= {
						interest_group = prev
					}
				}
			}
		}
		set_global_variable = {
			name = apoyo_anexion_spain_santo_domingo_next
			value = 9
		}
	}
	else = {
		set_global_variable = {
			name = apoyo_anexion_spain_santo_domingo_next
			value = -9
		}
	}
	set_global_variable = {
		name = apoyo_anexion_spain_santo_domingo_next_dependencia
		value = "root.economic_dependence(c:SPA)"
	}
	set_global_variable = {
		name = apoyo_anexion_spain_santo_domingo_next_relaciones
		value = "c:SPA.relations:root"
	}
	change_global_variable = {
		name = apoyo_anexion_spain_santo_domingo_next_relaciones
		subtract = 50
	}
	change_global_variable = {
		name = apoyo_anexion_spain_santo_domingo_next_relaciones
		divide = 10
	}
}

limpiar_variables_anexion = {
	hidden_effect = {
		every_country = {
			limit = {
				has_journal_entry = je_incorporacion_republica_dominicana_esp
			}
			je:je_incorporacion_republica_dominicana_esp = {
				set_bar_progress = {
					value = 0
					name = je_faccion_anexionista_con_espanna_bar
				}
			}
		}
		every_country = {
			limit = {
				has_journal_entry = je_faccion_anexionista_con_espanna_dom
			}
			je:je_faccion_anexionista_con_espanna_dom = {
				set_bar_progress = {
					value = 0
					name = je_faccion_anexionista_con_espanna_bar
				}
			}
		}
	}
	remove_global_variable = apoyo_anexion_spain_santo_domingo_next_relaciones
	remove_global_variable = apoyo_anexion_spain_santo_domingo_next_dependencia
	remove_global_variable = apoyo_anexion_spain_santo_domingo_next
	remove_variable = peticion_de_anexion
}
