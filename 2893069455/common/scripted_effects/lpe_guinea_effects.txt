apoyar_reclamos_espannoles_guinea = {
	custom_tooltip = apoyar_reclamos_espannoles_guinea_tt
	root = {
		save_scope_as = scope_yo
	}
	set_variable = apoyo_reclamo_guinea_espanna
	if = {
		limit = {
			has_variable = rechazo_apoyo_guinea_espanna
		}
		remove_variable = rechazo_apoyo_guinea_espanna
		change_global_variable = {
			name = votos_negativos_espana
			subtract = 1
		}
	}
	change_global_variable = {
		name = votos_positivos_espana
		add = 1
	}
	set_variable = {
		name = ha_tomado_decision
		months = 12
	}
	hidden_effect = {
		c:SPA ?= {
			post_notification = voto_positivo
		}
	}
}

limpiar_variables_guinea_esp = {
	remove_variable = ultimatum_1
	remove_variable = ultimatum_2
	remove_variable = ultimatum_3
	s:STATE_GABON = {
		remove_claim = root
	}
	remove_global_variable = votos_positivos_espana
	remove_global_variable = votos_negativos_espana
	remove_global_variable = meses_restantes_conferencia
	remove_variable = presento_exigencia
	every_country = {
		limit = {
			has_variable = rechazo_la_exigencia_de_entregar_guinea
		}
		remove_variable = rechazo_la_exigencia_de_entregar_guinea
	}
}
