default_je_la_granja_effect = {
	# Limpiar variables
	if = {
		limit = {
			has_variable = constitucion_redactada
		}
		remove_variable = constitucion_redactada
	}
	custom_tooltip = {
		text = je_la_granja_esp_tt1
		every_scope_state = {
			limit = {
				has_modifier = junta_provincial
			}
			remove_modifier = junta_provincial
		}
		remove_variable = multiplicador2
	}
	custom_tooltip = {
		text = je_la_granja_esp_tt3
		random_political_movement = {
			limit = {
				has_modifier = peridoicos_liberales
			}
			remove_modifier = peridoicos_liberales
		}
		if = {
			limit = {
				has_modifier = caza_periodicos
			}
			remove_modifier = caza_periodicos
		}
	}
}

fin_je_la_granja_effect = {
	# Nuevlo lider intelectuales
	if = {
		limit = {
			ig:ig_intelligentsia = {
				leader = {
					OR = {
						has_ideology = ideology:ideology_radical
						has_ideology = ideology:ideology_progresista
						has_ideology = ideology:ideology_republican_leader
					}
				}
			}
		}
		custom_tooltip = {
			text = je_la_granja_esp_tt4
			create_character = {
				template = evaristo_perez_castro
			}
		}
	}
	else = {
		custom_tooltip = je_la_granja_esp_tt5
	}
	default_je_la_granja_effect = yes
}

incrementar_modificador_junta_provincial = {
	if = {
		limit = {
			NOT = {
				exists = var:multiplicador2
			}
		}
		set_variable = {
			name = multiplicador2
			value = 0
		}
	}
	change_variable = {
		name = multiplicador2
		add = 1
	}
	if = {
		limit = {
			has_journal_entry = je_la_granja_esp
		}
		if = {
			limit = {
				je:je_la_granja_esp = {
					has_modifier = junta_provincial_estado
				}
			}
			je:je_la_granja_esp = {
				remove_modifier = junta_provincial_estado
			}
		}
		je:je_la_granja_esp = {
			add_modifier = {
				name = junta_provincial_estado
				multiplier = var:multiplicador2
			}
		}
	}
}
