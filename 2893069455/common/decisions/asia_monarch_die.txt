chi_empiredie1 = {
	#咸丰去世
	is_shown = {
		is_ai = yes
		OR = {
			AND = {
				exists = c:CHI
				root = c:CHI
			}
			AND = {
				exists = c:ROC
				c:ROC = ROOT
			}
		}
		country_has_primary_culture = cu:manchu
		has_law_or_variant = law_type:law_monarchy
		NOT = {
			has_variable = is_empiredie_1
		}
		game_date < 1861.8.22
		any_scope_character = {
			has_template = chi_xianfeng_template
		}
	}
	possible = {
		custom_tooltip = {
			text = 27_chi_empiredie1_king
			any_scope_character = {
				#专业锁咸丰人物
				has_template = chi_xianfeng_template
			}
		}
		country_has_primary_culture = cu:manchu
		has_law_or_variant = law_type:law_monarchy
		game_date > 1861.1.1
		game_date < 1862.1.1
	}
	when_taken = {
		every_scope_character = {
			limit = {
				has_template = chi_xianfeng_template
			}
			kill_character = yes
		}
		set_variable = {
			name = is_empiredie_1
			value = yes
		}
	}
	ai_chance = {
		value = 100
	}
}

taiping_empire = {
	#太平天国锁定洪秀全和洪贵天福
	is_shown = {
		exists = c:TPG
		root = c:TPG
		NOT = {
			has_variable = is_tpg_empiredie_1
		}
		game_date < 1890.7.30		#也许多活十年？
	}
	possible = {
		game_date < 1890.7.30
		OR = {
			has_law_or_variant = law_type:law_monarchy
			has_law_or_variant = law_type:law_theocracy
		}
	}
	when_taken = {
		set_variable = is_tpg_empiredie_1
		if = {
			limit = {
				any_diplomatic_play = {
					is_war = no
					is_diplomatic_play_participant = ROOT
					is_diplomatic_play_participant = c:CHI
					initiator_is = ROOT
					target_is = c:CHI
				}
			}
			random_diplomatic_play = {
				limit = {
					target = c:CHI
					initiator = ROOT
				}
				add_war_goal = {
					holder = c:CHI
					target_country = ROOT
					type = annex_country
				}
			}
			hidden_effect = {
				c:CHI ?= {
					change_infamy = -150
				}
			}
		}
		create_character = {
			template = ecchi_tpg_hong_xiuquan_character_template
			ruler = yes
			on_created = {
				#原版设定，但是原版洪秀全没DNA
				set_variable = hong_var
				set_career_length = {
					years = 50
					random_range = {
						1
						1.5
					}
				}
				add_modifier = {
					name = modifier_brother_of_christ
					days = long_modifier_time
				}
			}
		}
		create_character = {
			first_name = Tianguifu
			last_name = Hong
			birth_date = 1846.6.6
			culture = cu:hakka
			historical = yes
			religion = rel:protestant
			interest_group = ig_devout
			heir = yes
			ideology = ideology_moderate
			traits = {
				reserved
			}
		}
	}
	ai_chance = {
		value = 100
	}
}
