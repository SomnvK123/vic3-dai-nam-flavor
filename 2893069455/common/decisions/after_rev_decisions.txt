decision_rev_triumphed = {
	is_shown = {
		is_at_war = no
		has_modifier = modifier_help_build_country
		NOT = {
			has_variable = rev_triumphed
		}
	}
	possible = {
		is_at_war = no
		has_modifier = modifier_help_build_country
		has_law_or_variant = law_type:law_council_republic
		NOT = {
			has_variable = rev_triumphed
		}
	}
	when_taken = {
		set_variable = rev_triumphed
		trigger_event = {
			id = after_rev_events.1
			days = 10
			popup = yes
		}
		trigger_event = {
			id = after_rev_events.7
			days = 14
			popup = yes
		}
	}
	ai_chance = {
		value = 100
	}
}
