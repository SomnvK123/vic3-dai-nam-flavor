WCR1_de_simple_westernization = {
	is_shown = {
		has_game_rule = WCR1_simple_westernization
		WCR1_is_westernized_trigger = no		# WCR1_is_westernized_trigger doesn't work right here for some reason?
		WCR1_is_using_vanilla_rules = no
		NOT = {
			has_variable = WCR1_decision_taken
		}
		NOT = {
                OR = {
			c:JAP ?= this
			c:SGN ?= this
                }
                }
		OR = {
			is_ai = no
			has_game_rule = WCR1_allow_AI_westernization
		}
	}
	possible = {
		trigger_if = {
			limit = {
				has_game_rule = WCR1_WCF_requirements
			}
			OR = {
				has_technology_researched = corporate_management
				WCR1_can_start_westernizing_trigger = yes
			}
		}
		trigger_else = {
			WCR1_snori_can_westernize_trigger = yes
		}
	}
	when_taken = {
		set_variable = {
			name = WCR1_decision_taken
			days = 120
		}
		if = {
			limit = {
				has_game_rule = WCR1_enable_influences
				WCR1_has_vanilla_uniform_rule = no
			}
			trigger_event = {
				id = WCR1_ev.2
				popup = yes
			}
		}
		else = {
			if = {
				limit = {
					has_game_rule = WCR1_enable_influences
					WCR1_has_vanilla_uniform_rule = yes
				}
				set_variable = {
					name = WCR1_ruler_influence
					value = -1
				}
				set_variable = {
					name = WCR1_army_influence
					value = -1
				}
				set_variable = {
					name = WCR1_navy_influence
					value = -1
				}
			}
			else = {
				set_variable = {
					name = WCR1_ruler_influence
					value = 0
				}
				set_variable = {
					name = WCR1_army_influence
					value = 0
				}
				set_variable = {
					name = WCR1_navy_influence
					value = 0
				}
			}
			trigger_event = {
				id = WCR1_ev.1
			}
			WCR1_se_set_IG_westernized_modifier = yes
		}
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				any_interest_group = {
					count >= 2
					is_powerful = yes
					WCR1_st_is_interest_group_progressive = yes
				}
			}
			add = 100
		}
		if = {
			limit = {
				game_date >= 1900.1.1
			}
			add = 100
		}
		if = {
			limit = {
				game_date >= 1915.1.1
			}
			add = 100
		}
		if = {
			limit = {
				game_date >= 1930.1.1
			}
			add = 100
		}
		if = {
			limit = {
				has_variable = japan_emperor_restored
			}
			add = 100
		}
		if = {
			limit = {
				has_variable = KOR_korean_empire_declared
			}
			add = 100
		}
		if = {
			limit = {
				any_interest_group = {
					filter = {
						WCR1_st_is_interest_group_conservative = yes
					}
					percent >= 0.75
					ig_counts_as_marginal = no
					ig_approval < define:NPolitics|APPROVAL_THRESHOLD_UNHAPPY
				}
			}
			add = -200
		}
	}
}
