chi_tib_decision = {
	#吞并西藏
	is_shown = {
		# 大清专属
		exists = c:CHI
		exists = c:TIB
		root = c:CHI
		game_date < 1840.1.1		#吃掉西藏会导致没有西藏探险的事件
		c:TIB ?= {
			is_ai = yes
			is_subject_of = ROOT
			lca_is_in_diplomatic_play_or_war = no
			is_subject_type = subject_type_vassal
		}
		NOT = {
			has_variable = chi_tib_decision
		}
	}
	possible = {
		has_technology_researched = centralization
		game_date < 1840.1.1
		OR = {
			AND = {
				is_player = no
			}
			AND = {
				custom_tooltip = {
					text = qinghai_zhuzha_tt
					any_military_formation = {
						is_army = yes
						any_scope_character = {
							has_role_of_type = general
						}
			      current_hq.region ?= sr:region_himalayas
					any_combat_unit = {
						count >= 20
					}
					}
				}
			}
		}
	}
	when_taken = {
		set_variable = chi_tib_decision
		annex = c:TIB
	}
	ai_chance = {
		value = 100
	}
}

annex_west_Africa = {
	is_shown = {
		exists = c:FRA
		root = c:FRA
		NOT = {
			has_variable = annex_west_Africa_taken
		}
		game_date < 1856.1.1
	}
	possible = {
		country_rank >= rank_value:great_power
		game_date >= 1846.4.1
		game_date < 1856.1.1
		owns_entire_state_region = STATE_ALGIERS
	}
	when_taken = {
		set_variable = annex_west_Africa_taken
		add_journal_entry = {
			type = ra_fra_je_ambiciones_guinea
		}
		every_country = {
			limit = {
				NOT = {
					this = root
				}
				country_rank >= rank_value:great_power
				OR = {
					c:RUS ?= this
					c:AUS ?= this
					c:PRU ?= this
					has_interest_marker_in_region = sr:region_north_africa
					has_interest_marker_in_region = sr:region_west_africa
					any_scope_state = {
						OR = {
							region = sr:region_north_africa
							region = sr:region_west_africa
						}
					}
					any_subject_or_below = {
						is_direct_subject_of = root
						any_scope_state = {
							OR = {
								region = sr:region_north_africa
								region = sr:region_west_africa
							}
						}
					}
				}
			}
			trigger_event = {
				id = france_flavor.5
				popup = yes
			}
		}
	}
	ai_chance = {
		value = 100
	}
}
