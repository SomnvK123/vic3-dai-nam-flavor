spa_burgundian_cross_enabled = {
	is_shown = {
		is_player = yes
		AND = {
			exists = c:SPA
			c:SPA ?= this
		}
		NOT = {
			has_variable = spa_burgundian_cross_war_flag_enabled
		}
		has_game_rule = allow_war_flags
	}
	when_taken = {
		set_variable = spa_burgundian_cross_war_flag_enabled
	}
	ai_chance = {
		value = 0
	}
}
