texan_csa = {
	is_shown = {
		exists = c:TEX
		root = c:TEX
		root = {
			is_subject = no
			is_at_war = no
		}
		exists = c:CSA
		NOT = {
			has_law_or_variant = law_type:law_slavery_banned
		}
	}

	possible = {
		any_neighbouring_state = {
			owner = c:CSA
		}
		c:CSA ?= {
			relations:root >= relations_threshold:cordial
			is_at_war = no
		}
		NOT = {
			has_variable = confederate_texas
		}
		prestige < c:CSA.prestige
	}

	when_taken = {
		set_variable = confederate_texas
		usfp_entrench_slavery_effect = yes
		c:CSA ?= {
			create_diplomatic_pact = {
				country = root
				type = puppet
			}
		}
		change_relations = {
			country = c:CSA
			value = 50
		}
	}

	ai_chance = {
		value = 100
	}
}
