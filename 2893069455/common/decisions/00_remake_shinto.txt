REPLACE_OR_CREATE:shinto_decision = {
	is_shown = {
		exists = c:JAP
		c:JAP ?= this
		NOT = { country_has_state_religion = rel:shinto }
		NOT = { has_dlc_feature = ep2_content }
	}
	possible = {
		custom_tooltip = {
			text = ruler_is_japanese_emperor_tt
			has_variable = japan_emperor_restored
		}
		ig:ig_devout ?= {
			ig_counts_as_marginal = no
		}
		country_has_monarchy_law = yes
		NOT = { has_law_or_variant = law_type:law_state_religion }
	}

	when_taken = {
		set_state_religion = rel:shinto
		state_religion_switch_effect = yes
		ig:ig_devout = {
			hidden_effect = {
				set_interest_group_name = ig_shinto_monks
			}
			remove_ideology = ideology_buddhist_moralist
			remove_ideology = ideology_mahayana_buddhist_moralist
			add_ideology = ideology_shinto_moralist
			set_ig_trait = ig_trait:ig_trait_haibutsu_kishaku
			set_ig_trait = ig_trait:ig_trait_secular_shrine_theory
			set_ig_trait = ig_trait:ig_trait_heavenly_sovereign
			leader ?= {
				change_character_religion = rel:shinto
			}
		}
		ig:ig_industrialists = {
			remove_ideology = ideology_plutocratic
			#add_ideology = ideology_dazheng_jituan
			add_ideology = ideology_japan_plutocratic
		}
		ig:ig_armed_forces = {
			remove_ideology = ideology_bakufu
			remove_ideology = ideology_anti_shogunate
			add_ideology = ideology_meiji_japan
		}
		random_scope_state = {
			limit = {
				owner = ROOT
				is_capital = yes
			}
			convert_population = {
				target = rel:shinto
				value = 0.4
			}
		}
		every_scope_state = {
			limit = {
				owner = ROOT
				is_incorporated = yes
				NOT = {
					is_capital = yes
				}
			}
			convert_population = {
				target = rel:shinto
				value = 0.2
			}
		}
	}
	ai_chance = {
		value = 100
	}
}
