ITA_AUS_war_decision = {
	is_shown = {
		# NOT = { has_global_variable = ITA_AUS_war_decision_flag }
		any_country = {
			country_has_primary_culture = cu:south_german
			any_scope_state = {
				OR = {
					state_region = s:STATE_LOMBARDY
					state_region = s:STATE_VENETIA
				}
			}
		}
		OR = {
			lca_this_is = {
				TARGET = c:ITA
			}
			AND = {
				lca_is_or_is_candidate = {
					TAG = ITA
				}
				OR = {
					exists = c:ITA
					any_country = {
						count <= 1
						is_unification_candidate = ITA
					}
				}
				has_technology_researched = pan-nationalism
			}
		}
		OR = {
			AND = {
				NOT = {
					owns_entire_state_region = STATE_LOMBARDY
				}
				has_claim = s:STATE_LOMBARDY
			}
			AND = {
				NOT = {
					owns_entire_state_region = STATE_VENETIA
				}
				has_claim = s:STATE_VENETIA
			}
		}
	}
	possible = {
		is_subject = no
		lca_is_in_diplomatic_play_or_war = no
		OR = {
			lca_has_alliance_with_and_available = {
				WHO = c:FRA
			}
			lca_has_alliance_with_and_available = {
				WHO = c:PRU
			}
			lca_has_alliance_with_and_available = {
				WHO = c:NGF
			}
			lca_has_alliance_with_and_available = {
				WHO = c:GER
			}
		}
		custom_tooltip = {
			text = ITA_AUS_war_decision_possible_tt
			any_scope_ally = {
				AND = {
					OR = {
						lca_this_is = {
							TARGET = c:FRA
						}
						lca_this_is = {
							TARGET = c:PRU
						}
						lca_this_is = {
							TARGET = c:NGF
						}
						lca_this_is = {
							TARGET = c:GER
						}
					}
					lca_is_in_diplomatic_play_or_war = no
					any_country = {
						country_has_primary_culture = cu:south_german
						any_scope_state = {
							OR = {
								state_region = s:STATE_LOMBARDY
								state_region = s:STATE_VENETIA
							}
						}
						NOR = {
							has_truce_with = root
							lca_has_alliance_or_defensive_pact_with = {
								WHO = root
							}
							has_truce_with = prev
							lca_has_alliance_or_defensive_pact_with = {
								WHO = prev
							}
						}
					}
				}
			}
		}
	}
	when_taken = {
		# set_global_variable = ITA_AUS_war_decision_flag
		hidden_effect = {
			random_scope_ally = {
				limit = {
					OR = {
						lca_this_is = {
							TARGET = c:FRA
						}
						lca_this_is = {
							TARGET = c:PRU
						}
						lca_this_is = {
							TARGET = c:NGF
						}
						lca_this_is = {
							TARGET = c:GER
						}
					}
					lca_is_in_diplomatic_play_or_war = no
					any_country = {
						country_has_primary_culture = cu:south_german
						any_scope_state = {
							OR = {
								state_region = s:STATE_LOMBARDY
								state_region = s:STATE_VENETIA
							}
						}
						NOR = {
							has_truce_with = root
							lca_has_alliance_or_defensive_pact_with = {
								WHO = root
							}
							has_truce_with = prev
							lca_has_alliance_or_defensive_pact_with = {
								WHO = prev
							}
						}
					}
				}
				random_country = {
					limit = {
						country_has_primary_culture = cu:south_german
						any_scope_state = {
							OR = {
								state_region = s:STATE_LOMBARDY
								state_region = s:STATE_VENETIA
							}
						}
						NOR = {
							has_truce_with = root
							lca_has_alliance_or_defensive_pact_with = {
								WHO = root
							}
							has_truce_with = prev
							lca_has_alliance_or_defensive_pact_with = {
								WHO = prev
							}
						}
					}
					save_scope_as = lca_target_country
					if = {
						limit = {
							any_scope_state = {
								state_region = s:STATE_LOMBARDY
							}
						}
						random_scope_state = {
							limit = {
								state_region = s:STATE_LOMBARDY
							}
							save_scope_as = lca_target_state
							set_variable = {
								name = lca_targeted_state
								days = 1
							}
						}
					}
					else_if = {
						limit = {
							any_scope_state = {
								state_region = s:STATE_VENETIA
							}
						}
						random_scope_state = {
							limit = {
								state_region = s:STATE_VENETIA
							}
							save_scope_as = lca_target_state
							set_variable = {
								name = lca_targeted_state
								days = 1
							}
						}
					}
				}
				save_scope_as = lca_ally_country
			}
		}
		if = {
			limit = {
				exists = scope:lca_target_country
				exists = scope:lca_target_state
				exists = scope:lca_ally_country
			}
			create_diplomatic_play = {
				name = italy_austria_war
				target_state = scope:lca_target_state
				escalation = 78
				type = dp_return_state
			}
			violate_sovereignty_join = {
				violator = root
				target = scope:lca_ally_country
				join_violator = yes
			}
			every_scope_ally = {
				limit = {
					NOT = {
						this = scope:lca_ally_country
					}
					lca_is_in_diplomatic_play_or_war = no
					any_scope_treaty = {
						binds = root
						any_scope_article = {
							has_type = alliance
						}
					}
					NOT = {
						has_truce_with = scope:lca_target_country
					}
					NOT = {
						lca_has_alliance_or_defensive_pact_with = {
							WHO = scope:lca_target_country
						}
					}
					lca_has_hostile_attitude = {
						WHO = scope:lca_target_country
					}
					scope:lca_target_country = {
						relations:prev < relations_threshold:cordial
					}
				}
				violate_sovereignty_join = {
					violator = root
					target = this
					join_violator = yes
				}
			}
			scope:lca_target_country = {
				every_scope_ally = {
					limit = {
						lca_is_in_diplomatic_play_or_war = no
					}
					violate_sovereignty_join = {
						violator = root
						target = this
						join_violator = no
					}
				}
			}
		}
	}
	ai_chance = {
		value = 100
		if = {
			limit = {
				infamy >= infamy_threshold:infamous
			}
			add = -100
		}
		if = {
			# 如果奥地利有列强盟友而我们只有一个盟友，那就算了
			limit = {
				any_country = {
					AND = {
						country_has_primary_culture = cu:south_german
						any_scope_state = {
							OR = {
								state_region = s:STATE_LOMBARDY
								state_region = s:STATE_VENETIA
							}
						}
						NOR = {
							has_truce_with = root
							lca_has_alliance_or_defensive_pact_with = {
								WHO = root
							}
						}
						any_scope_ally = {
							country_rank >= rank_value:great_power
							lca_is_in_diplomatic_play_or_war = no
							count >= 1
						}
					}
				}
				any_scope_ally = {
					country_rank >= rank_value:great_power
					lca_is_in_diplomatic_play_or_war = no
					count <= 1
				}
			}
			add = -100
		}
	}
}

ITA_unification_war_decision = {
	is_shown = {
		OR = {
			lca_this_is = {
				TARGET = c:ITA
			}
			AND = {
				lca_is_or_is_candidate = {
					TAG = ITA
				}
				OR = {
					exists = c:ITA
					any_country = {
						count <= 1
						is_unification_candidate = ITA
					}
				}
				has_technology_researched = pan-nationalism
			}
		}
		any_neighbouring_state = {
			owner ?= {
				AND = {
					OR = {
						country_has_primary_culture = cu:north_italian
						country_has_primary_culture = cu:south_italian
					}
					# is_non_customs_union_subject = no
					country_rank < root.country_rank
					any_scope_state = {
						count <= 3
					}
				}
			}
		}
	}
	possible = {
		is_subject = no
		lca_is_in_diplomatic_play_or_war = no
		custom_tooltip = {
			text = ITA_unification_war_decision_possible_tt
			any_neighbouring_state = {
				owner ?= {
					AND = {
						OR = {
							country_has_primary_culture = cu:north_italian
							country_has_primary_culture = cu:south_italian
						}
						# is_non_customs_union_subject = no
						NOT = {
							lca_has_same_market_with = {
								TARGET = root
							}
						}
						root = {
							NOT = {
								has_truce_with = prev
							}
						}
						country_rank < root.country_rank
						trigger_if = {
							limit = {
								is_non_customs_union_subject = yes
								exists = overlord
							}
							overlord = {
								relations:root < relations_threshold:cordial
								root = {
									NOT = {
										has_truce_with = prev
									}
								}
								NOT = {
									lca_has_alliance_or_defensive_pact_with = {
										WHO = root
									}
								}
							}
						}
						trigger_else = {
							relations:root < relations_threshold:neutral
						}
						any_scope_state = {
							count <= 3
						}
					}
				}
			}
		}
	}
	when_taken = {
		hidden_effect = {
			if = {
				limit = {
					any_neighbouring_state = {
						owner ?= {
							AND = {
								OR = {
									country_has_primary_culture = cu:north_italian
									country_has_primary_culture = cu:south_italian
								}
								is_non_customs_union_subject = no
								NOT = {
									lca_has_same_market_with = {
										TARGET = root
									}
								}
								root = {
									NOT = {
										has_truce_with = prev
									}
								}
								NOT = {
									any_scope_ally = {
										country_rank >= root.country_rank
									}
								}
								country_rank < root.country_rank
								trigger_if = {
									limit = {
										is_non_customs_union_subject = yes
										exists = overlord
									}
									overlord = {
										relations:root < relations_threshold:cordial
										root = {
											NOT = {
												has_truce_with = prev
											}
										}
										NOT = {
											lca_has_alliance_or_defensive_pact_with = {
												WHO = root
											}
										}
									}
								}
								trigger_else = {
									relations:root < relations_threshold:neutral
								}
								any_scope_state = {
									count <= 3
								}
							}
						}
					}
				}
				random_neighbouring_state = {
					limit = {
						owner ?= {
							AND = {
								OR = {
									country_has_primary_culture = cu:north_italian
									country_has_primary_culture = cu:south_italian
								}
								is_non_customs_union_subject = no
								NOT = {
									lca_has_same_market_with = {
										TARGET = root
									}
								}
								root = {
									NOT = {
										has_truce_with = prev
									}
								}
								NOT = {
									any_scope_ally = {
										country_rank >= root.country_rank
									}
								}
								country_rank < root.country_rank
								trigger_if = {
									limit = {
										is_non_customs_union_subject = yes
										exists = overlord
									}
									overlord = {
										relations:root < relations_threshold:cordial
										root = {
											NOT = {
												has_truce_with = prev
											}
										}
										NOT = {
											lca_has_alliance_or_defensive_pact_with = {
												WHO = root
											}
										}
									}
								}
								trigger_else = {
									relations:root < relations_threshold:neutral
								}
								any_scope_state = {
									count <= 3
								}
							}
						}
					}
					save_scope_as = lca_target_state
					owner ?= {
						save_scope_as = lca_target_country
					}
				}
			}
			else = {
				random_neighbouring_state = {
					limit = {
						owner ?= {
							AND = {
								OR = {
									country_has_primary_culture = cu:north_italian
									country_has_primary_culture = cu:south_italian
								}
								# is_non_customs_union_subject = no
								NOT = {
									lca_has_same_market_with = {
										TARGET = root
									}
								}
								root = {
									NOT = {
										has_truce_with = prev
									}
								}
								country_rank < root.country_rank
								trigger_if = {
									limit = {
										is_non_customs_union_subject = yes
										exists = overlord
									}
									overlord = {
										relations:root < relations_threshold:cordial
										root = {
											NOT = {
												has_truce_with = prev
											}
										}
										NOT = {
											lca_has_alliance_or_defensive_pact_with = {
												WHO = root
											}
										}
									}
								}
								trigger_else = {
									relations:root < relations_threshold:neutral
								}
								any_scope_state = {
									count <= 3
								}
							}
						}
					}
					save_scope_as = lca_target_state
					owner ?= {
						save_scope_as = lca_target_country
					}
				}
			}
		}
		if = {
			limit = {
				exists = scope:lca_target_country
				exists = scope:lca_target_state
			}
			create_diplomatic_play = {
				name = italy_unification_war
				target_state = scope:lca_target_state
				escalation = 78
				type = dp_annex_war
			}
			lca_call_allies_for_current_war = {
				ACTOR = root
				TARGET = scope:lca_target_country
			}
		}
	}
	ai_chance = {
		value = 100
		if = {
			limit = {
				infamy >= infamy_threshold:infamous
			}
			add = -100
		}
		if = {
			limit = {
				NOT = {
					any_neighbouring_state = {
						owner ?= {
							AND = {
								OR = {
									country_has_primary_culture = cu:north_italian
									country_has_primary_culture = cu:south_italian
								}
								# is_non_customs_union_subject = no
								NOT = {
									lca_has_same_market_with = {
										TARGET = root
									}
								}
								root = {
									NOT = {
										has_truce_with = prev
									}
								}
								NOT = {
									any_scope_ally = {
										AND = {
											country_rank >= root.country_rank
											army_size > root.army_size
										}
									}
								}
								trigger_if = {
									# 没有强大的宗主
									limit = {
										is_non_customs_union_subject = yes
										exists = overlord
									}
									overlord = {
										NAND = {
											country_rank >= root.country_rank
											army_size > root.army_size
										}
									}
								}
								trigger_else = {
									hidden_trigger = {
										always = yes
									}
								}
								country_rank < root.country_rank
								trigger_if = {
									limit = {
										is_non_customs_union_subject = yes
										exists = overlord
									}
									overlord = {
										relations:root < relations_threshold:cordial
										root = {
											NOT = {
												has_truce_with = prev
											}
										}
										NOT = {
											lca_has_alliance_or_defensive_pact_with = {
												WHO = root
											}
										}
									}
								}
								trigger_else = {
									relations:root < relations_threshold:neutral
								}
								any_scope_state = {
									count <= 3
								}
							}
						}
					}
				}
			}
			add = -100
		}
	}
}
