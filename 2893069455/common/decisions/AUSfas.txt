AUS_fascist_flag_enabled = {
	is_shown = {
		NOT = {
			any_country = {
				NOT = {
					this = root
				}
				is_unification_candidate = GER
			}
		}
		exists = c:AUS
		root = c:AUS
		NOT = {
			has_variable = coa_def_fascist_flag_trigger
		}
	}
	possible = {
		OR = {
			has_technology_researched = political_agitation
			has_law_or_variant = law_type:law_ethnostate
		}
	}
	when_taken = {
		custom_tooltip = AUS_fascist_flag_enabled.tt
		set_variable = coa_def_fascist_flag_trigger
	}
	ai_chance = {
		value = 0
	}
}

AUS_fascist_flag_disabled = {
	is_shown = {
		NOT = {
			any_country = {
				NOT = {
					this = root
				}
				is_unification_candidate = GER
			}
		}
		exists = c:AUS
		root = c:AUS
		has_variable = coa_def_fascist_flag_trigger
	}
	possible = {
		OR = {
			has_technology_researched = political_agitation
			has_law_or_variant = law_type:law_ethnostate
		}
	}
	when_taken = {
		custom_tooltip = AUS_fascist_flag_enabled.tt
		remove_variable = coa_def_fascist_flag_trigger
	}
	ai_chance = {
		value = 0
	}
}

AUS_fascist_to_GER = {
	is_shown = {
		NOT = {
			any_country = {
				NOT = {
					this = root
				}
				is_unification_candidate = GER
			}
		}
		exists = c:AUS
		root = c:AUS
		NOT = {
			has_variable = ger_coa_def_fascist_flag_trigger
		}
		has_variable = coa_def_fascist_flag_trigger
	}
	possible = {
		OR = {
			has_technology_researched = political_agitation
			has_law_or_variant = law_type:law_ethnostate
		}
	}
	when_taken = {
		custom_tooltip = AUS_fascist_to_GER.tt
		set_variable = ger_coa_def_fascist_flag_trigger
		c:GER ?= {
			set_variable = coa_def_fascist_flag_trigger
		}
	}
	ai_chance = {
		value = 0
	}
}
