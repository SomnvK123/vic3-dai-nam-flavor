berlin_conference_egy = {
	#埃及吞并达尔富尔
	is_shown = {
		year > 1883
		exists = c:EGY
		c:EGY ?= this
		is_player = no
		NOT = {
			has_variable = berlin_conference_egy_taken
		}
	}
	possible = {
		c:EGY ?= this
		is_player = no
		any_neighbouring_state = {
			state_region = s:STATE_DARFUR
			owner ?= {
				c:DFR ?= this
			}
		}
		exists = c:DFR
		c:DFR ?= {
			is_player = no
			is_subject = no
			lca_is_in_diplomatic_play_or_war = no
		}
	}
	when_taken = {
		s:STATE_DARFUR = {
			random_scope_state = {
				limit = {
					exists = c:DFR
					owner ?= {
						OR = {
							c:DFR ?= this
							is_subject_of = c:DFR
						}
					}
				}
				set_state_owner = c:EGY
			}
		}
		set_variable = {
			name = berlin_conference_egy_taken
			value = yes
		}
	}
	ai_chance = {
		value = 100
	}
}
