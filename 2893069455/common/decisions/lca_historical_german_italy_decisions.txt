italian_risorgimento_leader = {
	# （撒皮）复兴运动领袖
	is_shown = {
		NOT = {
			exists = c:ITA
		}
		OR = {
			country_has_primary_culture = cu:north_italian
			country_has_primary_culture = cu:south_italian
		}
		country_rank >= rank_value:minor_power		# 开局只有撒皮是这个等级（错误的，两西西里也能很快达到）
		OR = {
			AND = {
				#玩家是两西西里
				exists = c:SIC
				c:SIC ?= {
					is_player = yes
				}
			}
			AND = {
				#不是两西西里
				NOT = {
					c:SIC ?= this
				}
			}
		}
	}
	possible = {
		country_rank >= rank_value:minor_power
		has_technology_researched = nationalism
		NOT = {
			has_modifier = lca_italian_unification_movement
		}
		trigger_if = {
			limit = {
				any_country = {
					has_modifier = lca_italian_unification_movement
				}
			}
			any_country = {
				AND = {
					has_modifier = lca_italian_unification_movement
					NOT = {
						lca_is_or_is_candidate = {
							TAG = ITA
						}
					}
				}
			}
		}
		trigger_else = {
			hidden_trigger = {
				always = yes
			}
		}
	}
	when_taken = {
		if = {
			limit = {
				any_country = {
					has_modifier = lca_italian_unification_movement
				}
			}
			every_country = {
				limit = {
					has_modifier = lca_italian_unification_movement
					NOT = {
						lca_is_or_is_candidate = {
							TAG = ITA
						}
					}
				}
				remove_modifier = lca_italian_unification_movement
			}
		}
		if = {
			limit = {
				has_modifier = lca_refuse_italian_unification_movement
			}
			remove_modifier = lca_refuse_italian_unification_movement
		}
		add_modifier = {
			name = lca_italian_unification_movement
			years = 25			# 25年
		}
	}
	ai_chance = {
		value = 100
		if = {
			limit = {
				has_modifier = lca_refuse_italian_unification_movement
				NOT = {
					lca_is_or_is_candidate = {
						TAG = ITA
					}
				}
			}
			add = -100
		}
		if = {
			limit = {
				prestige < 50
			}
			add = -80
		}
	}
}

italian_prussia_treaty = {
	# 普鲁士-意大利反奥地利同盟
	is_shown = {
		c:SAR ?= this
		exists = c:PRU
		NOT = {
			has_variable = signed_italian_prussia_treaty
		}
	}
	possible = {
		has_technology_researched = nationalism
		is_subject = no
		OR = {
			has_technology_researched = multilateral_alliances
			NOT = {
				any_scope_ally = {
					country_rank >= rank_value:major_power
				}
			}
		}
		NOR = {
			trigger_if = {
				limit = {
					exists = c:AUS
				}
				lca_has_alliance_or_defensive_pact_with = {
					WHO = c:AUS
				}
			}
			trigger_if = {
				limit = {
					exists = c:SGF
				}
				lca_has_alliance_or_defensive_pact_with = {
					WHO = c:SGF
				}
			}
		}
		trigger_if = {
			limit = {
				country_has_primary_culture = cu:north_german
				lca_is_or_is_candidate = {
					TAG = GER
				}
			}
			any_country = {
				AND = {
					NOT = {
						any_scope_treaty = {
							binds = root
							any_scope_article = {
								has_type = alliance
							}
						}
					}
					NOT = {
						lca_has_hostile_attitude = {
							WHO = root
						}
					}
					lca_is_or_is_candidate = {
						TAG = ITA
					}
					relations:root >= relations_threshold:neutral
					OR = {
						has_technology_researched = multilateral_alliances
						NOT = {
							any_scope_ally = {
								country_rank >= rank_value:major_power
							}
						}
					}
					has_diplomatic_relevance = root
					NOR = {
						is_diplomatic_play_enemy_of = root
						has_war_with = root
					}
				}
			}
		}
		trigger_else_if = {
			limit = {
				lca_is_or_is_candidate = {
					TAG = ITA
				}
			}
			any_country = {
				AND = {
					NOT = {
						any_scope_treaty = {
							binds = root
							any_scope_article = {
								has_type = alliance
							}
						}
					}
					NOT = {
						lca_has_hostile_attitude = {
							WHO = root
						}
					}
					country_has_primary_culture = cu:north_german
					lca_is_or_is_candidate = {
						TAG = GER
					}
					relations:root >= relations_threshold:neutral
					OR = {
						has_technology_researched = multilateral_alliances
						NOT = {
							any_scope_ally = {
								country_rank >= rank_value:major_power
							}
						}
					}
					has_diplomatic_relevance = root
					NOR = {
						is_diplomatic_play_enemy_of = root
						has_war_with = root
					}
				}
			}
		}
	}
	when_taken = {
		save_scope_as = italian_prussia_alliance_target
		c:PRU ?= {
			trigger_event = {
				id = lca_german_italy.1
			}
			set_variable = {
				name = signed_italian_prussia_treaty
				years = 15				# signed treaty for 15 years
			}
		}
		set_variable = {
			name = signed_italian_prussia_treaty
			years = 15			# signed treaty for 15 years
		}
	}
	ai_chance = {
		value = 10
		if = {
			limit = {
				exists = c:AUS
				c:AUS ?= {
					lca_has_hostile_attitude = {
						WHO = root
					}
				}
			}
			add = 10
		}
		if = {
			limit = {
				exists = c:AUS
				lca_has_hostile_attitude = {
					WHO = c:AUS
				}
			}
			add = 10
		}
		if = {
			limit = {
				exists = c:AUS
				c:AUS ?= {
					relations:root < relations_threshold:neutral
				}
			}
			add = 10
		}
		if = {
			limit = {
				exists = c:AUS
				c:AUS ?= {
					relations:root >= relations_threshold:cordial
				}
			}
			add = 10
		}
		if = {
			limit = {
				exists = c:AUS
				c:AUS ?= {
					relations:root >= relations_threshold:amicable
				}
			}
			add = -40
		}
		if = {
			limit = {
				exists = c:SGF
				c:SGF ?= {
					lca_has_hostile_attitude = {
						WHO = root
					}
				}
			}
			add = 10
		}
		if = {
			limit = {
				exists = c:SGF
				lca_has_hostile_attitude = {
					WHO = c:SGF
				}
			}
			add = 10
		}
		if = {
			limit = {
				exists = c:SGF
				c:SGF ?= {
					relations:root < relations_threshold:neutral
				}
			}
			add = 10
		}
		if = {
			limit = {
				exists = c:SGF
				c:SGF ?= {
					relations:root >= relations_threshold:cordial
				}
			}
			add = 1
		}
		if = {
			limit = {
				exists = c:SGF
				c:SGF ?= {
					relations:root >= relations_threshold:amicable
				}
			}
			add = -40
		}
	}
}

			}
			lca_owns_any_homeland = {

				}
			}
