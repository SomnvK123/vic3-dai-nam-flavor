rev_on = {
	is_shown = {
		has_technology_researched = socialism
		has_variable = start_to_export_rev
		NOT = {
			has_variable = ready_to_sway
			has_variable = hide_rev_on
		}
	}
	possible = {
		is_at_war = no
		is_ai = no
		has_variable = start_to_export_rev
		has_technology_researched = political_agitation
		has_law_or_variant = law_type:law_council_republic
	}
	when_taken = {
		set_variable = hide_rev_on
		trigger_event = {
			id = agitate_rev_events.1
			popup = yes
		}
	}
}

		}

			add = 1.5
		}
		change_variable = {
			name = sway_TU_progress_var
			add = 1.5
		}
	}
}

decision_is_on_rev_kill_ruler ?= {
	is_shown = {
		is_revolutionary = yes
		has_modifier = modifier_help_build_country
	}
	possible = {
		is_at_war = no
		is_revolutionary = yes
		has_modifier = modifier_help_build_country
		ruler ?= {
			is_historical = no
			NOR = {
				has_ideology = ideology:ideology_communist
				has_ideology = ideology:ideology_vanguardist
				has_ideology = ideology:ideology_anarchist
			}
		}
	}
	when_taken = {
		every_scope_character = {
			limit = {
				is_ruler_of_own_country = yes
				is_historical = no
				NOR = {
					has_ideology = ideology:ideology_communist
					has_ideology = ideology:ideology_vanguardist
					has_ideology = ideology:ideology_anarchist
				}
			}
			kill_character = yes
		}
	}
	ai_chance = {
		value = 100
	}
}

decision_wartime_law = {
	is_shown = {
		is_revolutionary = yes
		has_law_or_variant = law_type:law_council_republic
		NOT = {
			has_variable = wartime_law
		}
	}
	possible = {
		is_revolutionary = yes
		has_law_or_variant = law_type:law_council_republic
		NOT = {
			has_variable = wartime_law
		}
	}
	when_taken = {
		set_variable = wartime_law
		activate_law = law_type:law_slavery_banned
		activate_law = law_type:law_militarized_police
		activate_law = law_type:law_appointed_bureaucrats
		activate_law = law_type:law_universal_suffrage
		activate_law = law_type:law_peasant_proprietorship
		activate_law = law_type:law_outlawed_dissent
		activate_law = law_type:law_mass_conscription
		activate_law = law_type:law_secret_police
		every_interest_group = {
			limit = {
				leader = {
					OR = {
						has_ideology = ideology:ideology_communist
						has_ideology = ideology:ideology_vanguardist
						has_ideology = ideology:ideology_anarchist
					}
				}
			}
			add_modifier = {
				name = modifier_wartime_law
				months = 6
			}
		}
		trigger_event = {
			id = revolution_events.4
			popup = yes
			days = 7
		}
	}
	ai_chance = {
		value = 100
	}
}
