ITA_claim_decision = {
	is_shown = {
		NOT = {
			has_variable = ITA_claim_decision_flag
		}
		lca_this_is = {
			TARGET = c:ITA
		}
		is_ai = yes
	}
	possible = {
		always = yes
	}
	when_taken = {
		set_variable = ITA_claim_decision_flag
		lca_add_claim_id = {
			STATE_ID = STATE_LOMBARDY
		}
		lca_add_claim_id = {
			STATE_ID = STATE_VENETIA
		}
		lca_add_claim_id = {
			STATE_ID = STATE_ISTRIA
		}
		lca_add_claim_id = {
			STATE_ID = STATE_SOUTH_TYROL
		}
		if = {
			limit = {
				trigger_if = {
					limit = {
						exists = c:FRA
					}
					NOR = {
						has_variable = signed_italian_france_treaty_accepted
						lca_has_alliance_or_defensive_pact_with = {
							WHO = c:FRA
						}
						lca_has_same_market_with = {
							TARGET = c:FRA
						}
						c:FRA ?= {
							relations:root >= relations_threshold:cordial
						}
					}
				}
				trigger_else = {
					always = yes
				}
			}
			lca_add_claim_id = {
				STATE_ID = STATE_SAVOY
			}
		}
		lca_add_claim_id = {
			STATE_ID = STATE_EMILIA
		}
		lca_add_claim_id = {
			STATE_ID = STATE_TUSCANY
		}
		lca_add_claim_id = {
			STATE_ID = STATE_ROMAGNA
		}
		lca_add_claim_id = {
			STATE_ID = STATE_LAZIO
		}
		lca_add_claim_id = {
			STATE_ID = STATE_PIEDMONT
		}
		lca_add_claim_id = {
			STATE_ID = STATE_SARDINIA
		}
		lca_add_claim_id = {
			STATE_ID = STATE_TUSCANY
		}
		lca_add_claim_id = {
			STATE_ID = STATE_UMBRIA
		}
		lca_add_claim_id = {
			STATE_ID = STATE_SICILY
		}
		lca_add_claim_id = {
			STATE_ID = STATE_LAZIO
		}
		lca_add_claim_id = {
			STATE_ID = STATE_CAMPANIA
		}
		lca_add_claim_id = {
			STATE_ID = STATE_CALABRIA
		}
		lca_add_claim_id = {
			STATE_ID = STATE_ABRUZZO
		}
		lca_add_claim_id = {
			STATE_ID = STATE_APULIA
		}
		if = {
			limit = {
				exists = c:AUS
				c:AUS ?= {
					any_scope_state = {
						OR = {
							state_region = s:STATE_LOMBARDY
							state_region = s:STATE_VENETIA
						}
					}
				}
			}
			c:AUS ?= {
				save_scope_as = lca_scope_country
			}
		}
		else_if = {
			limit = {
				exists = c:SGF
				c:SGF ?= {
					any_scope_state = {
						OR = {
							state_region = s:STATE_LOMBARDY
							state_region = s:STATE_VENETIA
						}
					}
				}
			}
			c:SGF ?= {
				save_scope_as = lca_scope_country
			}
		}
		else_if = {
			limit = {
				exists = c:GER
				c:GER ?= {
					any_scope_state = {
						OR = {
							state_region = s:STATE_LOMBARDY
							state_region = s:STATE_VENETIA
						}
					}
				}
			}
			c:GER ?= {
				save_scope_as = lca_scope_country
			}
		}
		if = {
			limit = {
				exists = scope:lca_scope_country
			}
			scope:lca_scope_country = {
				change_relations = {
					country = root
					value = -30
				}
			}
			if = {
				limit = {
					is_subject_of = scope:lca_scope_country
				}
				change_relations = {
					country = scope:lca_scope_country
					value = -30
				}
			}
			lca_remove_alliance_or_defensive_pact_with = {
				WHO = scope:lca_scope_country
			}
		}
		#lca_weak_gain_independence = yes
	}
	ai_chance = {
		value = 1000
	}
}
