ai_should_get_less_radical = {
	#AI获得革命者的妥协
	is_shown = {
		NOT = {
			has_variable = is_buff_1
		}
		has_game_rule = strong_buff_on
		is_ai = yes
		NOT = {
			has_modifier = dsk_ai_buff0
		}
	}
	possible = {
		is_ai = yes
		has_game_rule = strong_buff_on
		NOT = {
			has_modifier = dsk_ai_buff0
		}
	}
	when_taken = {
		remove_variable = is_dsk_ai_buff_1
		remove_variable = is_dsk_ai_buff_2
		remove_variable = is_dsk_ai_buff_3
		remove_variable = is_dsk_ai_buff_4
		remove_variable = is_dsk_ai_buff_5
		set_variable = is_buff_1
		set_government_wage_level = medium
		#set_tax_level = very_high
		remove_modifier = dsk_ai_buff1
		remove_modifier = dsk_ai_buff2
		remove_modifier = dsk_ai_buff3
		remove_modifier = dsk_ai_buff4
		remove_modifier = dsk_ai_buff5
		add_modifier = {
			name = dsk_ai_buff0
		}
	}
	ai_chance = {
		value = 100
	}
}

dsk_low_birth_rate = {
	is_shown = {
		NOT = {
			has_variable = renkou2
		}
		total_population >= 400000000
		has_game_rule = renkou_buff_on
	}
	possible = {
		total_population >= 400000000
	}
	when_taken = {
		set_variable = {
			name = renkou2
			years = 25
		}
		if = {
			limit = {
					has_modifier = renkou_l
				}
            remove_modifier = renkou_l
		}
		if = {
			limit = {
					has_modifier = renkou1
				}
            remove_modifier = renkou1
		}
		add_modifier = {
			name = renkou
			years = 5
		}
	}
	ai_chance = {
		value = 0
	}
}

too_many_pops_decisions = {
	#大清AI闭关锁国+吃空饷
	is_shown = {
		is_ai = yes
		game_date < 1837.1.1
		exists = c:CHI
		c:CHI ?= this
		total_population > 150000000
		NOT = {
			has_variable = what_the_qng_fuc
		}
	}
	possible = {
		is_ai = yes
		game_date < 1837.1.1
	}
	when_taken = {
		set_variable = what_the_qng_fuc
		if = {
			limit = {
				NOT = {
					has_strategy = ai_away_from_world
				}
			}
			set_strategy = ai_away_from_world
		}
		add_modifier = {
			name = qing_soldier_not_full
			months = 360
			is_decaying = yes
		}
	}
	ai_chance = {
		value = 100
	}
}

peace_somali_decisions = {
	#索马里提前一百年在欧加登地区打埃塞，甚至全吃埃塞不合理
	is_shown = {
		is_ai = yes
		exists = c:SM3
		c:SM3 ?= this
		NOT = {
			has_variable = what_the_qng_fuc
		}
		has_strategy = ai_strategy_unify_somalia
	}
	possible = {
		is_ai = yes
	}
	when_taken = {
		set_variable = what_the_qng_fuc
		if = {
			limit = {
				NOT = {
					has_strategy = ai_strategy_armed_isolationism
				}
			}
			set_strategy = ai_strategy_armed_isolationism
		}
		if = {
			limit = {
				exists = c:ETH
				c:ETH ?= {
					is_player = no
				}
			}
			change_relations = {
				country = c:ETH
				value = 80
			}
		}
	}
	ai_chance = {
		value = 100
	}
}
