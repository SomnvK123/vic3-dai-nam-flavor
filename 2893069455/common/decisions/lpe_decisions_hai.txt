acabar_descontento_dominicano = {
	is_shown = {
		any_scope_state = {
			state_region = s:STATE_SANTO_DOMINGO
			has_modifier = resistencia_caribenna
		}
	}
	possible = {
		NOT = {
			any_political_movement = {
				custom_tooltip = {
					text = je_independencia_rep_dominicana_hai_tt1_NOT
					exists = culture
					culture = cu:caribeno
				}
			}
		}
		OR = {
			cultural_acceptance_value = {
				target = cu:caribeno
				value >= acceptance_status_3
			}
			AND = {
				loyalist_fraction = {
					value >= 0.2
					culture = cu:caribeno
				}
				radical_fraction = {
					value <= 0.05
					culture = cu:caribeno
				}
				custom_tooltip = {
					text = coronacion_de_isabel_tt
					government_legitimacy > 74
				}
			}
		}
	}
	when_taken = {
		random_scope_state = {
			limit = {
				state_region = s:STATE_SANTO_DOMINGO
				has_modifier = resistencia_caribenna
			}
			remove_modifier = resistencia_caribenna
		}
	}
	ai_chance = {
		value = 100
	}
}
