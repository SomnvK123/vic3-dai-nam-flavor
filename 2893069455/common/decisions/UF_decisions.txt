end_british_concessions_decision = {
	#结束英国的让步（#购买烟草特许权日志--事件21选a获得50年buff）
	is_shown = {
		OR = {
			c:PER ?= this
			AND = {
				exists = c:QAJ
				c:QAJ ?= this
				NOT = {
					exists = c:PER
				}
			}
		}
		has_modifier = persia_tobacco_concessions_modifier
	}
	possible = {
		NOT = {
			has_journal_entry = je_tobacco_protests
		}
		#烟草抗议
	}
	when_taken = {
		custom_tooltip = historical_tt
		if = {
			limit = {
				has_journal_entry = je_persia_great_game
			}
			custom_tooltip = british_progress_increase_300_tt
			change_global_variable = {
				name = british_progress_persia
				add = 3
			}
		}
		remove_modifier = persia_tobacco_concessions_modifier
		c:GBR ?= {
			remove_modifier = britain_tobacco_concessions_modifier
		}
		trigger_event = {
			id = 03_persia_events.24
			popup = yes
		}
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				exists = c:GBR
				game_date > 1905.7.1
			}
			add = 100
		}
	}
}
