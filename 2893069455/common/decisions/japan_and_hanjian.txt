wwj_wjw = {
	#伪南京政府
	is_shown = {
		exists = c:WWJ
		root = c:WWJ
		game_date > 1900.1.1
		NOT = {
			has_variable = is_WWJ1
		}
	}
	possible = {
		game_date > 1900.1.1
	}
	when_taken = {
		activate_law = law_type:law_presidential_republic
		c:JAP ?= {
			create_diplomatic_pact = {
				country = c:WWJ
				type = puppet
			}
		}
		create_character = {
			first_name = ecchi_Jingwei
			last_name = ecchi_Wang_2
			historical = yes
			ruler = yes
			culture = cu:han
			female = no
			religion = rel:confucian
			ideology = ideology_jingoist_leader
			interest_group = ig_petty_bourgeoisie
			dna = ecchi_dna_wang_jingwei
			birth_date = 1883.5.4
			traits = {
				ambitious
				bandit
			}
		}
		set_variable = {
			name = is_WWJ1
			value = yes
		}
	}
	ai_chance = {
		value = 90
	}
}

lbs_lbs = {
	is_shown = {
		exists = c:LBS
		root = c:LBS
		NOT = {
			has_variable = is_PUPPET1
		}
		OR = {
			AND = {
				exists = c:CHI
				c:CHI ?= {
					has_variable = lingbei_sheng_end
				}
			}
			AND = {
				exists = c:ROC
				c:ROC ?= {
					has_variable = lingbei_sheng_end
				}
			}
		}
	}
	possible = {
		NOT = {
			has_law_or_variant = law_type:law_council_republic
		}
	}
	when_taken = {
		if = {
			limit = {
				exists = c:CHI
				c:CHI ?= {
					has_variable = lingbei_sheng_end
				}
			}
			c:CHI ?= {
				create_diplomatic_pact = {
					country = c:LBS
					type = colony
					is_initiator = yes
				}
			}
		}
		if = {
			limit = {
				exists = c:ROC
				c:ROC ?= {
					has_variable = lingbei_sheng_end
				}
			}
			c:ROC ?= {
				create_diplomatic_pact = {
					country = c:LBS
					type = colony
					is_initiator = yes
				}
			}
		}
		set_variable = {
			name = is_PUPPET1
			value = yes
		}
	}
	ai_chance = {
		value = 100
	}
}

DET_DET = {
	is_shown = {
		exists = c:CHI
		c:CHI ?= this
		c:DEI ?= {
			is_subject_of = c:CHI
		}
		NOT = {
			has_variable = is_DET1
		}
	}
	possible = {
	}
	when_taken = {
		if = {
			limit = {
				c:DEI ?= {
					country_has_primary_culture = cu:dutch
				}
			}
			c:DEI ?= {
				add_primary_culture = cu:han
				remove_primary_culture = cu:dutch
			}
		}
		set_variable = {
			name = is_DET1
			value = yes
		}
	}
	ai_chance = {
		value = 90
	}
}
