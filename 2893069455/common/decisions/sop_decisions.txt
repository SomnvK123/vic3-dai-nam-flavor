remove_aus_state_trait = {
	is_shown = {
		NOT = {
			OR = {
				c:AUS = ROOT
				was_formed_from = AUS
                                is_diplomatic_play_enemy_of = c:AUS
			        has_war_with = c:AUS
				top_overlord = {
					country_has_primary_culture = cu:south_german
				}
				}
		}
		any_state = {
			owner = ROOT
			custom_tooltip = {
				text = has_aus_state_trait_tt
				is_aus_crown = yes
			}
		}
	}
	possible = {
		NOT = {
			OR = {
				c:AUS = ROOT
				was_formed_from = AUS
                                is_diplomatic_play_enemy_of = c:AUS
			        has_war_with = c:AUS
				top_overlord = {
					country_has_primary_culture = cu:south_german
				}
				}
		}
		any_state = {
			owner = ROOT
			custom_tooltip = {
				text = has_aus_state_trait_tt
				is_aus_crown = yes
			}
		}
	}
	when_taken = {
		every_scope_state = {
			limit = {
				is_aus_crown = yes
				owner = ROOT
			}
			state_region = {
				custom_tooltip = {
					text = je_remove_aus_state_trait_tt
					soe_aus_remove_state_trait = yes
				}
			}
		}
	}
	ai_chance = {
		value = 100
	}
}
