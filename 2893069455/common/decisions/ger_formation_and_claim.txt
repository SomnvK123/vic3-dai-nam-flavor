join_german_puppet_decision = {
	# 德意志小国如果研究出了民族主义且加入了关税同盟，就变傀儡
	is_shown = {
		has_technology_researched = nationalism
		country_rank < rank_value:major_power
		NOT = {
			c:UBD ?= this
		}
		NOT = {
			c:AUS ?= this
		}
		NOT = {
			c:PRU ?= this
		}
		is_subject = no
		OR = {
			country_has_primary_culture = cu:north_german
			country_has_primary_culture = cu:south_german
		}
		is_ai = yes
		OR = {
			exists = c:NGF
			c:PRU ?= {
				NOT = {
					has_variable = supporting_great_German_var
				}
			}
			AND = {
				c:PRU ?= {
					has_variable = supporting_great_German_var
					has_modifier = germany_country_modifier
				}
			}
		}
	}
	possible = {
		is_subject = no
		is_ai = yes
		lca_is_in_diplomatic_play_or_war = no
		AND = {
			c:PRU ?= {
				is_unification_candidate = GER
			}
			c:AUS ?= {
				NOT = {
					is_unification_candidate = GER
				}
			}
		}
	}
	when_taken = {
		if = {
			limit = {
				exists = c:PRU
			}
			c:PRU ?= {
				create_diplomatic_pact = {
					country = root
					type = puppet
				}
			}
			change_relations = {
				country = root
				value = 50
			}
			add_modifier = {
				name = modifier_support_from_net
				years = 50
			}
		}
		if = {
			limit = {
				exists = c:NGF
			}
			c:NGF ?= {
				create_diplomatic_pact = {
					country = root
					type = puppet
				}
			}
			change_relations = {
				country = root
				value = 50
			}
			add_modifier = {
				name = modifier_support_from_net
				years = 50
			}
		}
	}
	ai_chance = {
		value = 100
	}
}
