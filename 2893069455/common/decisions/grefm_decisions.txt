grefm_corinth_canal_survey_decision = {
	is_shown = {
		owns_entire_state_region = STATE_PELOPONNESE
		NOR = {
			any_scope_state = {
				has_building = building_corinth_canal
			}
			has_variable = grefm_corinth_canal_var
			has_variable = grefm_corinth_survey_complete
		}
	}
	possible = {
		produced_bureaucracy > 200
		owns_entire_state_region = STATE_PELOPONNESE
		has_technology_researched = railways
		NOT = {
			has_journal_entry = je_grefm_corinth_canal_survey
		}
	}
	when_taken = {
		set_variable = {
			name = grefm_corinth_canal_var
			value = 0
		}
		add_journal_entry = {
			type = je_grefm_corinth_canal_survey
		}
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				produced_bureaucracy > 500
			}
			add = 100
		}
		if = {
			limit = {
				OR = {
					is_diplomatic_play_committed_participant = yes
					is_at_war = yes
				}
			}
			add = -1000
		}
	}
}

grefm_lake_kopais_decision = {
	is_shown = {
		has_variable = grefm_lake_kopais_option
		NOT = {
			has_variable = grefm_lake_kopais_option_taken
		}
		owns_entire_state_region = STATE_ATTICA
	}
	possible = {
		owns_entire_state_region = STATE_ATTICA
	}
	when_taken = {
		set_variable = grefm_lake_kopais_option_taken
		set_global_variable = {
			name = grefm_lake_kopais_var
			value = 0
		}
		add_modifier = {
			name = modifier_grefm_reclaiming_land
		}
		add_journal_entry = {
			type = je_grefm_land_reclaimation
		}
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				bureaucracy > 150
			}
			add = 100
		}
	}
}

grefm_annex_epirus_decision = {
	is_shown = {
		NOT = {
			has_variable = grefm_epirus_var
		}
		country_has_primary_culture = cu:greek
		is_subject = no
		exists = c:EPI
		c:EPI ?= {
			OR = {
				is_subject = no
				is_subject_of = ROOT
			}
			is_player = no
		}
	}
	possible = {
		is_at_war = no
		has_technology_researched = nationalism
		c:EPI ?= {
			relations:root >= relations_threshold:cordial
			is_at_war = no
		}
		c:EPI ?= {
			NOT = {
				has_truce_with = ROOT
			}
		}
	}
	when_taken = {
		set_variable = grefm_epirus_var
		custom_tooltip = treaty_of_eprius_event_tt
		hidden_effect = {
			trigger_event = {
				id = grefm.221				# EPI Annexation
				days = 0
				popup = yes
			}
		}
	}
	ai_chance = {
		value = 10
	}
}
