chile_get_easter_island = {
	is_shown = {
		game_date < 1900.1.1
		exists = c:CHL
		c:CHL = ROOT
		NOT = {
			has_variable = we_get_easter_island
		}
		NOT = {
			any_scope_state = {
				state_region = s:STATE_TAHITI
			}
		}
	}
	possible = {
		game_date <= 1900.1.1
		game_date >= 1888.1.1
		NOT = {
			any_scope_state = {
				state_region = s:STATE_TAHITI
			}
		}
		s:STATE_TAHITI = {
			any_scope_state = {
				owner ?= {
					relations:root >= relations_threshold:neutral
					owns_entire_state_region = STATE_TAHITI
					is_player = no
				}
			}
		}
	}
	when_taken = {
		set_variable = we_get_easter_island
		s:STATE_TAHITI = {
			set_owner_of_provinces = {
				country = ROOT
				provinces = {
					x9A79DC
				}
			}
		}
		if = {
			limit = {
				c:CHL ?= {
					is_player = no
				}
			}
			every_state = {
				limit = {
					owner ?= {
						c:CHL ?= this
					}
					OR = {
						region = sr:region_la_plata
						region = sr:region_andes
						region = sr:region_oceania
					}
				}
				set_state_type = incorporated
			}
		}
		hidden_effect = {
			every_country = {
				limit = {
					has_interest_marker_in_region = sr:region_oceania
				}
				NOT = {
					c:CHL ?= this
				}
				trigger_event = {
					id = clever_dsk.37
				}
			}
		}
	}
	ai_chance = {
		value = 100
	}
}

zhang_zuo_lin_great = {
	is_shown = {
		c:JAP ?= {
			NOT = {
				has_variable = jiuyibashibian_decision
			}
		}
		game_date < 1928.6.4
		exists = c:MCH
		c:MCH = ROOT
		NOT = {
			has_global_variable = zhang_zuo_lin_great
		}
		NOT = {
			any_scope_character = {
				has_template = ecchi_chi_zhang_zuolin_character_template
			}
		}
	}
	possible = {
		game_date < 1928.6.4
		game_date > 1911.1.1
	}
	when_taken = {
		set_global_variable = zhang_zuo_lin_great
                                add_primary_culture = cu:han
		activate_law = law_type:law_autocracy
		if = {
			limit = {
				any_country = {
					any_scope_character = {
						has_template = ecchi_chi_zhang_zuolin_character_template
					}
				}
			}
			random_country = {
				limit = {
					any_scope_character = {
						has_template = ecchi_chi_zhang_zuolin_character_template
					}
				}
				random_scope_character = {
					limit = {
						has_template = ecchi_chi_zhang_zuolin_character_template
					}
					transfer_character = ROOT
					set_character_as_ruler = yes
					set_as_interest_group_leader = yes
				}
			}
		}
		else_if = {
			limit = {
				any_character_in_exile_pool = {
					has_template = ecchi_chi_zhang_zuolin_character_template
				}
			}
			random_character_in_exile_pool = {
				limit = {
					has_template = ecchi_chi_zhang_zuolin_character_template
				}
				transfer_character = ROOT
				set_character_as_ruler = yes
				set_as_interest_group_leader = yes
			}
		}
		else = {
			create_character = {
				template = ecchi_chi_zhang_zuolin_character_template
				ruler = yes
				ig_leader = yes
								on_created = {
				set_career_length = {
						years = 50
						random_range = { 1.0 1.5 }
				}
				}
			}
		}
		if = {
			limit = {
                                                                exists = c:SMC
				c:SMC ?= {
					is_player = no
					lca_is_in_diplomatic_play_or_war = no
					OR = {
						is_subject = no
						is_subject_of = ROOT
					}
				}
			}
			annex = c:SMC
		}
		if = {
			limit = {
                                                                exists = c:NMC
				c:NMC ?= {
					is_player = no
					lca_is_in_diplomatic_play_or_war = no
					OR = {
						is_subject = no
						is_subject_of = ROOT
					}
				}
			}

		}
