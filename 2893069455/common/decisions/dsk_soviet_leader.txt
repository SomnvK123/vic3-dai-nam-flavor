clever_dsk.91_dec = {
	is_shown = {
		exists = c:RUS
		c:RUS = ROOT
		game_date > 1836.1.1
		game_date < 1858.1.1
		exists = c:AUS
		exists = c:HUN
		is_subject = no
		has_law_or_variant = law_type:law_monarchy
		NOT = {
			has_war_with = c:HUN
		}
		c:HUN ?= {
			has_war_with = c:AUS
			is_player = no
		}
		NOT = {
			has_variable = russia_kill_hungary
		}
	}
	possible = {
		c:AUS ?= {
			is_at_war = yes
			relations:root >= relations_threshold:cold
			any_scope_treaty = {
				binds = c:RUS
				any_scope_article = {
					has_type = alliance
				}
			}
		}
	}
	when_taken = {
		set_variable = russia_kill_hungary
		trigger_event = {
			id = clever_dsk.91
			days = 1
			popup = yes
		}
	}
	ai_chance = {
		value = 100
	}
}

back_to_moscow = {
	is_shown = {
		OR = {
			AND = {
				exists = c:RUS
				c:RUS = ROOT
				has_law_or_variant = law_type:law_council_republic
				NOT = {
					exists = c:SOV
				}
			}
			AND = {
				exists = c:SOV
				c:SOV = ROOT
				NOT = {
					exists = c:RUS
				}
			}
		}
		NOT = {
			has_variable = back_to_moscow
		}
		owns_entire_state_region = STATE_MOSCOW
	}
	possible = {
		owns_entire_state_region = STATE_MOSCOW
		is_at_war = no
	}
	when_taken = {
		set_variable = back_to_moscow
		set_capital = STATE_MOSCOW
		capital ?= {
			save_scope_as = russia_old_capital
		}
		scope:russia_old_capital ?= {
			add_modifier = {
				name = cradle_of_revolution
			}
		}
	}
	ai_chance = {
		value = 100
	}
}
