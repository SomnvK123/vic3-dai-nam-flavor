chi_republic_alt_roc_flag_enabled = {
	is_shown = {
		has_variable = whitesun
		NOT = {
			has_variable = chi_republic_alt_roc_flag_enabled_flag
		}
		NOT = {
			has_law_or_variant = law_type:law_council_republic
		}
		OR = {
			has_law_or_variant = law_type:law_presidential_republic
			has_law_or_variant = law_type:law_parliamentary_republic
			has_law_or_variant = law_type:law_corporate_state
		}
	}
	possible = {
		NOT = {
			has_variable = chi_republic_alt_roc_flag_enabled_flag
		}
		OR = {
			has_law_or_variant = law_type:law_presidential_republic
			has_law_or_variant = law_type:law_parliamentary_republic
			has_law_or_variant = law_type:law_corporate_state
		}
	}
	when_taken = {
		custom_tooltip = {
			text = wuzugonghe_bairiqi
			set_variable = chi_republic_alt_roc_flag_enabled_flag
			#remove_variable = chi_republic_vanilla_flag_enabled_flag
			remove_variable = chi_republic_alt_kmt_flag_enabled_flag
		}
	}
	ai_chance = {
		value = 100
	}
}

}
