je_usa_ruler_white_house_check_sgui = {
    scope = character
    saved_scopes = { usa_ruler_white_house_state }

    is_valid = {
        dsk_usa_ruler_white_house_trigger = yes
        dsk_gbr_ruler_buckingham_palace_trigger = no
        dsk_fra_ruler_versailles_trigger = no
        dsk_rus_ruler_kremlin_trigger = no
        trigger_if = {
            limit = {
                scope:usa_ruler_white_house_state = yes
            }
            owner ?= {
                c:USA ?= this
            }
        }
    }
}
