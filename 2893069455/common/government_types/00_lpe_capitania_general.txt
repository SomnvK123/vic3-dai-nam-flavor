REPLACE_OR_CREATE:gov_capitania_general = {
	# gov_colonial_administration 
	transfer_of_power = dictatorial
	new_leader_on_reform_government = no
	male_ruler = "RULER_TITLE_CAPITAN_GENERAL"
	female_ruler = "RULER_TITLE_CAPITAN_GENERAL"
	possible = {
		has_law_or_variant = law_type:law_presidential_republic
		exists = c:SPA
		is_subject_of = c:SPA
		is_subject_type = subject_type_colony
		OR = {
			is_country_type = colonial
			any_scope_treaty = {
				binds = c:SPA
				any_scope_article = {
					has_type = provincia_espannola
				}
			}
		}
		top_overlord = {
			NOT = {
				is_country_type = colonial
			}
		}
		any_primary_culture = {
					has_discrimination_trait = language_hispanophone
				}
		is_not_in_geographic_region = geographic_region_europe
	}
	on_government_type_change = {
		change_to_presidential_elective = yes
	}
	on_post_government_type_change = {
		post_change_to_presidential_elective = yes
	}
}
