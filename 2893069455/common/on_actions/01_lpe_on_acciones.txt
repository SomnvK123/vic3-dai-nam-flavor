					}
				}
			}

			}
		}
	}

lamour_camanna_termina = {
	effect = {
		if = {
			limit = {
				c:SPA ?= this
			}
			every_interest_group = {
				if = {
					limit = {
						has_modifier = apoyo_de_territorio_de_ultramar
					}
					remove_modifier = apoyo_de_territorio_de_ultramar
				}
				if = {
					limit = {
						has_modifier = apoyo_menor_de_territorio_de_ultramar
					}
					remove_modifier = apoyo_menor_de_territorio_de_ultramar
				}
				if = {
					limit = {
						has_modifier = sin_apoyo_de_ultramar
					}
					remove_modifier = sin_apoyo_de_ultramar
				}
				if = {
					limit = {
						has_modifier = apoyo_de_ultramar
					}
					remove_modifier = apoyo_de_ultramar
				}
			}
			every_subject_or_below = {
				limit = {
					is_direct_subject_of = root
					OR = {
						country_has_primary_culture = cu:caribeno
						country_has_primary_culture = cu:spanish
					}
					has_diplomatic_pact = {
						who = root
						type = representacion_en_cortes
					}
				}
				every_interest_group = {
					limit = {
						has_modifier = diputados_en_las_cortes
					}
					remove_modifier = diputados_en_las_cortes
				}
				top_overlord = {
					trigger_event = {
						id = esp.34
						days = 2
						popup = yes
					}
				}
			}
		}
	}
}

lamour_guerra_civil = {
	effect = {
		if = {
			limit = {
				exists = c:SPC
			}
			if = {
				# Cristinos
				limit = {
					root = c:SPA
					has_war_with = c:SPC
				}
				scope:target = {
					add_journal_entry = {
						type = je_cristininos_victory
					}
				}
				C:SPC ?= {
					create_diplomatic_play = {
						type = dp_revolution
						target_state = scope:target.capital
						annex_as_civil_war = yes
						escalation = 5
					}
				}
			}
			else_if = {
				# Carlistas 
				limit = {
					root = c:SPC
					has_war_with = c:SPA
				}
				scope:target = {
					add_journal_entry = {
						type = je_carlist_victory
					}
				}
				C:SPA ?= {
					create_diplomatic_play = {
						type = dp_revolution
						target_state = scope:target.capital
						annex_as_civil_war = yes
						escalation = 5
					}
				}
			}
		}
	}
}
