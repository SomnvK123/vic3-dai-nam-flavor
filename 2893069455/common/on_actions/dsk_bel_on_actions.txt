on_game_started = {
	on_actions = {
		dsk_bel_startup_action
	}
}

dsk_bel_startup_action = {
	effect = {
		c:BEL ?= {
			if = {
				limit = { NOT = { has_variable = dsk_bel_startup } }
				set_variable = dsk_bel_startup
				dsk_bel_initialize_complex_variables = yes
				add_journal_entry = { type = je_dsk_bel_national_question }
				add_journal_entry = { type = je_dsk_bel_independence }
				add_journal_entry = { type = je_dsk_bel_workshop_of_europe }
				dsk_bel_recalculate_monthly_display_deltas = yes
			}
		}
	}
}

on_monthly_pulse_country = {
	effect = {
		if = {
			limit = {
				exists = c:BEL
				c:BEL ?= this
			}
			dsk_bel_monthly_journal_unlocks = yes
			dsk_bel_monthly_royal_succession = yes
			dsk_bel_recalculate_monthly_display_deltas = yes
		}
	}
}
