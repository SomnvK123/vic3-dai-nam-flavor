corvee_pops = {
	value = 0
	if = {
		limit = {
			NOT = {
				has_law_or_variant = law_type:law_corvee
			}
		}
		every_scope_pop = {
			limit = {
				OR = {
					is_pop_type = peasants
					is_pop_type = farmers
					is_pop_type = clerks
					is_pop_type = shopkeepers
				}
				OR = {
					is_employed = no
					pop_employment_building_group = bg_subsistence_agriculture
				}
				this.culture = {
					is_primary_culture_of = ROOT
				}
			}
			add = 10
		}
	}
}

government_leader_james_polk_value = {
	value = 0
	if = {
		limit = {
                        NOT = {
			any_country = {
                                NOT = { THIS = ROOT }
				NOT = { is_subject_of = ROOT }
                                NOT = { is_country_type = decentralized }
                                any_scope_state = {
                                   OR = {
		                   state_region = s:STATE_WASHINGTON
		                   state_region = s:STATE_OREGON
		                   state_region = s:STATE_IDAHO
                                   }
		                   }
		}
		}
		}
		add = 1
	}
	if = {
		limit = {
                        NOT = {
			any_country = {
                                NOT = { THIS = ROOT }
				NOT = { is_subject_of = ROOT }
                                NOT = { is_country_type = decentralized }
                                any_scope_state = {
                                   OR = {
		                   state_region = s:STATE_CALIFORNIA
		                   state_region = s:STATE_NEVADA
		                   state_region = s:STATE_ARIZONA
		                   state_region = s:STATE_NEW_MEXICO
		                   state_region = s:STATE_COLORADO
		                   state_region = s:STATE_UTAH
                                   }
		                   }
		}
		}
		}
		add = 1
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_free_trade
			market_capital.market = {
				market_trade = {
				  value >= 8000
				}
			}
		}
		add = 1
	}
	if = {
		limit = {
                                owns_entire_state_region = STATE_TEXAS
                                any_scope_state = {
		                   state_region = s:STATE_TEXAS
				   is_incorporated = yes
		                }
		                }
		add = 1
	}
	if = {
		limit = {
			gold_reserves > 0
                        NOT = {
                                any_scope_state = {
				owner = ROOT
				is_incorporated = yes
                                NOT = { has_building = building_government_administration }
		                }
		                }
		                }
		add = 1
	}
}
