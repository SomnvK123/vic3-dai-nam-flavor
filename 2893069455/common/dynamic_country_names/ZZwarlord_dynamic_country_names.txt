REPLACE_OR_CREATE:FRS = {
	# 闽系
	dynamic_country_name = {
		name = FRS_warlord
		adjective = FRS_ADJ
		is_main_tag_only = yes
		priority = 0
		trigger = {
			scope:actor ?= {
				has_law_or_variant = law_type:law_presidential_republic
				has_law_or_variant = law_type:law_autocracy
				OR = {
					has_law_or_variant = law_type:law_peasant_levies
					has_law_or_variant = law_type:law_mass_conscription
					has_law_or_variant = law_type:law_professional_army
				}
			}
		}
	}
	dynamic_country_name = {
		name = FRS_warlord_great
		adjective = FRS_ADJ
		is_main_tag_only = yes
		priority = 10
		trigger = {
			scope:actor ?= {
				has_law_or_variant = law_type:law_presidential_republic
				has_law_or_variant = law_type:law_autocracy
				has_law_or_variant = law_type:law_peasant_levies
				any_scope_state = {
					OR = {
						#Maybe theres a better trigger?
						region = sr:region_north_china
						region = sr:region_south_china
						is_in_geographic_region = geographic_region_manchuria_old
						region = sr:region_central_asia
					}
					count >= 5
				}
			}
		}
	}
	dynamic_country_name = {
		name = FRS_republic
		adjective = FRS_ADJ
		is_main_tag_only = yes
		priority = 0
		trigger = {
			country_has_voting_franchise = yes
		}
	}
	dynamic_country_name = {
		name = FRS_imperial
		adjective = FRS_ADJ
		is_main_tag_only = yes
		priority = 0
		trigger = {
			scope:actor ?= {
				has_law_or_variant = law_type:law_monarchy
			}
			OR = {
				NOT = {
					exists = c:CHI
				}
				c:CHI ?= {
					NAND = {
						has_law_or_variant = law_type:law_monarchy
						country_has_primary_culture = cu:manchu
					}
				}
				has_variable = warlord_state
