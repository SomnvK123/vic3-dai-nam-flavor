pax_britannica_change = {
	type = country
	random_valid = no
	text = {
		trigger = {
			root.var:pax_britannica_points_change = 0
		}
		localization_key = pax_britannica_neutral
	}
	text = {
		trigger = {
			root.var:pax_britannica_points_change < 0
			root.var:pax_britannica_points_change > -0.05
		}
		localization_key = pax_britannica_negative
	}
	text = {
		trigger = {
			root.var:pax_britannica_points_change > 0
			root.var:pax_britannica_points_change < 0.05
		}
		localization_key = pax_britannica_positive
	}
	text = {
		trigger = {
			root.var:pax_britannica_points_change >= 0.05
		}
		localization_key = pax_britannica_very_positive
	}
	text = {
		trigger = {
			root.var:pax_britannica_points_change <= -0.05
		}
		localization_key = pax_britannica_very_negative
	}
}
