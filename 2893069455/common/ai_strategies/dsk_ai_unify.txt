ai_strategy_unify_yemen = {

	type = diplomatic
	wanted_army_ratio = 150000	# 1 barrack per 150k incorporated population
	undesirable_infamy_level = 50
	unacceptable_infamy_level = 100
	ideological_opinion_effect_mult = 0.5
	wargoal_maneuvers_fraction = {
		value = 0.1
	}
	diplomatic_play_neutrality = 25	# Added to base neutrality in diplomatic plays	
	diplomatic_play_boldness = 50	# Added to base boldness in diplomatic plays
	# Added to value from default strategy
	obligation_value = {
		value = 0
		if = {
			limit = {
				scope:target_country = {
					can_form_nation = YEM
				}
			}
			add = 25
		}
	}
	# Added to value from default strategy
	bravery = {
		value = 0.5
	}
	# Added to value from default strategy
	aggression = {
		value = 0.5
		if = {
			limit = {
				scope:target_country = {
					can_form_nation = YEM
				}
			}
			add = 10
		}
		# Less aggressive against countries we're just antagonistic against, rather than belligerent/domineering
		if = {
			limit = {
				has_attitude = {
					who = scope:target_country
					attitude = antagonistic
				}
			}
			multiply = 0.25
		}
	}
	building_group_weights = {
		bg_army = 1.5
	}
	wargoal_scores = {
		conquer_state = {
			if = {
				limit = {
					scope:target_state = {
						state_region = {
								is_homeland = cu:yemenite
						}
					}
					can_reach_target_state = yes
				}
				add = 100
			}
		}
		return_state = {
			if = {
				limit = {
					scope:target_state = {
						state_region = {
								is_homeland = cu:yemenite
						}
					}
					can_reach_target_state = yes
				}
				add = 100
			}
		}
	}
	secret_goal_scores = {
		befriend = {
			if = {
				limit = {
					scope:target_country = {
						can_form_nation = YEM
					}
				}
				add = 200
			}
		}
		dominate = {
			if = {
				limit = {
					scope:target_country = {
						can_form_nation = YEM
					}
				}
				add = 300
			}
		}
		conquer = {
			if = {
				limit = {
					scope:target_country = {
						is_subject = no
						can_form_nation = YEM
					}
				}
				add = 300
			}
		}
	}
	wargoal_weights = {
		conquer_state = 2
		return_state = 2
	}
	possible = {
		NOT = {
			exists = c:YEM
		}
		can_form_nation = YEM
		country_has_primary_culture = cu:yemenite
		has_game_rule = super_dsk_ai
	}
	weight = {
		value = 100
		if = {
			limit = {
				c:ARA ?= this
			}
			add = -1000
		}
		if = {
			limit = {
				is_subject = yes
			}
			add = -50
		}
		if = {
			limit = {
				NOT = {
					is_country_type = unrecognized
				}
			}
			add = 100
		}
		if = {
			limit = {
				country_rank = rank_value:unrecognized_regional_power
			}
			add = 25
		}
		if = {
			limit = {
				country_rank = rank_value:unrecognized_major_power
			}
			add = 50
		}
	}
}
