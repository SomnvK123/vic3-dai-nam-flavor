diet_of_hungary_promote_loyalists_button = {
	#提拔效忠派
	name = "diet_of_hungary_promote_loyalists_button"
	desc = "diet_of_hungary_promote_loyalists_button_desc"
	visible = {
		owner ?= {
			has_variable = aus_diet_of_hungary
		}
	}
	possible = {
		trigger_if = {
			limit = {
				has_variable = has_promote_loyalists_once_var
			}
			custom_tooltip = {
				text = cannot_promote_loyalists_twice_tt
				always = no
			}
		}
		custom_tooltip = {
			text = promote_loyalists_button_tt
			var:diet_of_hungary_loyalists < 200
		}
	}
	effect = {
		set_variable = {
			name = has_promote_loyalists_once_var
			months = 36
		}
		custom_tooltip = {
			text = promoting_loyalists_tt
		}
		custom_tooltip = {
			text = diet_of_hungary_loyalists_increase_25_tt
			change_variable = {
				name = diet_of_hungary_loyalists
				add = 25
			}
			change_variable = {
				name = diet_of_hungary_radicals
				subtract = 25
			}
		}
	}
	ai_chance = {
		value = 100
	}
}

bohemia_dual_monarchy_button = {
	#改组波西米亚议会
	name = "bohemia_dual_monarchy_button"
	desc = "bohemia_dual_monarchy_button_desc"
	visible = {
		has_variable = aus_dual_monarchy_var
	}
	possible = {
		custom_tooltip = {
			text = aus_dual_monarchy_region_reform_tt
			has_variable = aus_dual_monarchy_var
		}
		NOT = {
			has_modifier = obstruction_from_diet_of_hungary
		}
		any_scope_state = {
			custom_tooltip = {
				text = is_bohemia
				soe_aus_bohemia_trigger = yes
			}
			is_incorporated = yes
			any_scope_building = {
				is_building_type = building_urban_center
				level >= 15
			}
			any_scope_building = {
				is_building_type = building_railway
				level >= 10
			}
			NOT = {
				has_state_trait = state_trait_bohemian_crown_level_2
			}
		}
	}
	effect = {
		every_state_region = {
			limit = {
				is_homeland = cu:czech
			}
			custom_tooltip = {
				text = je_remove_aus_state_trait_tt
				remove_state_trait = state_trait_bohemian_crown_level_1
				remove_state_trait = state_trait_bohemian_crown_level_3
			}
			add_state_trait = state_trait_bohemian_crown_level_2
		}
		set_variable = bohemia_dual_monarchy_var
	}
}
