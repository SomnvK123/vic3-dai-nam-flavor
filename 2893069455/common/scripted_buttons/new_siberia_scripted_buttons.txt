develop_siberia_button_1 = {
	name = "develop_siberia_button_1"
	desc = "develop_siberia_button_1_desc"

	visible = { always = yes }

	cooldown = { days = 60 }

	possible = {
	any_scope_state = {
		OR = {
			region = sr:region_siberia
			region = sr:region_northeast_asia
		}
		is_incorporated = yes
		NOR = {
			has_state_trait = state_trait_siberian_frontier_level_1
			has_state_trait = state_trait_siberian_frontier_level_2
		}
		any_scope_building = {
			is_building_type = building_coal_mine
			level >= 5
		}
		any_scope_building = {
			is_building_type = building_logging_camp
			level >= 5
		}
	}
}

	ai_chance = {
		value = 20
		if = {
			limit = {
				OR = {
					is_at_war = yes
					in_default = yes
				}
			}
			add = -100
		}
	}

	effect = {
		random_scope_state = {
			limit = {
				OR = {
			    region = sr:region_siberia
			region = sr:region_northeast_asia
		}
		is_incorporated = yes
		NOR = {
				has_state_trait = state_trait_siberian_frontier_level_1
				has_state_trait = state_trait_siberian_frontier_level_2
			}
				any_scope_building = {
					is_building_type = building_coal_mine
					level >= 5
				}
				any_scope_building = {
					is_building_type = building_logging_camp
					level >= 5
				}
			}
			save_scope_as = siberia_development_level_one_state_scope

			show_as_tooltip = {
				state_region = {
					add_state_trait = state_trait_siberian_frontier_level_1
				}
			}

			root = {
				trigger_event = {
					id = siberia_development.1
					popup = yes
				}
			}
		}
	}
}

			}

				trigger_event = {
					id = siberia_development.2
					popup = yes

	}

je_develop_siberia_resettlement_button = {
    name = je_develop_siberia_resettlement_button
    desc = je_develop_siberia_resettlement_button_desc

    visible = {
        always = yes
    }

	cooldown = { years = 3 }

    possible = {
        custom_tooltip = {
            text = je_develop_siberia_resettlement_button_possible_tt

            hidden_trigger = {
                any_scope_state = {
                    is_incorporated = yes
                    NOT = { is_pan_siberia = yes }
                    state_population > 10000
                }

                any_scope_state = {
                    is_incorporated = yes
                    is_pan_siberia = yes
                }
            }
        }
    }

    effect = {
        random_scope_state = {
            limit = {
                is_incorporated = yes
                NOT = { is_pan_siberia = yes }
                state_population > 10000
            }
            save_scope_as = siberia_resettlement_source
        }

        random_scope_state = {
            limit = {
                is_incorporated = yes
                is_pan_siberia = yes
            }
            save_scope_as = siberia_resettlement_target
        }

        trigger_event = {
            id = new_siberia_events.1
            popup = yes
        }
    }


    ai_chance = { value = 0 }
}
