line_struggle_1_communist_button = {
	name = "communist_button"
	desc = "communist_button_desc"
	visible = {
		has_technology_researched = socialism
		#has_variable = chose_initial_line_struggle_1 
	}
	possible = {
		custom_tooltip = {
			text = not_already_supporting_communist
			NOT = {
				has_variable = supporting_communist_var
			}
		}
	}
	effect = {
		set_variable = supporting_communist_var
		remove_variable = supporting_anarchist_var
		remove_variable = supporting_social_democrat_var

		custom_tooltip = {
			text = supporting_communist_tt
		}
	}
}
