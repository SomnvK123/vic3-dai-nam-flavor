balkan_1877_small_crisis_button = {
	name = "balkan_1877_small_crisis_button"
	desc = "balkan_1877_small_crisis_button_desc"
	visible = {
		c:RUS ?= this
		exists = c:TUR
		c:TUR ?= {
			has_journal_entry = je_eqfm_main
		}
	}
	possible = {
		year >= 1875
		year < 1880
		NOT = {
			has_global_variable = eqfm_crisis_happened
		}
		NOT = {
			is_at_war = yes
		}
		custom_tooltip = {
			text = balkan_1877_small_crisis_button_tt
			hidden_trigger = {
				always = yes
			}
		}
	}
	effect = {
		if = {
			limit = {
				NOT = {
					has_journal_entry = je_eqfm_sub
				}
			}
			add_journal_entry = {
				type = je_eqfm_sub
			}
		}
		remove_variable = eqfm_pro_turk_var
		remove_variable = eqfm_neutral_var
		set_variable = eqfm_anti_turk_var
		set_variable = eqfm_accelerating_tensions_var
		if = {
			limit = {
				NOT = {
					has_global_variable = eqfm_crisis_var
				}
			}
			set_global_variable = {
				name = eqfm_crisis_var
				value = 75
			}
		}
		else = {
			change_global_variable = {
				name = eqfm_crisis_var
				add = 25
			}
		}
		change_global_variable = {
			name = eqfm_tension_var
			add = 40
		}
		clamp_global_variable = {
			name = eqfm_tension_var
			min = 0
			max = 500
		}
		trigger_event = {
			id = eqfm.100
		}
	}
	ai_chance = {
		value = 0
	}
}
