let_our_alliance_stronger_button = {
	name = "let_our_alliance_stronger_button"
	desc = "let_our_alliance_stronger_button_desc"
	visible = {
		exists = c:AUS
		exists = c:GER
		c:GER ?= this
	}
	possible = {
		custom_tooltip = {
			text = enough_aus_ger_union_75_var
			var:aus_ger_union_var >= 75
		}
		NOT = {
			is_diplomatic_play_enemy_of = c:AUS
			has_war_with = c:AUS
                                                has_law_or_variant = law_type:law_isolationism
		}
		c:AUS ?= {
                                                relations:root >= 60
			NOT = {
                                                                has_law_or_variant = law_type:law_isolationism
				has_modifier = modifier_let_our_alliance_stronger
			}
		}
	}
	effect = {
		c:AUS ?= {
			add_modifier = {
				name = modifier_let_our_alliance_stronger
				years = 5
				is_decaying = yes
			}
		}
		custom_tooltip = aus_ger_union_75_var
		change_variable = {
			name = aus_ger_union_var
			add = -75
		}
	}
	ai_chance = {
		value = 25
	}
}

hungarian_rebels_button = {
	name = "hungarian_rebels_button"
	desc = "hungarian_rebels_button_desc"
	visible = {
		always = yes
	}
	possible = {
		OR = {
			has_law_or_variant = law_type:law_ethnostate
			has_law_or_variant = law_type:law_national_supremacy
			custom_tooltip = {
				text = voting_franchise_tt
				country_has_voting_franchise = no
			}
		}
		OR = {
			country_or_subject_owns_entire_state_region = STATE_CENTRAL_HUNGARY
			country_or_subject_owns_entire_state_region = STATE_TRANSDANUBI
		}
		custom_tooltip = {
			text = hungarian_rebels_button_tt_2
			NOT = {
				has_variable = hungarian_rebels_cooldown
			}
			has_variable = The_situation_of_Hungary
		}
	}
	effect = {
		trigger_event = {
			id = dsk_hungary_events.4
			days = 0
			popup = yes
		}
		custom_tooltip = hungarian_rebels_button_tt_1
		set_variable = {
			name = hungarian_rebels_cooldown
			days = normal_modifier_time
		}
		show_as_tooltip = {
			add_modifier = {
				name = funding_hungarian_radicals
				days = normal_modifier_time
			}
		}
	}
}

join_the_aus_ger_war_button = {
	name = "join_the_aus_ger_war_button"
	desc = "join_the_aus_ger_war_button_desc"
	visible = {
		exists = c:AUS
		any_neighbouring_state = {
			owner = c:AUS
		}
	}
	cooldown = { years = 20 }
	possible = {
		custom_tooltip = {
			text = enough_aus_ger_union_100_var
			var:aus_ger_union_var >= 100
		}
		c:AUS ?= {
		                is_subject = no
                        is_player = no
                        relations:root >= 50
        }
		NOT = {
			is_diplomatic_play_enemy_of = c:AUS
		}
		NOT = {
			has_war_with = c:AUS
		}
	}
	effect = {
		custom_tooltip = aus_ger_union_100_var
		change_variable = {
			name = aus_ger_union_var
			add = -100
		}
		if = {
				limit = {
					NOT = {
						has_diplomatic_pact = {
							who = c:AUS
							type = qwf_interest_alliance
						}
					}
				}
				create_diplomatic_pact = {
					country = c:AUS
					type = qwf_interest_alliance
				}
		}
		c:AUS ?= {
			set_owes_obligation_to = {
				country = ROOT
				setting = yes
			}
		}
		set_owes_obligation_to = {
				country = c:AUS
				setting = yes
		}
	}
	ai_chance = {
		value = 10
	}
}
