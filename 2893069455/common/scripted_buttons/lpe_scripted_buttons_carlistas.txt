expedicion_gomez = {
	name = expedicion_gomez_name
	desc = expedicion_gomez_desc
	visible = {
		# is_revolutionary = no
		is_at_war = yes
		NOT = {
			has_variable = expedicion_gomez_vari
		}
	}
	possible = {
		has_war_with = c:SPA
		any_scope_general = {
			is_ruler_of_own_country = no
		}
	}
	effect = {
		set_variable = expedicion_gomez_vari
		trigger_event = {
			id = car.1
			popup = yes
		}
	}
	ai_chance = {
		value = 100		#都要被灭了还不发起远征？
		if = {
		limit = {
				has_variable = expedicion_real_vari
			}
			add = 30
		}
	}
}

expedicion_real = {
	name = expedicion_real_name
	desc = expedicion_real_desc
	visible = {
		is_revolutionary = no
		is_at_war = yes
		NOT = {
			has_variable = expedicion_real_vari
		}
	}
	possible = {
		has_war_with = c:SPA
		ruler ?= {
			has_role_of_type = general
		}
	}
	effect = {
		set_variable = expedicion_real_vari
		ruler ?= {
			add_modifier = {
				name = marcha_real
				months = 20
			}
		}
		# custom_tooltip = expedicion_real_tt
		c:SPA ?= {
			trigger_event = {
				id = esp.9
				popup = yes
			}
		}
	}
	ai_chance = {
		value = 7
		if = {
		limit = {
				has_variable = expedicion_gomez_vari
			}
			add = 30
		}
	}
}

alzamiento_carlista = {
	name = alzamiento_carlista_name
	desc = alzamiento_carlista_desc
	visible = {
		# is_revolutionary = no
		is_at_war = yes
	}
	possible = {
		has_war_with = c:SPA
		custom_tooltip = {
			text = alzamiento_carlista_tt
			je:je_carlist_victory = {
				scope:journal_entry = {
					"scripted_bar_progress(alzamientos_carlistas_bar)" >= 60
				}
			}
		}
		custom_tooltip = {
			text = alzamiento_carlista_tt2
			OR = {
				can_alzamiento_maestrazgo_car = yes
				can_alzamiento_aragon_car = yes
				can_alzamiento_cataluna_car = yes
			}
		}
	}
	effect = {
		custom_tooltip = {
			text = alzamiento_carlista_tt1
			je:je_carlist_victory = {
				add_progress = {
					value = -60
					name = alzamientos_carlistas_bar
				}
			}
			if = {
				limit = {
					exists = c:SPA
					c:SPA ?= {
						has_journal_entry = je_cristininos_victory
					}
				}
				c:SPA ?= {
					je:je_cristininos_victory ?= {
						add_progress = {
							value = -60
							name = alzamientos_carlistas_bar
						}
					}
				}
			}
		}
		trigger_event = {
			id = car.4
			popup = yes
		}
	}
	ai_chance = {
		value = 20
	}
}
