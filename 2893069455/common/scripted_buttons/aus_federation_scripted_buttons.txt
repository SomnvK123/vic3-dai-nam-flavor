bohemia_federation_button = {
	name = "bohemia_federation_button"
	desc = "bohemia_federation_button_desc"
	visible = {
		has_variable = aus_federation_var
	}
	possible = {
		custom_tooltip = {
			text = aus_federation_reform_tt
			has_variable = aus_federation_var
		}
		NOT = {
			has_modifier = obstruction_from_diet_of_hungary
		}
		any_scope_state = {
			custom_tooltip = {
				text = is_bohemia
				soe_aus_bohemia_trigger = yes
			}
			is_incorporated = yes
			any_scope_building = {
				is_building_type = building_urban_center
				level >= 20
			}
			any_scope_building = {
				is_building_type = building_railway
				level >= 8
			}
			NOT = {
				has_state_trait = state_trait_bohemian_crown_level_3
			}
		}
	}
	effect = {
		trigger_event = {
			id = dnf_events.2
			popup = yes
		}
		add_radicals = {
			culture = cu:hungarian
			value = 0.05
		}
		add_loyalists = {
			culture = cu:czech
			value = 0.05
		}
		every_scope_state = {
			limit = {
				soe_aus_bohemia_trigger = yes
			}
			state_region = {
				custom_tooltip = {
					text = je_remove_aus_state_trait_tt
					soe_aus_remove_state_trait = yes
				}
				add_state_trait = state_trait_bohemian_crown_level_3
			}
		}
		set_variable = bohemia_federation_var
	}
}
