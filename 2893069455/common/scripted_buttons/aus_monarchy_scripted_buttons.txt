		owner ?= {
			has_variable = aus_leading_ig_group
		}
	}
	possible = {
		trigger_if = {
			limit = {
				has_variable = has_selected_Klowrat_once_var
			}

				text = cannot_choose_Klowrat_twice_tt
				always = no
			}
		}

			is_in_government = yes

			text = not_already_supporting_Klowrat
			NOT = {
				has_variable = supporting_Klowrat_var
			}
		}
	}
	selected = {
		has_variable = supporting_Klowrat_var
	}
	effect = {
		set_variable = supporting_Klowrat_var
		set_variable = has_selected_Klowrat_once_var
		remove_variable = supporting_Metternich_var
		custom_tooltip = {
			text = supporting_Klowrat_tt
		}
		custom_tooltip = {
			text = Klowrat_progress_increase_50_tt
			je:je_habsburg_monarchy = {
				add_progress = {
					value = 50
					name = aus_intelligentsia_progress_bar
				}
			}
			change_variable = {
				name = aus_intelligentsia_progress_from_effects
				add = 5
			}
		}
	}
}

				has_variable = dissolution_of_parliament_var
			}
		}
	}
	ai_chance = {
		value = 0
	}
	effect = {
		every_interest_group = {
			limit = {
				is_in_government = yes
			}
			remove_ruling_interest_group = yes
		}
		add_modifier = {
			name = dissolution_of_parliament_modifier
			months = 12
		}

			text = dissolution_of_parliament_effect_tt

	}
