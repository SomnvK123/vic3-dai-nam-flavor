dsk_je_gbr_economy_button = {
	name = "dsk_je_gbr_economy_button"
	desc = "dsk_je_gbr_economy_button_desc"
	visible = {
		OR = {
			is_subject_of = c:GBR
			c:GBR ?= this
		}
	}
	cooldown = { years = 5 }
	possible = {
		OR = {
			AND = {
				is_subject_of = c:GBR
			                root.gdp >= 8000000
			                root.gdp < 150000000
				c:GBR ?= {
					#大妈有12M的钱，每周收入400k
					gold_reserves >= 12000000
					net_fixed_income > 400000
				}
				NOT = {
					gold_reserves > 0
				}
		                OR = {
                                                                is_player = yes
		                                has_modifier = declared_bankruptcy
		                                has_modifier = in_default
			}
			}
			AND = {
				c:GBR ?= this
				gold_reserves >= 12000000
				any_country = {
					is_subject_of = ROOT
				NOT = {
					gold_reserves > 0
				}
				}
			}
		}
	}
	effect = {
		if = {
			limit = {
				is_subject_of = c:GBR
			}
			trigger_event = {
				id = dsk_gbr.48
				popup = yes
			}
		}
		if = {
			limit = {
				c:GBR ?= this
			}
			trigger_event = {
				id = dsk_gbr.50
				popup = yes
			}
		}
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				c:GBR ?= this
				OR = {
					taking_loans = no
					gold_reserves >= 0
				}
			}
			add = 20
		}
		if = {
			limit = {
				is_subject_of = c:GBR
				c:GBR ?= {
					#大妈有5M的钱
					gold_reserves > 5000000					#2M英镑
					net_fixed_income > 0					#大妈在正盈利
					is_player = no					#鼓励英国玩家自己用资金转让条款	
				}
				OR = {
					#英联邦成员负债才找大妈借钱
					taking_loans = yes
					NOT = {
						gold_reserves >= 0
					}
				}
			}
			add = 70
		}
	}
}

support_nzl_ruler_button = {
	name = "support_nzl_ruler_button"
	desc = "support_nzl_ruler_button_desc"
	visible = {
		exists = c:NZL
                                c:NZL ?= { is_subject_of = ROOT }
		NOT = {
			has_global_variable = support_nzl_ruler_button
		}
	}
	possible = {
		game_date > 1892.3.23
		game_date < 1940.3.27
        c:NZL ?= { is_subject_of = ROOT }
		lca_is_in_diplomatic_play_or_war = no
	}
	effect = {
		hidden_effect = {
			c:GBR ?= {
				pax_britannica_update = {
					POINTS_CHANGE = root.var:pax_britannica_events_positive
				}
				pax_britannica_update = {
					POINTS_CHANGE = root.var:pax_britannica_events_positive
				}
                                }
                                }
		set_global_variable = support_nzl_ruler_button
		add_treasury = money_gbr_spend
		add_modifier = {
			name = modifier_yuanzhu_gbr
			years = 5
		}
	if = {
		c:NZL ?= {
                                                   is_player = no
                                }
		c:NZL ?= {
		activate_law = law_type:law_IJ_elementary
		activate_law = law_type:law_public_health_insurance
		activate_law = law_type:law_IJ_rescue
                }
                }
		c:NZL ?= {
	if = {
			limit = {
				any_country = {
					any_scope_character = {
						has_template = ecchi_nzl_savage_character_template
					}
				}
			}
			random_country = {
				limit = {
					any_scope_character = {
						has_template = ecchi_nzl_savage_character_template
					}
				}
				random_scope_character = {
					limit = {
						has_template = ecchi_nzl_savage_character_template
					}
					transfer_character = c:NZL
				                set_character_as_ruler = yes
				                set_as_interest_group_leader = yes
                                                                                add_trait = special_character_colonial_local
				}
			}
		}
		else_if = {
			limit = {
				any_character_in_exile_pool = {
					has_template = ecchi_nzl_savage_character_template
				}
			}
			random_character_in_exile_pool = {
				limit = {
					has_template = ecchi_nzl_savage_character_template
				}
					transfer_character = c:NZL
				                set_character_as_ruler = yes
				                set_as_interest_group_leader = yes
                                                                                add_trait = special_character_colonial_local
			}
		}
		else = {
			create_character = {
				template = ecchi_nzl_savage_character_template
                                                                ruler = yes
                                                                ig_leader = yes
			}
		}
			add_liberty_desire = -20
			add_treasury = money_gbr
			add_modifier = {
				name = modifier_support_from_gbr
				years = 5
			}
		}
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				#500K黄金和500行政力
				gold_reserves > 500000
				bureaucracy > 500
			}
			add = 90
		}
	}
}
