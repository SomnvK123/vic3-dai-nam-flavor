sb_green_magical_girl_begin_tour = {
	name = "sb_green_magical_girl_begin_tour"
	desc = "sb_green_magical_girl_begin_tour_desc"

	visible = {
		any_scope_character = {
			has_variable = dsk_go_travel
		}
	}

	possible = {
		custom_tooltip = {
			text = gmg_has_green_magical_girl_tt
			any_scope_character = {
				has_variable = dsk_go_travel
			}
		}

		custom_tooltip = {
			text = gmg_tour_not_on_cooldown_tt
			NOT = {
				has_variable = gmg_tour_cooldown
			}
		}

		custom_tooltip = {
			text = gmg_has_valid_tour_target_tt
			any_country = {
				NOT = { this = ROOT }

				ruler ?= {
					always = yes
				}

				NOT = {
					has_modifier = modifier_gmg_recently_visited_by_green_magical_girl
				}
			}
		}
	}

	effect = {
		# 找到探险家并缓存
		random_scope_character = {
			limit = {
				has_variable = dsk_go_travel
			}
			save_scope_as = green_magical_girl_scope
		}

		if = {
			limit = {
				exists = scope:green_magical_girl_scope
			}

			if = {
				limit = {
					has_variable = gmg_green_magical_girl_var
				}
				remove_variable = gmg_green_magical_girl_var
			}

			set_variable = {
				name = gmg_green_magical_girl_var
				value = scope:green_magical_girl_scope
			}
		}

		# 清理旧目标作用域与变量，防止沿用旧目标
		if = {
			limit = {
				has_variable = gmg_last_tour_country
			}
			remove_variable = gmg_last_tour_country
		}

		# ------------------------------------------------------------
		# 稳定版目的地选择逻辑
		# ------------------------------------------------------------
		# 不再使用 random_country 的 weight，也不使用 random_list。
		# 原因：
		# scripted_button 里 random_list + random_country + save_scope_as 容易导致目标作用域没保存成功。
		#
		# 现在逻辑：
		# 1. 有未冷却列强：必定从列强中随机
		# 2. 没有未冷却列强，但有未冷却主要强国：从主要强国随机
		# 3. 没有未冷却主要强国，但有未冷却次要国家：从次要国家随机
		# 4. 最后才从小国/无足轻重国家随机
		# ------------------------------------------------------------

		if = {
			limit = {
				any_country = {
					NOT = { this = ROOT }

					ruler ?= {
						always = yes
					}

					NOT = {
						has_modifier = modifier_gmg_recently_visited_by_green_magical_girl
					}

					country_rank >= rank_value:great_power
				}
			}

			random_country = {
				limit = {
					NOT = { this = ROOT }

					ruler ?= {
						always = yes
					}

					NOT = {
						has_modifier = modifier_gmg_recently_visited_by_green_magical_girl
					}

					country_rank >= rank_value:great_power
				}

				save_scope_as = gmg_tour_target
			}
		}

		else_if = {
			limit = {
				any_country = {
					NOT = { this = ROOT }

					ruler ?= {
						always = yes
					}

					NOT = {
						has_modifier = modifier_gmg_recently_visited_by_green_magical_girl
					}

					country_rank >= rank_value:major_power

					NOT = {
						country_rank >= rank_value:great_power
					}
				}
			}

			random_country = {
				limit = {
					NOT = { this = ROOT }

					ruler ?= {
						always = yes
					}

					NOT = {
						has_modifier = modifier_gmg_recently_visited_by_green_magical_girl
					}

					country_rank >= rank_value:major_power

					NOT = {
						country_rank >= rank_value:great_power
					}
				}

				save_scope_as = gmg_tour_target
			}
		}

		else_if = {
			limit = {
				any_country = {
					NOT = { this = ROOT }

					ruler ?= {
						always = yes
					}

					NOT = {
						has_modifier = modifier_gmg_recently_visited_by_green_magical_girl
					}

					country_rank >= rank_value:minor_power

					NOT = {
						country_rank >= rank_value:major_power
					}
				}
			}

			random_country = {
				limit = {
					NOT = { this = ROOT }

					ruler ?= {
						always = yes
					}

					NOT = {
						has_modifier = modifier_gmg_recently_visited_by_green_magical_girl
					}

					country_rank >= rank_value:minor_power

					NOT = {
						country_rank >= rank_value:major_power
					}
				}

				save_scope_as = gmg_tour_target
			}
		}

		else_if = {
			limit = {
				any_country = {
					NOT = { this = ROOT }

					ruler ?= {
						always = yes
					}

					NOT = {
						has_modifier = modifier_gmg_recently_visited_by_green_magical_girl
					}

					NOT = {
						country_rank >= rank_value:minor_power
					}
				}
			}

			random_country = {
				limit = {
					NOT = { this = ROOT }

					ruler ?= {
						always = yes
					}

					NOT = {
						has_modifier = modifier_gmg_recently_visited_by_green_magical_girl
					}

					NOT = {
						country_rank >= rank_value:minor_power
					}
				}

				save_scope_as = gmg_tour_target
			}
		}

		# ------------------------------------------------------------
		# 如果成功选到目标国家，开始记录旅行
		# ------------------------------------------------------------

		if = {
			limit = {
				exists = scope:gmg_tour_target
			}

			# 目标国家自身记录：被访问次数
			scope:gmg_tour_target = {
				if = {
					limit = {
						NOT = {
							exists = var:gmg_visit_count_by_green_magical_girl
						}
					}

					set_variable = {
						name = gmg_visit_count_by_green_magical_girl
						value = 0
					}
				}

				change_variable = {
					name = gmg_visit_count_by_green_magical_girl
					add = 1
				}

				# 目标国家获得 8 年冷却
				# 8 年 = 365 * 8 = 2920 天
				add_modifier = {
					name = modifier_gmg_recently_visited_by_green_magical_girl
					days = 2920
				}
			}

			# ROOT 国家记录：总旅行次数
			if = {
				limit = {
					NOT = {
						exists = var:gmg_total_tours
					}
				}

				set_variable = {
					name = gmg_total_tours
					value = 0
				}
			}

			change_variable = {
				name = gmg_total_tours
				add = 1
			}

			set_variable = {
				name = gmg_last_tour_country
				value = scope:gmg_tour_target
			}

			add_to_variable_list = {
				name = gmg_visited_countries
				target = scope:gmg_tour_target
			}

			# 清理旧旅行阶段变量
			if = { limit = { has_variable = gmg_is_travelling } remove_variable = gmg_is_travelling }
			if = { limit = { has_variable = gmg_mid_event_1_lock } remove_variable = gmg_mid_event_1_lock }
			if = { limit = { has_variable = gmg_mid_event_2_lock } remove_variable = gmg_mid_event_2_lock }
			if = { limit = { has_variable = gmg_return_lock } remove_variable = gmg_return_lock }
			if = { limit = { has_variable = gmg_return_pending } remove_variable = gmg_return_pending }
			if = { limit = { has_variable = gmg_mid_event_1_done } remove_variable = gmg_mid_event_1_done }
			if = { limit = { has_variable = gmg_mid_event_2_done } remove_variable = gmg_mid_event_2_done }

			# 旅行状态：330 天视为“人在外面旅行”
			set_variable = {
				name = gmg_is_travelling
				days = 330
			}

			# 中途事件锁：到期后由 weekly pulse 检测并触发
			set_variable = {
				name = gmg_mid_event_1_lock
				days = 90
			}

			set_variable = {
				name = gmg_mid_event_2_lock
				days = 210
			}

			# 回国结算锁
			set_variable = {
				name = gmg_return_lock
				days = 330
			}

			# 标记“这次旅行还没结算”
			set_variable = gmg_return_pending

			# 按钮冷却仍然 365 天
			if = {
				limit = {
					has_variable = gmg_tour_cooldown
				}
				remove_variable = gmg_tour_cooldown
			}

			set_variable = {
				name = gmg_tour_cooldown
				days = 365
			}

			# 出发事件
			trigger_event = {
				id = gmg_world_tour.1
				popup = yes
			}
		}
	}

	ai_chance = {
		value = 0
	}
}

stop_our_tour = {
	name = "stop_our_tour_button"
	desc = "stop_our_tour_button_desc"
	visible = {
			NOT = {
				has_variable = stop_our_tour_now
			}
	}
	possible = {
                      bureaucracy >= 100
	}
	effect = {
		set_variable = {
			name = stop_our_tour_now
			days = 30
		}
		add_radicals = {
			value = 0.01
			pop_type = aristocrats
		}
		custom_tooltip = {
			text = stop_our_tour_now_tt
		}
	}
	ai_chance = {
		value = 0
	}
}
