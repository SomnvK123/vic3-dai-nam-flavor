		years = 5
	}
	possible = {
		owns_entire_state_region = STATE_MANITOBA

		owns_entire_state_region = STATE_SASKATCHEWAN
		custom_tooltip = {
			text = enough_canadian_Confederation_var
			scope:journal_entry = {
				"scripted_bar_progress(canadian_confederation_bar)" >= 50

		s:STATE_SASKATCHEWAN = {
			random_scope_state = {
				limit = {
					owner ?= {
						THIS = ROOT
					}
				}
				add_modifier = {
					name = nafta_union_modifier_c
					days = 730
				}
			}
		}
		je:je_Canadian_Confederation = {
			add_progress = {
