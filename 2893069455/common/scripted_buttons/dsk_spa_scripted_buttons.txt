mex_button = {
	name = "mex_button"
	desc = "mex_button_desc"
	visible = {
		OR = {
			exists = c:MEX
			exists = c:SC1
		}
	}
	cooldown = {
		years = 5
	}
	possible = {
		bureaucracy >= 300
		OR = {
			AND = {
				exists = c:MEX
				c:MEX ?= {
					is_subject_of = root
				}
			}
			AND = {
				exists = c:SC1
				c:SC1 ?= {
					is_subject_of = root
				}
			}
		}
	}
	effect = {
		add_modifier = {
			name = modifier_spa_yuanzhu
			years = 4
		}
		if = {
			limit = {
				c:MEX ?= {
					is_subject_of = root
				}
			}
			c:MEX ?= {
				add_liberty_desire = -20
				add_modifier = {
					name = modifier_spa_support
					years = 4
				}
			}
		}
		if = {
			limit = {
				c:SC1 ?= {
					is_subject_of = root
				}
			}
			c:SC1 ?= {
				add_liberty_desire = -20
				add_modifier = {
					name = modifier_spa_support
					years = 4
				}
			}
		}
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				#500K黄金和500行政力
				gold_reserves >= 500000
				bureaucracy >= 500
			}
			add = 30
		}
	}
}
