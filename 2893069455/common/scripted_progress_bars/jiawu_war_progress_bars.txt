			desc = "jap_empire_progress_a"			# 明治政府推动宪政改革
			value = {
				if = {
					limit = {
						army_size >= 500
						navy_size >= 200
					}
					value = 2
				}
				else_if = {
					limit = {
						army_size >= 250
						navy_size >= 100

					value = 0.3
				}
			}
