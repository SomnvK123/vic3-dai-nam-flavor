china_Treaty_Port_bar = {
	name = "china_Treaty_Port_bar"
	desc = "china_Treaty_Port_bar_desc"	#
	default_green = yes	# 
	monthly_progress = {
		add = {
			desc = "china_Treaty_Port_progress_a"
			value = {
				if = {
					limit = {
						OR = {
							AND = {
								exists = c:ROC
								OR = {
									any_scope_treaty = {
										binds = c:ROC
										any_scope_article = {
											has_type = trade_privilege
										}
									}
									any_scope_treaty = {
										binds = c:ROC
										any_scope_article = {
											has_type = foreign_investment_rights
										}
									}
								}
							}
							AND = {
								exists = c:CHI
								OR = {
									any_scope_treaty = {
										binds = c:CHI
										any_scope_article = {
											has_type = trade_privilege
										}
									}
									any_scope_treaty = {
										binds = c:CHI
										any_scope_article = {
											has_type = foreign_investment_rights
										}
									}
								}
							}
						}
					}
					value = 1
				}
				else = {
					value = 0
				}
			}
		}
		add = {
			desc = "china_Treaty_Port_progress_b"
			value = {
				if = {
					limit = {
						any_scope_state = {
							lca_is_chinese_state = yes
							is_treaty_port = yes
							any_scope_building = {
								is_building_type = building_urban_center
								occupancy >= 0.5
								level >= 2
							}
						}
					}
					value = 1
				}
				else = {
					value = 0
				}
			}
		}
		# 港口
		add = {
			desc = "china_Treaty_Port_progress_c"
			value = {
				if = {
					limit = {
						any_scope_state = {
							lca_is_chinese_state = yes
							is_treaty_port = yes
							any_scope_building = {
								is_building_type = building_railway
								occupancy >= 0.5
								level >= 2
							}
							any_scope_building = {
								is_building_type = building_port
								occupancy >= 0.5
								level >= 2
							}
						}
					}
					value = 1
				}
				else = {
					value = 0
				}
			}
		}
	}
	start_value = 0
	min_value = 0
	max_value = 100
}
