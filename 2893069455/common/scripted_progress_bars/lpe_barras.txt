je_deuda_esp_bar = {
	name = "je_deuda_esp_bar"
	desc = "je_deuda_esp_bar_desc"
	default = yes
	monthly_progress = {
		if = {
			limit = {
				scaled_debt < 0.01
			}
			add = {
				desc = "je_deuda_esp_bar_deuda_good"
				value = 2
			}
		}
		else = {
			subtract = {
				desc = "je_deuda_esp_bar_deuda_bad"
				value = 1
			}
		}
		add = {
			desc = "je_deuda_esp_bar_aceleracion_deuda"
			if = {
				limit = {
					has_modifier = acelaracion_pago_deuda
				}
				value = 2
			}
			else = {
				value = 0
			}
		}
	}
	start_value = 0
	min_value = 0
	max_value = 300
}
