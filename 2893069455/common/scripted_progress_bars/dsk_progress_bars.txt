je_risorgimento_repubblicano_bar = {
	name = "je_risorgimento_repubblicano_bar"
	desc = "je_risorgimento_repubblicano_bar_desc_monarchia"	# 君主派描述
	second_desc = "je_risorgimento_repubblicano_bar_desc_repubblica"	# 共和派描述
	double_sided_bad = yes	# 双面金色进度条
	weekly_progress = {
		if = {
			limit = {
				has_variable = rep_popolarita_personaggi_pro_repubblica
				has_variable = rep_popolarita_personaggi_pro_monarchia
				has_variable = rep_popolarita_lider_elite_intellettuale
			}
			# 基础影响因素
			add = {
				desc = "unificazione_culturale"				# 文化统一进程
				value = 0.3				# 基础衰减值更陡峭
			}
			# 政府参与度影响
			add = {
				desc = "partecipazione_governativa"
				if = {
					limit = {
						ig:ig_intelligentsia = {
							is_in_government = no
						}
						has_law_or_variant = law_type:law_monarchy
					}
					value = root.var:rep_popolarita_personaggi_pro_repubblica					# 知识分子在野且君主制
					multiply = 0.9
				}
				else_if = {
					limit = {
						ig:ig_intelligentsia = {
							is_in_government = no
						}
						OR = {
							has_law_or_variant = law_type:law_presidential_republic
							has_law_or_variant = law_type:law_parliamentary_republic
						}
					}
					value = root.var:rep_popolarita_lider_elite_intellettuale					# 知识分子在野且共和制
					multiply = -1
				}
				else = {
					value = root.var:rep_popolarita_personaggi_pro_monarchia
					multiply = -1.1
				}
			}
			# 知识分子集团支持度
			add = {
				desc = "sostegno_intellighenzia"
				value = root.var:progresso_repubblicano_total
				if = {
					limit = {
						has_law_or_variant = law_type:law_monarchy
					}
					divide = 2					# 君主制下影响力更强
				}
				else = {
					divide = -6					# 共和制反向影响
				}
			}
			# 人物影响力
			add = {
				desc = "leader_repubblicani"
				value = root.var:rep_popolarita_personaggi_pro_repubblica				# 
				if = {
					limit = {
						OR = {
							any_scope_character = {
								has_template = garibaldi_character_template
							}
							any_scope_character = {
								has_template = agitator_giuseppe_mazzini
							}
						}
					}
					multiply = 1.6					# 单杰在影响力一倍
				}
				else_if = {
					limit = {
						AND = {
							any_scope_character = {
								has_template = garibaldi_character_template
							}
							any_scope_character = {
								has_template = agitator_giuseppe_mazzini
							}
						}
					}
					multiply = 3					# 双杰在影响力更强
				}
			}
			add = {
				desc = "leader_monarchici"
				if = {
					limit = {
						any_interest_group = {
							OR = {
								leader = {
									has_ideology = ideology:ideology_royalist
								}
								leader = {
									has_ideology = ideology:ideology_humanitarian_royalist
								}
							}
							is_in_government = yes
						}
					}
					value = 3					# 地主在政府且君主制
					multiply = 0.9
				}
				else_if = {
					limit = {
						any_interest_group = {
							NOR = {
								leader = {
									has_ideology = ideology:ideology_royalist
								}
								leader = {
									has_ideology = ideology:ideology_humanitarian_royalist
								}
							}
							is_in_government = yes
						}
					}
					value = -3					# 地主在野或不举足轻重
				}
				else = {
					value = 1
					multiply = -1.1
				}
			}
			add = {
				desc = "influenza_capo_intellettuale"
				value = 1
				if = {
					limit = {
						OR = {
							country_rank > country_rank.major_power
							country_has_primary_culture = cu:north_italian
							country_has_primary_culture = cu:south_italian
						}
						relations:root > relations_threshold:amicable
						OR = {
							has_law_or_variant = law_type:law_presidential_republic
							has_law_or_variant = law_type:law_parliamentary_republic
						}
					}
					multiply = -1					# 浪漫主义思潮加成
				}
			}
			# 地区特殊加成
			add = {
				desc = "sostegno_settentrionale"
				value = 0
				if = {
					limit = {
						owns_entire_state_region = STATE_LOMBARDY
					}
					add = -1					# 控制伦巴第额外加成
				}
			}
			# 地区特殊加成
			add = {
				desc = "south_settentrionale"
				value = 0
				if = {
					limit = {
						owns_entire_state_region = STATE_SICILY
					}
					add = 2					# 控制西西里额外加成
				}
			}
		}
	}
	start_value = 1000	# 初始值更偏向共和派
	min_value = 0
	max_value = 2000	# 总周期缩短
}
