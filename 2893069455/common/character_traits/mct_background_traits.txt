academic = {
	#学者
	type = skill
	texture = "gfx/interface/icons/character_trait_icons/academic.dds"
	character_modifier = {
		character_popularity_add = 10
		character_prominence_add = 2
	}
	command_modifier = {
	}
	country_modifier = {
		country_tech_spread_mult = 0.05		#学者科技扩散
		country_authority_mult = -0.05
		building_university_throughput_add = 0.1
	}
	interest_group_modifier = {
		country_academics_pol_str_mult = 0.1
	}
	agitator_modifier = {
		country_academics_pol_str_mult = 0.05
	}
	possible = {
		has_role_of_type = politician
		NAND = {
			is_ruler_of_own_country = yes
			owner ?= {
				has_law_or_variant = law_type:law_monarchy
			}
		}
		NAND = {
			is_heir_of_own_country = yes
			owner ?= {
				has_law_or_variant = law_type:law_monarchy
			}
		}
		interest_group ?= {
			OR = {
				is_interest_group_type = ig_landowners
				is_interest_group_type = ig_devout
				is_interest_group_type = ig_intelligentsia
				is_interest_group_type = ig_petty_bourgeoisie
			}
		}
		owner ?= {
			has_technology_researched = academia
		}
		NOT = {
			has_variable = has_historical_traits
		}
		NOT = {
			has_variable = david_copperfield_var
		}
		NOR = {
			has_trait = academic
			has_trait = banker
			has_trait = bureaucrat
			has_trait = clerk
			has_trait = doctor
			has_trait = technician
			has_trait = entrepreneur
			has_trait = mill_owner
			has_trait = farmer
			has_trait = journalist
			has_trait = labourer
			has_trait = lawyer
			has_trait = merchant
			has_trait = military_officer
			has_trait = landed_elite
			has_trait = police_chief
			has_trait = political_dynasty
			has_trait = religious_leader
			has_trait = soldier
			has_trait = teacher
		}
		age > define:NPortrait|GRACEFUL_AGING_START
	}
	weight = {
		value = 0
		if = {
			limit = {
				interest_group ?= {
					is_interest_group_type = ig_intelligentsia
				}
			}
			add = 490
		}
		if = {
			limit = {
				interest_group ?= {
					OR = {
						is_interest_group_type = ig_landowners
						is_interest_group_type = ig_devout
						is_interest_group_type = ig_petty_bourgeoisie
					}
				}
			}
			add = 430
		}
	}
	replace = {
	}
	value = 1
}
