je_dsk_bel_national_question = {
	icon = "gfx/interface/icons/event_icons/bel.png"
	group = je_group_dsk_belgium
	is_shown_in_lobby = {
		c:BEL ?= this
	}
	immediate = {
		dsk_bel_save_core_scopes = yes
		dsk_bel_initialize_complex_variables = yes
	}
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_dsk_bel_national_question_status_collapse
				trigger = {
					OR = {
						var:dsk_bel_neutrality_var <= 10
						var:dsk_bel_unity_var <= 10
						has_modifier = dsk_bel_broken_neutrality_modifier
					}
				}
			}
			triggered_desc = {
				desc = je_dsk_bel_national_question_status_confident
				trigger = {
					var:dsk_bel_independence_var >= 55
					var:dsk_bel_industry_var >= 55
					var:dsk_bel_unity_var >= 55
					var:dsk_bel_neutrality_var >= 55
					has_variable = dsk_bel_london_treaty
					dsk_bel_workshop_industrial_base_trigger = yes
					dsk_bel_finance_network_base_trigger = yes
					OR = {
						has_variable = dsk_bel_neutrality_done
						has_modifier = dsk_bel_fortress_neutrality_modifier
						has_modifier = dsk_bel_strict_neutrality_modifier
					}
					OR = {
						has_variable = dsk_bel_school_done
						has_variable = dsk_bel_unity_done
						has_modifier = dsk_bel_school_compromise_modifier
						has_modifier = dsk_bel_language_settlement_modifier
					}
				}
			}
			triggered_desc = {
				desc = je_dsk_bel_national_question_status_default
				trigger = {
					always = yes
				}
			}
		}
	}
	complete = {
		custom_tooltip = {
			text = dsk_bel_national_question_independence_tt
			var:dsk_bel_independence_var >= 55
		}
		custom_tooltip = {
			text = dsk_bel_national_question_industry_tt
			var:dsk_bel_industry_var >= 55
		}
		custom_tooltip = {
			text = dsk_bel_national_question_unity_tt
			var:dsk_bel_unity_var >= 55
		}
		custom_tooltip = {
			text = dsk_bel_national_question_neutrality_tt
			var:dsk_bel_neutrality_var >= 55
		}
		custom_tooltip = {
			text = dsk_bel_london_treaty_done_tt
			has_variable = dsk_bel_london_treaty
		}
		custom_tooltip = {
			text = dsk_bel_workshop_industrial_base_tt
			dsk_bel_workshop_industrial_base_trigger = yes
		}
		custom_tooltip = {
			text = dsk_bel_finance_network_base_tt
			dsk_bel_finance_network_base_trigger = yes
		}
		custom_tooltip = {
			text = dsk_bel_neutrality_institution_done_tt
			OR = {
				has_variable = dsk_bel_neutrality_done
				has_modifier = dsk_bel_fortress_neutrality_modifier
				has_modifier = dsk_bel_strict_neutrality_modifier
			}
		}
		custom_tooltip = {
			text = dsk_bel_school_or_language_done_tt
			OR = {
				has_variable = dsk_bel_school_done
				has_variable = dsk_bel_unity_done
				has_modifier = dsk_bel_school_compromise_modifier
				has_modifier = dsk_bel_language_settlement_modifier
			}
		}
	}
	on_complete = {
		set_variable = dsk_bel_national_question_done
		add_modifier = {
			name = dsk_bel_postwar_compromise_modifier
			years = 10
		}
		trigger_event = {
			id = dsk_belgium.127
			popup = yes
		}
	}
	fail = {
		OR = {
			has_modifier = declared_bankruptcy
			custom_tooltip = {
				text = dsk_bel_national_question_fail_tt
				OR = {
					var:dsk_bel_neutrality_var <= -1
					var:dsk_bel_unity_var <= -1
					NOT = {
						exists = c:BEL
					}
				}
			}
		}
	}
	on_fail = {
		set_variable = dsk_bel_national_question_done
		add_modifier = {
			name = dsk_bel_broken_neutrality_modifier
			years = 10
		}
		trigger_event = {
			id = dsk_belgium.128
			popup = yes
		}
	}
	weight = 11000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
