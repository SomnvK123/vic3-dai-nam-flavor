	}

			value = 0

	}
	complete = {
		scope:journal_entry = {
			is_goal_complete = yes
		}
		NOT = {
			has_law_or_variant = law_type:law_serfdom
		}
	}
	on_complete = {
		set_variable = donghak_complete

			}			#동학사태의 해결

			text = donghak_will_happen_tt

				id = donghak_je_events.1
			}
		}

		effect = {
			if = {
				limit = {
					NOT = {

					}
				}

					name = donghak_progress_var
