		dsk_bel_save_core_scopes = yes

			set_variable = { name = dsk_bel_independence_var value = 0 }
		}
		trigger_event = { id = dsk_belgium.1 days = 5 popup = yes }
	}
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_dsk_bel_independence_status_london_ready
				trigger = { var:dsk_bel_independence_var >= 3 }
			}
			triggered_desc = {
				desc = je_dsk_bel_independence_status_dutch_pressure
				trigger = {
					exists = c:NET
					relations:c:NET < relations_threshold:cold
				}
			}
			triggered_desc = {
				desc = je_dsk_bel_independence_status_default
				trigger = { always = yes }
			}
		}
	}
	on_monthly_pulse = {
		random_events = {
			300 = 0
			25 = dsk_belgium.2
			25 = dsk_belgium.3
			25 = dsk_belgium.4
		}
		effect = {
			dsk_bel_save_core_scopes = yes
			if = {
				limit = {
					var:dsk_bel_independence_var < 55
					OR = {
						AND = { exists = c:GBR relations:c:GBR >= relations_threshold:amicable }
						AND = { exists = c:FRA relations:c:FRA >= relations_threshold:amicable }
						AND = { exists = c:PRU relations:c:PRU >= relations_threshold:amicable }
					}
					NOT = { has_modifier = dsk_bel_broken_neutrality_modifier }
				}
				change_variable = { name = dsk_bel_independence_var add = 0.25 }
			}
			if = {
				limit = {
					var:dsk_bel_independence_var < 55
					NOT = { has_modifier = dsk_bel_broken_neutrality_modifier }
					OR = {
						AND = {
							exists = c:GBR
							relations:c:GBR >= relations_threshold:amicable
							exists = c:FRA
							relations:c:FRA >= relations_threshold:amicable
						}
						AND = {
							exists = c:GBR
							relations:c:GBR >= relations_threshold:amicable
							exists = c:PRU
							relations:c:PRU >= relations_threshold:amicable
						}
						AND = {
							exists = c:FRA
							relations:c:FRA >= relations_threshold:amicable
							exists = c:PRU
							relations:c:PRU >= relations_threshold:amicable
						}
					}
				}
				change_variable = { name = dsk_bel_independence_var add = 0.25 }
			}
			if = {
				limit = {
					var:dsk_bel_independence_var >= 5
					NOT = { has_variable = dsk_bel_london_deadline_event }
				}
				set_variable = dsk_bel_london_deadline_event
				trigger_event = { id = dsk_belgium.8 days = 10 popup = yes }
			}
		}
	}
	complete = {
		custom_tooltip = {
			text = dsk_bel_independence_progress_tt
			var:dsk_bel_independence_var >= 55
		}
		custom_tooltip = {
			text = dsk_bel_country_at_peace_tt
			is_at_war = no
		}
	}
	on_complete = {
		set_variable = dsk_bel_london_treaty
		if = {
			limit = {
				OR = {
					NOT = { has_variable = dsk_bel_neutrality_var }
					var:dsk_bel_neutrality_var < 3
				}
			}
			set_variable = { name = dsk_bel_neutrality_var value = 3 }
		}
		change_variable = { name = dsk_bel_neutrality_var add = 1 }
		if = {
			limit = { NOT = { has_journal_entry = je_dsk_bel_neutrality } }
			add_journal_entry = { type = je_dsk_bel_neutrality }
		}
		add_modifier = { name = dsk_bel_london_treaty_modifier years = 10 }
		dsk_bel_recalculate_monthly_display_deltas = yes
		trigger_event = { id = dsk_belgium.6 days = 10 popup = yes }
	}
	fail = {
		OR = {
			NOT = { exists = c:BEL }
			is_subject = yes
		}
	}
	on_fail = {
		trigger_event = { id = dsk_belgium.7 popup = yes }
	}
	weight = 10000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
