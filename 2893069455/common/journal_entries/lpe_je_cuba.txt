je_economia_agraria_cub = {
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	group = je_group_internal_affairs
	immediate = {
		add_modifier = economia_agricola_cubana
	}
	complete = {
		OR = {
			AND = {
                                              "goods_production_rank(g:tobacco)" <= 2
                                                market_capital.market = {
                                                                 mg:tobacco = {
                                                                     market_goods_production >= 2500
                                                                 }
                                                                 }
                                              }
			AND = {
                                              "goods_production_rank(g:coffee)" <= 2
                                                market_capital.market = {
                                                                 mg:coffee = {
                                                                     market_goods_production >= 2500
                                                                 }
                                                                 }
                                              }
			AND = {
                                              "goods_production_rank(g:sugar)" <= 2
                                                market_capital.market = {
                                                                 mg:sugar = {
                                                                     market_goods_production >= 2500
                                                                 }
                                                                 }
                                              }
			AND = {
				any_scope_building = {
					filter = {
						custom_tooltip = {
							text = je_economia_agraria_cub_tt
							is_plantation_building = yes
						}
					}
					cash_reserves_ratio >= 0.5
					custom_tooltip = {
						text = je_economia_agraria_cub_tt2
						earnings >= 5
					}
					percent >= 0.65
				}
				country_has_building_group_levels = {
					type = bg_plantations
					value >= 100
				}
			}
		}
	}
	on_complete = {
		hidden_effect = {
			remove_modifier = economia_agricola_cubana
			remove_modifier = industira_azucarera_cubana
		}
		add_modifier = economia_agricola_cubana2
		add_modifier = {
			name = industira_azucarera_cubana
			days = very_long_modifier_time
		}
	}
	fail = {
		OR = {
			NOR = {
				has_law_or_variant = law_type:law_traditionalism
				has_law_or_variant = law_type:law_agrarianism
				has_law_or_variant = law_type:law_industry_banned
			}
			NOR = {
				has_law_or_variant = law_type:law_serfdom
				has_law_or_variant = law_type:law_tenant_farmers
			}
			any_scope_building = {
				filter = {
					custom_tooltip = {
						text = je_economia_agraria_cub_tt
						is_plantation_building = yes
					}
				}
				cash_reserves_ratio <= 0.2
				custom_tooltip = {
					text = je_economia_agraria_cub_tt1
					earnings <= 4
				}
				percent >= 0.5
			}
		}
	}
	on_fail = {
		remove_modifier = economia_agricola_cubana
	}
	can_revolution_inherit = yes
	weight = 200
	should_be_pinned_by_default_uninvolved_or_context = yes
}

	on_fail = {
	}

	}

	weight = 200
}
