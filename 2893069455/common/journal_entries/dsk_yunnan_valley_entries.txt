# Yunnan Valley of Insects Expedition Journal Entry
# Based on Tibet Expedition design pattern
je_yunnan_valley_expedition = {
	group = je_group_objectives
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	progressbar = yes
	on_monthly_pulse = {
		random_events = {
			chance_to_happen = 33
			# Phase I: Preparations
			10 = dsk_yunnan_valley.4
			10 = dsk_yunnan_valley.5
			10 = dsk_yunnan_valley.6
			10 = dsk_yunnan_valley.7
			10 = dsk_yunnan_valley.14
			# Triggers Phase II
			10 = dsk_yunnan_valley.8
			# 9 is triggered by 8
			# Phase II: Incursion into Valley
			10 = dsk_yunnan_valley.9
			10 = dsk_yunnan_valley.10
			#10 = dsk_yunnan_valley.15
			#10 = dsk_yunnan_valley.16
			10 = dsk_yunnan_valley.17
			# Phase III: Discovery & Negotiation
			10 = dsk_yunnan_valley.11
			10 = dsk_yunnan_valley.18
			10 = dsk_yunnan_valley.19
			#10 = dsk_yunnan_valley.20
			10 = dsk_yunnan_valley.21
			# Phase IV: Final Phase
			#10 = dsk_yunnan_valley.22
			#10 = dsk_yunnan_valley.23
			10 = dsk_yunnan_valley.24
			10 = dsk_yunnan_valley.25
			10 = dsk_yunnan_valley.26
			10 = dsk_yunnan_valley.27
			10 = dsk_yunnan_valley.28
			10 = dsk_yunnan_valley.29
			10 = dsk_yunnan_valley.30
		}
	}
	immediate = {
		random_scope_state = {
			limit = {
				state_region = s:STATE_YUNNAN
			}
			save_scope_as = current_expedition_location
		}
		s:STATE_YUNNAN = {
			random_scope_state = {

				save_scope_as = current_expedition_location
			}
		}
		save_scope_as = expedition_country
		random_scope_character = {
			limit = {
				has_variable = yunnan_expedition_leader_var
			}
			save_scope_as = yunnan_expedition_leader
		}
		set_variable = {
			name = yunnan_expedition_progress_var
			value = 0
		}
		set_variable = on_expedition
	}
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = yunnan_expedition_preparing
				trigger = {
					exists = var:yunnan_expedition_progress_var
					var:yunnan_expedition_progress_var < 3
				}
			}
			triggered_desc = {
				desc = yunnan_expedition_entering
				trigger = {
					exists = var:yunnan_expedition_progress_var
					var:yunnan_expedition_progress_var <= 5
					NOT = {
						has_variable = negotiation_started
					}
				}
			}
			triggered_desc = {
				desc = yunnan_expedition_exploring
				trigger = {
					exists = var:yunnan_expedition_progress_var
					var:yunnan_expedition_progress_var <= 7
					has_variable = negotiation_started
				}
			}
			triggered_desc = {
				desc = yunnan_expedition_finalizing
				trigger = {
					exists = var:yunnan_expedition_progress_var
					var:yunnan_expedition_progress_var > 7
				}
			}
		}
	} 
	complete = {
		custom_tooltip = {
			text = je_yunnan_valley_expedition_success_tt
			AND = {
				#has_variable = expedition_concluded
				var:yunnan_expedition_progress_var >= 8
			}
		}
	}
	on_complete = {
       custom_tooltip = {
			text = je_yunnan_valley_expedition_success_event_tt
	   }
		random_scope_character = {
			limit = {
				has_variable = yunnan_expedition_leader_var
			}
			remove_variable = yunnan_expedition_leader_var
			set_character_busy_and_immortal = no
		}
		trigger_event = {
			id = dsk_yunnan_valley.12
			popup = yes
		}
		remove_variable = on_expedition
		remove_variable = entered_into_yunnan
		remove_variable = expedition_concluded
		remove_modifier = modifier_large_expedition_cost
		remove_variable = yunnan_expedition_progress_var
	}
	event_outcome_completed_desc = {
		desc = event_effects_yunnan_valley_expedition_success_tt
	}
	timeout = 1825	# five years
	on_timeout = {
		remove_variable = yunnan_expedition_progress_var
		remove_variable = on_expedition
		remove_variable = yunnan_has_started
		remove_variable = entered_into_yunnan
		remove_variable = expedition_concluded
		remove_modifier = modifier_large_expedition_cost
		trigger_event = {
			id = dsk_yunnan_valley.13
			popup = yes
		}
		random_scope_character = {
			limit = {
				has_variable = yunnan_expedition_leader_var
			}
			remove_variable = yunnan_expedition_leader_var
			set_character_busy_and_immortal = no
		}
		add_modifier = {
			name = valley_expedition_failure
			days = normal_modifier_time
		}
	}
	fail = {
		custom_tooltip = {
			text = je_yunnan_valley_expedition_failure_tt
			OR = {
				NOT = {
					any_scope_character = {
						has_variable = yunnan_expedition_leader_var
					}
				}
				NOT = {
					has_variable = yunnan_has_started
				}
			}
			
		}
	}
	on_fail = {
		remove_variable = yunnan_expedition_progress_var
		remove_variable = on_expedition
		remove_variable = yunnan_has_started
		remove_variable = entered_into_yunnan
		remove_variable = expedition_concluded
		remove_modifier = modifier_large_expedition_cost
		trigger_event = {
			id = dsk_yunnan_valley.13
			popup = yes
		}
		random_scope_character = {
			limit = {
				has_variable = yunnan_expedition_leader_var
			}
			remove_variable = yunnan_expedition_leader_var
		}
		add_modifier = {
			name = valley_expedition_failure
			days = normal_modifier_time
		}
	}
	current_value = {
		if = {
			limit = {
				has_variable = yunnan_expedition_progress_var
			}
			value = var:yunnan_expedition_progress_var
		}
	}
	goal_add_value = {
		add = 8
	}
	weight = 200
	transferable = no
	should_be_pinned_by_default_uninvolved_or_context = yes
}

		has_interest_marker_in_region = sr:region_south_china
		country_rank >= rank_value:major_power
		is_ai = no

	possible = {
		has_interest_marker_in_region = sr:region_south_china
		country_rank >= rank_value:major_power
		has_technology_researched = colonization
		is_at_war = no
		OR = {
			is_country_type = recognized
			country_is_colonial_or_company = yes
		}

			OR = {
				has_role_of_type = general

		}
	}

		custom_tooltip = {
		text = no_tales_of_the_strange_tt
		has_variable = no_tales_of_the_strange_var
	}
	}
	complete = {
	}
    weight = 100
	transferable = no

}
