# Protectionism vs Free Trade
swedish_the_trade_problems = {
	#贸易讨论
	icon = "gfx/interface/icons/invention_icons/screw_frigate.dds"
	group = je_group_historical_content
	is_shown_when_inactive = {
		exists = c:SWE
		c:SWE ?= this
	}
	possible = {
		game_date > 1836.1.1
	}
	complete = {
		OR = {
			has_law_or_variant = law_type:law_free_trade
			has_law_or_variant = law_type:law_protectionism
		}
		OR = {
			any_scope_state = {
				is_incorporated = yes
				any_scope_building = {
					is_building_type = building_port
					level >= 2
					occupancy > 0.75
					#weekly_profit > 0
				}
				count >= 5
			}
			any_scope_state = {
				#内陆国别想了
				any_scope_building = {
					is_building_type = building_port
					level >= 8
					occupancy > 0.95
				}
			}
		}
	}
	on_complete = {
		if = {
			limit = {
				has_law_or_variant = law_type:law_free_trade
			}
			trigger_event = {
				id = sweden_flavor.3
			}
		}
		if = {
			limit = {
				has_law_or_variant = law_type:law_protectionism
			}
			trigger_event = {
				id = sweden_flavor.4
			}
		}
		if = {
			# Only for tooltip 
			limit = {
				NOR = {
					has_law_or_variant = law_type:law_free_trade
					has_law_or_variant = law_type:law_protectionism
				}
			}
			trigger_event = {
				id = sweden_flavor.4
			}
		}
	}
	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}

	}
	}

		c:SWE ?= this
		has_modifier = sweden_outdated_military
	}
	possible = {
		custom_tooltip = {
			text = has_completed_champions_of_national_defense
			NOT = {
				has_modifier = sweden_land_based_taxation_policies
			}
		}
	}
	complete = {
		army_size >= 180		#放宽条件
		has_technology_researched = breech_loading_artillery
		has_technology_researched = repeaters
		is_at_war = yes
		NOT = {
			any_scope_building = {
				is_building_type = building_barrack
				OR = {
					building_has_goods_shortage = yes
					has_active_production_method = pm_no_organization
				}
			}
		}
		any_military_formation = {
			filter = {
				is_army = yes
			}
			any_combat_unit = {
				OR = {
					has_unit_type = unit_type:combat_unit_type_siege_artillery
					has_unit_type = unit_type:combat_unit_type_shrapnel_artillery
					has_unit_type = unit_type:combat_unit_type_mobile_artillery
				}
			}
			percent >= 0.9
		}
		any_military_formation = {
			filter = {
				is_army = yes
			}
			any_combat_unit = {
				OR = {
					has_unit_type = unit_type:combat_unit_type_skirmish_infantry
					has_unit_type = unit_type:combat_unit_type_trench_infantry
					has_unit_type = unit_type:combat_unit_type_squad_infantry
					has_unit_type = unit_type:combat_unit_type_mechanized_infantry
				}
			}
			percent >= 0.9
		}
	}
	on_complete = {
		remove_modifier = sweden_outdated_military
		add_modifier = sweden_new_strong_military
	}
	invalid = {
		game_date > 1900.1.1
		is_player = no
	}
	on_invalid = {
		remove_modifier = sweden_outdated_military
		add_modifier = sweden_new_strong_military
	}
	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}
