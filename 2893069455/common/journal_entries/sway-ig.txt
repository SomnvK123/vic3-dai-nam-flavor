		remove_variable = random_event_triggered_6

			if = {
				change_variable = {

				}
			}
		}
	}

		NOT = {
			any_country = {
				has_modifier = modifier_rev_target_country
			}

		value = root.var:sway_IG_progress_var

		value = 100
	}
	progressbar = yes
	weight = 120
	should_be_pinned_by_default_uninvolved_or_context = yes
	timeout = 180
}
