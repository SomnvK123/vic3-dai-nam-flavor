	group = je_group_internal_affairs
	is_shown_when_inactive = {
		exists = c:SER
		c:SER ?= this
	}
	possible = {
			any_country = {
					relations:ROOT >= 30
		                        country_rank >= rank_value:great_power
        }
	}
	complete = {
		lca_is_in_diplomatic_play_or_war = no
			any_country = {
					relations:ROOT >= 80
		            country_rank >= rank_value:great_power
        }
	}

		  country_rank >= rank_value:great_power
	}
	on_complete = {
		trigger_event = {
			id = serb_events.1
		}

				exists = c:GBR

				exists = c:SOV
				exists = c:RUS
				exists = c:PRU

			}
	}

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}
