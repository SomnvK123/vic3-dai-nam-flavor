					}
				}
			}

je_persia_great_game = {
	#波斯大博弈
	icon = "gfx/interface/icons/event_icons/gbrus.dds"
	group = je_group_historical_content
	immediate = {
		set_variable = {
			name = other_leading_great_game_progress
			value = 0
		}
		other_set_great_game_variables = yes
		trigger_event = {
			id = 03_persia_events.25
			days = 90
		}
	}
	scripted_button = great_game_britain_persia_button
	scripted_button = great_game_russia_persia_button
	scripted_button = british_loans_button
	scripted_button = russian_loans_button
	scripted_button = end_british_loans_button
	scripted_button = end_russian_loans_button
	modifiers_while_active = {
		great_game_diplomacy_modifier
	}
	on_monthly_pulse = {
		#events = {
		#	french_pretenders.5
		#}
		effect = {
			if = {
				limit = {
					NOT = {
						has_variable = other_great_game_freeze						#This variable stops iteration, and is granted by reaching 360 with one dynasty. It's reset when this JE is completed, or when je_cement_the_rightful_dynasty fails.
					}
				}
				other_weekly_pulse_britain_progress = yes				#Calculates how much the Legitimists will get next month. Updates weekly, but is not added until the monthly.
				other_weekly_pulse_russia_progress = yes				#Calculates how much the Orleanists will get next month. Updates weekly, but is not added until the monthly.
			}
			other_define_leading_empire = yes			#The progress bar in this JE is always equivalent to the progress of the dynasty that's currently doing best. This scripted effect determines which dynasty that is, and sets fra_leading_dynasty_progress to be equal to that dynasty's progress variable.
		}
		random_events = {
			150 = 0
			1 = 03_persia_events.18			#Missionaries
			1 = 03_persia_events.16			#Study Abroad
			1 = 03_persia_events.17			#Extravagent Trips			
			1 = 03_persia_events.62			#Ship Inspection
			1 = 03_persia_events.63			#British Monopolies
			1 = 03_persia_events.64			#Trade Take Over
			1 = 03_persia_events.65			#Russian Advisors
			1 = 03_persia_events.66			#Anchorage Rights
		}
	}
	on_yearly_pulse = {
		effect = {
			if = {
				limit = {
					NOT = {
						has_variable = other_great_game_freeze
					}
				}
				change_global_variable = {
					name = british_progress_persia
					add = global_var:british_progress_persia_next
				}
				#On the monthly tick, the value calculated in the fra_weekly_pulse_bonapartist_progress effect is added to the master progress variable.
				change_global_variable = {
					name = russian_progress_persia
					add = global_var:russian_progress_persia_next
				}
				clamp_global_variable = {
					name = british_progress_persia
					max = 100
					min = 0
				}
				#These variables cannot exceed 360 or 0, to prevent bad and ugly displays.
				clamp_global_variable = {
					name = russian_progress_persia
					max = 100
					min = 0
				}
				other_define_leading_empire = yes				#See above - this ensures that the monthly tick will not momentarily send the status loc out of sync with what the bar is representing.
			}
		}
	}
	complete = {
		is_subject = no
		OR = {
			AND = {
				exists = c:GBR
				c:GBR ?= {
					country_rank >= rank_value:major_power
					lca_is_in_diplomatic_play_or_war = no
					NOT = {
						has_truce_with = ROOT
					}
				}
				custom_tooltip = {
					text = britain_great_game_full_tt
					global_var:british_progress_persia >= 100
				}
				any_neighbouring_state = {
					OR = {
						owner = c:GBR
						owner ?= {
							is_subject_of = c:GBR
						}
					}
				}
				NOR = {
					any_scope_treaty = {
						binds = c:GBR
						any_scope_article = {
							has_type = defensive_pact
						}
					}
					any_scope_treaty = {
						binds = c:GBR
						any_scope_article = {
							has_type = alliance
						}
					}
				}
			}
			AND = {
				exists = c:RUS
				c:RUS ?= {
					country_rank >= rank_value:major_power
					lca_is_in_diplomatic_play_or_war = no
					NOT = {
						has_truce_with = ROOT
					}
				}
				custom_tooltip = {
					text = russia_great_game_full_tt
					global_var:russian_progress_persia >= 100
				}
				any_neighbouring_state = {
					OR = {
						owner = c:RUS
						owner ?= {
							is_subject_of = c:RUS
						}
					}
				}
				NOR = {
					any_scope_treaty = {
						binds = c:RUS
						any_scope_article = {
							has_type = defensive_pact
						}
					}
					any_scope_treaty = {
						binds = c:RUS
						any_scope_article = {
							has_type = alliance
						}
					}
				}
			}
		}
	}
	on_complete = {
		if = {
			limit = {
				global_var:british_progress_persia >= global_var:russian_progress_persia
			}
			c:GBR ?= {
				save_scope_as = great_britain_sec_scope
				trigger_event = {
					id = 03_persia_events.80
					days = 3
				}
			}
			set_variable = {
				name = british_threat_complete_var
				value = yes
			}
		}
		if = {
			limit = {
				global_var:british_progress_persia < global_var:russian_progress_persia
			}
			c:RUS ?= {
				save_scope_as = russia_sec_scope
				trigger_event = {
					id = 03_persia_events.80
					days = 3
				}
			}
		}
		set_variable = repeat_great_game_var
		hidden_effect = {
			c:RUS ?= {
				create_bidirectional_truce = {
					country = c:GBR
					months = 4
				}
				create_bidirectional_truce = {
					country = c:BIC
					months = 4
				}
				change_relations = {
					country = c:GBR
					value = -20
				}
				change_relations = {
					country = c:BIC
					value = -20
				}
			}
		}
		persia_clear_great_game_variables = yes		#Removes all variables used for math in this JE, to save memory and performance.	
		custom_tooltip = again_great_game_tt
	}
	current_value = {
		value = root.var:other_leading_great_game_progress
	}
	goal_add_value = {
		add = 100
	}
	fail = {
		OR = {
			country_rank >= rank_value:great_power
			custom_tooltip = {
				text = has_anglo_russian_con_tt
				has_variable = anglo_russian_con_var
			}
			custom_tooltip = {
				text = has_anglo_russian_con_ttt
				c:FRA ?= {
					has_variable = three_country_xieyue
				}
			}
		}
	}
	on_fail = {
		if = {
			limit = {
				NOT = {
					has_variable = anglo_russian_con_var
				}
			}
			add_modifier = {
				name = survived_game_modifier
				years = 5
			}
		}
		add_loyalists = {
			value = 0.05
		}
		set_variable = no_more_great_game_var
		if = {
			limit = {
				has_variable = other_great_game_freeze
			}
			remove_variable = other_great_game_freeze			#Clears the freeze variable so that progress can be made if this JE is somehow activated again.
		}
		persia_clear_great_game_variables = yes		#Removes all variables used for math in this JE, to save memory and performance.
	}
	invalid = {
		NOT = {
			exists = c:GBR
		}
		NOT = {
			exists = c:RUS
		}
	}
	on_invalid = {
		set_variable = no_more_great_game_var
		add_modifier = {
			name = survived_game_modifier
			years = 5
		}
	}
	progressbar = yes
	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
