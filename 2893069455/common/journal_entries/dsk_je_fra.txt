			}
		}

				has_law_or_variant = law_type:law_presidential_republic

					text = je_fall_of_july_monarchy_goal.1

						is_powerful = yes
					}

				year = 1848
			}
		}
	}
	invalid = {

			any_scope_character = {
				is_ruler_of_own_country = yes

					has_ideology = ideology:ideology_orleanist
