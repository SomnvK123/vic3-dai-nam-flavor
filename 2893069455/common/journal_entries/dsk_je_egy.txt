je_muhammad_alis_reforms = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	group = je_group_historical_content
	is_shown_in_lobby = {
		c:EGY ?= this
	}
	complete = {
		scope:journal_entry = {
			is_goal_complete = yes
		}
	}
	on_complete = {
		set_variable = {
			name = muhammad_alis_reforms_complete
		}
		trigger_event = {
			id = muhammad_alis_reform_events.7
			popup = yes
		}
	}
	timeout = 8800
	on_timeout = {
		set_variable = {
			name = muhammad_alis_reforms_complete
		}
		trigger_event = {
			id = muhammad_alis_reform_events.8
			popup = yes
		}
	}
	current_value = {
		value = var:egypt_reform_var
	}
	goal_add_value = {
		add = 4
	}
	fail = {
			NOT = {
				has_law_or_variant = law_type:law_monarchy
			}
	}
	progressbar = yes
	can_revolution_inherit = yes	#内战保护
	weight = 10000
	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_muhammad_alis_trade = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	group = je_group_historical_content
	complete = {
		GDP >= 20000000
		OR = {
			AND = {
				is_junior_in_customs_union = no
				OR = {
					any_scope_state = {
						is_incorporated = yes
						any_scope_building = {
							is_building_type = building_trade_center
							level >= 20
							occupancy > 0.75
							weekly_profit > 0
						}
					}
					market = {
						mg:fabric = {
							market_goods_exports >= 500
						}
					}
					market = {
						mg:clothes = {
							market_goods_exports >= 500
						}
					}
				}
			}
			AND = {
				is_junior_in_customs_union = yes
				OR = {
					any_scope_state = {
						is_incorporated = yes
						any_scope_building = {
							is_building_type = building_trade_center
							level >= 30
							occupancy > 0.75
							weekly_profit > 0
						}
					}
					market = {
						mg:fabric = {
							market_goods_exports >= 800
						}
					}
					market = {
						mg:clothes = {
							market_goods_exports >= 800
						}
					}
				}
			}
		}
	}
	on_complete = {
		custom_tooltip = je_muhammad_alis_advance
		change_variable = {
			name = egypt_reform_var
			add = 1
		}
		trigger_event = {
			id = muhammad_alis_reform_events.2
		}
	}
	invalid = {
		custom_tooltip = {
			text = finished_egypt_reforms_tt
			has_variable = muhammad_alis_reforms_complete
		}
	}
	can_revolution_inherit = yes	#内战保护
	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 100
}
