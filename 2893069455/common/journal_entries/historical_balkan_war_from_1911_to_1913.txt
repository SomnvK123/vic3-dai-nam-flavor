historical_balkan_war_1912 = {
	#塞尔维亚发起第一次巴尔干战争，按历史的话，黑山可能活不到1912，让塞带头

	group = je_group_internal_affairs
	is_shown_when_inactive = {
		exists = c:SER
		c:SER ?= this
		game_date > 1905.1.1
		game_date < 1912.12.1
		exists = c:TUR
		c:TUR ?= {
			#土鸡只要没被逐出巴尔干，战争必然会发生
			is_player = no
			is_subject = no
			any_scope_state = {
				OR = {
		                        state_region = s:STATE_KOSOVO
					state_region = s:STATE_SKOPIA
					state_region = s:STATE_EASTERN_SERBIA
					state_region = s:STATE_MACEDONIA
					state_region = s:STATE_THESSALIA
					state_region = s:STATE_WESTERN_THRACE
					state_region = s:STATE_DOBRUDJA
					state_region = s:STATE_NORTHERN_THRACE
					state_region = s:STATE_BULGARIA
					state_region = s:STATE_ALBANIA
					state_region = s:STATE_WEST_AEGEAN_ISLANDS
					state_region = s:STATE_EAST_AEGEAN_ISLANDS
				}
			}
		}
	}
	possible = {
		game_date > 1912.7.5
		game_date < 1912.12.1
		c:TUR ?= {
			is_player = no
			is_subject = no
		}
		OR = {
			AND = {
				c:SER ?= {
					is_player = no
					OR = {
						is_subject = no
						is_subject_of = c:TUR
					}
				}
			}
			AND = {
				c:SER ?= {
					is_player = yes
				}
				has_technology_researched = pan-nationalism
				c:TUR ?= {
					relations:root < 30
					any_scope_state = {
						OR = {
		                                                                                state_region = s:STATE_KOSOVO
							state_region = s:STATE_SKOPIA
							state_region = s:STATE_EASTERN_SERBIA
							state_region = s:STATE_WESTERN_SERBIA
						}
					}
				}
			}
		}
	}
	complete = {
		hidden_trigger = {
			always = yes
		}
	}
	immediate = {
		set_variable = kill_one_turkey
		every_country = {
			limit = {
				is_player = yes
			}
			trigger_event = {
				id = clever_dsk.108
				days = {
					3
					5
				}
			}
		}
		c:SER ?= {
			create_diplomatic_play = {
				target_country = c:TUR
				type = dp_return_state
				war = no
				add_war_goal = {
						holder = c:TUR
						type = regime_change
						target_country = c:SER
			}
			}
		}
	}
}
