re1 = {
	#尽管有两个三国同盟，但意大利才是导致同盟的原因
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	group = je_group_internal_affairs
	is_shown_when_inactive = {
		exists = c:ITA
		c:ITA ?= this
		game_date > 1882.5.20
		game_date < 1914.1.1
		has_game_rule = shijie_buff_on
	}
	possible = {
		game_date > 1882.5.20
		game_date < 1914.1.1
		NOT = {
			is_diplomatic_play_enemy_of = c:AUS
		}
		NOT = {
			has_war_with = c:AUS
		}
		NOT = {
			is_diplomatic_play_enemy_of = c:GER
		}
		NOT = {
			has_war_with = c:GER
		}
		custom_tooltip = {
			text = aus_ger_not_hit_other
			exists = c:AUS
			c:AUS ?= {
				NOT = {
					is_diplomatic_play_enemy_of = c:GER
				}
				NOT = {
					has_war_with = c:GER
				}
			}
		}
		custom_tooltip = {
			text = ger_alive
			exists = c:GER
		}
		custom_tooltip = {
			text = ita_now_alive
			exists = c:ITA
		}
	}
	complete = {
		hidden_trigger = {
			always = yes
		}
	}
	immediate = {
		every_country = {
			limit = {
				is_player = yes
			}
			trigger_event = {
				id = clever_dsk.107
				days = {
					3
					5
				}
			}
		}
		s:STATE_TUNISIA = {
			add_claim = c:ITA
		}
		if = {
			limit = {
				c:ITA ?= {
					is_player = no
				}
			}
			s:STATE_SOMALILAND = {
				add_claim = c:ITA
			}
			s:STATE_ERITREA = {
				add_claim = c:ITA
				random_scope_state = {
					limit = {
						exists = c:EGY
						c:EGY ?= {
							is_player = no
							is_subject = no
							lca_is_in_diplomatic_play_or_war = no
						}
						owner ?= {
							OR = {
								c:EGY ?= this
								is_subject_of = c:EGY
							}
						}
					}
					set_state_owner = c:ITA
				}
			}
		}
		c:ITA ?= {
			change_relations = {
				#避免利益同盟关系太差自动断裂
				country = c:AUS
				value = 130
			}
			remove_diplomatic_pact = {
				#意大利不会移除对奥地利的宿敌
				country = c:GER
				type = rivalry
			}
			if = {
				limit = {
					NOT = {
						has_diplomatic_pact = {
							who = c:AUS
							type = qwf_interest_alliance
						}
					}
				}
				create_diplomatic_pact = {
					country = c:AUS
					type = qwf_interest_alliance
				}
				create_treaty = {
					name = central_powers
					first_country = c:ITA
					second_country = c:AUS
					is_draft = no
					binding_period = {
						years = 30
					}
					articles_to_create = {
						{
							article = alliance
						}
					}
				}
			}
		}
		c:GER ?= {
			change_relations = {
				country = c:AUS
				value = 130
			}
			remove_diplomatic_pact = {
				country = c:AUS
				type = rivalry
			}
			if = {
				limit = {
					NOT = {
						has_diplomatic_pact = {
							who = c:AUS
							type = qwf_interest_alliance
						}
					}
				}
				create_diplomatic_pact = {
					country = c:AUS
					type = qwf_interest_alliance
				}
				create_treaty = {
					name = central_powers
					first_country = c:GER
					second_country = c:AUS
					is_draft = no
					binding_period = {
						years = 30
					}
					articles_to_create = {
						{
							article = alliance
						}
					}
				}
			}
			remove_diplomatic_pact = {
				country = c:ITA
				type = rivalry
			}
			change_relations = {
				country = c:ITA
				value = 130
			}
		}
			c:GER ?= {
			if = {
				limit = {
					NOT = {
						has_diplomatic_pact = {
							who = c:ITA
							type = qwf_interest_alliance
						}
					}
				}
				create_diplomatic_pact = {
					country = c:ITA
					type = qwf_interest_alliance
				}
				create_treaty = {
					name = central_powers
					first_country = c:GER
					second_country = c:ITA
					is_draft = no
					binding_period = {
						years = 30
					}
					articles_to_create = {
						{
							article = alliance
						}
					}
				}
			}
		}
			change_relations = {
				country = c:GBR
				value = -20
			}
			change_relations = {
				country = c:FRA
				value = -20
			}
			change_relations = {
				country = c:RUS
				value = -20
			}
		
	}
}
