je_hawaii_usa = {
	#America only
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	group = je_group_internal_affairs
	is_shown_when_inactive = {
		c:USA ?= this
		country_rank >= rank_value:great_power
		exists = c:HAW
		year > 1865
		c:HAW ?= {
			is_player = no
		}
	}
	possible = {
		c:USA ?= this
		country_rank >= rank_value:great_power
		custom_tooltip = {
			text = hawaii_reconstruction_over.tt
			has_variable = reconstruction_over
		}
		exists = c:HAW
		c:HAW ?= {
			relations:root >= relations_threshold:cordial
			NOT = {
				is_subject_of = c:USA
			}
		}
	}
	complete = {
		custom_tooltip = {
			text = je_hawaii_usa_annexed.tt
			OR = {
				has_variable = hawaii_puppet
				has_variable = hawaii_annexation
			}
		}
	}
	on_complete = {
		trigger_event = {
			id = usa_hawaii.1			#The Newlands Resolution: Hawaiian Islands secured, becomes Yankee homeland
		}
	}
	invalid = {
		OR = {
			c:HAW ?= {
				is_subject = yes
			}
			NOT = {
				exists = c:HAW
			}
		}
	}
	on_monthly_pulse = {
		random_events = {
			60 = 0
			20 = usa_hawaii.2			#Committee of Safety
			20 = usa_hawaii.3			#Yankee settlers on hawaii
			100 = usa_hawaii.4			#Rebellion in Hawaii
			100 = usa_hawaii.5			#Approached for annexation
		}
	}
	should_be_pinned_by_default_uninvolved_or_context = yes
}
