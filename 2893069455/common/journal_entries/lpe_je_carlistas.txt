je_carlist_victory = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	group = je_group_historical_content
	is_shown_in_lobby = {
                                OR = {
		c:SPA ?= this
		c:SPC ?= this
                                was_formed_from = SPA
	}
	}
	modifiers_while_active = {
		escaramuzas_guerra_carlista
		victoria_carlsita
		unidad_carlista
	}
	immediate = {
		if = {
			limit = {
				NOT = {
					has_variable = espanna_carlista
				}
			}
			set_variable = espanna_carlista
		}
		if = {
			limit = {
				NOT = {
					has_modifier = guerrilla_carlista
				}
			}
			add_modifier = {
				name = guerrilla_carlista
				days = short_modifier_time
				is_decaying = yes
			}
		}
		root = {
			save_scope_as = car_scope
		}
		#c:SPA ?= {
		#	save_scope_as = spa_scope # estado_expedicion_gomez
		#}
	}
	progress_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_cristininos_victory_desc_1
				trigger = {
					c:SPC ?= {
						NOT = {
							any_scope_general = {
								has_variable = general_expedicion
							}
						}
					}
				}
			}
			triggered_desc = {
				desc = je_cristininos_victory_desc_3
				trigger = {
					c:SPC ?= {
						any_scope_general = {
							has_variable = general_expedicion
						}
					}
					c:SPA ?= {
						any_scope_general = {
							has_variable = expedition_contra
						}
					}
				}
			}
			triggered_desc = {
				desc = je_cristininos_victory_desc_2
				trigger = {
					c:SPC ?= {
						any_scope_general = {
							has_variable = general_expedicion
						}
					}
				}
			}
		}
	}
	on_weekly_pulse = {
		#effect = {
		#	c:SPA ?= {
		#		random_scope_state = {
		#			limit = { has_variable = var_estado_expedicion_gomez }
		#			save_scope_as = estado_expedicion_gomez
		#		}
		#	}
		#}
		events = {
			car.3			# Exilio de la familia real carlista
		}
	}
	complete = {
		custom_tooltip = {
			text = win_civil_war
                                                OR = {
				c:SPA ?= this
			NOT = {
				exists = c:SPA
			}
			}
		}
		custom_tooltip = {
			text = no_guerra_civil
			is_in_civil_war = no
		}
	}
	scripted_button = expedicion_gomez
	scripted_button = expedicion_real
	scripted_button = alzamiento_carlista
	scripted_progress_bar = alzamientos_carlistas_bar
	on_complete = {
		add_modifier = {
			name = victoria_carlsita
			days = long_modifier_time
			is_decaying = yes
		}
		custom_tooltip = {
			text = je_carlist_victory_1
			add_journal_entry = {
				type = je_deuda_esp
			}
			add_journal_entry = {
				type = je_infraestructura_esp
			}
			add_journal_entry = {
				type = je_falta_capital_esp
			}
			#if = {
			#	limit = { 
			#		any_direct_subject = {
			#			has_journal_entry = je_economia_agraria_cub
			#		}
			#	}
			#	add_journal_entry = { type = je_economia_agraria_cub_esp }
			#}
		}
		# Limpiar Variables
		limpiar_variables_guerra_carlista = yes
		hidden_effect = {
			every_country = {
				limit = {
					NOT = {
						is_country_type = decentralized
					}
				}
				c:SPA ?= {
					save_temporary_scope_as = car_scope
				}
				post_notification = fin_guerra_carlista_carlsitas
			}
			every_scope_general = {
				limit = {
					is_busy = yes
				}
				set_character_busy_and_immortal = no
				remove_variable = general_expedicion
			}
			if = {
				limit = {
					NOT = {
						has_variable = espanna_carlista
					}
				}
				set_variable = espanna_carlista
			}
			if = {
				limit = {
					has_modifier = escaramuzas_guerra_carlista
				}
				remove_modifier = escaramuzas_guerra_carlista
			}
			#if = {
			#	limit = { has_modifier = reclutamiento_de_expediciones }
			#	remove_modifier = reclutamiento_de_expediciones
			#} 
			if = {
				limit = {
					has_modifier = guerrilla_carlista
				}
				remove_modifier = guerrilla_carlista
			}
		}
	}
	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 100
}

je_cristininos_victory = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	group = je_group_historical_content
	modifiers_while_active = {
		escaramuzas_guerra_carlista
		victoria_isabelina
	}
	complete = {
		custom_tooltip = {
			text = derrota_carlistas
			hidden_trigger = {
				NOT = {
					exists = c:SPC
				}
			}
		}
		custom_tooltip = {
			text = no_guerra_civil
			is_in_civil_war = no
			is_revolutionary = no
		}
	}
	immediate = {
		#if = {
		#	limit = { 
		#		NOT = { has_modifier = reclutamiento_de_expediciones_esp }
		#	}
		#	add_modifier = {
		#		name = reclutamiento_de_expediciones_esp
		#		days = short_modifier_time
		#		is_decaying = yes
		#	}
		#}
		c:SPC ?= {
			save_scope_as = car_scope
		}
	}
	progress_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_cristininos_victory_desc_1
				trigger = {
					c:SPC ?= {
						NOT = {
							any_scope_general = {
								has_variable = general_expedicion
							}
						}
					}
				}
			}
			triggered_desc = {
				desc = je_cristininos_victory_desc_3
				trigger = {
					c:SPC ?= {
						any_scope_general = {
							has_variable = general_expedicion
						}
					}
					c:SPA ?= {
						any_scope_general = {
							has_variable = expedition_contra
						}
					}
				}
			}
			triggered_desc = {
				desc = je_cristininos_victory_desc_2
				trigger = {
					c:SPC ?= {
						any_scope_general = {
							has_variable = general_expedicion
						}
					}
				}
			}
		}
	}
	scripted_button = expedición_gomez_but
	scripted_progress_bar = alzamientos_carlistas_bar
	on_monthly_pulse = {
		random_events = {
			40 = 0			# Nada
			15 = esp.20			# descontento expedicion Gomez 
			6 = esp.17			# salarios ejercito
			2 = esp.4			# Sitio de bilbao
			1 = esp.21			# captura de gomez
			10 = esp.40			# Cagada general
		}
	}
	on_complete = {
                                remove_modifier = deslegitmacion_regimen
		add_modifier = {
			name = victoria_isabelina
			days = long_modifier_time
			is_decaying = yes
		}
		if = {
			limit = {
				exists = global_var:derrotas_carlistas
			}
			if = {
				limit = {
					global_var:derrotas_carlistas = 0
				}
				trigger_event = {
					id = esp.23
					popup = yes
				}
				hidden_effect = {
					set_strategy = ai_consolidar_dominio_filipinas
				}
			}
			change_global_variable = {
				name = derrotas_carlistas
				add = 1
			}
		}
		else = {
			# Es la primera guerra carlista
			set_global_variable = {
				name = derrotas_carlistas
				value = 1
			}
		}
		# Variables
		limpiar_variables_guerra_carlista = yes
		custom_tooltip = {
			text = je_cristininos_victory_tt
			remove_modifier = escaramuzas_guerra_carlista
			remove_modifier = bilbao_asegurado
			remove_modifier = bilbao_perdido
			eliminar_modificador = yes
			if = {
				limit = {
					has_modifier = parar_expedicion
				}
				remove_modifier = parar_expedicion
			}
			if = {
				limit = {
					any_scope_general = {
						has_variable = expedition_contra
						is_busy = yes
					}
				}
				random_scope_general = {
					limit = {
						has_variable = expedition_contra
						is_busy = yes
					}
					remove_variable = expedition_contra
					set_character_busy_and_immortal = no
				}
			}
			every_scope_state = {
				limit = {
					has_modifier = expedicion_gomez
				}
				remove_modifier = expedicion_gomez
			}
			every_scope_general = {
				remove_modifier = salarios_bajos
				remove_modifier = efectos_marcha_real
				remove_modifier = guerreando_por_bilbao
			}
			every_scope_state = {
				limit = {
					is_incorporated = no
					OR = {
						state_region = s:STATE_BASQUE_COUNTRY
						state_region = s:STATE_ARAGON
						state_region = s:STATE_CATALONIA
					}
				}
				set_state_type = incorporated
			}
			every_scope_state = {
				every_scope_building = {
					limit = {
						OR = {
							is_building_type = building_barrack
							is_building_type = building_conscription_center
						}
						has_modifier = expedicion_gomez2
					}
					remove_modifier = expedicion_gomez2
				}
			}
		}
	}
	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 100
}

je_cristinos_victory_fuera = {
	icon = "gfx/interface/icons/event_icons/spa2.dds"
	group = je_group_historical_content
	immediate = {
		if = {
			limit = {
				ROOT ?= c:FRA
			}
			c:FRA ?= {
				trigger_event = {
					id = caw.1
					days = 1
					popup = yes
				}
			}
		}
		else_if = {
			limit = {
				ROOT ?= c:GBR
			}
			c:GBR ?= {
				trigger_event = {
					id = caw.1
					days = 2
					popup = yes
				}
			}
		}
		c:FRA ?= {
			save_scope_as = fra_scope
		}
		c:GBR ?= {
			save_scope_as = gbr_scope
		}
		c:POR ?= {
			save_scope_as = por_scope
		}
		c:SPA ?= {
			random_scope_war = {
				save_scope_as = guerra_carlista
			}
			save_scope_as = spa_scope
		}
		c:SPC ?= {
			save_scope_as = car_scope
		}
		add_modifier = {
			name = efectivos_en_la_guerra_carlista
			days = 730
                        is_decaying = yes
		}
	}
	complete = {
		custom_tooltip = {
			text = je_cristinos_victory_fuera_tt
			hidden_trigger = {
				NOT = {
					exists = c:SPC
				}
			}
		}
	}
	on_complete = {
		change_relations = {
			country = scope:spa_scope
			value = 10
		}
		trigger_event = {
			id = caw.6
			popup = yes
		}
		if = {
			limit = {
				is_ai = yes
			}
			every_scope_treaty = {
				limit = {
					binds = scope:spa_scope
					any_scope_article = {
						has_type = military_assistance
					}
				}
				withdraw = {
					country = root
				}
			}
		}
	}
	fail = {
		OR = {
			custom_tooltip = {
				text = je_cristinos_victory_fuera_tt1
				c:SPA ?= {
					has_variable = espanna_carlista
				}
			}
			#NOT = {
			#	has_diplomatic_pact = {
			#		type = military_assistance
			#		who = scope:spa_scope
			#	}
			#}
		}
	}
	on_fail = {
		if = {
			limit = {
				ROOT ?= c:FRA
			}
			if = {
				limit = {
					has_modifier = efectivos_en_la_guerra_carlista
				}
				remove_modifier = efectivos_en_la_guerra_carlista
			}
			scope:spa_scope = {
				if = {
					limit = {
						has_journal_entry = je_cristininos_victory
						je:je_cristininos_victory = {
							has_modifier = apoyo_de_francia
						}
					}
					show_as_tooltip = {
						je:je_cristininos_victory = {
							remove_modifier = apoyo_de_francia
						}
					}
					trigger_event = {
						id = caw.4
						popup = yes
					}
				}
			}
		}
		else_if = {
			limit = {
				ROOT ?= c:GBR
			}
			if = {
				limit = {
					has_modifier = efectivos_en_la_guerra_carlista
				}
				remove_modifier = efectivos_en_la_guerra_carlista
			}
			scope:spa_scope = {
				if = {
					limit = {
						has_journal_entry = je_cristininos_victory
						je:je_cristininos_victory = {
							has_modifier = apoyo_de_gran_bretanna
						}
					}
					show_as_tooltip = {
						je:je_cristininos_victory = {
							remove_modifier = apoyo_de_gran_bretanna
						}
					}
					trigger_event = {
						id = caw.5
						popup = yes
					}
				}
			}
		}
		change_relations = {
			country = scope:spa_scope
			value = -20
		}
	}
	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 100
}
