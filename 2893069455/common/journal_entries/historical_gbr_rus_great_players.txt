je_rus_great_game_japan = {
	#大博弈日本
	icon = "gfx/interface/icons/event_icons/gbrus.dds"
	group = je_group_internal_affairs
	complete = {
		scope:journal_entry = {
			is_goal_complete = yes
		}
		custom_tooltip = {
			text = rus_japan_win
		}
	}
	fail = {
		OR = {
			custom_tooltip = {
				has_global_variable = gg_japan_win_gbr
				text = rus_japan_loss
			}
			custom_tooltip = {
				text = jap_not_existnow_ttt
				NOT = {
					exists = c:JAP
				}
			}
			custom_tooltip = {
				text = japan_stalemate
				OR = {
					c:JAP ?= {
						country_rank > rank_value:major_power
					}
					AND = {
						NOT = {
							has_global_variable = gg_japan_win_gbr
						}
						game_date >= 1900.1.1
					}
				}
			}
		}
	}
	on_complete = {
		trigger_event = {
			id = russia_expand.8
		}
		set_global_variable = gg_japan_win_rus
		custom_tooltip = {
				text = rus_votos_positivos_espana_text
		change_global_variable = {
				name = rus_votos_positivos_espana

		}
		}
	}
	current_value = {
		value = var:great_game_rus_japan
	}
	goal_add_value = {
		add = 600

	immediate = {
		set_variable = {
			name = great_game_rus_japan

		}
	}
	progressbar = yes
	on_weekly_pulse = {
		effect = {
			if = {
