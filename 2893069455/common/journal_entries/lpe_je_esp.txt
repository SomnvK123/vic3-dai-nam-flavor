je_regencia_esp = {
	icon = "gfx/interface/icons/law_icons/monarchy.dds"
	group = je_group_historical_content
	scripted_button = coronacion_de_isabel
	immediate = {
		# Regencia de Maria Cristina
		set_variable = {
			name = agitacion_regencia
			value = 141
		}		# Meses de regencia en teoria
		# Contador de la baja legitimidad
		set_variable = {
			name = contador
			value = 0
		}
	}
	modifiers_while_active = {
		regencia_esp
	}
	on_monthly_pulse = {
		effect = {
			change_variable = {
				name = agitacion_regencia
				subtract = 1
			}
		}
	}
	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					government_legitimacy < 26
					root.var:contador < 40
				}
				change_variable = {
					name = contador
					add = 1
				}
			}
			else = {
				set_variable = {
					name = contador
					value = 0
				}
			}
		}
	}
	complete = {
		trigger_complete_regencia = yes
	}
	on_complete = {
		limpiar_variables_regencia_complete = yes
		trigger_event = {
			id = esp.3
			popup = yes
		}
	}
	fail = {
		has_law_or_variant = law_type:law_monarchy
		OR = {
			ruler ?= {
				custom_tooltip = {
					text = je_regencia_esp_indicacion11
					has_template = spa_isabel_de_borbon_template
				}
			}
			AND = {
				custom_tooltip = {
					text = je_regencia_esp_indicacion5
					NOT = {
						exists = c:SPC
					}
				}
				trigger_fail_personaje_regencia = yes
				OR = {
					custom_tooltip = {
						text = je_regencia_esp_indicacion3
						government_legitimacy < 26
						root.var:contador >= 40
					}
					trigger_fail_revolucion_regencia = yes
				}
			}
		}
	}
	on_fail = {
		# Cuantos mas regentes peor se vuelve el multiplicador
		set_variable = {
			name = numero_de_nuevos_regentes
			value = 0
		}
		# Añadir el trait a isabel
		random_scope_character = {
			limit = {
				has_template = spa_isabel_de_borbon_template
			}
			add_trait = compliant			# political_appointee
			set_interest_group = root.ig:ig_armed_forces
		}
		# Contener la revuelta. Si hay
		if = {
			limit = {
				any_civil_war = {
					is_civil_war_type = revolution
				}
			}
			random_civil_war = {
				limit = {
					is_civil_war_type = revolution
				}
				this.political_movement ?= {
					add_modifier = {
						name = espartero_maneja_la_situacion
						days = short_modifier_time
						is_decaying = yes
					}
				}
			}
		}
		# Efecto genérico
		fail_regencia_elegir_nuevo_regente = yes
	}
	invalid = {
		# o no es una monarquia o isabel esta muerta
		trigger_invalid_regencia = yes
	}
	on_invalid = {
		invalid_regencia = yes
	}
	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_regencia_espartero_esp = {
	icon = "gfx/interface/icons/law_icons/monarchy.dds"
	group = je_group_historical_content
	scripted_button = coronacion_de_isabel
	immediate = {
		ruler ?= {
			save_scope_as = espartero_scope
		}
	}
	modifiers_while_active = {
		regencia_esp
	}
	on_monthly_pulse = {
		effect = {
			change_variable = {
				name = agitacion_regencia
				subtract = 1
			}
		}
	}
	complete = {
		trigger_complete_regencia = yes
	}
	on_complete = {
		limpiar_variables_regencia_complete = yes
		trigger_event = {
			id = esp.3
			popup = yes
		}
	}
	fail = {
		has_law_or_variant = law_type:law_monarchy
		OR = {
			ruler ?= {
				custom_tooltip = {
					text = je_regencia_espartero_esp_tt
					has_template = spa_isabel_de_borbon_template
				}
			}
			AND = {
				custom_tooltip = {
					text = je_regencia_espartero_esp_tt1
					NOT = {
						has_variable = periodo_de_gracia_regencia
					}
				}
				trigger_fail_personaje_regencia = yes
				trigger_fail_revolucion_regencia = yes
			}
		}
	}
	on_fail = {
		fail_regencia_elegir_nuevo_regente = yes
		# Contener la revuelta. Si hay
		if = {
			limit = {
				any_civil_war = {
					is_civil_war_type = revolution
				}
			}
			random_civil_war = {
				limit = {
					is_civil_war_type = revolution
				}
				this.political_movement ?= {
					add_modifier = {
						name = espartero_maneja_la_situacion
						days = short_modifier_time
						multiplier = 0.5
						is_decaying = yes
					}
				}
			}
		}
		if = {
			limit = {
				var:numero_de_nuevos_regentes >= 1
			}
			if = {
				limit = {
					has_modifier = demasiados_regentes
				}
				remove_modifier = demasiados_regentes
			}
			add_modifier = {
				name = demasiados_regentes
				multiplier = var:numero_de_nuevos_regentes
			}
		}
	}
	invalid = {
		# o no es una monarquia o isabel esta muerta
		trigger_invalid_regencia = yes
	}
	on_invalid = {
		invalid_regencia = yes
	}
	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
