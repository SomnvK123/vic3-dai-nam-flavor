	group = je_group_historical_content
	complete = {
		OR = {
			AND = {
				is_junior_in_customs_union = no
				OR = {
					AND = {
						country_rank >= rank_value:great_power
						market_capital.market = {
							mg:transportation = {
								market_goods_production >= 5000
							}
						}
					}
					AND = {
						country_rank = rank_value:major_power
						market_capital.market = {
							mg:transportation = {
								market_goods_production >= 3000
							}
						}
					}
					AND = {
						country_rank = rank_value:minor_power
						market_capital.market = {
							mg:transportation = {
								market_goods_production >= 2000
							}
						}
					}
					AND = {
						country_rank < rank_value:minor_power
						market_capital.market = {
							mg:transportation = {
								market_goods_production >= 1000
							}
						}
					}
				}
			}
			AND = {
				#去别人市场骗吃骗喝，那就老实修路          
				is_junior_in_customs_union = yes
				OR = {
					any_scope_state = {
						is_incorporated = yes
						any_scope_building = {
							is_building_type = building_railway
							level >= 6
							occupancy > 0.75
						}
						count >= 3
					}
					any_scope_state = {
						is_incorporated = yes
						any_scope_building = {
							is_building_type = building_railway
							level >= 15
							occupancy > 0.85
						}
					}

		}
	}
