je_against_shougun = {
	#倒幕运动
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	group = je_group_historical_content
	immediate = {
		c:USA ?= {
			save_scope_as = usa_scope
		}
		set_variable = {
			name = shogunate_var
			value = 0
		}
	}
	is_shown_in_lobby = {
		c:JAP ?= this
	}
	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					ig:ig_landowners = {
						is_powerful = no
					}
					has_insurrectionary_interest_groups = no
					is_at_war = no
					owns_entire_state_region = STATE_KANTO
				}
				change_variable = {
					name = shogunate_var
					add = 1
				}
			}
		}
	}
	complete = {
		# If-else clause to avoid errors because of missing variable before JE activates
		trigger_if = {
			limit = {
				exists = var:shogunate_var
			}
			OR = {
				custom_tooltip = {
					text = defeat_shogunate_civil_war_tt
					scope:journal_entry = {
						is_goal_complete = yes
					}
				}
				scope:journal_entry = {
					is_goal_complete = yes
				}
			}
		}
		trigger_else = {
			custom_tooltip = je_meiji_restoration_goal
		}
		ig:ig_landowners = {
			is_powerful = no
		}
		has_insurrectionary_interest_groups = no
		is_at_war = no
		owns_entire_state_region = STATE_KANTO
	}
	on_complete = {
		remove_modifier = shogun_ig_forced_to_open_market
		if = {
		limit = {
		has_law = law_type:law_bakufu
		}
		activate_law = law_type:law_oligarchy
		}
		if = {
		limit = {
		NOT = { has_variable = japan_emperor_restored }	
		}
		trigger_event = {
			id = meiji.1
		}
		set_variable = japan_emperor_restored
	    }
		set_variable = japan_restoration_complete
		set_global_variable = dsk_saigo_takamori_happen
		if = {
			limit = {
				any_country = {
					any_scope_character = {
						has_template = agitator_saigo_takamori
					}
				}
			}
			random_country = {
				limit = {
					any_scope_character = {
						has_template = agitator_saigo_takamori
					}
				}
				random_scope_character = {
					limit = {
						has_template = agitator_saigo_takamori
					}
					transfer_character = ROOT
					set_as_interest_group_leader = yes
					add_character_role = general
				}
			}
		}
		else_if = {
			limit = {
				any_character_in_exile_pool = {
					has_template = agitator_saigo_takamori
				}
			}
			random_character_in_exile_pool = {
				limit = {
					has_template = agitator_saigo_takamori
				}
				transfer_character = ROOT
				set_as_interest_group_leader = yes
				add_character_role = general
			}
		}
		else = {
			create_character = {
				template = agitator_saigo_takamori
				ig_leader = yes
			    is_general = yes		
		on_created = {
				set_career_length = {
						years = 50
						random_range = { 1.0 1.5 }
				}
				}
			}
		}
	}
	fail = {
		NOT = {
			has_law_or_variant = law_type:law_monarchy
		}
	}
	current_value = {
		value = root.var:shogunate_var
	}
	goal_add_value = {
		value = 60
	}
	progressbar = yes
	weight = 10000
	should_be_pinned_by_default_uninvolved_or_context = yes
}

				limit = {
					has_law = law_type:law_IJ_kami
					has_technology_researched = Ise_rebuild_tech_2
				}
				change_variable = {
					name = shinto_influence_var
					add = 2
				}
			}
			else_if = {
				limit = {
					has_law = law_type:law_IJ_kami
					has_technology_researched = Ise_rebuild_tech_1
				}
				change_variable = {
					name = shinto_influence_var
					add = 1
				}
			}
			else = {
				# 不满足任何技术条件时的基础增长
				change_variable = {
					name = shinto_influence_var
					add = 0.5
				}
			}
		}
		random_events = {
			80 = 0
			15 = open_japan.11			# 伊势神宫大祭
			10 = sino_event.82			# 国防妇人会
			10 = sino_event.263         
			15 = sino_event.260         #仁孝去世
		}
	}
	fail = {
		OR = {
			custom_tooltip = {
				text = japan_has_fail_tt
				has_global_variable = japan_fail_var
			}
			AND = {
				NOT = {
					has_law_or_variant = law_type:law_monarchy
				}
				NOT = {
					has_law = law_type:law_IJ_kami
				}
			}
		}
	}
	on_fail = {
		trigger_event = {
			id = open_japan.10
			popup = yes
		}
		# 人间宣言
		remove_variable = shinto_influence_var
	}

		value = root.var:shinto_influence_var
	}

		value = 200
	}

		shintoism_development_modifier
	}
