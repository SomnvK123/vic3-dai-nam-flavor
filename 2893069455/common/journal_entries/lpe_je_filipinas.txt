je_filipinas_esp = {
	icon = "gfx/interface/icons/event_icons/phi_map_now.png"
	group = je_group_foreign_affairs
	is_shown_when_inactive = {
		exists = c:PHI
                                c:PHI ?= {
                                 is_subject_of = ROOT
                }
                NOT = { has_global_variable = je_filipinas_esp_finished }
	}
	possible = {
		exists = c:PHI
                                c:PHI ?= {
                                 is_subject_of = ROOT
                }
	}
	immediate = {
		if = {
			limit = {
				any_subject_or_below = {
					c:PHI ?= this
				}
			}
			random_subject_or_below = {
				limit = {
					c:PHI ?= this
				}
				save_scope_as = phim_scope
			}
		}
		else = {
			save_scope_as = phim_scope
		}
	}
	# scripted_button = oferta_de_compra
	# scripted_button = integrar_marioneta
	complete = {
		c:PHI ?= {
			owns_entire_state_region = STATE_VISAYAS
			owns_entire_state_region = STATE_LUZON
			owns_entire_state_region = STATE_MINDANAO
		}
	}
	on_complete = {
                                                set_global_variable = je_filipinas_esp_finished
			trigger_event = {
					id = phi.3
					popup = yes
			}
			custom_tooltip = je_filipinas_esp_tt
			c:PHI ?= {
				add_modifier = {
					name = dominio_consolidado
					days = 3650
			}
			}
	}
	fail = {
		OR = {
			is_subject = yes
			custom_tooltip = {
				text = dsk_phi_no_exist_tt
				NOT = {
					exists = c:PHI
				}
			}
			c:PHI ?= {
				NOT = {
					is_direct_subject_of = root
				}
			}
		}
	}
	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_pirateria_mar_jolo_duennos_phi = {
	#摩尔海盗
	icon = "gfx/interface/icons/event_icons/sulu_priate.png"
	group = je_group_crises
	is_shown_when_inactive = {
		NOT = {
			has_global_variable = victoria_sobre_la_pirateria
		}
		NOT = {
			c:MGD ?= this
		}
		NOT = {
			c:SUL ?= this
		}
		OR = {
			any_subject_or_below = {
				any_scope_state = {
					is_coastal = yes
					region = sr:region_indonesia
				}
			}
			any_scope_state = {
				is_coastal = yes
				region = sr:region_indonesia
			}
		}
	}
	possible = {
		custom_tooltip = {
			text = victoria_sobre_la_pirateria_tt
			NOT = {
				has_global_variable = victoria_sobre_la_pirateria
			}
		}
		trigger_if = {
			OR = {
				exists = c:MGD
				exists = c:SUL
			}
			OR = {
				any_subject_or_below = {
					has_journal_entry = je_pirateria_mar_jolo_duennos_phi
				}
				tiene_estado_valido_para_piratear = yes
			}
		}
	}
	scripted_button = luchar_contra_pirateria_button
	scripted_button = stop_luchar_contra_pirateria_button
	immediate = {
		if = {
			limit = {
				NOT = {
					has_modifier = pirateria
				}
				OR = {
					AND = {
						exists = c:MGD
						root ?= c:MGD
					}
					AND = {
						exists = c:SUL
						root ?= c:SUL
					}
				}
			}
			je:je_pirateria_mar_jolo_duennos_phi = {
				add_modifier = pirateria
			}
		}
		else_if = {
			limit = {
				capital.region = sr:region_indonesia
				NOT = {
					has_modifier = asaltos_moros
				}
			}
			je:je_pirateria_mar_jolo_duennos_phi = {
				add_modifier = asaltos_moros
			}
		}
		else_if = {
			limit = {
				NOT = {
					has_modifier = asaltos_moros_overlord
				}
			}
			je:je_pirateria_mar_jolo_duennos_phi = {
				add_modifier = asaltos_moros_overlord
			}
		}
		trigger_event = {
			id = phi.7
			days = 1
			popup = yes
		}
		calculo_pirateria_mora_paises = yes
	}
	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					je:je_pirateria_mar_jolo_duennos_phi = {
						NOT = {
							has_modifier = pirateria
						}
					}
					OR = {
						AND = {
							exists = c:MGD
							root ?= c:MGD
						}
						AND = {
							exists = c:SUL
							root ?= c:SUL
						}
					}
				}
				je:je_pirateria_mar_jolo_duennos_phi = {
					remove_modifier = asaltos_moros
					remove_modifier = asaltos_moros_overlord
					add_modifier = pirateria
				}
			}
			else_if = {
				limit = {
					capital.region = sr:region_indonesia
					je:je_pirateria_mar_jolo_duennos_phi = {
						NOT = {
							has_modifier = asaltos_moros
						}
					}
				}
				je:je_pirateria_mar_jolo_duennos_phi = {
					remove_modifier = pirateria
					remove_modifier = asaltos_moros_overlord
					add_modifier = asaltos_moros
				}
			}
			else_if = {
				limit = {
					NOR = {
						capital.region = sr:region_indonesia
						je:je_pirateria_mar_jolo_duennos_phi = {
							has_modifier = asaltos_moros_overlord
						}
					}
				}
				je:je_pirateria_mar_jolo_duennos_phi = {
					remove_modifier = pirateria
					remove_modifier = asaltos_moros
					add_modifier = asaltos_moros_overlord
				}
			}
		}
	}
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_pirateria_mar_jolo_duennos_phi_status_1
			}
		}
	}
	on_yearly_pulse = {
		effect = {
			random_list = {
				10 = {
				}
				3 = {
					trigger = {
						scope:journal_entry = {
							has_modifier = pirateria
						}
						pais_algun_estado_valido_para_piratear = yes
					}
					random_state = {
						limit = {
							estado_valido_para_piratear = yes
							owner ?= {
								has_journal_entry = je_pirateria_mar_jolo_duennos_phi
							}
							NOT = {
								owner = root
							}
						}
						owner ?= {
							trigger_event = {
								id = phi.4
								popup = yes
							}
						}
					}
				}
			}
		}
	}
	complete = {
		OR = {
			AND = {
				custom_tooltip = {
					text = je_pirateria_mar_fucking_jolo_duennos_phi_tt
					OR = {
					has_global_variable = victoria_sobre_la_pirateria
				    game_date > 1838.1.1
				}
				}
			}
			AND = {
				OR = {
					c:MGD ?= {
						is_subject = yes
					}
					custom_tooltip = {
						text = je_pirateria_mar_jolo_phi_tt1
						NOT = {
							exists = c:MGD
						}
					}
				}
				OR = {
					c:SUL ?= {
						is_subject = yes
					}
					custom_tooltip = {
						text = je_pirateria_mar_jolo_phi_tt1
						NOT = {
							exists = c:SUL
						}
					}
				}
			}
		}
	}
	on_complete = {
		custom_tooltip = victoria_sobre_la_pirateria_tt_1
		set_global_variable = victoria_sobre_la_pirateria
		trigger_event = {
			id = clever_dsk.186
		}
		if = {
			limit = {
				has_modifier = luchando_contra_la_pirateria_mod
			}
			remove_modifier = luchando_contra_la_pirateria_mod
		}
		limpiar_variables_globales = yes
		limipar_variables_locales = yes
	}
	invalid = {
		NOR = {
			any_subject_or_below = {
				has_journal_entry = je_pirateria_mar_jolo_duennos_phi
			}
			any_scope_state = {
				is_coastal = yes
				region = sr:region_indonesia
			}
		}
	}
	on_invalid = {
		limipar_variables_locales = yes
	}
	can_revolution_inherit = yes
	weight = 200
	should_be_pinned_by_default_uninvolved_or_context = yes
}
