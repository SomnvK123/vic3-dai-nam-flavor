		exists = scope:ocupador_santo_domingo

			has_state_in_state_region = STATE_SANTO_DOMINGO
			trigger_if = {
				limit = {
					NOT = {
						scope:ocupador_santo_domingo = root
					}
				}
				scope:ocupador_santo_domingo = {
					is_direct_subject_of = root

je_capitania_general_republica_dominicana_esp = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	group = je_group_historical_content
	immediate = {
		limpiar_modificadores_resistencia_santo_domingo = yes
		s:STATE_SANTO_DOMINGO = {
			random_scope_state = {
				limit = {
					owner ?= {
						is_direct_subject_of = root
					}
					is_capital = yes
				}
				owner ?= {
					save_scope_as = ocupador_santo_domingo
				}
				save_scope_as = santo_domingo_scope
			}
		}
	}
	complete = {
		scope:ocupador_santo_domingo = {
			santo_domingo_es_capitania_general_interno = yes
		}
	}
	on_complete = {
		custom_tooltip = je_incorporacion_republica_dominicana_esp_tt1
		add_modifier = {
			name = reconquista_republica_dominicana
			days = long_modifier_time
			is_decaying = yes
		}
		add_modifier = {
			name = defensa_de_provincias_de_ultramar
			days = very_long_modifier_time
		}
		change_infamy = 2
	}
	fail = {
		OR = {
			custom_tooltip = {
				text = je_incorporacion_republica_dominicana_esp_tt
				NOT = {
					exists = scope:ocupador_santo_domingo
				}
			}
			scope:ocupador_santo_domingo = {
				NOT = {
					is_direct_subject_of = root
				}
			}
			s:STATE_SANTO_DOMINGO = {
				any_scope_state = {
					owner ?= {
						NOT = {
							country_has_primary_culture = cu:caribeno
						}
						country_rank > root.country_rank
					}
					is_capital = no
					is_treaty_port = no
				}
			}
		}
	}
	on_fail = {
	}
	invalid = {
		NOT = {
			any_subject_or_below = {
				country_has_primary_culture = cu:caribeno
			}
			any_scope_state = {
				state_region = {
					is_homeland = cu:caribeno
				}
			}
		}
	}
	on_invalid = {
	}
	transferable = yes
	weight = 200
	should_be_pinned_by_default_uninvolved_or_context = yes
}
