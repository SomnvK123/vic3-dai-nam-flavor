sway_to_communist = {
	#拉拢你成为共产主义者
	icon = "gfx/interface/icons/ideology_icons/ideology_leader/ideology_leader_communist.dds"
	potential = {
		owner ?= scope:actor
		owner ?= {
			has_journal_entry = je_line_struggle_1
			has_variable = supporting_communist_var
		}
		NOT = {
			has_ideology = ideology:ideology_communist
		}
	}
	possible = {
		interest_group ?= {
			#is_in_government = yes
			OR = {
				is_interest_group_type = ig_rural_folk				#农民
				is_interest_group_type = ig_armed_forces
				is_interest_group_type = ig_red_army				#红军或军队
				is_interest_group_type = ig_intelligentsia
				is_interest_group_type = ig_Marxism_scholar				#知识分子或左翼学者
				is_interest_group_type = ig_trade_unions				#工会
			}
		}
		NOR = {
			has_ideology = ideology:ideology_bonapartist
			has_ideology = ideology:ideology_orleanist
			has_ideology = ideology:ideology_legitimist
			has_ideology = ideology:ideology_royalist			#不可以有保皇主义系列
			has_ideology = ideology:ideology_slaver			#不可以是蓄奴主义者
			has_ideology = ideology:ideology_social_democrat
			has_ideology = ideology:ideology_communist
			has_ideology = ideology:ideology_anarchist
		}
	}
	effect = {
		set_ideology = ideology:ideology_communist
	}
	cooldown = {
		months = 3
	}
	show_confirmation_box = yes
	ai_chance = {
		value = 0
		if = {
			limit = {
				is_historical = no
			}
			add = 25
		}
	}
}
