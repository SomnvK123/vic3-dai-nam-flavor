santo_domingo_es_capitania_general = {
	s:STATE_SANTO_DOMINGO = {
		random_scope_state = {
			limit = {
				owner ?= {
					is_direct_subject_of = root
				}
				is_capital = yes
			}
			owner ?= {
				santo_domingo_es_capitania_general_interno = yes
			}
		}
	}
}

santo_domingo_es_capitania_general_interno = {
	is_direct_subject_of = root
	OR = {
		is_subject_type = subject_type_puppet
		is_subject_type = subject_type_colony
	}
	OR = {
	has_law_or_variant = law_type:law_presidential_republic
	has_law_or_variant = law_type:law_monarchy
	}
    custom_tooltip = {
    text = dom_provincia_espannola_ttt
	has_diplomatic_pact = {
		who = ROOT
		type = provincia_espannola
	}
    }
	trigger_if = {
		limit = {
			root = {
				OR = {
					has_law_or_variant = law_type:law_landed_voting
					has_law_or_variant = law_type:law_wealth_voting
					has_law_or_variant = law_type:law_census_voting
					has_law_or_variant = law_type:law_universal_suffrage
				}
			}
		}
                custom_tooltip = {
                text = dom_provincia_espannola_tttt
	has_diplomatic_pact = {
		who = ROOT
		type = representacion_en_cortes
	}
                }
	}
	trigger_else = {
		custom_tooltip = je_capitania_general_republica_dominicana_esp_tt
	}
}
