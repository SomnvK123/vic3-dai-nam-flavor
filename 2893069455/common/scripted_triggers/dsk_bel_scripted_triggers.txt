dsk_bel_continental_war_pressure_trigger = {
	custom_tooltip = {
		text = dsk_bel_continental_war_pressure_tt
		has_variable = dsk_bel_london_treaty
		c:FRA = {
			has_war_with = c:PRU
		}
		OR = {
			has_variable = dsk_bel_neutrality_done
			has_modifier = dsk_bel_fortress_neutrality_modifier
			has_modifier = dsk_bel_strict_neutrality_modifier
			var:dsk_bel_neutrality_var <= 18
			has_variable = dsk_bel_lowlands_interest
			var:dsk_bel_lowlands_var >= 2
			AND = {
				exists = c:FRA
				relations:c:FRA < relations_threshold:cordial
			}
			AND = {
				OR = {
					exists = c:PRU
					exists = c:GER
				}
				OR = {
					AND = { exists = c:PRU relations:c:PRU < relations_threshold:cordial }
					AND = { exists = c:GER relations:c:GER < relations_threshold:cordial }
				}
			}
		}
	}
}
