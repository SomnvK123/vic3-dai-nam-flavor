# Westernization influence triggers

# Country scope
WCR1_has_domestic_influence_available_trigger = {
    OR = {
        # Using trigger_if blocks for nicer tooltipping
        trigger_if = {
            limit = {
                WCR1_is_korea_trigger = yes
            }
            WCR1_is_korea_trigger = yes
        }
        trigger_else_if = {
            limit = {
                WCR1_is_japan_trigger = yes
            }
            WCR1_is_japan_trigger = yes
        }
        trigger_else_if = {
            limit = {
                OR = {
                WCR1_st_is_country_tag = { TAG = TUR }
                WCR1_st_is_country_tag = { TAG = TRU }
            }
            }
                OR = {
                WCR1_st_is_country_tag = { TAG = TUR }
                WCR1_st_is_country_tag = { TAG = TRU }
            }
        }
        trigger_else_if = {
            limit = {
                OR = {
                WCR1_st_is_country_tag = { TAG = PER }
                WCR1_st_is_country_tag = { TAG = QAJ }
            }
                has_dlc_feature = ep1_cosmetics # Vanilla DLC locks them or something
            }
                OR = {
                WCR1_st_is_country_tag = { TAG = PER }
                WCR1_st_is_country_tag = { TAG = QAJ }
            }
        }
        trigger_else_if = {
            limit = {
                OR = {
                WCR1_st_is_country_tag = { TAG = EGY }
                WCR1_st_is_country_tag = { TAG = ARA }
            }
            }
                OR = {
                WCR1_st_is_country_tag = { TAG = EGY }
                WCR1_st_is_country_tag = { TAG = ARA }
            }
        }
        trigger_else_if = {
            limit = {
                OR = {
                WCR1_st_is_country_tag = { TAG = CHI }
                WCR1_st_is_country_tag = { TAG = ROC }
            }
            }
                OR = {
                WCR1_st_is_country_tag = { TAG = CHI }
                WCR1_st_is_country_tag = { TAG = ROC }
            }
        }
        trigger_else_if = { # Siam Renaissance mod interop
            limit = {
                srm_is_active = yes
                WCR1_st_is_country_tag = { TAG = SIA }
            }
            WCR1_st_is_country_tag = { TAG = SIA }
        }
    }
}

# Character/pop scope
WCR1_st_owner_or_home_var_check = {
    trigger_if = {
        limit = {
            exists = owner
        }
        owner ?= {
            has_variable   = $VAR$
            THIS.var:$VAR$ = $IS_EQ_TO$
        }
    }
    trigger_else = {
        home_country ?= {
            has_variable   = $VAR$
            THIS.var:$VAR$ = $IS_EQ_TO$
        }
    }
}

WCR1_st_owner_or_home_uses_vanilla_rules = {
    trigger_if = {
        limit = {
            OR = { # If the game rules aren't on then don't even bother to save CPU time
                has_game_rule = WCR1_vanilla_KOR_westernization
                has_game_rule = WCR1_vanilla_JAP_westernization
            }
        }
        trigger_if = {
            limit = {
                exists = owner
            }
            owner ?= {
                WCR1_is_using_vanilla_rules = yes
            }
        }
        trigger_else = {
            home_country ?= {
                WCR1_is_using_vanilla_rules = yes
            }
        }
    }
    trigger_else = {
        always = no
    }
}

    scope:character ?= {
