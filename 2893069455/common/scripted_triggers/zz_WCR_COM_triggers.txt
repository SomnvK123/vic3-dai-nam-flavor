# Triggers dedicated to driving seamless compatibility with Community Outfit Mod
#
# Provides two scripted triggers COM should expect to have defined when WCR1 is active that WCR1 will use to signal intent
# to enable or disable an outfit.
#
# ==================================================================================================================== #
#
# COM_WCR1_enable_outfit
#     When true, signals intent to have an outfit be enabled; i.e., WCR1 is not trying to apply an outfit and is
#     expecting COM to provide one. 
#
# COM_WCR1_disable_outfit
#     When true, signals intent to have an outfit be disabled; i.e., WCR1 is trying to apply its own outfit and wants
#     COM to not apply a potentially conflicting outfit.
#
# Both triggers should be implemented directly in the modifier block of an outfit. No other checks with regards to
# country, character/pop, etc. are needed by COM and WCR1 will handle that on its end. Best usage for both triggers would
# be to use them in separate modifier blocks away from the normal logic (unless if otherwise acceptable). Either trigger
# returning false should be interpreted as WCR1 not wishing to interrupt the functionality of COM, and trigger logic
# should be set up such that either trigger returning false has zero impact on any outfits the triggers are used in.
#
# ==================================================================================================================== #
#
# List of WCR1-controlled COM outfits:
#   Monarchs
#     Korea
#     Japan
#
#   Militaries
#     Ottomans/Turkey
#
# ==================================================================================================================== #

REPLACE_OR_CREATE:COM_WCR1_enable_outfit = {
    com_is_active = yes
    
    # Korean monarch
    trigger_if = {
        limit = {
            monarchy_clothes_trigger = yes
            scope:character ?= {
                exists = owner
                owner  = {
                    WCR1_is_korea_trigger = yes
                }
            }
        }
        
        OR = {
            scope:character.owner = {
                WCR1_is_using_vanilla_rules = yes
                WCR1_westernization_allowed_by_rules_trigger = yes
            }
            AND = {
                scope:character ?= {
                    age > 12
                }
                OR = {
                    has_game_rule = WCR1_modded_KOR_westernization
                    has_game_rule = WCR1_mixed_KOR_westernization
                }
                WCR1_has_completed_westernization_trigger = yes
                WCR1_ruler_has_domestic_influence_trigger = yes
            }
        }
    }
    
    # Japanese monarch
    trigger_else_if = {
        limit = {
            monarchy_clothes_trigger = yes
            scope:character ?= {
                exists = owner
                owner  = {
                    WCR1_is_japan_trigger = yes
                }
            }
        }
        
        OR = {
            scope:character.owner = {
                WCR1_is_using_vanilla_rules = yes
                WCR1_westernization_allowed_by_rules_trigger = yes
            }
            AND = {
                scope:character ?= {
                    age > 12
                }
                OR = {
                    has_game_rule = WCR1_modded_JAP_westernization
                    has_game_rule = WCR1_mixed_JAP_westernization
                }
                WCR1_has_completed_westernization_trigger = yes
                WCR1_ruler_has_domestic_influence_trigger = yes
            }
        }
    }
    
    # Ottoman military
    trigger_else_if = {
        limit = {
            OR = {
                scope:character ?= {
                    owner ?= {
                    OR = {
                        WCR1_st_is_country_tag = { TAG = TUR }
                        WCR1_st_is_country_tag = { TAG = TRU }
                    }
                    }
                }
                scope:pop ?= {
                    owner ?= {
                    OR = {
                        WCR1_st_is_country_tag = { TAG = TUR }
                        WCR1_st_is_country_tag = { TAG = TRU }
                    }
                    }
                }
            }
        }
        OR = {
            AND = {
                ruler_clothes_trigger    = yes
                military_clothes_trigger = yes
                WCR1_late_army_uniform_trigger = no
                WCR1_ruler_has_domestic_influence_trigger = yes
            }
            AND = {
                WCR1_army_clothes_trigger = yes
                WCR1_late_army_uniform_trigger = no
                WCR1_army_has_domestic_influence_trigger = yes
            }
            AND = {
                WCR1_navy_clothes_trigger = yes
                WCR1_navy_has_domestic_influence_trigger = yes
                
                # [2026-03-29] Temporary fix for Turkish admirals wearing nothing after late-era uniforms are phased in
                # COM drops navy uniforms after researching Bolt Action Rifles for some reason, which conflicts with
                # WCR1's assumption that navy uniforms don't change into a late-game variety. Fixing this on COM's end
                # would be more correct technically, but it's easier to patch it here.
                # This should be removed once COM changes that behavior or adds late-game Ottoman uniforms.
                WCR1_late_army_uniform_trigger = no
            }
        }
    }
    
    trigger_else = {
        always = no
    }
}

                            WCR1_has_completed_westernization_trigger = yes
