# Eventos
hay_estado_valido_para_proclamar_junta = {
	any_scope_state = {
		estado_valido_para_proclamar_junta = yes
	}
}

estado_valido_para_proclamar_junta = {
	save_temporary_scope_as = estado_posible_para_proclamar_scope
	is_capital = no
	is_incorporated = yes
	NOT = {
		has_modifier = junta_provincial
	}
	turmoil >= 0.1
	owner ?= {
		any_political_movement = {
			OR = {
				is_political_movement_type = movement_radical
				is_political_movement_type = movement_liberal
				is_political_movement_type = movement_progresista
			}
			OR = {
				political_movement_support > 0.05
				political_movement_radicalism > 0.5
			}
			any_influenced_interest_group = {
				ig_counts_as_marginal = no
				leader = {
					has_ideology = ideology:ideology_progresista
				}
				scope:estado_posible_para_proclamar_scope = {
					ig_state_pol_strength_share = {
						target = prev
						value >= 0.15
					}
				}
			}
		}
	}
}
