es_un_general_valido_para_la_gloriosa = {
	OR = {
		has_role_of_type = general
		has_role_of_type = admiral
	}
	popularity > 0
	law_stance = {
		law = law_type:law_council_republic
		value <= neutral
	}
	interest_group ?= {
		law_stance = {
			law = law_type:law_council_republic
			value <= neutral
		}
	}
	law_stance = {
		law = law_type:law_autocracy
		value <= neutral
	}
	law_stance = {
		law = law_type:law_technocracy
		value <= neutral
	}
	law_stance = {
		law = law_type:law_oligarchy
		value <= neutral
	}
	interest_group ?= {
		ig_approval <= unhappy
	}
}

	}
