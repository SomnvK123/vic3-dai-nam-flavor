foreign_influence_trigger = {
	# add min year
	c:EGY ?= this
	NOT = {
		exists = var:foreign_influence_enabled
	}
	ruler ?= {
		OR = {
			has_culture = cu:albanian
			has_culture = cu:misri
		}
	}
	c:GBR ?= {
		country_rank >= rank_value:great_power
		has_interest_marker_in_region = sr:region_nile_basin
	}
	c:FRA ?= {
		country_rank >= rank_value:great_power
		has_interest_marker_in_region = sr:region_nile_basin
	}
}
