lca_this_is = {
	trigger_if = {
		limit = {
			exists = $TARGET$
		}
		this = $TARGET$
	}
	trigger_else = {
		hidden_trigger = {
			always = no
		}
	}
}

lca_ARoAI_is_chinese_country = {
	OR = {
		lca_this_is = {
			TARGET = c:TPG
		}
		capital.region = sr:region_north_china
		capital.region = sr:region_south_china
		capital.state_region = s:STATE_TUVA
		capital.state_region = s:STATE_JETISY
		capital.state_region = s:STATE_DZUNGARIA
		capital.state_region = s:STATE_TIANSHAN
		capital.state_region = s:STATE_QINGHAI
		capital.state_region = s:STATE_LHASA
		capital.state_region = s:STATE_NGARI
		capital.state_region = s:STATE_OUTER_MANCHURIA
		capital.state_region = s:STATE_AMUR
		capital.state_region = s:STATE_SOUTHERN_MANCHURIA
		capital.state_region = s:STATE_NORTHERN_MANCHURIA
		capital.state_region = s:STATE_SHENGJING
		capital.state_region = s:STATE_HINGGAN
	}
}

	OR = {

			TARGET = c:USA
		}
		lca_this_is = {
			TARGET = c:CSA
		}
		lca_this_is = {
			TARGET = c:ASA
		}
		capital.state_region = {
			is_homeland = cu:yankee
		}
		capital.state_region = {
			is_homeland = cu:dixie
		}
		capital.state_region = {
			is_homeland = cu:afro_american
		}
	}
}
