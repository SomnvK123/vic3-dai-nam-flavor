lca_this_is = {
	trigger_if = {
		limit = {
			exists = $TARGET$
		}
		this = $TARGET$
	}
	trigger_else = {
		hidden_trigger = {
			always = no
		}
	}
}

lca_owns_any_homeland = {
	trigger_if = {
		limit = {
			exists = $COUNTRY$
		}
		$COUNTRY$ = {
			any_scope_state = {
				state_region = {
					is_homeland = $CULTURE$
				}
			}
		}
	}
	trigger_else = {
		hidden_trigger = {
			always = no
		}
	}
}

lca_is_or_is_candidate = {
	OR = {
		lca_this_is = {
			TARGET = c:$TAG$
		}
		is_unification_candidate = $TAG$
	}
}

lca_has_hostile_attitude = {
	OR = {
		has_attitude = {
			who = $WHO$
			attitude = antagonistic
		}
		has_attitude = {
			who = $WHO$
			attitude = belligerent
		}
		has_attitude = {
			who = $WHO$
			attitude = domineering
		}
		has_attitude = {
			who = $WHO$
			attitude = rebellious
		}
	}
}

lca_is_leader_in_diplomatic_play_or_war = {
	OR = {
		is_diplomatic_play_initiator = yes
		is_diplomatic_play_target = yes
		has_revolution = yes
		has_any_secessionists_broken_out = yes
	}
}

lca_is_in_diplomatic_play_or_war = {
	OR = {
		is_at_war = yes
		is_diplomatic_play_initiator = yes
		is_diplomatic_play_target = yes
		is_diplomatic_play_committed_participant = yes
		# is_diplomatic_play_undecided_participant = yes # 未确定不算
	}
}

lca_has_alliance_or_defensive_pact_with = {
	trigger_if = {
		limit = {
			exists = $WHO$
		}
		OR = {
			any_scope_treaty = {
				binds = $WHO$
				any_scope_article = {
					has_type = alliance
				}
			}
			any_scope_treaty = {
				binds = $WHO$
				any_scope_article = {
					has_type = defensive_pact
				}
			}
		}
	}
	trigger_else = {
		hidden_trigger = {
			always = no
		}
	}
}

lca_is_weak_subject = {
	OR = {
		AND = {
			is_subject = no
			is_junior_in_customs_union = yes
		}
		is_subject_type = subject_type_customs_union
		is_subject_type = subject_type_protectorate
		is_subject_type = subject_type_tributary
	}
}

lca_has_alliance_with_and_available = {
	trigger_if = {
		limit = {
			exists = $WHO$
		}
		any_scope_treaty = {
			binds = $WHO$
			any_scope_article = {
				has_type = alliance
			}
		}
		$WHO$ = {
			lca_is_in_diplomatic_play_or_war = no
		}
	}
	trigger_else = {
		hidden_trigger = {
			always = no
		}
	}
}

lca_has_same_market_with = {
	trigger_if = {
		limit = {
			exists = market
			exists = $TARGET$.market
		}
		market = $TARGET$.market
	}
	trigger_else = {
		hidden_trigger = {
			always = no
		}
	}
}

lca_is_chinese_country = {
	custom_tooltip = lca_is_chinese_country_tt
	hidden_trigger = {
		AND = {
			NOT = {
				is_country_type = decentralized
			}
			OR = { #版图以民国秋海棠为准，不含东喜马拉雅是保护不丹
				lca_this_is = {
					TARGET = c:TPG
				}
				capital.region = sr:region_north_china
				capital.region = sr:region_south_china
				capital.state_region = s:STATE_TUVA #包括唐努
				capital.state_region = s:STATE_DZUNGARIA
				capital.state_region = s:STATE_TIANSHAN
				capital.state_region = s:STATE_QINGHAI
				capital.state_region = s:STATE_LHASA
				capital.state_region = s:STATE_NGARI
				capital.state_region = s:STATE_SOUTHERN_MANCHURIA
				capital.state_region = s:STATE_NORTHERN_MANCHURIA
				capital.state_region = s:STATE_SHENGJING
				capital.state_region = s:STATE_HINGGAN
			}
		}
	}
}

lca_is_ethiopia_country = {
	OR = {
		lca_this_is = {
			TARGET = c:ETH
		}
		                can_form_nation = ETH
                                country_has_primary_culture = cu:amhara
                                country_has_primary_culture = cu:tigray
                                country_has_primary_culture = cu:oromo
                                country_has_primary_culture = cu:sidama
		                c:ANU ?= this
		                c:OGD ?= this                            
                }
	OR = {
		capital.state_region = s:STATE_OROMIA
		capital.state_region = s:STATE_AMHARA
		capital.state_region = s:STATE_GONDER
		capital.state_region = s:STATE_ERITREA
	}
}
