			name = modifier_korean_empire

			ig:ig_landowners = {
				set_interest_group_name = ig_landowners
			}
		}
		hidden_effect = {
			set_variable = korean_empire_1
			set_variable = korean_empire_2
			set_variable = korean_empire_3
		}
		if = {
			limit = {
				exists = c:CHI
			}
			change_relations = {
				country = c:CHI
				value = -50
			}
		}
	}
