			}

				annex_with_incorporation = prev
			}
		}
	}

		name = 1848_hungary_revolution.2.a
		ai_chance = {
			base = 1
		}
		default_option = yes
		add_journal_entry = {
			type = je_postwar_reconstruction
		}
		add_modifier = {
			name = our_victory
			months = 60
			is_decaying = yes
		}
		add_loyalists = {
			value = 0.05
			culture = cu:hungarian
		}
		add_loyalists = {
			value = 0.1
			culture = cu:slovak
		}
		add_loyalists = {
			value = 0.1
			culture = cu:serb
		}
		c:AUS ?= {
			if = {
				limit = {
					is_ai = yes
				}
				add_loyalists = {
					value = 0.3
					culture = cu:hungarian
				}
				every_scope_state = {
					limit = {
						is_incorporated = no
					}
					set_state_type = incorporated
				}
			}
			else_if = {
				limit = {
					is_ai = no
				}
				every_scope_state = {
					limit = {
						is_incorporated = no
						state_region = {
							OR = {
								is_homeland = cu:slovak
								is_homeland = cu:serb
								is_homeland = cu:romanian
								is_homeland = cu:croat
							}
						}
					}
					set_state_type = incorporated
				}
			}
		}
	}
}
