namespace = car
car.1 = {
	type = country_event
	placement = ROOT
	title = car.1.t
	desc = car.1.d
	flavor = car.1.f
	event_image = {
		video = "ep1_redcoats"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3
     
	trigger = {
	}
	immediate = {
		set_variable = 27_spa_carlos_5
		if = {
			limit = {
				any_neighbouring_state = {
					is_capital = no
				}
			}
			random_neighbouring_state = {
				limit = {
					is_capital = no
				}
				set_variable = var_estado_expedicion_gomez
				save_scope_as = estado_expedicion_gomez
			}
		}
		else = {
			random_neighbouring_state = {
				set_variable = var_estado_expedicion_gomez
				save_scope_as = estado_expedicion_gomez
			}
		}
		if = {
			limit = {
				any_scope_general = {
					has_variable = is_gomez_damas
					is_ruler_of_own_country = no
				}
			}
			random_scope_general = {
				limit = {
					has_variable = is_gomez_damas
					is_ruler_of_own_country = no
				}
				save_scope_as = gomez_damas
			}
		}
		else = {
			random_scope_general = {
				limit = {
					is_ruler_of_own_country = no
				}
				save_scope_as = gomez_damas
			}
		}
		root = {
			save_scope_as = spa_car
		}
	}
	option = {
		name = car.1.a
		custom_tooltip = {
			text = car.1.tt
			scope:gomez_damas = {
				set_character_busy_and_immortal = yes
				set_variable = general_expedicion
			}
		}
		c:SPA ?= {
			trigger_event = {
				id = esp.18
				popup = yes
			}
		}
	}
}

car.2 = {
	type = country_event
	placement = ROOT
	title = car.2.t

	flavor = car.2.f
	event_image = {
		video = "votp_monarch_holding_court"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 3
     
	trigger = {
	}
	# Por el esp.2
	immediate = {
		random_scope_character = {
			limit = {
				is_ruler_of_own_country = no
				es_un_reclamante_carlista = yes
			}
			transfer_character = c:SPC
			free_from_exile_effect = yes
			set_character_as_ruler = yes
			add_character_role = politician
			if = {
				limit = {
					has_role_of_type = agitator
				}
				remove_character_role = agitator
			}
		}
		c:SPC ?= {
			ruler ?= {
				save_scope_as = lider_carlista_exiliado
			}
			save_scope_as = car_scope
		}
		c:SPA ?= {
			save_scope_as = spa_scope
		}
	}
	option = {
		name = car.2.a
		add_modifier = {
			name = lider_carlista_acogido
			days = short_modifier_time
		}
		change_relations = {
			country = c:SPA
			value = -10
		}
	}
}

car.3 = {
	type = country_event
	placement = ROOT
	title = car.3.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = car.3.d_infante_huido
				trigger = {
					exists = scope:heredero_scope
				}
			}
			triggered_desc = {
				desc = car.3.d
				trigger = {
					always = yes
				}
			}
		}
	}
	flavor = car.3.f
	event_image = {
		video = "ep1_transfer_of_authority"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 1
	trigger = {
		has_law_or_variant = law_type:law_monarchy
		NOT = {
			has_variable = evento_huida_lider_carlista
		}
		any_scope_war = {
			has_war_goal = annex_country
			any_war_participant = {
				this ?= c:SPA
			}
			OR = {
				has_war_support = {
					target = ROOT
					value <= -80
				}
				root = {
					enemy_occupation >= 0.9
				}
			}
		}
		any_scope_general = {
			is_ruler_of_own_country = no
		}
	}
	immediate = {
		set_variable = evento_huida_lider_carlista
		if = {
			limit = {
				exists = heir
				heir.age >= 17
			}
			heir = {
				exile_character_with_martyrdom = yes
			}
			random_character_in_exile_pool = {
				limit = {
					home_country ?= root
					has_ideology = ideology:ideology_carlist
				}
				save_scope_as = heredero_scope
			}
		}
		random_scope_general = {
			limit = {
				is_ruler_of_own_country = no
			}
			save_scope_as = general_regente_scope
		}
		random_scope_war = {
			limit = {
				has_war_goal = annex_country
				any_war_participant = {
					this ?= c:SPA
				}
				OR = {
					has_war_support = {
						target = ROOT
						value <= -80
					}
					root = {
						enemy_occupation >= 0.9
					}
				}
			}
			save_scope_as = guerra_carlista_scope
		}
		ruler ?= {
			save_scope_as = ruler_carlista_scope
		}
		c:SPA ?= {
			save_scope_as = spa_scope
		}
	}
	option = {
		name = car.3.a
		ai_chance = {
			base = 1
		}
		scope:guerra_carlista_scope = {
			add_war_war_support = {
				target = ROOT
				value = 20
			}
		}
		if = {
			limit = {
				exists = scope:heredero_scope
				scope:heredero_scope = {
					this.age >= 17
				}
			}
			every_country = {
				limit = {
					NOT = {
						is_country_type = decentralized
					}
				}
				post_notification = exilio_heredero_real_carlista
			}
		}
	}
	option = {
		name = car.3.b
		default_option = yes
		ai_chance = {
			base = 10
		}
		trigger = {
			exists = scope:general_regente_scope
		}
		every_country = {
			limit = {
				NOT = {
					is_country_type = decentralized
				}
			}
			post_notification = exilio_familia_real_carlista
		}
		if = {
			limit = {
					any_scope_character = {
						has_template = SPC_carlos_v_de_borbon_template
						is_ruler_of_own_country = yes
					}
			}
		ruler ?= {
			exile_character_with_martyrdom = yes
		}
		}
		activate_law = law_type:law_presidential_republic
		scope:general_regente_scope = {
			set_character_as_ruler = yes
		}
	}
}

car.5 = {
	# Evento para España
	type = country_event
	placement = ROOT
	title = car.5.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = car.5.d_maestrazgo
				trigger = {
					has_variable = alzamiento_maestrazgo
				}
			}
			triggered_desc = {
				desc = car.5.d_aragon
				trigger = {
					has_variable = alzamiento_aragon
				}
			}
			triggered_desc = {
				desc = car.5.d_cataluna
				trigger = {
					has_variable = alzamiento_cataluna
				}
			}
			triggered_desc = {
				desc = car.5.d
				trigger = {
					always = yes
				}
			}
		}
	}
	flavor = car.5.f
	event_image = {
		video = "ep1_redcoats"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3
     
	trigger = {
	}
	immediate = {
		if = {
			limit = {
				OR = {
					has_variable = alzamiento_aragon
					has_variable = alzamiento_cataluna
				}
			}
			c:SPC ?= {
				random_scope_state = {
					limit = {
						state_region = s:STATE_ARAGON
					}
					save_scope_as = aragon_espa
				}
			}
		}
		if = {
			limit = {
				has_variable = alzamiento_cataluna
			}
			c:SPC ?= {
				random_scope_state = {
					limit = {
						state_region = s:STATE_CATALONIA
					}
					save_scope_as = catalunna_espa
				}
			}
		}
		c:SPA ?= {
			save_scope_as = spa_scope
		}
		c:SPC ?= {
			save_scope_as = car_scope
		}
	}
	option = {
		name = car.5.a
		default_option = yes
		trigger = {
			has_variable = alzamiento_aragon
		}
		remove_variable = alzamiento_aragon
		show_as_tooltip = {
			alzamiento_aragon_car = yes
		}
		custom_tooltip = car.4.a_tt
	}
	option = {
		name = car.5.b
		trigger = {
			has_variable = alzamiento_cataluna
		}
		remove_variable = alzamiento_cataluna
		show_as_tooltip = {
			alzamiento_cataluna_car = yes
		}
	}
	option = {
		name = car.5.c
		trigger = {
			has_variable = alzamiento_maestrazgo
		}
		remove_variable = alzamiento_maestrazgo
		show_as_tooltip = {
			alzamiento_maestrazgo_car = yes
		}
	}
}
