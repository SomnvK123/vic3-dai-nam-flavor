		name = dnf_events.2.a

			base = 50
		}

			culture = cu:czech
			value = 0.15

			culture = cu:slovak
			value = 0.15
		}
		add_radicals = {
			culture = cu:south_german
			value = 0.05
		}

			culture = cu:hungarian
			value = 0.05
		}

			limit = {
				any_state = {
					state_region = {
						this = s:STATE_LOWER_SILESIA
					}
					owner = ROOT
				}
			}
			every_state = {
				limit = {
					state_region = {
						this = s:STATE_LOWER_SILESIA
					}
				}
				state_region = {
					add_homeland = cu:czech
				}
			}
		}
		if = {
			limit = {
				any_state = {
					state_region = {
						this = s:STATE_UPPER_SILESIA
					}
					owner = ROOT
				}
			}
			every_state = {
				limit = {
					state_region = {
						this = s:STATE_UPPER_SILESIA
					}
				}
				state_region = {
					add_homeland = cu:czech
				}
			}
		}
	}
	option = {
		name = dnf_events.2.b
		ai_chance = {
			base = 50
		}
		add_primary_culture = cu:czech
		add_primary_culture = cu:slovak
		trigger = {
			any_state = {
				state_region = {
					this = s:STATE_UPPER_SILESIA
				}
				NOT = {
					owner = ROOT
				}
			}
		}
		every_state = {
			limit = {
				state_region = {
					this = s:STATE_UPPER_SILESIA
				}
			}
			state_region = {
				add_homeland = cu:czech
			}
			owner ?= {
				change_relations = {
					country = ROOT
					value = -20
				}
			}
		}
		add_loyalists = {
			culture = cu:czech
			value = 0.15
		}
		add_loyalists = {
			culture = cu:slovak
			value = 0.15
		}
		add_radicals = {
			culture = cu:south_german
			value = 0.05
		}

	type = country_event
	event_image = {
		video = "gfx/event_pictures/europenorthamerica_springtime_of_nations.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/europenorthamerica/springtime_of_nation"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	title = dnf_events.9.t
	desc = dnf_events.9.d
	flavor = dnf_events.9.f
	duration = 3
    dlc = dlc_ra_plus
	trigger = {
	}
	immediate = {
		set_global_variable = {
			name = austrian_germany
		}
	}
	option = {
		name = dnf_events.9.a
		default_option = yes
		custom_tooltip = dnf_release
		ai_chance = {
			base = 80
		}
		create_country = {
			tag = AUS
			origin = ROOT
			state = s:STATE_CENTRAL_HUNGARY.region_state:GER
			on_created = {
				c:GER ?= {
					every_scope_state = {
						limit = {
							OR = {
								state_region = {
									is_homeland = cu:hungarian
								}
								state_region = {
									is_homeland = cu:croat
								}
								state_region = {
									is_homeland = cu:romanian
								}
								state_region = {
									is_homeland = cu:czech
								}
								state_region = {
									is_homeland = cu:slovak
								}
								state_region = {
									is_homeland = cu:polish
								}
								state_region = {
									is_homeland = cu:ukrainian
								}
							}
							NOT = {
								OR = {
									state_region = {
										is_homeland = cu:north_german
									}
									state_region = {
										is_homeland = cu:south_german
									}
								}
							}
						}
						set_state_owner = c:AUS
					}
					every_scope_state = {
						limit = {
							state_region = {
								this = s:STATE_TRANSDANUBIA
							}
						}
						set_state_owner = c:AUS
					}
				}
				c:AUS ?= {
					every_scope_state = {
						limit = {
							state_region = {
								this = s:STATE_ISTRIA
							}
						}
						set_state_owner = c:GER
					}
					ig:ig_intelligentsia = {
						add_ruling_interest_group = yes
					}
					add_primary_culture = cu:hungarian
					add_primary_culture = cu:croat
					add_primary_culture = cu:romanian
					add_primary_culture = cu:czech
					add_primary_culture = cu:north_italian
					add_primary_culture = cu:slovak
					add_primary_culture = cu:polish
					add_primary_culture = cu:ukrainian
				}
			}
		}
		c:GER ?= {
			if = {
				limit = {
					has_law_or_variant = law_type:law_monarchy
				}
				create_diplomatic_pact = {
					country = c:AUS
					type = personal_union
				}
			}
			else = {
				create_diplomatic_pact = {
					country = c:AUS
					type = puppet
				}
			}
		}
	}
	option = {
		name = dnf_events.9.b
		ai_chance = {
			base = 20
		}
		add_radicals = {
			value = 0.3
			culture = cu:romanian
		}
		add_radicals = {
			value = 0.3
			culture = cu:czech
		}
		add_radicals = {
			value = 0.3
			culture = cu:hungarian
		}
		add_radicals = {
			value = 0.3
			culture = cu:slovak
		}
		add_radicals = {
			value = 0.3
			culture = cu:croat
		}
