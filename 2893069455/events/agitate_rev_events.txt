namespace = agitate_rev_events
agitate_rev_events.1 = {
	type = country_event
	title = agitate_rev_events.1.t
	desc = agitate_rev_events.1.d
	flavor = agitate_rev_events.1.f
	event_image = {
		video = "europenorthamerica_political_extremism"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_election.dds"
	duration = 1
	immediate = {
		every_scope_character = {
			limit = {
				is_ruler_of_own_country = yes
			}
			save_scope_as = are_1_ruler_scope			#正常显示人名
		}
		random_country = {
			limit = {
				has_modifier = modifier_rev_target_country
			}
			save_scope_as = rev_target_country
		}
		scope:rev_target_country = {
			random_political_movement = {
				limit = {
					is_political_movement_type = movement_communist
				}
				save_scope_as = communist_now_movement
			}
			random_political_movement = {
				limit = {
					is_political_movement_type = movement_socialist
				}
				save_scope_as = socialist_now_movement
			}
		}
	}
	option = {
		name = agitate_rev_events.1.a
		#外部煽动
		trigger = {
			any_country = {
				has_modifier = modifier_rev_target_country
			}
		}
		add_modifier = {
			name = modifier_foreign_forces
			days = 90
		}
		scope:rev_target_country = {
			scope:socialist_now_movement = {
				add_modifier = {
					name = communism_big_big_now_modifier
					months = 20
				}
			}
			add_modifier = {
				name = modifier_outside_agitation_effect
				years = 1
			}
			add_modifier = {
				name = modifier_collapsing_citadel
				months = 3
			}
			create_character = {
				is_agitator = yes
				interest_group = ig_trade_unions
				ideology = ideology_vanguardist
			}
			if = {
				limit = {
					any_revolution_ongoing = yes
				}
				trigger_event = {
					id = agitate_rev_events.6
					days = 7
					popup = yes
				}
			}
			add_technology_researched = mass_communication
			add_technology_researched = dialectics
			add_technology_researched = egalitarianism
			add_technology_researched = labor_movement
			add_technology_researched = human_rights
		}
		remove_variable = start_to_export_rev
		remove_variable = start_to_sway
		remove_variable = ready_to_sway
		remove_variable = hide_rev_on
	}
	option = {
		name = agitate_rev_events.1.b
		default_option = yes
		trigger = {
			any_country = {
				has_modifier = modifier_rev_target_country
			}
		}
		#派遣煽动家
		add_modifier = {
			name = modifier_foreign_forces
			years = 1
		}
		scope:rev_target_country = {
			scope:communist_now_movement = {
				add_modifier = {
					name = communism_big_big_now_modifier
					months = 20
				}
			}
			add_modifier = {
				name = modifier_foreign_agitators_effect
				years = 1
			}
			add_modifier = {
				name = modifier_collapsing_citadel
				months = 3
			}
			add_technology_researched = mass_communication
			add_technology_researched = dialectics
			add_technology_researched = egalitarianism
			add_technology_researched = labor_movement
			add_technology_researched = human_rights
			add_technology_researched = socialism
			add_technology_researched = anarchism
			add_technology_researched = feminism
			create_character = {
				is_agitator = yes
				interest_group = ig_rural_folk
				ideology = ideology_communist
			}
			create_character = {
				is_agitator = yes
				interest_group = ig_intelligentsia
				ideology = ideology_anarchist
			}
			create_character = {
				is_agitator = yes
				interest_group = ig_trade_unions
				ideology = ideology_vanguardist
			}
		}
		trigger_event = {
			id = agitate_rev_events.2
			days = 14
			popup = yes
		}
	}
	option = {
		name = NO_OBJECTIVE		#你未选择目标
		trigger = {
			not = {
				any_country = {
					has_modifier = modifier_rev_target_country
				}
			}
		}
		hidden_effect = {
			trigger_event = {
				id = agitate_rev_events.1
				popup = yes
			}
		}
		remove_modifier = modifier_rev_investment
	}
	option = {
		name = quit
		add_modifier = {
			name = modifier_quit
			days = normal_modifier_time
		}
		every_country = {
			limit = {
				has_modifier = modifier_rev_target_country
			}
			add_modifier = {
				name = modifier_red_in_dark
				years = 1
			}
			remove_modifier = modifier_rev_target_country
		}
	}
	remove_variable = ready_to_sway
	remove_variable = start_to_sway
	remove_variable = start_to_export_rev
	remove_variable = hide_rev_on
}

agitate_rev_events.3 = {
	# 万众擎炬火
	type = country_event
	title = agitate_rev_events.3.t
	desc = agitate_rev_events.3.d
	flavor = agitate_rev_events.3.f
	event_image = {
		video = "europenorthamerica_springtime_of_nations"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_election.dds"
	duration = 1
	# 拉拢完就会触发
	immediate = {
		random_country = {
			limit = {
				has_modifier = modifier_rev_target_country
			}
			save_scope_as = rev_target_country
			random_scope_state = {
				limit = {
					radical_fraction = {
						value <= 0.98
					}
					is_incorporated = yes
				}
				save_scope_as = loyalist_province
			}
			random_scope_state = {
				limit = {
					radical_fraction = {
						value <= 0.99
					}
					is_incorporated = yes
				}
				save_scope_as = loyalist_state
			}
			random_interest_group = {
				limit = {
					is_powerful = yes
				}
				save_scope_as = powerful_ig
			}
		}
	}
	option = {
		name = agitate_rev_events.3.a
		random_country = {
			limit = {
				has_modifier = modifier_rev_target_country
			}
			add_modifier = {
				name = modifier_collapsing_citadel
				years = 1
			}
		}
		trigger_event = {
			id = agitate_rev_events.4
			days = 3
			popup = yes
		}
		scope:loyalist_province = {
			add_radicals_in_state = {
				value = 1
				strata = lower
			}
		}
		scope:loyalist_state = {
			add_radicals_in_state = {
				value = 1
				strata = lower
			}
		}
	}
}
