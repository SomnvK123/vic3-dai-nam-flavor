			limit = {

			}
			owner ?= {
				c:PHI ?= THIS
			}
			}
			set_state_type = incorporated
		                }
			}
		custom_tooltip = {
			text = bufo_filipina

				ruler ?= {
					add_modifier = {
						name = popular_autocrat
					}
				}
				add_modifier = {
					name = warlord_legitimacy
					days = normal_modifier_time
				}
			}
		}
	}
}

phi.4 = {
	type = country_event
	placement = scope:estado_raideado_scope
	title = phi.4.t
	desc = phi.4.d
	flavor = phi.4.f
	event_image = {
		video = "votp_gunboat_diplomacy"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/unspecific/military_parade"
	icon = "gfx/interface/icons/event_icons/event_fire.dds"
	duration = 3
     
	trigger = {
		any_scope_state = {
			estado_valido_para_piratear = yes
		}
	}
	immediate = {
		random_scope_state = {
			limit = {
				estado_valido_para_piratear = yes
			}
			save_scope_as = estado_raideado_scope
		}
		random_country = {
			limit = {
				pais_tiene_piratas = yes
			}
			random_scope_state = {
				limit = {
					is_coastal = yes
				}
				save_scope_as = estado_destino_piratas
			}
			save_scope_as = destino_piratas
		}
		# set_variable = ataque_piratas_moros
	}
	option = {
		name = phi.4.a
		scope:estado_raideado_scope = {
			add_modifier = {
				name = asalto_state_moro_raided
				days = short_modifier_time
				is_decaying = yes
			}
			state_region = {
				add_devastation = 5
			}
			add_radicals_in_state = {
				value = medium_radicals
				strata = lower
			}
			add_radicals_in_state = {
				value = small_radicals
				strata = middle
			}
			if = {
				limit = {
					is_slave_state = yes
				}
				ordered_scope_pop = {
					limit = {
						is_pop_type = slaves
					}
					order_by = total_size
					move_partial_pop = {
						state = scope:estado_destino_piratas
						population = {
							# move in at most 5% of the target state's population, max 5000, or the max size of the pop, whichever is smaller
							add = scope:estado_raideado_scope.state_population
							multiply = 0.1
							max = 400
						}
					}
				}
			}
		}
		change_relations = {
			country = scope:destino_piratas
			value = -10
		}
		scope:destino_piratas = {
			trigger_event = {
				id = phi.5
				popup = yes
			}
			#show_as_tooltip = {
			#	# custom_tooltip = esclavos_nuevos_destino
			#	add_modifier = {
			#		name = asalto_state_moro_raided_beneficio
			#		days = short_modifier_time
			#	}
			#}
		}
		hidden_effect = {
			if = {
				limit = {
					is_subject = yes
				}
				overlord = {
					trigger_event = {
						id = phi.6
						popup = yes
						days = 2
					}
				}
			}
		}
	}
}

phi.7 = {
	type = country_event
	placement = ROOT
	title = phi.7.t

	flavor = phi.7.f
	event_image = {
		video = "votp_gunboat_diplomacy"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/unspecific/military_parade"
	icon = "gfx/interface/icons/event_icons/sulu_priate.png"
	duration = 3
     
	trigger = {
	}
	immediate = {
		if = {
			limit = {
				NOR = {
					root ?= c:SUL
					root ?= c:MGD
				}
				any_country = {
					OR = {
						this ?= c:SUL
						this ?= c:MGD
					}
					has_journal_entry = je_pirateria_mar_jolo_duennos_phi
				}
			}
			random_country = {
				limit = {
					OR = {
						this ?= c:SUL
						this ?= c:MGD
					}
					has_journal_entry = je_pirateria_mar_jolo_duennos_phi
				}
			}
			remove_global_variable = progreso_piratas
		}
	}
	option = {
		name = phi.7.a
		if = {
			limit = {
				exists = c:MGD
			}
			change_relations = {
				country = c:MGD
				value = -50
			}
		}
		if = {
			limit = {
				exists = c:MGD
				is_player = no
			}
			change_relations = {
				country = c:MGD
				value = -50
			}
			set_secret_goal = {
				country = c:MGD
				secret_goal = conquer
			}
		}
		if = {
			limit = {
				exists = c:SUL
				is_player = no
			}
			change_relations = {
				country = c:SUL
				value = -50
			}
			set_secret_goal = {
				country = c:SUL
				secret_goal = conquer
			}
		}
		if = {
			limit = {
				exists = c:SUL
			}
			change_relations = {
				country = c:SUL
				value = -50
			}
		}
		ai_chance = {
			base = 90
		}
	}
}
