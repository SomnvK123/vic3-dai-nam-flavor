	}
	immediate = {
		c:KOR ?= {
			save_scope_as = jap_restoration_167_ruler_scope
		}
		c:JAP ?= {
			save_scope_as = jap_restoration_168_ruler_scope
		}
		scope:jap_restoration_167_ruler_scope = {
			every_scope_character = {
				limit = {
					is_ruler_of_own_country = yes
					NOT = {
						has_trait = meticulous
					}
				}
				exile_character_with_martyrdom = yes
			}
		}
		add_modifier = modifier_regency
		ruler ?= {
				if = {
					limit = {
						is_immortal = yes
					}
					set_character_immortal = no
				}
			}
		remove_regency_modifier = yes
		scope:jap_restoration_168_ruler_scope = {
			every_scope_character = {
				limit = {
					is_ruler_of_own_country = yes
					is_historical = yes
				}
				exile_character_with_martyrdom = yes
			}
			every_scope_character = {
				limit = {
					is_ruler_of_own_country = no
					is_heir_of_own_country = yes
					is_historical = yes
				}
				exile_character_with_martyrdom = yes
			}
		}
	}
	option = {
		# 만세! 만세! 만만세!
		default_option = yes
		name = jku.15.a
		if = {
			limit = {
				any_character_in_exile_pool = {
					is_historical = yes
					culture = cu:japanese
				}
			}
			random_character_in_exile_pool = {
				limit = {
					is_historical = yes
					culture = cu:japanese
				}
				transfer_character = ROOT
				add_character_role = general
			}
		}
		if = {
			limit = {
				any_character_in_exile_pool = {
					is_historical = yes
					culture = cu:korean
				}
			}
			random_character_in_exile_pool = {
				limit = {
					is_historical = yes
					culture = cu:korean
				}
				transfer_character = ROOT
				add_character_role = general
			}
		}
		scope:jap_restoration_168_ruler_scope = {
			every_scope_character = {
				limit = {
					is_ruler_of_own_country = no
					is_heir_of_own_country = no
					is_historical = yes
					has_role_of_type = politician
				}
				transfer_character = ROOT
				set_character_as_ruler = no
				set_as_interest_group_leader = yes
			}
			every_scope_character = {
				limit = {
					is_ruler_of_own_country = no
					is_heir_of_own_country = no
					is_historical = yes
					NOT = {
						has_role_of_type = politician
					}
