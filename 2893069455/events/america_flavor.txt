america_flavor.1 = {
	type = country_event
	placement = root
	event_image = {
		video = "gfx/event_pictures/unspecific_military_parade.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "texan_war_of_independence.1"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	title = america_flavor.1.t
	desc = america_flavor.1.d
	flavor = america_flavor.1.f
	duration = 3
	trigger = {
		exists = c:USA
		c:USA = ROOT
		NOT = {
			has_variable = Texsa_join_US
		}
		c:TEX ?= {
			is_at_war = no
		}
	}
	immediate = {
		set_variable = Texsa_join_US
	}
	option = {
		name = america_flavor.1.a
		default_option = yes
		c:USA ?= {
			annex = c:TEX
			hidden_effect = {
				add_modifier = {
					name = Texas_join
					years = 5
					is_decaying = yes
				}
			}
		}
		ai_chance = {
			base = 100
		}
	}
}
