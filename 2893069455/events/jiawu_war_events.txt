jiawu_war_events.6 = {
	#开化党/日本参与政变
	type = country_event
	placement = ROOT
	title = jiawu_war_events.6.t
	desc = jiawu_war_events.6.d
	flavor = jiawu_war_events.6.f
	event_image = {
		video = "europenorthamerica_political_extremism"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	duration = 3
	immediate = {
		c:CHI ?= {
			save_scope_as = china_scope
		}
		c:KOR ?= {
			save_scope_as = korea_scope
		}
		c:JAP ?= {
			save_scope_as = japan_scope
		}
		random_diplomatic_play = {
			limit = {
				target = c:CHI
			}
			save_scope_as = kor_duli_diplo_play
		}
	}
	option = {
		name = jiawu_war_events.6.a
		default_option = yes
		if = {
			limit = {
				exists = c:CHI
				exists = c:JAP
			}
			c:JAP ?= {
				save_temporary_scope_value_as = {
					name = war_reparations_amount
					value = {
						value = income
						multiply = define:NDiplomacy|DEFAULT_WAR_REPARATIONS_MONEY_TRANSFER
					}
				}
			}
			scope:kor_duli_diplo_play = {
				#add_target_backers = { root } #目标
				add_initiator_backers = {
					root
				}
				#发起方
				add_war_goal = {
					holder = c:CHI
					target_country = c:JAP
					type = enforce_treaty_article
					article = {
						article = money_transfer
						source_country = c:JAP
						target_country = c:CHI
						inputs = {
							{
								quantity = scope:war_reparations_amount
							}
						}
					}
				}
			}
		}
		#custom_tooltip = {
		#	text = jaiwu_war_tt.1
		#}
		#custom_tooltip = {
		#	text = jaiwu_war_tt.2
		#}
		add_treasury = {
			#加金
			value = -200000
		}
		c:KOR ?= {
			add_treasury = {
				value = 200000
			}
		}
		#add_journal_entry = {
		#	type = je_chi_baowei_kor1_war
		#}
	}
	option = {
		# 不感兴趣
		name = jiawu_war_events.6.b
		trigger = {
			scope:korea_scope = {
				NOT = {
					is_subject_of = ROOT
				}
			}
		}
		add_modifier = {
				name = warispolitics_backed_down
				days = 365
				is_decaying = yes
		}
		#custom_tooltip = {
		#	text = jaiwu_war_tt.6
		#}
		ai_chance = {
			base = 0
		}
	}
}
