### Choose Commander ###
dsk_yunnan_valley.2 = {
	type = country_event
	placement = ROOT
	title = dsk_yunnan_valley.2.t
	desc = dsk_yunnan_valley.2.d
	flavor = dsk_yunnan_valley.2.f
	duration = 3
	dlc = dlc_ra_plus
	event_image = {
		video = "ep1_ca_mountain_pass_civilian"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	trigger = {
		any_scope_character = {
			OR = {
				has_role = character_role_general
				has_role = character_role_admiral
			}
			is_ruler = no
			is_busy = no
		}
	}
	cancellation_trigger = {
		NOR = {
			exists = scope:yunnan_expedition_leader_1
			exists = scope:yunnan_expedition_leader_2
			exists = scope:yunnan_expedition_leader_3
		}
	}
	immediate = {
		if = {
			limit = {
				any_scope_character = {
					is_busy = no
					is_ruler = no
					OR = {
						has_role = character_role_general
						has_role = character_role_admiral
					}
					has_trait = explorer
				}
			}
			random_scope_character = {
				limit = {
					is_busy = no
					is_ruler = no
					OR = {
						has_role = character_role_general
						has_role = character_role_admiral
					}
					has_trait = explorer
				}
				save_scope_as = yunnan_expedition_leader_1
			}
		}
		random_scope_character = {
			limit = {
				is_busy = no
				is_ruler = no
				OR = {
					has_role = character_role_general
					has_role = character_role_admiral
				}
				trigger_if = {
					limit = {
						exists = scope:yunnan_expedition_leader_1
					}
					NOT = {
						this = scope:yunnan_expedition_leader_1
					}
				}
			}
			save_scope_as = yunnan_expedition_leader_2
		}
		random_scope_character = {
			limit = {
				is_busy = no
				is_ruler = no
				OR = {
					has_role = character_role_general
					has_role = character_role_admiral
				}
				NOR = {
					AND = {
						exists = scope:yunnan_expedition_leader_1
						this = scope:yunnan_expedition_leader_1
					}
					this = scope:yunnan_expedition_leader_2
				}
			}
			save_scope_as = yunnan_expedition_leader_3
		}
	}
	option = {
		name = dsk_yunnan_valley.2.a
		trigger = {
			exists = scope:yunnan_expedition_leader_1
		}
		scope:yunnan_expedition_leader_1 = {
			set_character_busy_and_immortal = yes
			set_variable = yunnan_expedition_leader_var
			ROOT = {
				set_variable = {
					name = yunnan_expedition_leader_storage_var
					value = prev
				}
			}
		}
		trigger_event = {
			id = dsk_yunnan_valley.3
			days = 30
		}
	}
	option = {
		name = dsk_yunnan_valley.2.b
		default_option = yes
		trigger = {
			exists = scope:yunnan_expedition_leader_2
		}
		scope:yunnan_expedition_leader_2 = {
			set_character_busy_and_immortal = yes
			set_variable = yunnan_expedition_leader_var
			ROOT = {
				set_variable = {
					name = yunnan_expedition_leader_storage_var
					value = prev
				}
			}
		}
		trigger_event = {
			id = dsk_yunnan_valley.3
			days = 30
		}
	}
	option = {
		name = dsk_yunnan_valley.2.c
		trigger = {
			exists = scope:yunnan_expedition_leader_3
		}
		scope:yunnan_expedition_leader_3 = {
			set_character_busy_and_immortal = yes
			set_variable = yunnan_expedition_leader_var
			ROOT = {

	dlc = dlc_ra_plus

		video = "ep1_ca_mountain_pass_military"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	immediate = {
		random_scope_character = {
			limit = {
				has_variable = yunnan_expedition_leader_var
			}
			save_scope_as = yunnan_expedition_leader
		}
	}
	cancellation_trigger = {
		OR = {
			NOT = {
				any_scope_character = {
					has_variable = yunnan_expedition_leader_var
				}
			}
		}
	}
	option = {
		name = dsk_yunnan_valley.3.a
		default_option = yes
		if = {
			limit = {
				has_journal_entry = je_yunnan_valley_expedition
			}
			show_as_tooltip = {
				add_journal_entry = {
					type = je_yunnan_valley_expedition
				}
			}
		}
		else = {
			add_journal_entry = {
				type = je_yunnan_valley_expedition
			}
		}
		custom_tooltip = expedition_costs_tt
		add_modifier = {
			name = modifier_large_expedition_cost
			multiplier = money_amount_multiplier_medium
		}
		set_variable = yunnan_has_started
	}
}

### Phase 1: Preparing for the Expedition ###
dsk_yunnan_valley.4 = {
	type = country_event
	placement = scope:current_expedition_location
	title = dsk_yunnan_valley.4.t
	desc = dsk_yunnan_valley.4.d
	flavor = dsk_yunnan_valley.4.f
	duration = 3
	dlc = dlc_ra_plus
	event_image = {
		video = "middleeast_jungle_expedition"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	cooldown = {
		years = 100
	}
	trigger = {
		NOT = {
			any_interest_group = {
				OR = {
					has_modifier = hostility_resisted
					has_modifier = resisted_being_hostile
				}
			}
		}
		var:yunnan_expedition_progress_var < 4
	}
	immediate = {
		random_scope_character = {
			limit = {
				has_variable = yunnan_expedition_leader_var
			}
			save_scope_as = yunnan_expedition_leader
		}
	}
	cancellation_trigger = {
		OR = {
			NOT = {
				any_scope_character = {
					has_variable = yunnan_expedition_leader_var
				}
			}
		}
	}
	option = {
		name = dsk_yunnan_valley.4.a
		default_option = yes
		ai_chance = {
			base = 1
		}
		every_interest_group = {
			limit = {
				OR = {
					is_interest_group_type = ig_armed_forces
					has_ideology = ideology:ideology_jingoist
					leader = {
						has_ideology = ideology:ideology_jingoist_leader
					}
				}
			}
			add_modifier = {
				name = hostility_resisted
				days = normal_modifier_time
			}
		}
		random_list = {
			1 = {
				custom_tooltip = {
					text = je_yunnan_valley_expedition_advance
					change_variable = {
						name = yunnan_expedition_progress_var
						add = 1
					}
				}
			}
			1 = {
				scope:yunnan_expedition_leader = {
					add_trait = scarred
				}
			}
			1 = {
				scope:yunnan_expedition_leader = {
					kill_character = yes
				}
			}
		}
	}
	option = {
		name = dsk_yunnan_valley.4.b
		ai_chance = {
			base = 0
		}
		custom_tooltip = {
			text = yunnan_expedition_delayed
			set_variable = {
				name = delayed_expedition_events
				months = 3
			}
		}
		every_interest_group = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_isolationist
					leader = {
						has_ideology = ideology:ideology_pacifist
					}
				}
			}
			add_modifier = {
				name = resisted_being_hostile
				days = normal_modifier_time
			}
		}
	}
}
