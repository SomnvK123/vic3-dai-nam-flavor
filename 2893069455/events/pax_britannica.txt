	trigger = {

		any_diplomatic_play = {

				OR = {
					country_rank = rank_value:minor_power
					country_rank = rank_value:major_power
				}
			}
			initiator = {
				country_rank = rank_value:great_power

					this = root
				}
			}
			NAND = {
				initiator = root.var:pax_britannica_initiator_var
				target = root.var:pax_britannica_target_var
			}
			any_scope_target_ally = {
				this = root
			}
		}
	}
	immediate = {
		#pax_britannica_event_start = yes
		random_diplomatic_play = {
			limit = {

						country_rank = rank_value:minor_power

					}
				}
				initiator = {
					country_rank = rank_value:great_power
					NOT = {
						this = root
					}
				}
				NAND = {
					initiator = root.var:pax_britannica_initiator_var
					target = root.var:pax_britannica_target_var
				}
				any_scope_target_ally = {
					this = root
				}
			}
			root = {
				set_variable = {
					name = pax_britannica_target_var

				}

					name = pax_britannica_initiator_var

				}
			}
			initiator = {
				save_scope_as = pax_britannica_initiator
			}
			target = {
				save_scope_as = pax_britannica_target
			}
