		if = {
			limit = {
				OR = {
					has_law_or_variant = law_type:law_national_militia

				}

		}

			limit = {

					this = s:STATE_BAJO_PARAGUAY

			}

				limit = {

					stalwart_defender

				}
			}

				first_name = Bernardino
				last_name = Caballero
				is_general = yes

				female = no

				traits = {
					resupply_commander
					surveyor
					expert_artillery_commander
				}
			}
		ruler ?= {
				set_variable = {
					name = ecchi_27_judao
					months = 24
				}
		}
			add_modifier = {
				name = prg_not_lose_the_war				#sets up +15% defense for Paraguay for 1 year, needs balance
				months = 12
			}
			every_scope_building = {
				remove_modifier = pm_offense_defense_readjustment
			}
		}

			#HERE WE GIVE PRG A LITTLE ADVANTAGE WITH MACHINE GUN to PAR IN CASE TAKE THIS LONG for war to start

				c:BRZ ?= {
					has_technology_researched = repeaters
