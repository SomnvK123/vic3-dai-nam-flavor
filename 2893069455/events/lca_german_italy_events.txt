lca_german_italy.1 = {

	type = country_event
	placement = ROOT
	title = lca_german_italy.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = lca_german_italy.1.d1
				trigger = {
					c:PRU ?= this
				}
			}
			triggered_desc = {
				desc = lca_german_italy.1.d2
				trigger = {
					c:SAR ?= this
				}
			}
		}
	}
	flavor = lca_german_italy.1.f
	event_image = {
		video = "gfx/event_pictures/unspecific_signed_contract.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/unspecific/signed_contract"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 1
    dlc = dlc_ra_plus
	trigger = {
	}
	immediate = {
		save_scope_as = italian_prussia_alliance_from
	}
	option = {
		# 同意
		name = lca_german_italy.1.a
		default_option = yes
		if = {
			limit = {
				NOT = {
					has_technology_researched = multilateral_alliances
				}
			}
			every_scope_ally = {
				limit = {
					country_rank < rank_value:major_power
				}
				every_scope_treaty = {
					limit = {
						binds = root
						any_scope_article = {
							has_type = alliance
						}
					}
					withdraw = {
						country = root
					}
				}
			}
		}
		scope:italian_prussia_alliance_target = {
			change_relations = {
				country = root
				value = 50
			}
		}
		create_treaty = {
			name = pru_sar_allie
			first_country = c:SAR
			second_country = c:PRU
			is_draft = no
			binding_period = {
				years = 5
			}
			articles_to_create = {
				{
					article = alliance
				}
			}
		}
		if = {
			limit = {
				country_has_primary_culture = cu:north_italian
			}
			lca_add_claim_id = {
				STATE_ID = STATE_LOMBARDY
			}
		}
		set_variable = {
			name = signed_italian_prussia_treaty_accepted
			years = 15
		}
		scope:italian_prussia_alliance_target = {
			set_variable = {
				name = signed_italian_prussia_treaty_accepted
				years = 15
			}
		}
		if = {
			limit = {
				exists = c:AUS
			}
			c:AUS ?= {
				change_relations = {
					country = root
					value = -30
				}
			}
		}
		else_if = {
			limit = {
				exists = c:SGF
			}
			c:SGF ?= {
				change_relations = {
					country = root
					value = -30
				}
			}
		}
		scope:italian_prussia_alliance_target = {
			trigger_event = {
				id = lca_german_italy.2
				popup = yes
			}
		}
		ai_chance = {
			base = 20
		}
	}
	option = {
		# 拒绝
		name = lca_german_italy.1.b
		if = {
			limit = {
				exists = c:AUS
			}
			c:AUS ?= {

					country = root
					value = 30
				}
			}

			}

					country = root
					value = 30

			}
		}

					binds = c:FRA
