				value = 0.3
				culture = cu:han
			}
			add_loyalists = {
				value = 0.3
				culture = cu:yue
			}
			add_loyalists = {
				value = 0.3
				culture = cu:min
			}
			add_loyalists = {
				value = 0.3
				culture = cu:hakka
			}
			every_scope_pop = {
				limit = {
					OR = {
						culture = cu:han
						culture = cu:yue
						culture = cu:min
						culture = cu:hakka
					}
				}
				change_pop_culture = {
					target = cu:chinese_culture
					value = 1
				}
			}
			every_scope_state = {
				limit = {
					state_region = {
						is_homeland = cu:han
					}
				}
				state_region = {
					add_homeland = cu:chinese_culture
					remove_homeland = cu:han
					remove_homeland = cu:miao
					remove_homeland = cu:min
					remove_homeland = cu:yue
					remove_homeland = cu:hakka
					remove_homeland = cu:zhuang
					remove_homeland = cu:tibetan
					remove_homeland = cu:yi
					remove_homeland = cu:mongol
					remove_homeland = cu:manchu
					remove_homeland = cu:polynesian
				}
			}
		}
		add_journal_entry = {
			type = je_culture_and_nationality_2
		}
	}
	option = {
		#民族分化
		name = culture_and_nationality.1.b
		add_primary_culture = cu:manchu
		add_primary_culture = cu:tibetan
		add_primary_culture = cu:uighur
		add_primary_culture = cu:mongol
		add_primary_culture = cu:hakka
		add_primary_culture = cu:yi
		add_primary_culture = cu:zhuang
		add_primary_culture = cu:yue
		add_primary_culture = cu:min
		add_primary_culture = cu:miao
		if = {
			limit = {
				owns_entire_state_region = STATE_RYUKYU_ISLANDS
			}
			add_primary_culture = cu:ryukyuan
		}
		custom_tooltip = {
			text = national_autonomy_will_begin
		}
	}
}
