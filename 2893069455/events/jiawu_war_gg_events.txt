namespace = jiawu_war_gg_core
jiawu_war_gg_core.1 = {
	# Great Game victory 胜利
	type = country_event
	placement = ROOT
	title = gg_core.1.t
	desc = gg_core.1.d
	flavor = gg_core.1.f
	event_image = {
		video = "ep1_power_blocs"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	duration = 3
  
	cooldown = {
		days = short_modifier_time
	}
	immediate = {
		if = {
			limit = {
				c:CHI ?= this
			}
			c:JAP ?= {
				save_scope_as = opponent_scope
			}
		}
		if = {
			limit = {
				c:JAP ?= this
			}
			c:CHI ?= {
				save_scope_as = opponent_scope
			}
		}
	}
	option = {
		name = gg_core.1.a
		default_option = yes
		set_global_variable = je_jiawu_war_wancheng_var
		if = {
			limit = {
				OR = {
					AND = {
						c:CHI ?= THIS
						has_global_variable = jiawu_chi_war
						global_var:jiawu_chi_war <= 40
					}
					AND = {
						c:JAP ?= THIS
						has_global_variable = jiawu_chi_war
						global_var:jiawu_chi_war >= 160
					}
				}
			}
			add_modifier = {
				name = modifier_great_game_victor
				days = very_long_modifier_time
				is_decaying = yes
			}
		}
		else_if = {
			limit = {
				OR = {
					AND = {
						c:CHI ?= THIS
						has_global_variable = jiawu_chi_war
						global_var:jiawu_chi_war <= 80
					}
					AND = {
						c:RUS ?= THIS
						has_global_variable = jiawu_chi_war
						global_var:jiawu_chi_war >= 120
					}
				}
			}
			add_modifier = {
				name = modifier_great_game_victor
				days = very_long_modifier_time
				is_decaying = yes
			}
		}
		else = {
			# Should never happen
			add_modifier = {
				name = modifier_great_game_victor
				days = very_long_modifier_time
				is_decaying = yes
				multiplier = 0.5
			}
		}
		if = {
			limit = {
				is_player = yes
			}
			clear_jiawu_war_variables = yes
		}
		scope:opponent_scope ?= {
			create_diplomatic_catalyst = {
				type = catalyst_event_positive
				target = root
			}
		}
	}
}
