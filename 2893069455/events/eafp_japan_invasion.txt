japan_invasion.7 = {
	#한?전쟁
	type = country_event
	event_image = {
		video = "asia_westerners_arriving"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	#on_opened_soundeffect = "event:/SFX/Events/asia/confucianism_shinto" 
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3
 
	title = japan_invasion.7.t
	desc = japan_invasion.7.d
	flavor = japan_invasion.7.f
	immediate = {
		random_country = {
			limit = {
				country_rank >= rank_value:great_power
				has_interest_marker_in_region = sr:region_northeast_asia
				NOT = {
					c:KOR ?= this
				}
			}
			save_scope_as = country_123
		}
	}
	option = {
		name = japan_invasion.7.a
		set_variable = ji_1
		country_123 = {
			change_relations = {
				country = c:KOR
				value = -50
			}
		}
		if = {
			limit = {
				c:JAP ?= {
					is_subject = yes
				}
			}
			c:KOR ?= {
				create_diplomatic_play = {
					target_country = c:JAP
					war = no
					type = dp_annex_war
				}
			}
		}
		if = {
			limit = {
				c:JAP ?= {
					is_subject = no
				}
			}
			c:KOR ?= {
				create_diplomatic_play = {
					target_country = c:JAP
					war = no
					type = dp_make_protectorate
				}
			}
		}
		change_infamy = -80
		ai_chance = {
			base = 0
		}
	}
	option = {
		name = japan_invasion.7.b
		default_option = yes
		add_radicals = {
			value = 0.25
			pop_type = officers
			culture = cu:korean
		}
		ai_chance = {
			base = 100
		}
	}
}
