dsk_spa.7 = {
	type = country_event
	placement = ROOT
	event_image = {
		video = "europenorthamerica_political_extremism"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/middleeast/battlefield_trenches"
	cooldown = {
		days = normal_modifier_time
	}
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	title = dsk_spa.7.t
	desc = dsk_spa.7.d
	flavor = dsk_spa.7.f
	duration = 3
	dlc = dlc_ra_plus
	trigger = {
	}
	immediate = {
		activate_law = law_type:law_monarchy
		remove_variable = spain_civil_war_revolt_var
		remove_variable = 27_spa_juan_3		#卡洛斯派全部暴毙
		remove_variable = 27_spa_carlos_5
		remove_variable = 27_spa_carlos_7
		remove_variable = 27_spa_carlist_jaime_3
		set_variable = 27_spa_alphonse_12
		if = {
			limit = {
				OR = {
					any_scope_character = {
						has_template = spa_isabel_de_borbon_template
						is_ruler_of_own_country = yes
					}
					any_scope_character = {
						has_template = spa_amadeo_i_de_savoy_template
						is_ruler_of_own_country = yes
					}
				}
			}
			create_character = {
				template = spa_alfonso_xii_de_borbon_template
				heir = yes
			}
		}
		if = {
			limit = {
				NOT = {
					any_scope_character = {
						has_template = spa_isabel_de_borbon_template
						is_ruler_of_own_country = yes
					}
					any_scope_character = {
						has_template = spa_amadeo_i_de_savoy_template
						is_ruler_of_own_country = yes
					}
				}
			}
			if = {
				limit = {
					any_country = {
						any_scope_character = {
							has_template = spa_alfonso_xii_de_borbon_template
						}
					}
				}
				random_country = {
					limit = {
						any_scope_character = {
							has_template = spa_alfonso_xii_de_borbon_template
						}
					}
					random_scope_character = {
						limit = {
							has_template = spa_alfonso_xii_de_borbon_template
						}
						transfer_character = ROOT
						set_character_as_ruler = yes
					}
				}
			}
			else_if = {
				limit = {
					any_character_in_exile_pool = {
						has_template = spa_alfonso_xii_de_borbon_template
					}
				}
				random_character_in_exile_pool = {
					limit = {
						has_template = spa_alfonso_xii_de_borbon_template
					}
					transfer_character = ROOT
					set_character_as_ruler = yes
				}
			}
			else = {
				add_modifier = modifier_regency
				ruler = {
					if = {
						limit = {
							is_immortal = yes
						}
						set_character_immortal = no
					}
				}
				create_character = {
					template = spa_alfonso_xii_de_borbon_template
					ruler = yes
					heir = no
					on_created = {
						set_variable = regency_complete
					}
				}
				remove_regency_modifier = yes
			}
		}
	}
	option = {
		name = dsk_spa.7.a
		default_option = yes
		set_variable = Modern_Spain		#变现代西班牙国旗
		#add_modifier = {
		#	name = modifier_Modern_Spain #现代建设
		#	months = 180
		#}
		add_loyalists = {
			value = 0.1
		}
		add_journal_entry = {
			type = je_Iberia_empire			#伊比利亚联合王国
		}
	}
}
