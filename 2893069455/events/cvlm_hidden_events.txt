namespace = cvlm_hidden

cvlm_hidden.10 = {
	type = country_event
	hidden = yes

    immediate = {
        if = { 
            # if no amicable great power
            limit = {
                NOT = {
                    any_country = {
                        country_rank >= rank_value:great_power
                        relations:root >= relations_threshold:amicable
                    }
                }
            }
            random_country = {
                limit = { country_rank >= rank_value:great_power }
                change_relations = { country = prev value = 500 }
            }
        }
        je:je_earn_recognition ?= {
            add_progress = { value = 500 name = earn_recognition_progress_bar }
        }
	}
}

cvlm_hidden.11 = {
	type = country_event
	hidden = yes

    immediate = {
        random_scope_character ={
            limit = { has_modifier = cvlm_modifier_character_governor_candidate_mark }
            if = {
                limit = { is_ruler_of_own_country = no }
                set_character_as_ruler = yes
            }
            remove_modifier = cvlm_modifier_character_governor_candidate_mark
        }
	}
}
