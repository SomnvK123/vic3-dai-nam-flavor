namespace = caw
caw.1 = {
	type = country_event
	placement = ROOT
	title = caw.1.t
	desc = caw.1.d
	flavor = caw.1.f
	event_image = {
		video = "middleeast_battlefield_trenches"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3
    dlc = dlc_ra_plus
	immediate = {
		c:SPA ?= {
			save_scope_as = spa_scope
		}
		c:SPC ?= {
			save_scope_as = car_scope
		}
	}
	cancellation_trigger = {
		NAND = {
			has_modifier = efectivos_en_la_guerra_carlista
			any_scope_treaty = {
				any_scope_article = {
					has_type = military_assistance
				}
			}
			has_journal_entry = je_cristinos_victory_fuera
		}
	}
	option = {
		name = caw.1.a
		default_option = yes
		show_as_tooltip = {
			change_relations = {
				country = c:SPA
				value = 10
			}
		}
		if = {
			limit = {
				ROOT = c:FRA
			}
			scope:spa_scope = {
				trigger_event = {
					id = caw.2
					days = 3
				}
			}
		}
		else = {
			scope:spa_scope = {
				trigger_event = {
					id = caw.3
					days = 5
				}
			}
		}
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					OR = {
						has_attitude = {
							who = scope:spa_scope
							attitude = protective
						}
						has_attitude = {
							who = scope:spa_scope
							attitude = genial
						}
						has_attitude = {
							who = scope:spa_scope
							attitude = cooperative
						}
						has_attitude = {
							who = scope:spa_scope
							attitude = conciliatory
						}
					}
				}
				add = 1
			}
		}
	}
	option = {
		name = caw.1.b
		remove_modifier = efectivos_en_la_guerra_carlista
		show_as_tooltip = {
			change_relations = {
				country = c:SPA
				value = -10
			}
		}
		if = {
			limit = {
				ROOT = c:FRA
			}
			scope:spa_scope = {
				trigger_event = {
					id = caw.4
					days = 3
				}
			}
		}
		else = {
			scope:spa_scope = {
				trigger_event = {
					id = caw.5
					days = 5
				}
			}
		}
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					OR = {
						has_attitude = {
							who = scope:spa_scope
							attitude = belligerent
						}
						has_attitude = {
							who = scope:spa_scope
							attitude = antagonistic
						}
					}
				}
				add = 1
			}
		}
	}
}
