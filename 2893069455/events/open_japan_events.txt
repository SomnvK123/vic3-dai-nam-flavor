open_japan.1 = {
	type = country_event
	placement = root
	title = open_japan.1.t
	desc = open_japan.1.d
	flavor = open_japan.1.f
	duration = 3
	dlc = dlc_ra_plus
	minor_left_icon = root
	minor_right_icon = c:USA
	cooldown = {
		days = very_long_modifier_time
	}
	icon = "gfx/interface/icons/event_icons/waving_flag_usa.dds"
	event_image = {
		video = "gfx/event_pictures/asia_westerners_arriving.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/asia/westeners_arriving_in_east_asia"
	trigger = {
		year >= 1853
		has_law_or_variant = law_type:law_isolationism
		any_country = {
			country_rank >= rank_value:great_power
			has_interest_marker_in_region = sr:region_northeast_asia
			#has_strategy = ai_strategy_monroe_doctrine
		}
	}
	immediate = {
		c:USA ?= {
			save_scope_as = usa_scope
		}
		set_variable = {
			name = japan_forced_to_open_market
			value = 0
		}
		random_scope_state = {
			limit = {
				has_port = yes
			}
			save_scope_as = infringed_port_state
		}
		ig:ig_landowners = {
			save_scope_as = shougun_ig
		}
		ruler ?= {
			save_scope_as = shogun_scope
		}
	}
	option = {
		name = open_japan.1.a
		scope:infringed_port_state = {
			add_devastation = 55
		}
		c:USA ?= {
			if = {
				limit = {
					is_ai = yes
				}
			}
			create_diplomatic_play = {
				target_country = c:JAP
				type = dp_take_treaty_port
				target_state = s:STATE_KYUSHU.region_state:JAP
				war = no
			}
		}
		change_relations = {
			country = c:USA
			value = -50
		}
		ai_chance = {
			base = 0
		}
	}
	option = {
		name = open_japan.1.b
		default_option = yes
		activate_law = law_type:law_free_trade
		create_treaty = {
			name = Kanagawa_Treaty
			first_country = c:USA
			second_country = c:JAP
			is_draft = no
			binding_period = {
				years = 5
			}
			articles_to_create = {
				{
					# 
					article = foreign_investment_rights
					source_country = c:JAP
					target_country = c:USA
				}
			}
		}
		scope:shougun_ig = {
			add_modifier = {
				name = shogunate_fall_1
				days = very_long_modifier_time
			}
		}
		add_modifier = {
			name = shogun_ig_forced_to_open_market
			days = very_long_modifier_time
		}
		scope:shogun_scope = {
			add_modifier = shogun_forced_to_open_market
		}
		every_interest_group = {
			limit = {
				OR = {
					has_ideology = ideology_paternalistic
					has_ideology = ideology_bakufu
					has_ideology = ideology_agrarian
				}
			}
			add_modifier = {
				name = undermined_ig_modifier
				days = long_modifier_time
			}
		}
		every_interest_group = {
			limit = {
				OR = {
					has_ideology = ideology_laissez_faire
					has_ideology = ideology_individualist
					has_ideology = ideology_meritocratic
					has_ideology = ideology_liberal
					has_ideology = ideology_republican
					has_ideology = ideology_egalitarian
				}
			}
			add_modifier = {
				name = ig_gaining_support
				days = long_modifier_time
			}
		}
		ai_chance = {
			base = 90
		}
	}
}

	type = country_event
	placement = root
	title = open_japan.9.t
	desc = open_japan.9.d
	flavor = open_japan.9.f
	duration = 3
	dlc = dlc_ra_plus
	gui_window = event_window_superevent_modern_full
	event_image = {
		texture = "gfx/event_pictures/soe_event_pictures/japan4.dds"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/unspecific/military_parade"
	icon = "gfx/interface/icons/event_icons/jap1.dds"
	trigger = {
	}
	minor_left_icon = c:SGN
	minor_right_icon = c:JAP
	immediate = {
		ig:ig_landowners = {
			save_scope_as = shogun_ig			# 幕府派（德川幕府）
		}
		ig:ig_intelligentsia = {
			save_scope_as = sonno_ig			# 尊攘派（长州藩主导）
		}
		s:STATE_CHUGOKU = {
			# 长州藩所在的中国地区
			save_scope_as = choshu_region
		}
		s:STATE_KANSAI = {
			# 京都所在的关西地区
			save_scope_as = kyoto_region
		}
		create_country = {
			tag = SGN
			origin = c:JAP
			state = s:STATE_TOHOKU.region_state:JAP
			state = s:STATE_HOKKAIDO.region_state:JAP
		}
		if = {
			limit = {
				owns_entire_state_region = STATE_KANTO
			}
			s:STATE_KANTO = {
				set_owner_of_provinces = {
					country = c:SGN
					provinces = {
						xEF9040
						x709040
						xD350FB
					}
				}
			}
		}
		if = {
			limit = {
				c:EZO ?= {
					is_ai = yes
				}
			}
			c:SGN = {
				annex = c:EZO
			}
		}
		c:SGN = {
			activate_law = law_type:law_monarchy
			activate_law = law_type:law_bakufu
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_KANTO
					}
					capital.state_region = {
						NOT = {
							this = s:STATE_KANTO
						}
					}
				}
				set_capital = STATE_KANTO
			}
			if = {
				limit = {
					any_country = {
						any_scope_character = {
							has_template = JAP_yoshinobu_tokugawa
						}
					}
				}
				random_country = {
					limit = {
						any_scope_character = {
							has_template = JAP_yoshinobu_tokugawa
						}
					}
					random_scope_character = {
						limit = {
							has_template = JAP_yoshinobu_tokugawa
						}
						exile_character_with_martyrdom = yes
						transfer_character = c:SGN
						set_character_as_ruler = yes
					}
				}
			}
			else_if = {
				limit = {
					any_character_in_exile_pool = {
						has_template = JAP_yoshinobu_tokugawa
					}
				}
				random_character_in_exile_pool = {
					limit = {
						has_template = JAP_yoshinobu_tokugawa
					}
					transfer_character = c:SGN
					set_character_as_ruler = yes
				}
			}
			else = {
