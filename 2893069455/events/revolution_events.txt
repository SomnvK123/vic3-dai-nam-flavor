	duration = 1
	immediate = {
		random_country = {
			limit = {
				has_modifier = modifier_rev_target_country
			}
			save_scope_as = rev_target_country
		}
	}
	option = {
		set_variable = rr_batch_3
		name = revolution_events.1.a

			any_country = {

			}
		}

		default_option = yes
		every_country = {
			limit = {
				has_modifier = modifier_rev_target_country
			}

				name = modifier_contact_party_branches
				years = 1
			}
		}
		trigger_event = {
			id = revolution_events.3
			days = 14
			popup = yes
		}
	}
	option = {
		set_variable = rr_batch_4
		name = revolution_events.1.b

			limit = {
				has_modifier = modifier_rev_target_country

			}
		}

			popup = yes

	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_election.dds"
	duration = 1
	immediate = {
		random_country = {
			limit = {
				has_modifier = modifier_rev_target_country
			}
			save_scope_as = rev_target_country
			random_scope_state = {
				save_scope_as = entrance_state
			}
		}
	}
	option = {
		name = revolution_events.2.a
		trigger = {
			any_country = {
				has_modifier = modifier_rev_target_country
			}
		}
		#避避风头，到此为止
		default_option = yes
		random_country = {
			limit = {
				has_modifier = modifier_rev_target_country
			}
			change_relations = {
				country = root
				value = -20
			}
			add_modifier = {
				name = modifier_arrest_leftists
				months = 7
			}
			remove_modifier = modifier_rev_target_country
		}
		remove_modifier = modifier_rev_investment
	}
	option = {
		set_variable = rr_batch_5
		name = revolution_events.2.b
		#再派点人来
		every_country = {
			limit = {
				has_modifier = modifier_rev_target_country
			}
			change_relations = {
				country = root
				value = -40
			}
		}
		add_modifier = {
			name = modifier_additional_expenses
			months = 3
		}
		trigger_event = {
			id = revolution_events.3
			days = 7
			popup = yes
		}
	}
}
