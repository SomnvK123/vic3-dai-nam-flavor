# ------------------------------------------------------------
# 旅行途中随机调度事件
# 日志 on_weekly_pulse 中建议触发：
# trigger_event = { id = gmg_world_tour.30 popup = no }
# ------------------------------------------------------------

gmg_world_tour.30 = {
	type = country_event
	placement = root

	title = gmg_world_tour.30.t
	desc = gmg_world_tour.30.d

	event_image = {
		video = "unspecific_signed_contract"
	}

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	duration = 1
    dlc = dlc_ra_plus
	trigger = {
		exists = var:gmg_last_tour_country
	}

	immediate = {
		var:gmg_last_tour_country = {
			save_scope_as = gmg_last_tour_country_scope
		}

		# ------------------------------------------------------------
		# 国家特色事件
		# 注意：
		# 不使用 tag = USA
		# 不使用 tag = "USA"
		# 直接用 this = c:USA 判断国家实例
		# ------------------------------------------------------------

		if = {
			limit = {
				exists = scope:gmg_last_tour_country_scope
				exists = c:USA
				scope:gmg_last_tour_country_scope = {
					this = c:USA
				}
			}
			trigger_event = {
				id = gmg_world_tour.31
				popup = yes
			}
		}

		else_if = {
			limit = {
				exists = scope:gmg_last_tour_country_scope
				exists = c:EGY
				scope:gmg_last_tour_country_scope = {
					this = c:EGY
				}
			}
			trigger_event = {
				id = gmg_world_tour.32
				popup = yes
			}
		}

		else_if = {
			limit = {
				exists = scope:gmg_last_tour_country_scope
				exists = c:JAP
				scope:gmg_last_tour_country_scope = {
					this = c:JAP
				}
			}
			trigger_event = {
				id = gmg_world_tour.33
				popup = yes
			}
		}

		else_if = {
			limit = {
				exists = scope:gmg_last_tour_country_scope
				scope:gmg_last_tour_country_scope = {
					OR = {
						AND = {
							exists = c:RUS
							this = c:RUS
						}
						AND = {
							exists = c:SOV
							this = c:SOV
						}
					}
				}
			}
			trigger_event = {
				id = gmg_world_tour.34
				popup = yes
			}
		}

		# ------------------------------------------------------------
		# 印度圈特色事件
		# ------------------------------------------------------------

		else_if = {
			limit = {
				exists = scope:gmg_last_tour_country_scope
				scope:gmg_last_tour_country_scope = {
					OR = {
						AND = { exists = c:BIC this = c:BIC }
						AND = { exists = c:IND this = c:IND }
						AND = { exists = c:HND this = c:HND }
						AND = { exists = c:PJB this = c:PJB }
						AND = { exists = c:SIK this = c:SIK }
						AND = { exists = c:MYS this = c:MYS }
						AND = { exists = c:HYD this = c:HYD }
						AND = { exists = c:AWA this = c:AWA }
						AND = { exists = c:BER this = c:BER }
					}
				}
			}
			trigger_event = {
				id = gmg_world_tour.35
				popup = yes
			}
		}

		# ------------------------------------------------------------
		# 东亚特色事件
		# ------------------------------------------------------------

		else_if = {
			limit = {
				exists = scope:gmg_last_tour_country_scope
				scope:gmg_last_tour_country_scope = {
					OR = {
						AND = { exists = c:CHI this = c:CHI }
						AND = { exists = c:KOR this = c:KOR }
					}
				}
			}
			trigger_event = {
				id = gmg_world_tour.36
				popup = yes
			}
		}

		# ------------------------------------------------------------
		# 南非特色事件
		# ------------------------------------------------------------

		else_if = {
			limit = {
				exists = scope:gmg_last_tour_country_scope
				scope:gmg_last_tour_country_scope = {
					OR = {
						AND = {
							exists = c:SAF
							this = c:SAF
						}
						any_scope_state = {
							region = sr:region_south_africa
						}
					}
				}
			}
			trigger_event = {
				id = gmg_world_tour.39
				popup = yes
			}
		}

		# ------------------------------------------------------------
		# 非洲内陆地区特色事件
		# ------------------------------------------------------------

		else_if = {
			limit = {
				exists = scope:gmg_last_tour_country_scope
				scope:gmg_last_tour_country_scope = {
					any_scope_state = {
						OR = {
							region = sr:region_niger
							region = sr:region_congo
							region = sr:region_zanj
						}
					}
				}
			}
			random_list = {
				50 = {
					trigger_event = {
						id = gmg_world_tour.37
						popup = yes
					}
				}
				50 = {
					trigger_event = {
						id = gmg_world_tour.38
						popup = yes
					}
				}
			}
		}

		# ------------------------------------------------------------
		# 战争、法律特色事件
		# ------------------------------------------------------------

		else_if = {
			limit = {
				exists = scope:gmg_last_tour_country_scope
				scope:gmg_last_tour_country_scope = {
					is_at_war = yes
				}
			}
			trigger_event = {
				id = gmg_world_tour.42
				popup = yes
			}
		}

		else_if = {
			limit = {
				exists = scope:gmg_last_tour_country_scope
				scope:gmg_last_tour_country_scope = {
					has_law = law_monarchy
				}
			}
			trigger_event = {
				id = gmg_world_tour.40
				popup = yes
			}
		}

		else_if = {
			limit = {
				exists = scope:gmg_last_tour_country_scope
				scope:gmg_last_tour_country_scope = {
					OR = {
						has_law = law_presidential_republic
						has_law = law_parliamentary_republic
						has_law = law_council_republic
					}
				}
			}
			trigger_event = {
				id = gmg_world_tour.41
				popup = yes
			}
		}

		# ------------------------------------------------------------
		# 没有任何特色时，才走通用随机事件
		# ------------------------------------------------------------

		else = {
			random_list = {
				10 = {
					trigger_event = {
						id = gmg_world_tour.43
						popup = yes
					}
				}

				10 = {
					trigger_event = {
						id = gmg_world_tour.44
						popup = yes
					}
				}

				10 = {
					trigger_event = {
						id = gmg_world_tour.45
						popup = yes
					}
				}

				20 = {
					trigger_event = {
						id = gmg_world_tour.10
						popup = yes
					}
				}
			}
		}
	}

	option = {
		name = gmg_world_tour.30.a
		default_option = yes
	}
}
