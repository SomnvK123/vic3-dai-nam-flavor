namespace = rev_sway_events
# 每个拉拢了之后触发
rev_sway_events.1 = {
	type = country_event
	title = rev_sway_events.1.t
	desc = rev_sway_events.1.d
	flavor = rev_sway_events.1.f
	event_image = {
		video = "unspecific_factory_closed"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_election.dds"
	duration = 1
	immediate = {
		random_country = {
			limit = {
				has_modifier = modifier_rev_target_country
			}
			save_scope_as = rev_target_country
		}
		scope:rev_target_country = {
			random_political_movement = {
				limit = {
					is_political_movement_type = movement_communist
				}
				save_scope_as = communist_now_movement
			}
		}
	}
	option = {
		name = rev_sway_events.1.a
		scope:rev_target_country = {
			scope:communist_now_movement = {
				add_modifier = {
					name = communism_big_big_now_modifier
					months = 20
				}
			}
			every_interest_group = {
				limit = {
					is_interest_group_type = ig_trade_unions
				}
				leader = {
					set_ideology = ideology:ideology_vanguardist
				}
			}
			add_modifier = {
				name = modifier_IG_leave_gov
				months = 10
			}
			ig:ig_trade_unions = {
				join_revolution = yes
				add_modifier = {
					name = ig_unappeased
					months = 6
				}
			}
		}
	}
}
