hai.3 = {
	type = country_event
	placement = ROOT
	title = hai.3.t
	desc = hai.3.d
	flavor = hai.3.f
	event_image = {
		video = "southamerica_aristocrats"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3
     
	trigger = {
	}
	immediate = {
		cu:afro_caribeno = {
			save_scope_as = afro_caribenno_scope
		}
		cu:caribeno = {
			save_scope_as = caribenno_scope
		}
		s:STATE_SANTO_DOMINGO = {
			if = {
				limit = {
					any_scope_state = {
						owner ?= {
							is_direct_subject_of = root
							is_subject_type = subject_type_protectorate
						}
						is_capital = yes
					}
				}
				random_scope_state = {
					limit = {
						owner ?= {
							is_direct_subject_of = root
							is_subject_type = subject_type_protectorate
						}
						is_capital = yes
					}
					owner ?= {
						save_scope_as = ocupador_santo_domingo
					}
					save_scope_as = santo_domingo_scope
				}
			}
			else = {
				random_scope_state = {
					limit = {
						owner = root
						is_treaty_port = no
					}
					save_scope_as = santo_domingo_scope
					owner ?= {
						save_scope_as = ocupador_santo_domingo
					}
				}
			}
		}
	}
	option = {
		name = hai.3.a		# Declarar Capitania General
		default_option = yes
		trigger = {
			exists = scope:ocupador_santo_domingo
		}
		# 10 de infamia
		change_infamy = 10
		# Bajar relacion grandes potencias
		every_country = {
			limit = {
				has_interest_marker_in_region = sr:region_central_america
				this.country_rank >= rank_value:great_power
				NOR = {
					root = this
					has_attitude = {
						who = root
						attitude = genial
					}
					has_attitude = {
						who = root
						attitude = cooperative
					}
				}
			}
			if = {
				limit = {
					has_variable = monroe_doctrine
				}
				change_relations = {
					country = root
					value = -40
				}
			}
			else = {
				change_relations = {
					country = root
					value = -10
				}
			}
		}
		# Ocupa la provincia de Santo Domingo
		if = {
			limit = {
				root = scope:ocupador_santo_domingo
			}
			# Crea la república dominicana
			create_country = {
				tag = DOM
				origin = root
				state = scope:santo_domingo_scope
				on_created = {
					activate_law = law_type:law_presidential_republic
				}
			}
			create_diplomatic_pact = {
				country = c:DOM
				type = colony
			}
			create_diplomatic_pact = {
				country = c:DOM
				type = provincia_espannola
			}
			custom_tooltip = hai.3_tt
		}
		else = {
			# No ocupa Santo Domingo
			# Informa al jugador de que tendrá que converitirlo en marioneta si no lo es
			if = {
				limit = {
					scope:ocupador_santo_domingo = {
						NAND = {
							is_subject_type = subject_type_colony
							is_subject_of = root
						}
					}
				}
				custom_tooltip = hai.3_tt1
			}
		}
		# Añade entrada de Capitania
		add_journal_entry = {
			type = je_capitania_general_republica_dominicana_esp
		}
		ai_chance = {
			base = 30
		}
	}
	option = {
		name = hai.3.b
		default_option = yes
		trigger = {
			exists = scope:ocupador_santo_domingo
			scope:ocupador_santo_domingo = {
				NOR = {
					this = root
					is_subject_type = subject_type_colony
				}
			}
		}
		every_country = {
			limit = {
				has_interest_marker_in_region = sr:region_central_america
				this.country_rank >= rank_value:great_powerd
				NOT = {
					root = this
					has_attitude = {
						who = THIS
						attitude = genial
					}
				}
			}
			if = {
				limit = {
					has_variable = monroe_doctrine
				}
				change_relations = {
					country = root
					value = -20
				}
			}
			else = {
				change_relations = {
					country = root
					value = -5
				}
			}
		}
		change_infamy = 5
		custom_tooltip = hai.3_tt
		ai_chance = {
			base = 1
		}
	}
	option = {
		name = hai.3.c
		trigger = {
			root = scope:ocupador_santo_domingo
		}
		every_country = {
			limit = {
				has_interest_marker_in_region = sr:region_central_america
				this.country_rank >= rank_value:great_power
				NOT = {
					root = this
					has_attitude = {
						who = THIS
						attitude = genial
					}
				}
			}
			if = {
				limit = {
					has_variable = monroe_doctrine
				}
				change_relations = {
					country = root
					value = -60
				}
			}
			else = {
				change_relations = {
					country = root
					value = -30
				}
			}
		}
		change_infamy = 20
		custom_tooltip = hai.3_tt
		ai_chance = {
			base = 1
		}
	}
}
