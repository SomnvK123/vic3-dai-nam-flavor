#Newlands Resolution
usa_hawaii.1 = {
	type = country_event
	event_image = {
		video = "gfx/event_pictures/unspecific_signed_contract.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/unspecific/signed_contract"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	title = usa_hawaii.1.t
	desc = usa_hawaii.1.d
	flavor = usa_hawaii.1.f
	duration = 1
	trigger = {
		#Triggered by completing usa_hawaii journal entry 
	}
	immediate = {
		c:USA ?= {
			save_scope_as = united_states
			random_scope_state = {
				limit = {
					is_incorporated = yes
				}
				save_scope_as = newlands_state
			}
		}
		c:HAW ?= {
			save_scope_as = hawaii
			random_scope_state = {
				limit = {
					is_incorporated = yes
				}
				save_scope_as = hawaiian_islands
			}
		}
	}
	option = {
		#Annex Hawaii, Yankee homeland immediately 
		name = usa_hawaii.1.a
		c:USA ?= {
			annex = c:HAW
		}
		s:STATE_HAWAIIAN_ISLANDS = {
			add_homeland = yankee
		}
	}
}

#Rising Tensions in Hawaii
usa_hawaii.2 = {
	type = country_event
	event_image = {
		video = "africa_public_protest"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	title = usa_hawaii.2.t
	desc = usa_hawaii.2.d
	flavor = usa_hawaii.2.f
	duration = 3
	trigger = {
		exists = c:HAW
		c:HAW ?= {
			has_law_or_variant = law_type:law_monarchy
		}
		NOT = {
			has_variable = hawaii_tensions_rising
		}
	}
	immediate = {
		c:USA ?= {
			save_scope_as = united_states
		}
		c:HAW ?= {
			set_variable = fucking_american_o
			save_scope_as = hawaii
			random_scope_state = {
				limit = {
					is_incorporated = yes
				}
				save_scope_as = hawaiian_islands
			}
			create_character = {
				template = ecchi_haw_dole_character_template
				ig_leader = yes
				on_created = {
					set_variable = {
						name = sanford_dole
					}
					save_scope_as = sanford_dole
					set_as_interest_group_leader = yes
				}
			}
		}
		random_scope_character = {
			limit = {
				has_variable = sanford_dole
			}
			save_scope_as = sanford_dole
		}
		set_variable = {
			name = hawaii_tensions_rising
			value = yes
		}
	}
	option = {
		#Concerning
		name = usa_hawaii.2.a
		scope:hawaiian_islands = {
			add_radicals_in_state = {
				value = 0.25
			}
		}
	}
}

		}

		}

				}

			}
		}
	}
	option = {
		#More Americans moving to Hawaii
		name = usa_hawaii.3.a
		scope:hawaiian_islands = {
			create_pop = {
				pop_type = peasants
				culture = yankee
				size = 2000
			}
			add_radicals_in_state = {
				value = 0.15
				culture = cu:yankee
			}
			add_radicals_in_state = {
				value = 0.25
				interest_group = ig:ig_intelligentsia
			}
		}
	}
}

#Republic of Hawaii approaches us for annexation
usa_hawaii.5 = {
	type = country_event
	event_image = {
		video = "gfx/event_pictures/unspecific_politicians_arguing.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/asia/politician_parliament_motion"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	title = usa_hawaii.5.t
	desc = usa_hawaii.5.d
	flavor = usa_hawaii.5.f
	duration = 3
	trigger = {
		exists = c:HAW
		has_variable = hawaii_tensions_rising
		has_variable = yankees_in_hawaii
		has_variable = hawaiian_revolution
		c:HAW ?= {
			is_at_war = no
			NOT = {
				has_law_or_variant = law_type:law_monarchy
			}
		}
		NOT = {
			has_variable = hawaii_approach_annexation
		}
	}
	immediate = {
		set_variable = {
			name = hawaii_approach_annexation
			value = yes
		}
		c:USA ?= {
			save_scope_as = united_states
		}
		c:HAW ?= {
			save_scope_as = hawaii
			random_scope_state = {
				limit = {
					is_incorporated = yes
				}
				save_scope_as = hawaiian_islands
			}
		}
	}
	option = {
		#Outright annexation
		name = usa_hawaii.5.a
		custom_tooltip = {
			text = hawaii_annexation.tt
			set_variable = {
				name = hawaii_annexation
				value = yes
			}
		}
	}
	option = {
		#Establish a puppet government
		name = usa_hawaii.5.b
		trigger = {
			c:HAW ?= {
				relations:root >= relations_threshold:cordial
				NOT = {
					is_subject_of = c:USA
				}
			}
		}
		custom_tooltip = {
			text = hawaii_puppet.tt
			set_variable = {
				name = hawaii_puppet
				value = yes
			}
		}
	}
}
