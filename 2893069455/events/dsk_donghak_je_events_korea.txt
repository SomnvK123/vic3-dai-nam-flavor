donghak_je_events.1 = {
	type = country_event
	placement = ROOT
	title = donghak_je_events.1.t1
	desc = donghak_je_events.1.d1
	flavor = donghak_je_events.1.f1
	event_image = {
		video = "ep1_donghak_rebellion"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/asia/union_leader"
	icon = "gfx/interface/icons/pops_icons/farmers.dds"
	duration = 3
 
	trigger = {
		NOT = {
			is_subject_of = c:JAP
		}
	}
	immediate = {
		create_country = {
			tag = KO5
			origin = this
			state = s:STATE_YANGHO.region_state:KOR
			state = s:STATE_BUSAN.region_state:KOR
			on_created = {
				activate_law = law_type:law_universal_suffrage
				activate_law = law_type:law_peasant_proprietorship
				activate_law = law_type:law_elected_bureaucrats
				ig:ig_rural_folk = {
					add_ruling_interest_group = yes
				}
				set_state_religion = rel:cheondoism
			}
		}
	}
	option = {
		name = donghak_je_events.1.a
		default_option = yes
		add_journal_entry = {
			type = je_donghak_war
		}
		c:KO5 ?= {
			add_journal_entry = {
				type = je_donghak_war_2
			}
			add_loyalists = {
				pop_type = peasants
				culture = cu:korean
				value = 0.25
			}
			add_loyalists = {
				pop_type = farmers
				culture = cu:korean
				value = 0.25
			}
			create_diplomatic_play = {
				name = huguo_war
				war = yes
				type = dp_revolution
				annex_as_civil_war = yes
				target_country = c:KOR
			}
			change_infamy = -80
			every_scope_character = {
				limit = {
					OR = {
						is_ruler_of_own_country = yes
						is_heir_of_own_country = yes
						has_role_of_type = politician
						has_role_of_type = general
						has_role_of_type = admiral
					}
				}
				change_character_religion = rel:cheondoism
			}
		}
		random_scope_state = {
			convert_population = {
				target = rel:cheondoism
				culture = cu:korean
				pop_type = peasants
				value = 0.03
			}
			convert_population = {
				target = rel:cheondoism
				culture = cu:korean
				pop_type = farmers
				value = 0.03
			}
		}
		change_infamy = -50
		ai_chance = {
			base = 100
		}
	}
	option = {
		name = donghak_je_events.1.b
		add_journal_entry = {
			type = je_donghak_war
		}
		c:KO5 ?= {
			add_journal_entry = {
				type = je_donghak_war_2
			}
			add_loyalists = {
				pop_type = peasants
				culture = cu:korean
				value = 0.35
			}
			add_loyalists = {
				pop_type = farmers
				culture = cu:korean
				value = 0.35
			}
			create_diplomatic_play = {
				name = huguo_war
				war = no
				escalation = 50
				type = dp_revolution
				annex_as_civil_war = yes
				target_country = c:KOR
			}
			change_infamy = -100
			every_scope_character = {
				limit = {
					OR = {
						is_ruler_of_own_country = yes
						is_heir_of_own_country = yes
						has_role_of_type = politician
						has_role_of_type = general
						has_role_of_type = admiral
					}
				}
				change_character_religion = rel:cheondoism
			}
		}
		play_as = c:KO5
		ai_chance = {
			base = 0
		}
	}
}
