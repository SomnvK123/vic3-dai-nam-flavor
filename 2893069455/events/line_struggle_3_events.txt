line_struggle_3_events.1 = {
	# Startup event
	type = country_event
	placement = ROOT
	title = line_struggle_3_events.1.t
	desc = line_struggle_3_events.1.desc
	flavor = line_struggle_3_events.1.f
	#left_icon = scope:french_monarch
	event_image = {
		video = "europenorthamerica_political_extremism"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/political_parties/communists.dds"
	duration = 3
    dlc = dlc_ra_plus
	immediate = {
		set_variable = line_struggle_3_startup_var
		every_scope_character = {
			limit = {
				is_ruler_of_own_country = yes
			}
			save_scope_as = ls3_ruler_scope_1
		}
	}
	option = {
		name = line_struggle_3_events.1.a
		default_option = yes
		add_journal_entry = {
			#添加日志：第三次路线斗争-第一阶段
			type = je_line_struggle_3_1p
		}
		add_journal_entry = {
			#添加日志：公社政府的探索
			type = je_line_struggle_3_1p_commune
		}
		add_journal_entry = {
			#添加日志：工业组织中的革命
			type = je_line_struggle_3_1p_communal_priciple
		}
		trigger_event = {
			id = line_struggle_3_events.12
		}
		#利益集团领袖改变派系
		#地主、实业家、宗教、工会、小市民成为走资派
		every_interest_group = {
			limit = {
				OR = {
					is_interest_group_type = ig_landowners
					#is_interest_group_type = ig_trade_unions
					is_interest_group_type = ig_devout
					is_interest_group_type = ig_petty_bourgeoisie
					is_interest_group_type = ig_industrialists
				}
			}
			leader = {
				set_ideology = ideology:ideology_capitalist_roader
			}
		}
		create_character = {
			#创建一个工会走资派领袖
			is_politician = yes
			interest_group = ig_trade_unions
			ideology = ideology_capitalist_roader
		}
		#红军/军队、知识分子成为极左派
		every_interest_group = {
			limit = {
				OR = {
					is_interest_group_type = ig_armed_forces
					is_interest_group_type = ig_intelligentsia
					is_interest_group_type = ig_red_army
					is_interest_group_type = ig_Marxism_scholar
				}
			}
			leader = {
				set_ideology = ideology:ideology_ultraleftist
			}
		}
		#农民成为毛派
		ig:ig_rural_folk = {
			leader = {
				set_ideology = ideology:ideology_maoist
			}
		}
		#添加毛派煽动者，分属于工会、乡村群众
		create_character = {
			is_agitator = yes
			interest_group = ig_trade_unions
			ideology = ideology_maoist
		}
		create_character = {
			is_agitator = yes
			interest_group = ig_rural_folk
			ideology = ideology_maoist
		}
		every_scope_character = {
			limit = {
				is_ruler_of_own_country = yes
			}
			set_ideology = ideology:ideology_maoist
		}
		#统治者成为毛派
	}
}

	option = {
		#全部改为走资派

			limit = {

			limit = {

			text = no_more_ultraleftist
