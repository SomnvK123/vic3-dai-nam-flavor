france_flavor.2 = {
	#拿破仑参与总统竞选事件
	type = country_event
	placement = root
	title = france_flavor.2.t
	desc = france_flavor.2.d
	flavor = france_flavor.2.f
	duration = 3
    dlc = dlc_ra_plus
	event_image = {
		video = "gfx/event_pictures/unspecific_politicians_arguing.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/europenorthamerica/springtime_of_nation"
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	trigger = {
	}
	immediate = {
	}
	option = {
		name = france_flavor.2.a
		default_option = yes
		set_global_variable = napoleon_iii_spawn
		set_variable = bonapartist_restoration_var
		if = {
			limit = {
				any_country = {
					any_scope_character = {
						has_template = fra_napoleon_iii_character_template
					}
				}
			}
			random_country = {
				limit = {
					any_scope_character = {
						has_template = fra_napoleon_iii_character_template
					}
				}
				random_scope_character = {
					limit = {
						has_template = fra_napoleon_iii_character_template
					}
					transfer_character = ROOT
					set_character_as_ruler = yes
				}
			}
		}
		else_if = {
			limit = {
				any_character_in_exile_pool = {
					has_template = fra_napoleon_iii_character_template
				}
			}
			random_character_in_exile_pool = {
				limit = {
					has_template = fra_napoleon_iii_character_template
				}
				transfer_character = ROOT
				set_character_as_ruler = yes
			}
		}
		else = {
			create_character = {
				#直接成为大总统，但是非利益集团领袖，后面有寡头制保护他
				template = fra_napoleon_iii_character_template
				ruler = yes
				is_agitator = no
				on_created = {
					set_variable = {
						name = bonapartist_leader
					}
					set_variable = napoleon_iii_var
				}
			}
		}
		trigger_event = {
			#尽快触发变寡头事件
			id = french_flavor.2
			days = 10
		}
		ai_chance = {
			base = 100
		}
	}
	option = {
		name = france_flavor.2.b
		ai_chance = {
			base = 0
		}
	}
}
