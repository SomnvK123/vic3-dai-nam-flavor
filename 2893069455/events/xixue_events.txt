xixue_events.2 = {
	type = country_event
	placement = scope:xili_sheng
	title = xixue_events.2.t
	desc = xixue_events.2.d
	flavor = xixue_events.2.f
	event_image = {
		video = "gfx/event_pictures/asia_union_leader.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/unspecific/asia_union_leader"
	icon = "gfx/interface/icons/objectives/egalitarian_society.dds"
	duration = 3
	cooldown = {
		days = short_modifier_time
	}
	trigger = {
	}
	immediate = {
		random_country = {
			limit = {
				religion = {
					has_discrimination_trait = heritage_christian
				}
				country_rank >= rank_value:major_power
			}
			save_scope_as = xuezhe_xixue
		}
		random_scope_state = {
			limit = {
				OR = {
					is_coastal = yes
					any_neighbouring_state = {
						has_modifier = xixue_chuanjiao
					}
				}
			}
			save_scope_as = xili_sheng
		}
	}
	option = {
		name = xixue_events.2.a
		default_option = yes
		change_relations = {
			country = scope:xuezhe_xixue
			value = -25
		}
		if = {
			limit = {
				any_scope_pop = {
					religion = rel:protestant
				}
			}
			add_radicals = {
				religion = rel:protestant
				value = small_radicals
			}
		}
		if = {
			limit = {
				any_scope_pop = {
					religion = rel:catholic
				}
			}
			add_radicals = {
				religion = rel:catholic
				value = small_radicals
			}
		}
		if = {
			limit = {
				any_scope_pop = {
					religion = rel:orthodox
				}
			}
			add_radicals = {
				religion = rel:orthodox
				value = small_radicals
			}
		}
		ig:ig_devout = {
			add_modifier = {
				name = paixizhichi
				multiplier = 2
				days = normal_modifier_time
			}
		}
		ig:ig_landowners = {
			add_modifier = {
				name = paixizhichi
				multiplier = 2
				days = normal_modifier_time
			}
		}
	}
	option = {
		name = xixue_events.2.b
		default_option = yes
		scope:xili_sheng = {
			add_modifier = {
				name = xixue_chuanjiao
				years = 5
			}
			convert_population = {
				target = rel:protestant
				value = 0.05
			}
		}
		ig:ig_devout = {
			add_modifier = {
				name = paixizhichi
				multiplier = -2
				days = normal_modifier_time
			}
		}
	}
}
