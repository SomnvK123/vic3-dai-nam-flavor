		exists = c:GER

		NOT = {
			exists = c:PBM
		}
		owns_entire_state_region = STATE_BOHEMIA
		owns_entire_state_region = STATE_MORAVIA

	option = {
		name = Lebensraum_event.1.o1
		change_variable = {
			name = nazi_german_empire_var
			add = 1
		}
		create_country = {
			tag = PBM
			origin = c:GER
			state = s:STATE_BOHEMIA.region_state:GER
			state = s:STATE_MORAVIA.region_state:GER
			state = s:STATE_RUTHENIA.region_state:GER
			state = s:STATE_SLOVAKIA.region_state:GER
			on_created = {
				c:GER ?= {
					create_diplomatic_pact = {
						country = PREV
						type = colony
					}
				}
				c:PBM ?= {
					add_modifier = {
						name = sick_lebensraum
					}
				}
			}
		}
	}
	option = {
		name = Lebensraum_event.1.o2
		default_option = yes
	}
