denmark_flavor.4 = {
	type = country_event
	title = denmark_flavor.4.t
	desc = denmark_flavor.4.d
	flavor = denmark_flavor.4.f
	event_image = {
		video = "gfx/event_pictures/unspecific_politicians_arguing.bk2"
	}
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/asia/politician_parliament_motion"
	duration = 3
    dlc = dlc_ra_plus
	trigger = {
		any_scope_character = {
			is_ruler_of_own_country = yes
			has_trait = imperious			# Just to make sure it's him, unlikely to occur otherwise
		}
	}
	cooldown = {
		days = normal_modifier_time
	}
	immediate = {
		random_scope_character = {
			limit = {
				is_ruler_of_own_country = yes
			}
			save_scope_as = frederik_the_monarch
		}
	}
	option = {
		name = denmark_flavor.4.a
		default_option = yes
		scope:frederik_the_monarch = {
			kill_character = yes
		}
		if = {
			limit = {
				c:HOL ?= {
					is_subject_of = c:DEN
				}
			}
			c:HOL ?= {
				change_relations = {
					country = c:DEN
					value = -10
				}
				change_relations = {
					country = c:PRU
					value = 10
				}
			}
		}
		if = {
			limit = {
				c:SCH ?= {
					is_subject_of = c:DEN
				}
			}
			c:SCH ?= {
				change_relations = {
					country = c:DEN
					value = -10
				}
				change_relations = {
					country = c:PRU
					value = 10
				}
			}
		}
	}
}

# First Schleswig War
denmark_flavor.5 = {
	type = country_event
	title = denmark_flavor.5.t
	desc = denmark_flavor.5.d
	flavor = denmark_flavor.5.f
	event_image = {
		video = "gfx/event_pictures/europenorthamerica_before_the_battle.bk2"
	}
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/asia/politician_parliament_motion"
	duration = 3
                dlc = dlc_ra_plus
	cooldown = {
		days = normal_modifier_time
	}
	# If there was just a freaking way to have a country properly join a diplomatic play
	immediate = {
		c:DEN ?= {
				save_scope_as = den_overlord_scope
		}
		c:SCH ?= {
		create_diplomatic_play = {
			name = first_schleswig_war
			escalation = 40
			target_country = scope:den_overlord_scope
			type = dp_independence
			add_initiator_backers = { 
				    c:HOL
			}
			add_war_goal = {
				holder = scope:den_overlord_scope
				type = conquer_state
				target_state = s:STATE_SCHLESWIG_HOLSTEIN.region_state:SCH
			}
		}
        }
		random_diplomatic_play = {
			limit = {
				any_scope_play_involved = {
					THIS = ROOT
				}
			}
			save_scope_as = cur_play
		}
		c:PRU ?= {
			# Should help
			set_secret_goal = {
				country = c:HOL
				secret_goal = protect
			}
			set_secret_goal = {
				country = c:SCH
				secret_goal = protect
			}
			set_secret_goal = {
				country = c:DEN
				secret_goal = antagonize
			}
		}
	}
	option = {
		name = denmark_flavor.5.a
		default_option = yes
		show_as_tooltip = {
			create_diplomatic_play = {
				name = first_schleswig_war
				target_country = c:DEN
				type = dp_independence
			}
		}
		# Add alternative later
		add_loyalists = {
			culture = cu:north_german
			value = 0.25
		}
		change_relations = {
			country = c:DEN
			value = -50
		}
		c:SCH ?= {
			if = {
				limit = { is_ai = no }
				trigger_event = { id = denmark_flavor.6 }
			}
			else = {
				change_relations = {
					country = c:DEN
					value = -50
				}
			}
		}
	}
}
