# set strategy - adm
cvlm_misc.2 = { 
	type = country_event
	dlc = dlc_cvlm
	placement = scope:target_country
	duration = 1

	title = cvlm_misc.1.t
	desc = cvlm_misc.1.d
	flavor = cvlm_misc.1.f

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    event_image = {
		video = "africa_construction_colony"
	}

	icon = "gfx/interface/icons/lens_toolbar_icons/unused/diplomatic_actions.dds"

	option = {
		name = ai_strategy_colonial_extraction
		scope:target_country = { set_strategy = ai_strategy_colonial_extraction }
		hidden_effect = { trigger_event = { id = cvlm_misc.1 popup = yes } }
	}

	option = {
		name = ai_strategy_plantation_economy
		scope:target_country = { set_strategy = ai_strategy_plantation_economy }
		hidden_effect = { trigger_event = { id = cvlm_misc.1 popup = yes } }
	}

	option = {
		name = ai_strategy_agricultural_expansion
		scope:target_country = { set_strategy = ai_strategy_agricultural_expansion }
		hidden_effect = { trigger_event = { id = cvlm_misc.1 popup = yes } }
	}

	option = {
		name = ai_strategy_placate_population
		scope:target_country = { set_strategy = ai_strategy_placate_population }
		hidden_effect = { trigger_event = { id = cvlm_misc.1 popup = yes } }
	}

	option = {
		name = ai_strategy_industrial_expansion
		scope:target_country = { set_strategy = ai_strategy_industrial_expansion }
		hidden_effect = { trigger_event = { id = cvlm_misc.1 popup = yes } }
        highlighted_option = yes
		show_as_unavailable = { always = yes }
        trigger = {
			country_rank >= rank_value:major_power
			has_technology_researched = civilizing_mission
			scope:target_country = { 	
				# ruler
				custom_tooltip = {
					text = cvlm_misc.1.6
					ruler ?= {
						NOR = {
							has_ideology = ideology:ideology_traditionalist
							has_ideology = ideology:ideology_luddite
						}
					}
				}
				# ig
				ig:ig_industrialists = {
					OR = {
						is_in_government = yes
						is_powerful = yes
					} 
				}
				# law
				NOR = {
					has_law_or_variant = law_type:law_traditionalism
					has_law_or_variant = law_type:law_agrarianism
					has_law_or_variant = law_type:law_industry_banned
				}
			}
		}
	}

	option = {
		name = cvlm_misc.1.5
		hidden_effect = { trigger_event = { id = cvlm_misc.1 popup = yes } }
		default_option = yes
	}
}

cvlm_misc.4 = { 
	type = country_event
	dlc = dlc_cvlm
	placement = scope:target_country
	duration = 1

	title = cvlm_misc.1.t
	desc = cvlm_misc.1.d
	flavor = cvlm_misc.1.f

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    event_image = {
		video = "africa_construction_colony"
	}

	icon = "gfx/interface/icons/lens_toolbar_icons/unused/diplomatic_actions.dds"

	option = {
		name = ai_strategy_conservative_agenda
		scope:target_country = { set_strategy = ai_strategy_conservative_agenda }
		hidden_effect = { trigger_event = { id = cvlm_misc.1 popup = yes } }
        highlighted_option = yes
		show_as_unavailable = { always = yes }
        trigger = {
			scope:target_country = { 
				custom_tooltip = {
					text = cvlm_misc.1.6
					ruler ?= {
						NOT = { has_ideology = ideology:ideology_radical }
						has_socialist_ideology = no
					}
				}
			}
		}
	}

	option = {
		name = ai_strategy_reactionary_agenda
		scope:target_country = { set_strategy = ai_strategy_reactionary_agenda }
		hidden_effect = { trigger_event = { id = cvlm_misc.1 popup = yes } }
        highlighted_option = yes
		show_as_unavailable = { always = yes }
        trigger = {
			scope:target_country = { 
				custom_tooltip = {
					text = cvlm_misc.1.6
					ruler ?= {
						OR = {
							has_conservative_ideology = yes
							has_reactionary_ideology = yes
							has_monarchist_ideology = yes
						}
					}
				}
			}
		}
	}

	option = {
		name = ai_strategy_progressive_agenda
		scope:target_country = { set_strategy = ai_strategy_progressive_agenda }
		hidden_effect = { trigger_event = { id = cvlm_misc.1 popup = yes } }
        highlighted_option = yes
		show_as_unavailable = { always = yes }
        trigger = {
			scope:target_country = { 
				custom_tooltip = {
					text = cvlm_misc.1.6
					ruler ?= {
						NOR = {
							has_conservative_ideology = yes
							has_reactionary_ideology = yes
							has_monarchist_ideology = yes
						}
					}
				}
			}
		}
	}

	option = {
		name = ai_strategy_egalitarian_agenda
		scope:target_country = { set_strategy = ai_strategy_egalitarian_agenda }
		hidden_effect = { trigger_event = { id = cvlm_misc.1 popup = yes } }
        highlighted_option = yes
		show_as_unavailable = { always = yes }
        trigger = {
			# law
			has_technology_researched = egalitarianism 
			scope:target_country = { has_technology_researched = egalitarianism }
			# ruler
			scope:target_country = { 
				custom_tooltip = {
					text = cvlm_misc.1.6
					ruler ?= {
						OR = {
							has_liberal_ideology = yes
							has_progressive_ideology = yes
							has_socialist_ideology = yes
						}
					}
				}
			}
		}
	}

	option = {
		name = ai_strategy_nationalist_agenda
		scope:target_country = { set_strategy = ai_strategy_nationalist_agenda }
		hidden_effect = { trigger_event = { id = cvlm_misc.1 popup = yes } }
        highlighted_option = yes
		show_as_unavailable = { always = yes }
        trigger = {
			# law
			has_technology_researched = nationalism
			scope:target_country = { has_technology_researched = nationalism }
			# ruler
			scope:target_country = { 
				custom_tooltip = {
					text = cvlm_misc.1.6
					ruler ?= {
						OR = {
							has_conservative_ideology = yes
							has_reactionary_ideology = yes
							has_monarchist_ideology = yes
						}
					}
				}
			}
		}
	}

	option = {
		name = cvlm_misc.1.5

		default_option = yes
	}

# customize - governemnt type
cvlm_misc.230 = { 
	type = country_event
	dlc = dlc_cvlm
	placement = scope:target_country
	duration = 1

	title = cvlm_misc.200.t
	desc = cvlm_misc.200.d
	flavor = cvlm_misc.200.f

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    event_image = {
		video = "africa_construction_colony"
	}

	icon = "gfx/interface/icons/event_icons/event_customize.dds"

	# Colonial Government
	option = {
		name = cvlm_misc.200.colonial_government
        highlighted_option = yes
        show_as_unavailable = { always = yes }
		trigger = {
			scope:target_country = { 
				has_law_or_variant = law_type:law_presidential_republic
				is_country_type = colonial
				OR = {
					is_subject_type = subject_type_colony
					is_subject_type = subject_type_dominion
				}
			}
        }
		scope:target_country = { cvlm_customize_flavor_editor_effect = { VARIABLE = cvlm_customize_flavor_colonial_government } }
		custom_tooltip = cvlm_misc.200.colonial_government.tooltip.a
		custom_tooltip = cvlm_misc.200.colonial_government.tooltip.b
	}

	# Crown Colony
	option = {
		name = cvlm_misc.200.crown_colony
        highlighted_option = yes
        show_as_unavailable = { always = yes }
		trigger = {
			has_law_or_variant = law_type:law_monarchy
			scope:target_country = { 
				has_law_or_variant = law_type:law_presidential_republic
				is_country_type = colonial
				is_subject_type = subject_type_colony
			}
        }
		scope:target_country = { cvlm_customize_flavor_editor_effect = { VARIABLE = cvlm_customize_flavor_crown_colony } }
		custom_tooltip = cvlm_misc.200.crown_colony.tooltip.a
		custom_tooltip = cvlm_misc.200.crown_colony.tooltip.b
	}

	# Viceroyalty
	option = {
		name = cvlm_misc.200.viceroyalty
        highlighted_option = yes
        show_as_unavailable = { always = yes }
		trigger = {
			has_law_or_variant = law_type:law_monarchy
			scope:target_country = { 
				has_law_or_variant = law_type:law_presidential_republic
			}
        }
		scope:target_country = { cvlm_customize_flavor_editor_effect = { VARIABLE = cvlm_customize_flavor_viceroyalty } }
		custom_tooltip = cvlm_misc.200.viceroyalty.tooltip.a
		custom_tooltip = cvlm_misc.200.viceroyalty.tooltip.b
	}

	# captaincy
	option = {
		name = cvlm_misc.200.captaincy
        highlighted_option = yes
        show_as_unavailable = { always = yes }
		trigger = {
			has_law_or_variant = law_type:law_monarchy
			scope:target_country = { 
				has_law_or_variant = law_type:law_presidential_republic
			}
        }
		scope:target_country = { cvlm_customize_flavor_editor_effect = { VARIABLE = cvlm_customize_flavor_captaincy } }
		custom_tooltip = cvlm_misc.200.captaincy.tooltip.a
		custom_tooltip = cvlm_misc.200.captaincy.tooltip.b
	}

	# Overseas Territory
	option = {
		name = cvlm_misc.200.overseas_territory
        highlighted_option = yes
        show_as_unavailable = { always = yes }
		trigger = {
			scope:target_country = { 
				has_law_or_variant = law_type:law_presidential_republic
			}
        }
		scope:target_country = { cvlm_customize_flavor_editor_effect = { VARIABLE = cvlm_customize_flavor_overseas_territory } }
		custom_tooltip = cvlm_misc.200.overseas_territory.tooltip.a
		custom_tooltip = cvlm_misc.200.overseas_territory.tooltip.b
	}

	option = {
		name = cvlm_misc.200.0
		scope:target_country = { cvlm_customize_flavor_cleaner_effect = yes }
	}

	option = {
		name = cvlm_misc.1.4
		hidden_effect = { trigger_event = { id = cvlm_misc.200 popup = yes } }
		default_option = yes
	}
