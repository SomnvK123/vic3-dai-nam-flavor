esp.2 = {
	# Retorno de los carlistas
	type = country_event

	flavor = esp.2.f

		video = "europenorthamerica_american_civil_war"
	}
	duration = 3

	icon = "gfx/interface/icons/event_icons/event_military.dds"
	immediate = {
		if = {
			limit = {
				exists = c:SPC
			}
			c:SPC ?= {
				# Impuestos altos y te interesa importar armamento
				set_tax_level = very_high
				add_taxed_goods = g:liquor
				set_tariffs_import_priority = g:artillery
				set_tariffs_import_priority = g:small_arms

				ig:ig_devout = {
					set_interest_group_name = ig_catholic_church
					add_ruling_interest_group = yes
				}
				ig:ig_landowners = {
					add_ideology = ideology_pious
					add_ruling_interest_group = yes
				}
				add_journal_entry = {
					type = je_carlist_victory
				}
				activar_leyes_default_carlistas = yes
				if = {
					limit = {
						any_character_in_exile_pool = {
							es_un_reclamante_carlista = yes
						}
					}
					random_character_in_exile_pool = {
						limit = {
							es_un_reclamante_carlista = yes
						}
						transfer_character = c:SPC
						free_from_exile_effect = yes
						set_character_as_ruler = yes
						add_character_role = politician
						if = {
							limit = {
								has_role_of_type = agitator
							}
							remove_character_role = agitator
						}
					}
					root = {
						post_notification = lider_carlista_regresa
					}
				}
				else_if = {
					limit = {
						any_character = {
							is_ruler_of_own_country = no
							es_un_reclamante_carlista = yes
						}
					}
					random_character = {
						limit = {
							is_ruler_of_own_country = no
							es_un_reclamante_carlista = yes
						}
						owner ?= {
							trigger_event = {
								id = car.2
								popup = yes
							}
						}
					}
					root = {
						post_notification = lider_carlista_regresa
					}
				}
				else = {
					create_character = {
						first_name = "Carlos"
						last_name = "de_Borbon"
						ruler = yes
						ideology = ideology_carlist
						traits = {
						}
					}
				}
				set_variable = espanna_carlista
				# Bourbon for Everyone - Permitir el logro
				set_variable = bourbon_for_everyone_ach_possible
				# trigger_event = { id = esp.14 popup = yes months = 1 }
			}
		}
		# trigger_event = { id = esp.14 popup = yes months = 1 }
		add_journal_entry = {
			type = je_cristininos_victory
		}
		c:SPC ?= {
			save_scope_as = car_scope
		}
		save_scope_as = spa_scope_scope
		every_country = {
			limit = {
				NOT = {
					is_country_type = decentralized
				}
			}
			post_notification = segunda_guerra_carlista
		}
		if = {
			limit = {
				exists = c:SPC
				c:SPC ?= {
					has_journal_entry = je_carlist_victory
				}
			}
			c:SPC ?= {
				je:je_carlist_victory ?= {
					add_progress = {
						value = 20
						name = alzamientos_carlistas_bar
					}
				}
			}
		}
		if = {
			limit = {
				exists = c:SPA
				c:SPA ?= {
					has_journal_entry = je_cristininos_victory
				}
			}
			c:SPA ?= {
				je:je_cristininos_victory ?= {
					add_progress = {
						value = 20
						name = alzamientos_carlistas_bar
					}
				}
			}
		}
	}
	option = {
		name = esp.2.a
		default_option = yes
	}
	option = {
		name = esp.2.b

			is_ai = no
