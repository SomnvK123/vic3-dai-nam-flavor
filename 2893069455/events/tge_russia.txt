DSK_russia.8 = {
	type = country_event
	hidden = yes
	trigger = {
		any_scope_state = {
			dsk_russian_east_zone = yes
			dsk_westward_expansion_criteria = yes
		}
	}
	immediate = {
		random_scope_state = {
			limit = {
				dsk_russian_east_zone = yes
				dsk_westward_expansion_criteria = yes
			}
			save_scope_as = new_homeland_state
		}
		scope:new_homeland_state = {
			if = {
				limit = {
					owner ?= {
						country_has_primary_culture = cu:russian
					}
				}
				state_region = {
					add_homeland = cu:russian
				}
				cu:russian = {
					save_scope_as = new_homeland_culture
				}
			}
			else_if = {
				limit = {
					OR = {
						state_region = s:STATE_SAKHALIN
						state_region = s:STATE_ALASKA
						region = sr:region_central_asia
					}
				}
				state_region = {
					add_homeland = cu:russian
				}
				cu:russian = {
					save_scope_as = new_homeland_culture
				}
			}
			else = {
				state_region = {
					add_homeland = cu:russian
				}
				cu:russian = {
					save_scope_as = new_homeland_culture
				}
			}
		}
		post_notification = dsk_new_russian_homeland
	}
}
