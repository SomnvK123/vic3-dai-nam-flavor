namespace = jpf_joseon
jpf_joseon.1 = {
	type = country_event
	placement = ROOT
	title = jpf_joseon.1.t
	desc = jpf_joseon.1.d
	flavor = jpf_joseon.1.f
	event_image = {
		video = "ep1_donghak_rebellion"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/asia/union_leader"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3
 
	trigger = {
		exists = c:KOR
		c:KOR ?= this
		game_date > 1839.1.1
		game_date < 1840.1.1
		NOT = {
			has_variable = fucking_joseon_1
		}
	}
	immediate = {
		set_variable = fucking_joseon_1
	}
	option = {
		name = jpf_joseon.1.a
		default_option = yes
		custom_tooltip = jpf_joseon.1.a.tt
		hidden_effect = {
			trigger_event = {
				id = jpf_joseon.2
				days = 30
			}
		}
	}
	option = {
		name = jpf_joseon.1.b
		trigger = {
			NOT = {
				has_law_or_variant = law_type:law_state_religion
			}
		}
		custom_tooltip = jpf_joseon.1.b.tt
		every_interest_group = {
			limit = {
				law_stance = {
					law = law_type:law_state_religion
					value > neutral
				}
			}
			add_modifier = {
				name = jpf_ignored
				days = short_modifier_time
			}
		}
		every_interest_group = {
			limit = {
				law_stance = {
					law = law_type:law_state_religion
					value < neutral
				}
			}
			add_modifier = {
				name = jpf_repected
				days = short_modifier_time
			}
		}
	}
}
