namespace = acw_peace_treaty
#Treaty of Appomattox, USA victory over CSA
acw_peace_treaty.1 = {
	type = country_event
	placement = s:STATE_VIRGINIA.region_state:CSA
	event_image = {
		video = "gfx/event_pictures/unspecific_signed_contract.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/unspecific/signed_contract"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	title = acw_peace_treaty.1.t
	desc = acw_peace_treaty.1.d
	flavor = acw_peace_treaty.1.f
	duration = 3
	trigger = {
		any_scope_war = {
			AND = {
				is_war_participant = c:USA
				has_war_support = {
					target = c:USA
					value >= 0
				}
				has_war_support = {
					target = c:CSA
					value <= 0
				}
			}
			war_has_active_peace_deal = no
		}
		#triggered by completing je_acw_war against csa
	}
	immediate = {
		c:USA ?= {
			ruler ?= {
				set_variable = {
					name = abraham_lincoln
					value = yes
				}
				save_scope_as = president
			}
			every_scope_state = {
				limit = {
					OR = {
						has_modifier = usa_southern_supremacy
						has_modifier = usa_deep_southern_supremacy
					}
				}
				set_state_type = unincorporated
			}
		}
		c:CSA ?= {
			ruler ?= {
				save_scope_as = confederate_president
			}
			save_scope_as = confederacy
		}
	}
	option = {
		name = acw_peace_treaty.1.a
		custom_tooltip = {
			text = acw_peace_treaty.1.tt
		}
	}
}

#Treaty of Washington, CSA victory over USA
acw_peace_treaty.2 = {
	type = country_event
	event_image = {
		video = "gfx/event_pictures/europenorthamerica_capitalists_meeting.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/europenorthamerica/capitalists_meeting"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	title = acw_peace_treaty.2.t
	desc = acw_peace_treaty.2.d
	flavor = acw_peace_treaty.2.f
	duration = 3
	trigger = {
		#triggered by failing je_acw_war against csa
	}
	immediate = {
		c:USA ?= {
			ruler ?= {
				save_scope_as = president
			}
		}
		c:CSA ?= {
			ruler ?= {
				save_scope_as = confederate_president
			}
			save_scope_as = confederacy
		}
	}
	option = {
		name = acw_peace_treaty.2.a
		custom_tooltip = {
			text = acw_peace_treaty.2.tt
			s:STATE_TEXAS = {
				add_claim = c:CSA
			}
			s:STATE_NEW_MEXICO = {
				add_claim = c:CSA
			}
			s:STATE_ARIZONA = {
				add_claim = c:CSA
			}
			s:STATE_MISSOURI = {
				add_claim = c:CSA
			}
			s:STATE_OKLAHOMA = {
				add_claim = c:CSA
			}
			s:STATE_KENTUCKY = {
				add_claim = c:CSA
			}
			s:STATE_MARYLAND = {
				add_claim = c:CSA
			}
			s:STATE_DISTRICT_OF_COLUMBIA = {
				add_claim = c:CSA
			}
			s:STATE_DELAWARE = {
				add_claim = c:CSA
			}
		}
		c:USA ?= {
			add_modifier = {
				name = acw_lost_civil_war
				years = 20
				is_decaying = yes
			}
		}
	}
}

#Treaty of Washington, USA victory over FSA
acw_peace_treaty.3 = {
	type = country_event
	event_image = {
		video = "gfx/event_pictures/unspecific_signed_contract.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/unspecific/signed_contract"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	title = acw_peace_treaty.3.t
	desc = acw_peace_treaty.3.d
	flavor = acw_peace_treaty.3.f
	duration = 3
	trigger = {
		#triggered by completing je_acw_war against fsa
	}
	immediate = {
		c:USA ?= {
			ruler ?= {
				save_scope_as = president
			}
			every_scope_state = {
				limit = {
					OR = {
						has_modifier = usa_iron_belt
						has_modifier = usa_new_england
						has_modifier = usa_northern_supremacy
					}
				}
				set_state_type = unincorporated
			}
		}
		c:FSA ?= {
			ruler ?= {
				save_scope_as = free_president
			}
			save_scope_as = free_states
		}
	}
	option = {
		name = acw_peace_treaty.3.a
		custom_tooltip = {
			text = acw_peace_treaty.3.tt
			c:USA ?= {
				annex = c:FSA
			}
			trigger_event = {
				id = acw_reconstruction.15
				days = 3
			}
		}
	}
}
