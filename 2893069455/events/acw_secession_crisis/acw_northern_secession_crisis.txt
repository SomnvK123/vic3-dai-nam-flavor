namespace = acw_northern_secession_crisis
#NORTHERN SECESSION CRISIS
#New England Secedes, Boston Convention
acw_northern_secession_crisis.1 = {
	type = country_event
	event_image = {
		video = "gfx/event_pictures/europenorthamerica_capitalists_meeting.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/europenorthamerica/capitalists_meeting"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	title = acw_northern_secession_crisis.1.t
	desc = acw_northern_secession_crisis.1.d
	flavor = acw_northern_secession_crisis.1.f
	duration = 1
	trigger = {
		#triggered by event
	}
	immediate = {
		c:USA ?= {
			every_scope_state = {
				if = {
					limit = {
						is_incorporated = yes
						has_modifier = usa_new_england
					}
					set_state_owner = c:FSA
				}
			}
			save_scope_as = united_states
		}
		s:STATE_MASSACHUSETTS.region_state:FSA = {
			save_scope_as = massachusetts
		}
		c:FSA ?= {
			activate_law = law_type:law_dedicated_policing
			activate_law = law_type:law_census_suffrage
			add_modifier = {
				#Abolitionist supremacy, keeps the intelligentsia and the devout in power for a while
				name = fsa_abolitionist_supremacy
				years = 20
				is_decaying = yes
			}
			ig:ig_landowners = {
				set_interest_group_name = ig_moderate_democrats
				remove_ideology = ideology_paternalistic
				add_ideology = ideology_republican_paternalistic
			}
			ig:ig_intelligentsia = {
				set_interest_group_name = ig_northern_abolitionists
			}
			ig:ig_devout = {
				add_ideology = ideology_anti_slavery
			}
			every_scope_character = {
				limit = {
					OR = {
						has_ideology = ideology:ideology_slaver
						has_ideology = ideology:ideology_authoritarian
					}
				}
				kill_character = yes
			}
			set_ruling_interest_groups = {
				ig_intelligentsia
				ig_devout
			}
		}
	}
	option = {
		name = acw_northern_secession_crisis.1.a
		default_option = yes
		custom_tooltip = {
			text = acw_northern_secession_crisis.1.tt
			c:USA ?= {
				trigger_event = {
					id = acw_northern_secession_crisis.2
					days = 30
				}
			}
			c:FSA ?= {
				every_scope_state = {
					if = {
						limit = {
							has_modifier = usa_new_england
						}
						set_state_type = incorporated
					}
				}
			}
		}
	}
}

#The War Begins
acw_northern_secession_crisis.3 = {
	type = country_event
	event_image = {
		video = "gfx/event_pictures/europenorthamerica_american_civil_war.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/europenorthamerica/american_civil_war"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	title = acw_northern_secession_crisis.3.t
	desc = acw_northern_secession_crisis.3.d
	flavor = acw_northern_secession_crisis.3.f
	duration = 1
	trigger = {
		#triggered by event
	}
	immediate = {
		c:USA ?= {
			every_scope_state = {
				if = {
					limit = {
						OR = {
							has_modifier = usa_northern_supremacy
							has_modifier = usa_iron_belt
						}
					}
					set_state_owner = c:FSA
				}
			}
			ruler ?= {
				save_scope_as = president
			}
		}
		s:STATE_MASSACHUSETTS.region_state:FSA = {
			save_scope_as = massachusetts
		}
		c:FSA ?= {
			set_capital = STATE_MASSACHUSETTS
			if = {
				limit = {
					NOT = {
						has_law_or_variant = law_type:law_slavery_banned
					}
				}
				activate_law = law_type:law_slavery_banned
			}
			save_scope_as = free_states
			add_loyalists_in_state = {
				culture = cu:yankee
				value = 0.75
			}
			py:conservative_party = {
				add_momentum = -0.5
			}
			py:liberal_party = {
				add_momentum = 0.5
			}
			py:radical_party = {
				add_momentum = 0.15
			}
		}
		set_global_variable = {
			name = american_civil_war
			value = yes
		}
		set_global_variable = {
			name = midwest_seceded
			value = yes
		}
	}
	option = {
		name = acw_northern_secession_crisis.3.a
		default_option = yes
		custom_tooltip = {
			text = acw_northern_secession_crisis.3.tt
			c:USA ?= {
				create_diplomatic_play = {
					set_key = "The American Civil War"
					target_country = c:FSA
					war = yes
					type = dp_make_protectorate
					add_war_goal = {
						holder = c:FSA
						type = revoke_all_claims
						target_country = c:USA
						primary_demand = yes
					}
				}
			}
			set_global_variable = {
				name = american_civil_war
				value = yes
			}
			change_infamy = -100
			c:FSA ?= {
				#####FSA EDITS######
				every_scope_state = {
					if = {
						limit = {
							OR = {
								has_modifier = usa_northern_supremacy
								has_modifier = usa_iron_belt
								has_modifier = usa_new_england
							}
						}
						set_state_type = incorporated
						state_region = {
							add_claim = c:USA
						}
					}
				}
				change_infamy = -75
				add_treasury = 3000000
				change_institution_investment_level = {
					institution = institution_schools
					investment = -5
				}
				change_institution_investment_level = {
					institution = institution_police
					investment = -5
				}
				change_institution_investment_level = {
					institution = institution_colonial_affairs
					investment = -5
				}
				change_institution_investment_level = {
					institution = institution_social_security
					investment = -5
				}
				change_institution_investment_level = {
					institution = institution_home_affairs
					investment = -5
				}
				change_institution_investment_level = {
					institution = institution_health_system
					investment = -5
				}
				change_institution_investment_level = {
					institution = institution_workplace_safety
					investment = -5
				}
				every_scope_state = {
					limit = {
						is_incorporated = yes
					}
					create_building = {
						building = "building_government_administration"
						level = 2
						reserves = 1
					}
					create_building = {
						building = "building_barrack"
						level = 5
					}
				}
			}
			#######END FSA EDITS###########
			set_global_variable = {
				name = free_states_seceded
				value = yes
			}
			set_variable = union_side_acw
		}
		add_journal_entry = {
			type = je_acw_war
			target = c:USA
		}
	}
	#######PLAY AS THE FREE STATES###############################
	option = {
		name = acw_northern_secession_crisis.3.b
		custom_tooltip = {
			text = acw_northern_secession_crisis.3.tt
			c:USA ?= {
				create_diplomatic_play = {
					set_key = "The American Civil War"
					target_country = c:FSA
					war = yes
					type = dp_make_protectorate
					add_war_goal = {
						holder = c:FSA
						type = revoke_all_claims
						target_country = c:USA
						primary_demand = yes
					}
				}
			}
			set_global_variable = {
				name = american_civil_war
				value = yes
			}
			change_infamy = -100
			c:FSA ?= {
				#####FSA EDITS######
				add_journal_entry = {
					type = je_fsa_acw_war
				}
				every_scope_state = {
					if = {
						limit = {
							OR = {
								has_modifier = usa_northern_supremacy
								has_modifier = usa_iron_belt
								has_modifier = usa_new_england
							}
						}
						set_state_type = incorporated
						state_region = {
							add_claim = c:USA
						}
					}
				}
				change_infamy = -75
				add_treasury = 3000000
				change_institution_investment_level = {
					institution = institution_schools
					investment = -5
				}
				change_institution_investment_level = {
					institution = institution_police
					investment = -5
				}
				change_institution_investment_level = {
					institution = institution_colonial_affairs
					investment = -5
				}
				change_institution_investment_level = {
					institution = institution_social_security
					investment = -5
				}
				change_institution_investment_level = {
					institution = institution_home_affairs
					investment = -5
				}
				change_institution_investment_level = {
					institution = institution_health_system
					investment = -5
				}
				change_institution_investment_level = {
					institution = institution_workplace_safety
					investment = -5
				}
				every_scope_state = {
					limit = {
						is_incorporated = yes
					}
					create_building = {
						building = "building_government_administration"
						level = 2
						reserves = 1
					}
					create_building = {
						building = "building_barrack"
						level = 5
					}
				}
			}
			#######END FSA EDITS###########
			set_global_variable = {
				name = free_states_seceded
				value = yes
			}
			set_global_variable = {
