russia_expand.6 = {
	type = country_event
	placement = ROOT
	event_image = {
		video = "ep1_power_blocs"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/europenorthamerica/capitalists_meeting"
	icon = "gfx/interface/icons/event_icons/event_russkie.dds"
	title = russia_expand.6.t
	desc = russia_expand.6.d
	flavor = russia_expand.6.f
	duration = 3
	dlc = dlc_ra_plus
	trigger = {
	}
	immediate = {
		c:GBR ?= {
			save_scope_as = gbr_scope
		}
		c:PER ?= {
			save_scope_as = per_scope
		}
	}
	option = {
		name = russia_expand.6.a
		default_option = yes
		if = {
			limit = {
				exists = c:ALK
				exists = c:ATB
				c:ATB ?= {
					has_state_in_state_region = STATE_ALASKA
				}
				c:ALK ?= {
					is_subject_of = ROOT
				}
			}
			s:STATE_ALASKA = {
				every_scope_state = {
					limit = {
						owner ?= {
							c:ATB ?= this
							is_country_type = decentralized
						}
					}
					set_state_owner = c:ALK
				}
			}
		}
		if = {
			limit = {
				exists = c:ATB
				NOT = {
					c:USA ?= {
						has_state_in_state_region = STATE_YUKON_TERRITORY
					}
				}
				c:ATB ?= {
					has_state_in_state_region = STATE_ALASKA
				}
				OR = {
					NOT = {
						exists = c:ALK
					}
					AND = {
						exists = c:ALK
						NOT = {
							c:ALK ?= {
								is_subject_of = ROOT
							}
						}
					}
				}
			}
			s:STATE_ALASKA = {
				every_scope_state = {
					limit = {
						owner ?= {
							c:ATB ?= this
							is_country_type = decentralized
						}
					}
					set_state_owner = ROOT
				}
			}
		}
		if = {
			limit = {
				exists = c:USA
				exists = c:ATB
				c:ATB ?= {
					has_state_in_state_region = STATE_YUKON_TERRITORY
				}
				c:USA ?= {
					has_state_in_state_region = STATE_YUKON_TERRITORY
				}
			}
			s:STATE_YUKON_TERRITORY = {
				every_scope_state = {
					limit = {
						owner ?= {
							c:ATB ?= this
							is_country_type = decentralized
						}
					}
					set_state_owner = c:USA
				}
			}
		}
		if = {
			limit = {
				exists = c:HBC
				exists = c:ATB
				c:ATB ?= {
					has_state_in_state_region = STATE_YUKON_TERRITORY
				}
				c:HBC ?= {
					is_subject_of = c:GBR
				}
			}
			s:STATE_YUKON_TERRITORY = {
				every_scope_state = {
					limit = {
						owner ?= {
							c:ATB ?= this
							is_country_type = decentralized
						}
					}
					set_state_owner = c:HBC
				}
			}
		}
		if = {
			limit = {
				exists = c:ATB
				exists = c:CAN
				c:ATB ?= {
					has_state_in_state_region = STATE_YUKON_TERRITORY
				}
				OR = {
					NOT = {
						exists = c:HBC
					}
					AND = {
						exists = c:CAN
						NOT = {
							c:CAN ?= {
								is_subject_of = c:GBR
							}
						}
					}
				}
			}
			s:STATE_YUKON_TERRITORY = {
				every_scope_state = {
					limit = {
						owner ?= {
							c:ATB ?= this
							is_country_type = decentralized
						}
					}
					set_state_owner = c:CAN
				}
			}
		}
		create_diplomatic_pact = {
			country = c:GBR
			type = rivalry
		}
		custom_tooltip = {
			text = gbr_attack_1_tt
			c:GBR ?= {
				change_relations = {
					country = c:RUS
					value = -5
				}
				set_secret_goal = {
					country = c:RUS
					secret_goal = conquer
				}
			}
		}
		add_journal_entry = {
			type = je_rus_great_game_japan
		}
		add_journal_entry = {
			type = je_rus_great_game_china
		}
		add_journal_entry = {
			type = je_rus_great_game_india
		}
		add_journal_entry = {
			type = je_rus_great_game_persia
		}
		add_journal_entry = {
			type = je_rus_great_game_turkey
		}
	}
}
