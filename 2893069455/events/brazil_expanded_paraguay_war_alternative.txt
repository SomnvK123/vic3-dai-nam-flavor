namespace = par_war_alternative
#THESE ARE THE ALTER HISTORY EVENTS OF PARAGUAY WAR
par_war_alternative.1 = {
	#PARAGUAY DECLARES WAR ON ARGENTINA AFTER PUPPET URU
	type = country_event
	placement = ROOT
	title = par_war_alternative.1.t
	desc = par_war_alternative.1.d
	flavor = par_war_alternative.1.f
	event_image = {
		video = "gfx/event_pictures/europenorthamerica_before_the_battle.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/europenorthamerica/before_the_battle"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3
    dlc = dlc_ra_plus
	trigger = {
	}
	immediate = {
		c:PRG ?= {
			create_diplomatic_pact = {
				country = c:URU
				type = puppet
			}
		}
		if = {
			limit = {
				exists = c:PRG
				exists = c:ARG
			}
			c:PRG ?= {
				save_temporary_scope_value_as = {
					name = war_reparations_amount
					value = {
						value = income
						multiply = define:NDiplomacy|DEFAULT_WAR_REPARATIONS_MONEY_TRANSFER
					}
				}
			}
			c:PRG ?= {
				create_diplomatic_play = {
					name = paraguay_war
					target_state = s:STATE_CORRIENTES.region_state:ARG
					war = yes
					type = dp_conquer_state
					add_war_goal = {
						# paraguay claim Corrientes
						holder = c:PRG
						type = conquer_state
						target_state = s:STATE_CHACO.region_state:ARG
					}
					add_war_goal = {
						# paraguay claim to santa fé
						holder = c:PRG
						type = conquer_state
						target_state = s:STATE_SANTA_FE.region_state:ARG
					}
					add_war_goal = {
						holder = c:ARG
						target_country = c:PRG
						type = enforce_treaty_article
						article = {
							article = money_transfer
							source_country = c:PRG
							target_country = c:ARG
							inputs = {
								{
									quantity = scope:war_reparations_amount
								}
							}
						}
					}
				}
			}
		}
	}
	option = {
		default_option = yes
		name = par_war_alternative.1.a		#OH OH FROM BR
		c:BRZ ?= {
			add_modifier = {
				name = br_failed_diplo
				months = 25
			}
		}
	}
}
