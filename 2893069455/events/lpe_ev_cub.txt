namespace = cub
cub.1 = {
	type = country_event
	placement = ROOT
	title = cub.1.t
	desc = cub.1.d
	flavor = cub.1.f
	gui_window = event_window_1char_tabloid
	left_icon = scope:cespedes_scope
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/europenorthamerica/london_center"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 3
     
	trigger = {
		exists = c:CUB
	}
	immediate = {
		c:CUB ?= {
			save_scope_as = cuba_scope
			random_scope_character = {
				limit = {
					has_template = CUB_carlos_manuel_de_cespedes
				}
				save_scope_as = cespedes_scope
				interest_group = {
					save_scope_as = grupo_indepe
				}
			}
			overlord = {
				save_scope_as = overlord_scope
			}
		}
	}
	option = {
		name = cub.1.a
		if = {
			limit = {
				this ?= c:CUB
			}
			scope:cespedes_scope = {
				add_modifier = cabecilla_independentista
			}
			scope:grupo_indepe = {
				add_modifier = {
					name = ig_independentista
					days = very_long_modifier_time
				}
			}
			if = {
				limit = {
					any_political_lobby = {
						is_political_lobby_type = lobby_anti_overlord
						target = overlord
					}
				}
				random_political_lobby = {
					limit = {
						is_political_lobby_type = lobby_anti_overlord
						target = overlord
					}
					add_lobby_member = scope:grupo_indepe
				}
			}
			else = {
				create_political_lobby = {
					type = lobby_anti_overlord
					target = overlord
					add_interest_group = scope:grupo_indepe
				}
			}
			overlord = {
				trigger_event = cub.1
			}
		}
		else = {
			show_as_tooltip = {
				c:CUB ?= {
					scope:cespedes_scope = {
						add_modifier = cabecilla_independentista
					}
					scope:grupo_indepe = {
						add_modifier = {
							name = ig_independentista
							days = very_long_modifier_time
						}
					}
				}
			}
		}
	}
}

cub.2 = {
	type = country_event
	placement = ROOT
	title = cub.2.t
	desc = cub.2.d
	flavor = cub.2.f
	gui_window = event_window_1char_tabloid
	left_icon = scope:maximo_gomez_scope
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 3
     
	trigger = {
		exists = c:CUB
	}
	immediate = {
		c:CUB ?= {
			save_scope_as = cuba_scope
			random_scope_character = {
				limit = {
					has_template = CUB_maximo_gomez
				}
				save_scope_as = maximo_gomez_scope
				interest_group = {
					save_scope_as = grupo_indepe
				}
			}
			overlord = {
				save_scope_as = overlord_scope
			}
		}
	}
	option = {
		name = cub.2.a
		if = {
			limit = {
				this ?= c:CUB
			}
			custom_tooltip = cub.2_tt
			add_journal_entry = {
				type = je_general_independentista_cub
			}
			scope:maximo_gomez_scope = {
				add_modifier = lider_independista
			}
			add_modifier = {
				name = agitadores_independentistas
				days = very_long_modifier_time
			}
			add_liberty_desire = 5
			overlord = {
				trigger_event = cub.2
			}
		}
		else = {
			show_as_tooltip = {
				c:CUB ?= {
					custom_tooltip = cub.2_tt
					scope:maximo_gomez_scope = {
						add_modifier = lider_independista
					}
					add_modifier = {
						name = agitadores_independentistas
						days = very_long_modifier_time
					}
					add_liberty_desire = 5
				}
			}
		}
	}
}
