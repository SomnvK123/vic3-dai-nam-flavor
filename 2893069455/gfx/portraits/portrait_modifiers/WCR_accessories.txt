WCR1_sash_accessories = {
    usage = game
    selection_behavior = weighted_random
    
    WCR1_generic_sashes = {
        dna_modifiers = {
            accessory = {
                mode = add
                gene = sashes
                template = all_sashes
                range = { 0 1 }
            }
        }
        weight = {
            base = 0
            modifier = {
                add = 100
                COM_WCR1_enable_outfit = no # COM compatibility
                agitator_character_trigger = no
                WCR1_has_completed_westernization_trigger = yes
                scope:character ?= {
                    age > 12
                    owner ?= {
                        WCR1_is_using_vanilla_rules = no
                    }
                }
                OR = {
                    AND = { # Generals
                        WCR1_army_clothes_trigger = yes
                        OR = {
                            WCR1_army_has_domestic_influence_trigger = yes
                            WCR1_army_has_russian_influence_trigger  = yes
                            WCR1_army_has_no_influence_trigger       = yes
                        }
                    }
                    AND = { # Admirals
                        WCR1_navy_clothes_trigger = yes
                        OR = {
                            WCR1_navy_has_domestic_influence_trigger = yes
                            WCR1_navy_has_russian_influence_trigger  = yes
                            WCR1_navy_has_no_influence_trigger       = yes
                        }
                    }
                    AND = { # Rulers
                        ruler_clothes_trigger = yes
                        OR = {
                            military_clothes_trigger = yes
                            monarchy_clothes_trigger = yes
                        }
                        OR = {
                            WCR1_ruler_has_domestic_influence_trigger = yes
                            WCR1_ruler_has_russian_influence_trigger  = yes
                            WCR1_ruler_has_no_influence_trigger       = yes
                        }
                    }
                }
            }
            modifier = { # Chinese navy has special logic to account for clipping problems
                add = -2000
                WCR1_has_completed_westernization_trigger = yes
                WCR1_navy_clothes_trigger = yes
                WCR1_navy_has_domestic_influence_trigger = yes
                scope:character ?= {
                    owner ?= {
                        OR = {
                        WCR1_st_is_country_tag = { TAG = CHI }
                        WCR1_st_is_country_tag = { TAG = ROC }
                    }
                    }
                }
            }
            modifier = { # CUP compatibility/deconflict
                add = -2000
                md_chinese_uniform_pack_active_trigger = yes
                scope:character ?= {
                    owner ?= {
                        OR = {
                        WCR1_st_is_country_tag = { TAG = CHI }
                        WCR1_st_is_country_tag = { TAG = ROC }
                    }
                    }
                }
            }
        }
    }
}
